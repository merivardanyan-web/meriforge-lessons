app/Models/Product.php — scopeFilter()
    /** Առաջադրանք 1-ի buildWhere()-ը՝ Eloquent-ով։ Ամեն պայման՝ միայն եթե ֆիլտրը կա */
    public function scopeFilter(Builder $query, array $filters): void
    {
        $query
            ->when($filters['category_id'] ?? null, fn (Builder $q, int $id) => $q->where('category_id', $id))
            ->when($filters['q'] ?? null, fn (Builder $q, string $text) => $q->where(fn (Builder $w) => $w
                ->where('name', 'like', '%' . addcslashes($text, '%_\\') . '%')
                ->orWhere('specs', 'like', '%' . addcslashes($text, '%_\\') . '%')))
            ->when(isset($filters['min']), fn (Builder $q) => $q->whereRaw('ROUND(price * (100 - discount) / 100) >= ?', [$filters['min']]))
            ->when(isset($filters['max']), fn (Builder $q) => $q->whereRaw('ROUND(price * (100 - discount) / 100) <= ?', [$filters['max']]))
            ->when(($filters['stock'] ?? 'all') === 'in', fn (Builder $q) => $q->where(fn (Builder $w) => $w->where('stock', '>', 0)->orWhere('type', 'digital')))
            ->when(($filters['stock'] ?? 'all') === 'sale', fn (Builder $q) => $q->where('discount', '>', 0));
    }
app/Http/Controllers/CheckoutController.php — տրանզակցիա
        $order = DB::transaction(function () use ($data, $items, $summary, $request) {
            $order = new Order(collect($data)->except('agree')->all() + [
                'shipping' => $summary['shipping'],
                'total'    => $summary['total'],
            ]);
            $order->user_id = $request->user()?->id;      // ⭐ ոչ ֆորմայից
            $order->save();

            foreach ($items as $item) {
                $product = $item['product'];

                $order->items()->create([
                    'product_id' => $product->id,
                    'quantity'   => $item['qty'],
                    'price'      => $product->finalPrice(),
                ]);

                if ($product->isDigital()) {
                    continue;
                }

                // UPDATE ... WHERE stock >= qty — վերջին ապրանքը երկու հոգու չի վաճառվի
                $updated = Product::whereKey($product->id)->where('stock', '>=', $item['qty'])->decrement('stock', $item['qty']);

                if ($updated === 0) {
                    // Բացառությունը → rollBack, իսկ ValidationException-ը → վերադարձ ֆորմա՝ սխալով
                    throw ValidationException::withMessages(['cart' => 'Պահեստում բավարար չէ՝ ' . $product->name]);
                }
            }

            return $order;
        });

class DemoSeeder extends Seeder
{
    public function run(): void
    {
        $password = env('DEMO_PASSWORD');

        if (! $password) {
            $this->command->error('Լրացրու DEMO_PASSWORD-ը .env-ում');
            return;
        }

        $user = User::firstOrNew(['email' => 'demo@technomarket.example.am']);
        $user->name = 'Demo';
        $user->password = $password;
        $user->role = 'manager';               // ոչ admin ⭐
        $user->save();

        $this->command->info('Demo հաշիվը պատրաստ է՝ manager դերով');
    }
}

05Ընդունման ստուգաթերթ

Դաս 18–21-ի պարտադիր փորձարկումները՝ քո նախագծում

  • /product/գոյություն-չունի՝ 404, /?sort=price;DROP TABLE products՝ սովորական ցուցակ
  • Ապրանք <script>alert(1)</script> անունով՝ տեքստ, ոչ alert։ {!! !!}-ը նախագծում՝ միայն nl2br(e(...))-ի հետ
  • Հանիր with('category')-ը ադմինի ցուցակից․ LazyLoadingViolationException։ Վերադարձրու
  • Անջատված ապրանքի հանրային հասցեն՝ 404, ջնջվածը՝ «Ջնջվածներ»-ում, վերականգնվում է
  • Ադմինի ֆորմայում դատարկ անուն և discount=150՝ հայերեն սխալներ, մյուս դաշտերը մնացել են
  • category_id=99999 curl-ով՝ Rule::exists-ը բռնում է
  • Ելքից հետո /admin/products/1/edit → մուտք → վերադարձ նույն էջ
  • Վեց սխալ գաղտնաբառ, ապա cookie-ների ջնջում՝ արգելափակումը մնում է (RateLimiter-ը՝ ըստ հասցեի և IP-ի)
  • Գրանցում DevTools-ով ավելացված role=admin-ով՝ customer, գլխավոր էջ, /admin/products՝ 403
  • Մենեջերը չի տեսնում «Ջնջել»-ը, իսկ ճիշտ token-ով DELETE-ը տալիս է 403
  • Անջատված JS-ով «Գնել»՝ էջը վերբեռնվում է, flash-ը երևում է, հաշվիչը աճել է

Հրապարակում

  • HTTPS, APP_DEBUG=false, /.env և /composer.json՝ 404
  • Demo հաշիվը (մենեջեր) մուտք է գործում, խմբագրում է, բայց չի կարող ջնջել
  • README՝ կենդանի հղում, demo հաշիվ, սքրինշոթ կամ GIF, «Security decisions», «How I used AI»
  • Պահոցում՝ առնվազն 15 իմաստալից commit մի քանի օրվա ընթացքում, ոչ մեկ երեկոյում
  • Lighthouse Accessibility՝ առնվազն 90 գլխավոր էջում և կատալոգում

06Պաշտպանություն · Դաս 22–24

Ամեն ուսանողին՝ 10 րոպե։ Սա նաև Դաս 24-ի փորձնական հարցազրույցի նախապատրաստություն է․ նույն հարցերը կտան աշխատանքի հարցազրույցում, երբ բացես քո GitHub-ը։

Ինչ ցույց տալՐոպե
Ցուցադրություն՝ հրապարակված կայքում․ կատալոգ → ապրանք → զամբյուղ → պատվեր, ապա demo հաշվով ադմին3
Մեկ հարցում՝ սկզբից մինչև վերջ․ POST /cart/items-ը՝ ուղի → middleware → կոնտրոլեր → Cart → JSON2
Անվտանգության երեք որոշում և թե ինչ կլիներ առանց դրանց2
Մեկ բան, որ դասընթացում չկար (Դաս 22-ի տնային․ վարկանիշ, wishlist, email...)2
Հարցեր1

Հնարավոր հարցեր

  • Ինչու՞ է դասավորման սյունակը սպիտակ ցուցակով, եթե Eloquent-ը պաշտպանում է SQL injection-ից (Դաս 10, 18)
  • Ինչու՞ role-ը $fillable-ում չէ։ Ինչ կլիներ, եթե լիներ (Դաս 20)
  • Որտեղ է ստուգվում, որ մենեջերը չի կարող ջնջել։ Ինչու՞ @can-ը բավական չէ (Դաս 21)
  • Ինչ է N+1-ը, և ինչպես ես համոզվել, որ քո ցուցակներում այն չկա (Դաս 20)
  • Ինչու՞ է պատվերի տողում պահվում գինը, եթե այն կա products-ում (Դաս 9)
  • Ինչ կլինի, եթե երկու գնորդ միաժամանակ գնեն վերջին ապրանքը (Դաս 11)
  • Ինչպես է CSRF token-ը հասնում fetch հարցմանը, և ինչու՞ GET-ը դրա կարիքը չունի (Դաս 16, 17)
  • Ինչու՞ է «Գնել»-ը ֆորմա, ոչ հղում, նույնիսկ JS-ով (Դաս 7, 17)
  • Ինչ ես փոխել Laravel-ի լռելյայն կարգավորումներից և ինչու (fortify.home, shouldBeStrict, APP_LOCALE)
  • Ինչ կանեիր այլ կերպ, եթե սկսեիր նորից
