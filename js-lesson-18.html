03export և import

js/utils.js — արտահանում
// Տարբերակ 1 — ուղիղ հայտարարման ժամանակ
export const formatPrice = n => n.toLocaleString("ru-RU") + " ֏";

export function debounce(fn, delay = 300) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

// Ներքին ֆունկցիա — ՉԻ արտահանվում, դրսից անհասանելի է
function pad(n) {
  return String(n).padStart(2, "0");
}

export function formatTime(date) {
  return `${pad(date.getHours())}:${pad(date.getMinutes())}`;
}

export function save(key, value) {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch (e) {
    console.warn("Չհաջողվեց պահել", e);
  }
}

export function load(key, fallback = null) {
  try {
    const raw = localStorage.getItem(key);
    return raw === null ? fallback : JSON.parse(raw);
  } catch (e) {
    return fallback;
  }
}

08Bundler-ների մասին

Մեր նախագիծն ունի 6 մոդուլ։ Բրաուզերը դրանք ներբեռնում է առանձին՝ 6 հարցում։ Իսկ իրական նախագծերում մոդուլները հարյուրավոր են։

10Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Մոդուլներ», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

Նախագիծ · Առաջադրանք 2

ՏեխնոՄարկետը մոդուլներով, ապրանքի էջ, ֆորմաներ՝ Առաջադրանք 2։

  1. Տեղադրիր Live Server-ը VS Code-ում և բացիր քո կատալոգը դրանով։ Համոզվիր, որ հասցեն սկսվում է http://-ով։
  2. Ստեղծիր երկու ֆայլ՝ math.js (արտահանում է add, multiply) և main.js։ Ստուգիր, որ import-ը աշխատում է։
  3. Ավելացրու math.js-ում ներքին ֆունկցիա, որը չես արտահանում։ Փորձիր դիմել դրան main.js-ից և գրիր՝ ինչ սխալ ստացար։
  4. Դիտավորյալ գրիր import { add } from "math.js" (առանց ./)։ Գրիր՝ ինչ սխալ ստացար։ Հետո ուղղիր։
  5. Փորձիր export default։ Ներմուծիր այն երկու տարբեր անուններով երկու ֆայլում։
  6. Վերակառուցիր քո կատալոգը մոդուլներով՝ data.js, storage.js, utils.js, catalog.js, cart.js, main.js։
  7. Համոզվիր, որ ամեն մոդուլ արտահանում է միայն այն, ինչ իսկապես պետք է դրսում։ Ներքին օգնականները թողիր առանց export-ի։
  8. Նկարիր (թղթի վրա կամ ֆայլում) քո նախագծի կախվածությունների գրաֆը։ Համոզվիր, որ օղակ չկա։
  9. Ավելացրու նոր մոդուլ՝ wishlist.js, հավանածների ցուցակի համար՝ իր localStorage բանալիով։
  10. Commit արա և push արա GitHub։ Ստուգիր, որ GitHub Pages-ի վրա էլ ամեն ինչ աշխատում է (Pages-ը սերվեր է, ուստի մոդուլները պիտի աշխատեն)։
