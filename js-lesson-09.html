01Ի՞նչ է DOM-ը

DOM = Document Object Model։ Երբ բրաուզերը կարդում է քո HTML ֆայլը, նա տեքստից կառուցում է օբյեկտների ծառ հիշողության մեջ։ JS-ը աշխատում է հենց այդ ծառի հետ, ոչ թե ֆայլի։

քո HTML-ը
<main>
  <section class="hero">
    <div class="row">
      <div class="col">
        <div class="wrapper">
          <h1>Վերնագիր</h1>
          <p>Տեքստ</p>
        </div>
      </div>
    </div>
  </section>
</main>

Ընտանեկան բառապաշար

div.col-ը ծնողն է div.wrapper-ի։
h1-ը և p-ն երեխաներ են և իրար հարևաններ (sibling)։
main-ը h1-ի նախնին է (ancestor)։

Ամեն հանգույց ծառում սովորական JS օբյեկտ է՝ հատկություններով և մեթոդներով։ Հենց դրա համար էր պետք Դաս 7-ը։

03querySelector — մեկ էլեմենտ

Ամենակարևոր մեթոդն է։ Ներսում գրում ես ճիշտ նույն սելեկտորը, ինչ CSS-ում։ Ոչ մի նոր բան սովորելու կարիք չկա։

CSS-ի սելեկտորները՝ JS-ում
document.querySelector("h1");              // թեգով
document.querySelector(".wrapper");        // դասով
document.querySelector("#header");         // id-ով
document.querySelector(".row .col");       // ներդրված
document.querySelector("section.hero .wrapper");
document.querySelector("input[type='email']");
document.querySelector(".col:first-child");
Կարևոր

// 1. Քանի՞ ապրանք կա
const cards = document.querySelectorAll(".col");
console.log(`Ապրանքների քանակը՝ ${cards.length}`);

// 2. Բոլոր անունները
const titles = [...document.querySelectorAll(".title")];
console.log(titles.map(t => t.textContent));

// 3. Բոլոր գները՝ թվերով (textContent-ը միշտ տող է)
const prices = [...document.querySelectorAll(".price")]
  .map(p => Number(p.textContent));
console.log(prices);

// 4. Ընդհանուր և միջին արժեքը (Դաս 6)
const total = prices.reduce((a, b) => a + b, 0);
console.log(`Ընդհանուր՝ ${total.toLocaleString("hy-AM")} ֏`);
console.log(`Ամենաթանկը՝ ${Math.max(...prices)} ֏`);

// 5. Ամեն քարտի ներսում որոնում
cards.forEach((card, i) => {
  const name  = card.querySelector(".title").textContent;
  const price = card.querySelector(".price").textContent;
  console.log(`${i + 1}. ${name} — ${price} ֏`);
});

// 6. Կոճակից բարձրանալ դեպի իր քարտը
const firstBtn = document.querySelector(".btn");
const itsCard = firstBtn.closest(".col");
console.log(itsCard.querySelector(".title").textContent);

Կոճակից closest(".col")-ով բարձրանալ, հետո querySelector-ով ներս իջնել — սա ամենակարևոր օրինաչափությունն է DOM-ի հետ աշխատելիս։ Դաս 11–12-ում այս երկու տողը կդառնա քո ամենահաճախ գրածը։

09Խաղահրապարակ

Այս անգամ խաղահրապարակը աշխատում է իսկական DOM-ի հետ՝ հենց այս էջի։ Փորձիր տարբեր սելեկտորներ և տես, թե ինչ է գտնում։

10Տնային աշխատանք

Օգտագործիր քո սեփական HTML/CSS նախագիծը նախորդ բլոկից։ Ամեն պատասխան տպիր console.log-ով։

  1. Ընտրիր էջի h1-ը և տպիր դրա տեքստը։
  2. Հաշվիր՝ քանի section, քանի .row և քանի .col կա էջում։
  3. Տպիր բոլոր h2-ների տեքստերը զանգվածով (օգտագործիր [...] և map)։
  4. Ընտրիր վերջին .col-ը և տպիր դրա ներսի ամբողջ տեքստը։
  5. Ընտրիր առաջին .wrapper-ը։ Տպիր դրա երեխաների քանակը և ամեն երեխայի tagName-ը։
  6. Ընտրիր ցանկացած ներքին էլեմենտ և closest-ով գտիր դրա section-ը։ Տպիր section-ի դասը։
  7. Ցիկլով անցիր բոլոր .col-երով և ամեն մեկի ներսում գտիր վերնագիրը՝ col.querySelector(...)։
  8. Փորձիր ընտրել գոյություն չունեցող դաս։ Տպիր արդյունքը և ապահով կերպով մշակիր ?.-ով։
  9. Դիտավորյալ հանիր defer-ը <script>-ից և տես, թե ինչ է լինում։ Գրիր մեկնաբանության մեջ՝ ինչ սխալ ստացար։ Հետո վերադարձրու։
  10. Եթե էջում գներ կան՝ հավաքիր դրանք զանգվածում, վերածիր թվերի և տպիր ընդհանուրը, միջինը և ամենաթանկը։

11Հիշելու թերթիկ

ԳրառումԻնչ է վերադարձնում
document.querySelector(".x")Առաջին էլեմենտը կամ null
document.querySelectorAll(".x")NodeList (կարող է դատարկ լինել)
[...document.querySelectorAll(".x")]Իսկական զանգված — map/filter հասանելի են
el.querySelector(".y")Որոնում միայն el-ի ներսում
el.parentElementՈւղիղ ծնողը
el.closest(".x")Մոտակա նախնին, որ համապատասխանում է
el.childrenՈւղիղ երեխաները
el.firstElementChildԱռաջին երեխան
el.lastElementChildՎերջին երեխան
el.nextElementSiblingՀաջորդ հարևանը
el.previousElementSiblingՆախորդ հարևանը
el.textContentՆերսի տեքստը (միշտ տող)
el.tagNameԹեգի անունը մեծատառերով
el.classNameԴասերը մեկ տողով

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել՝ ինչ է DOM ծառը
  • Գիտե՞մ տարբերությունը querySelector-ի և querySelectorAll-ի արդյունքների միջև
  • Հասկանո՞ւմ եմ, ինչու NodeList-ի վրա map չկա
  • Կարո՞ղ եմ բացատրել, ինչու է defer-ը պարտադիր
  • Գիտե՞մ, ե՞րբ օգտագործել closest-ը parentElement-ի փոխարեն
Հաջորդ դասին

DOM-ի փոփոխում։ Այսօր միայն գտնում էինք էլեմենտները — հաջորդ դասին կսկսենք դրանք փոխել․ տեքստ, դասեր, ոճեր, և ամենակարևորը՝ նոր էլեմենտներ ստեղծել JS-ով և դնել էջում։
