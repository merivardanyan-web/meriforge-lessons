02debounce-ի գաղտնիքը

Հիմա վերադառնանք այն ֆունկցիային, որը Դաս 16-ից օգտագործում ենք։

վերջապես պարզ է
function debounce(fn, delay = 300) {
  let timer;                       // ← ապրում է փակույթում

  return function(...args) {
    clearTimeout(timer);          // տեսնում է նախորդ կանչի timer-ը
    timer = setTimeout(() => fn(...args), delay);
  };
}

Ամեն կանչի ժամանակ վերադարձված ֆունկցիան հիշում է նույն timer-ը։ Հենց դրա համար կարող է չեղարկել նախորդը։ Առանց փակույթի timer-ը պիտի լիներ գլոբալ։

ևս երեք օգտակար օրինակ
// 1. Կատարել միայն մեկ անգամ
function once(fn) {
  let called = false;
  let result;

  return function(...args) {
    if (called) return result;
    called = true;
    result = fn(...args);
    return result;
  };
}

const init = once(() => {
  console.log("Գործարկում");
  return "պատրաստ";
});

console.log(init());
console.log(init());

function memoize(fn) {
  const cache = {};

  return function(n) {
    if (n in cache) {
      console.log("քեշից՝", n);
      return cache[n];
    }
    cache[n] = fn(n);
    return cache[n];
  };
}

function parseConfig(json) {
  try {
    return JSON.parse(json);

  } catch (error) {
    if (error instanceof SyntaxError) {
      console.warn("Փչացած JSON, օգտագործում ենք լռելյայնը");

    throw error;      // չգիտենք՝ ինչ է — փոխանցում ենք վերև

  } finally {
    console.log("Վերլուծությունն ավարտվեց");
  }
}

05Սեփական սխալների դասեր

Դաս 22-ի ժառանգումը կիրառելու հիանալի առիթ։

extends Error
class ValidationError extends Error {
  constructor(field, message) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

class ApiError extends Error {
  constructor(status, message) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }

  get isServerError() {
    return this.status >= 500;
  }
}

function validate(user) {
  if (!user.email) {
    throw new ValidationError("email", "Էլ. հասցեն պարտադիր է");
  }
  return true;
}

try {
  validate({ name: "Անի" });
} catch (error) {
  if (error instanceof ValidationError) {
    console.log("Դաշտ՝", error.field);
    console.log("Հաղորդագրություն՝", error.message);
  } else {
    throw error;
  }
}

կիրառում Դաս 21-ի API շերտում
async function request(path, options) {
  const response = await fetch(path, options);

  if (!response.ok) {
    throw new ApiError(response.status, `Հարցումը ձախողվեց՝ ${path}`);
  }

  return response.json();
}

try {
  await request("/users");
} catch (error) {
  if (error instanceof ApiError && error.isServerError) {
    showMessage("Սերվերի խնդիր։ Փորձեք ավելի ուշ։");
  } else if (error instanceof ApiError) {
    showMessage("Տվյալները չգտնվեցին։");
  } else {
    showMessage("Ստուգեք ինտերնետը։");
  }
}
