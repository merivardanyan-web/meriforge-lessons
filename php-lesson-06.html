Ահա գլխավոր միտքը

HTML-ի վալիդացիան օգտագործողի հարմարության համար է, ոչ անվտանգության։

Այն ասում է բարեխիղճ օգտագործողին․ «մոռացել ես դաշտը»։ Դա արժեքավոր է — բայց հարձակվողին այն ընդհանրապես չի վերաբերում։

Սերվերը երբեք չի կարող վստահել այն ամենին, ինչ գալիս է բրաուզերից։ Ոչ ֆորմայի դաշտերին, ոչ cookie-ներին, ոչ hidden դաշտերին, ոչ URL-ի պարամետրերին։

Երկուսն էլ պետք են

Սա չի նշանակում՝ հրաժարվել required-ից։

Բրաուզերի վալիդացիա → արագ արձագանք, առանց էջի վերբեռնման, լավ UX
Սերվերային վալիդացիա → իրական պաշտպանություն

Գրիր երկուսն էլ։ Բայց երբեք մի՛ ապավինիր միայն առաջինին։

Երբ արժեքը պիտի լինի սահմանափակ բազմությունից՝ կատեգորիա, տեսակավորման ձև, կարգավիճակ — միշտ ստուգիր ցուցակով․

Այս օրինաչափությունը Դաս 10-ում կփրկի SQL injection-ից, երբ տեսակավորման դաշտը գնա ուղիղ հարցման մեջ։

04Վալիդացիայի ֆունկցիա

validate.php
<?php
declare(strict_types=1);

/**
 * Վերադարձնում է սխալների զանգված։
 * Դատարկ զանգված = ամեն ինչ կարգին է։
 */
function validateContact(array $data): array
{
    $errors = [];

    // --- Անուն ---
    $name = trim($data['name'] ?? '');

    if ($name === '') {
        $errors['name'] = 'Խնդրում ենք նշել ձեր անունը';
    } elseif (mb_strlen($name) < 2) {
        $errors['name'] = 'Անունը շատ կարճ է';
    } elseif (mb_strlen($name) > 60) {
        $errors['name'] = 'Անունը շատ երկար է (առավելագույնը 60 նիշ)';
    }

    // --- Էլ. հասցե ---
    $email = trim($data['email'] ?? '');

    if ($email === '') {
        $errors['email'] = 'Խնդրում ենք նշել էլ. հասցեն';
    } elseif (filter_var($email, FILTER_VALIDATE_EMAIL) === false) {
        $errors['email'] = 'Էլ. հասցեն վավեր չէ';
    }

    // --- Թեմա՝ ցուցակից ---
    $allowed = ['order', 'support', 'other'];
    $topic = $data['topic'] ?? '';

    if (!in_array($topic, $allowed, true)) {
        $errors['topic'] = 'Ընտրեք թեման';
    }

    // --- Հաղորդագրություն ---
    $message = trim($data['message'] ?? '');
    $len = mb_strlen($message);

    if ($len < 10) {
        $errors['message'] = 'Հաղորդագրությունը պիտի լինի առնվազն 10 նիշ';
    } elseif ($len > 2000) {
        $errors['message'] = 'Հաղորդագրությունը շատ երկար է';
    }

    return $errors;
}
Ինչո՞ւ վերադարձնում ենք զանգված, ոչ bool

true/false-ը ասում է միայն «սխալ կա»։ Օգտագործողը պիտի կռահի, թե որտեղ։

Զանգվածը՝ ['email' => 'Էլ. հասցեն վավեր չէ'], թույլ է տալիս ցույց տալ սխալը հենց այդ դաշտի կողքին։

Բանալիները համընկնում են դաշտերի name-երին — դա հեշտացնում է ցուցադրումը և հետագայում կհեշտացնի Laravel-ին անցումը, որտեղ ճիշտ նույն կառուցվածքն է։

Առավելագույն երկարությունը մի՛ մոռացիր

<textarea>-ում կարելի է տեղադրել 50 ՄԲ տեքստ։ Առանց վերին սահմանի՝

· Բազայի դաշտը կլցվի կամ կկտրվի
· Էջը կդանդաղի ցուցադրելիս
· Հեշտ է կազմակերպել ծանրաբեռնում

Ամեն տեքստային դաշտ պիտի ունենա վերին սահման։ Միշտ։

06POST/Redirect/GET

Խնդիրը

Օգտագործողը ուղարկեց ֆորման, տեսավ «Շնորհակալություն», ապա սեղմեց F5։

Բրաուզերը հարցնում է․ «Կրկին ուղարկե՞լ տվյալները»։ Եթե սեղմի «Այո» — հաղորդագրությունը կուղարկվի երկրորդ անգամ։

Պատվերի դեպքում դա երկու պատվեր է։ Վճարման դեպքում՝ երկու վճարում։

Առանց PRG PRG-ով POST /contact.php POST /contact.php │ │ Ցուցադրում ենք էջը 302 Redirect │ │ F5 → կրկին POST ⚠️ GET /contact.php?sent=1 │ Ցուցադրում ենք էջը │ F5 → կրկին GET ✅
contact.php — կոնտրոլեր
<?php
declare(strict_types=1);

session_start();

require_once __DIR__ . '/functions.php';
require_once __DIR__ . '/validate.php';

$topics = [
    'order'   => 'Պատվեր',
    'support' => 'Տեխնիկական աջակցություն',
    'other'   => 'Այլ',
];

$errors  = [];
$old     = [];
$success = false;

// --- Մշակում ենք POST-ը ---
if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    $errors = validateContact($_POST);

    if ($errors === []) {

        // Այստեղ կուղարկեինք նամակ կամ կպահեինք բազայում

        $_SESSION['flash'] = 'sent';

        header('Location: /contact.php', true, 303);
        exit;                       // ⭐ պարտադիր
    }

    // Սխալ կա — պահում ենք լրացրածը
    $old = $_POST;
}

// --- Ցուցադրում ---
if (isset($_SESSION['flash'])) {
    $success = $_SESSION['flash'] === 'sent';
    unset($_SESSION['flash']);   // ցույց տալիս ջնջում ենք
}

$content = render('pages/contact', compact('errors', 'old', 'success', 'topics'));

echo render('layout', [
    'title'   => 'Կապ — ՏեխնոՄարկետ',
    'content' => $content,
]);
Երկու պարտադիր դետալ

1. exit header()-ից հետո
header()-ը միայն ուղարկում է վերնագիր։ Սկրիպտը շարունակում է աշխատել։ Առանց exit-ի կոդը կկատարվի մինչև վերջ — և եթե ներքևում բազայի գրառում կա, այն կկատարվի։

2. Ոչ մի ելք header()-ից առաջ
Եթե որևէ բան արդեն ուղարկվել է բրաուզերին՝ նույնիսկ մեկ բացատ <?php-ից առաջ — կստանաս «Cannot modify header information — headers already sent»։

Հենց դրա համար Դաս 1-ում ասացինք՝ մաքուր PHP ֆայլերում փակող ?> մի՛ գրիր։

Ինչո՞ւ 303, ոչ 302

303 See Other-ը բացահայտ ասում է․ «կատարիր GET այս հասցեով»։

302-ը պատմականորեն երկիմաստ է — որոշ հին client-ներ կրկնում են POST-ը։ 303-ը հենց PRG-ի համար է ստեղծված։

Երբ ինքդ ես շրջանցում սեփական պաշտպանությունը և տեսնում, որ սերվերը դիմանում է — դա այն պահն է, երբ սերվերային վալիդացիան դադարում է «ևս մեկ քայլ» լինելուց։

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ ««Առաջարկիր գիրք» ֆորմա», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Կատարիր դասարանականի 10 քայլը։
  2. Ուղարկիր curl-ի հրամանը և պատասխանը՝ սքրինշոթով։
  3. Ցուցադրիր, ինչ է լինում, երբ exit-ը բացակայում է header()-ից հետո։
  4. Դիտմամբ դիր բացատ <?php-ից առաջ և բացատրիր սխալը։
  5. Ավելացրու հեռախոսի դաշտ preg_match-ով վալիդացիայով։
  6. Ավելացրու checkbox՝ «Համաձայն եմ պայմաններին»։ Ինչպե՞ս ես ստուգում այն PHP-ում։
  7. Ցուցադրիր, ինչու in_array-ի երրորդ արգումենտը կարևոր է։ Բեր օրինակ, որտեղ առանց դրա ստուգումն անցնում է սխալմամբ։
  8. Ցուցադրիր filter_var-ի 0-ի թակարդը՝ !-ով և === false-ով։
  9. Ավելացրու ամեն դաշտի վերին սահման։ Փորձարկիր 5000 նիշանոց տեքստով։
  10. Ստուգիր ֆորման միայն ստեղնաշարով և էկրանընթերցիչով (NVDA կամ VoiceOver)։ Սխալները կարդացվո՞ւմ են։

Վալիդացիայի օրինաչափությունը

1. Կարդալ $x = trim($_POST['x'] ?? '') 2. Ստուգել $errors['x'] = '...' 3. Եթե սխալ չկա → գործողություն → redirect → exit 4. Եթե սխալ կա → $old = $_POST → ցուցադրել ֆորման

Ստուգաթերթ

  • Ամեն դաշտ ստուգվում է սերվերում
  • Ամեն տեքստային դաշտ ունի վերին սահման
  • Ընտրովի արժեքները ստուգվում են in_array-ով
  • Սխալները ցուցադրվում են դաշտերի կողքին
  • Լրացրած արժեքները չեն կորչում
  • Կա aria-invalid և aria-describedby
  • Հաջողությունից հետո՝ redirect
  • exit-ը կա header()-ից հետո
  • F5-ը չի կրկնում ուղարկումը
  • Ստուգված է curl-ով

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու required-ը անվտանգություն չէ
  • Գիտե՞մ, ե՞րբ GET, ե՞րբ POST
  • Հասկանո՞ւմ եմ, ինչու $_REQUEST-ը վտանգավոր է
  • Կարո՞ղ եմ բացատրել PRG-ն
  • Գիտե՞մ, ինչու exit-ը պարտադիր է
Հաջորդ դասին

Սեսիաներ, cookie և ֆայլերի վերբեռնում։ Այսօր արդեն օգտագործեցինք $_SESSION-ը flash-հաղորդագրության համար՝ առանց բացատրության։ Վաղը կհասկանանք, ինչպես է այն աշխատում։

Կկառուցենք նաև զամբյուղ սեսիայի վրա և կսովորենք ֆայլերի ապահով վերբեռնում — որտեղ կան իրենց խոցելիությունները։
