01Ինչո՞ւ է պետք օբյեկտը

Փորձենք զանգվածով նկարագրել մեկ աշակերտի։

զանգվածով — վատ գաղափար
const student = ["Անի", 14, "Երևան", true];

console.log(student[0]);
console.log(student[3]);   // ի՞նչ է սա նշանակում
Console
Անի
true

student[3]-ը true է։ Բայց ի՞նչ է true։ Ակտիվ է։ Ամուսնացած է։ Դասը հանձնե՞լ է։ Ոչ ոք չգիտի։ Բացի այդ՝ պիտի անգիր անես, որ 1-ը տարիքն է, 2-ը՝ քաղաքը։

Զանգված — շատ նույնատիպ բան, հերթականությունը կարևոր է, հասցեն՝ թիվ։
Օբյեկտ — մեկ բանի տարբեր հատկություններ, հերթականությունը կարևոր չէ, հասցեն՝ անուն։

02Կարդալ, փոխել, ավելացնել, ջնջել

հիմնական գործողությունները
const user = {
  name: "Դավիթ",
  age: 20
};

// կարդալ
console.log(user.name);

// փոխել
user.age = 21;

// ավելացնել նոր հատկություն
user.city = "Գյումրի";
user.isAdmin = false;

// ջնջել
delete user.isAdmin;

console.log(user);
console.log(user.email);      // չկա այդպիսի հատկություն
Console
Դավիթ
{"name":"Դավիթ","age":21,"city":"Գյումրի"}
undefined
const-ը այստեղ էլ չի խանգարում

Ինչպես զանգվածների դեպքում՝ const-ը արգելում է փոխարինել ամբողջ օբյեկտը, բայց ոչ նրա հատկությունները։ user = {} → սխալ։ user.age = 21 → կարելի է։

Կետ, թե՞ քառակուսի փակագծեր

երկու ձև
const user = { name: "Անի", "favorite color": "կապույտ" };

console.log(user.name);                // կետով — սովորական դեպք

console.log(user["favorite color"]);

const key = "name";
console.log(user[key]);
console.log(user.key);                 // փնտրում է "key" անունով հատկություն

կապույտ

03Ներդրված օբյեկտներ

Հատկության արժեքը կարող է լինել ցանկացած բան՝ նաև ուրիշ օբյեկտ կամ զանգված։ Իրական տվյալները միշտ այսպիսի տեսք ունեն։

օբյեկտ օբյեկտի մեջ
const user = {
  name: "Անի",
  age: 14,

    building: 25
  },
  hobbies: ["նկարչություն", "շախմատ", "լող"]
};

console.log(user.address.city);
console.log(user.hobbies[0]);
console.log(user.hobbies.length);
console.log(user.hobbies.join(", "));

նկարչություն

Երբ խորությունը վտանգավոր է դառնում

դասական սխալ
const user = { name: "Դավիթ" };   // address չկա

console.log(user.address);
console.log(user.address.city);

Սա ամենահաճախ հանդիպող սխալն է իրական նախագծերում։ user.address-ը undefined է, իսկ undefined-ից «քաղաք» կարդալ չի լինում։

?. — ապահով դիմում
const user = { name: "Դավիթ" };

console.log(user.address?.city);
console.log(user.address?.city ?? "Քաղաքը նշված չէ");
console.log(user.hobbies?.[0]);

?. կարդացվում է այսպես

«Եթե ձախ կողմը կա — շարունակիր։ Եթե չկա — կանգնիր և վերադարձրու undefined՝ առանց սխալ գցելու»։ Հիշիր Դաս 2-ի ??-ը — երկուսը հիանալի աշխատում են միասին։

04Մեթոդներ և this

Հատկության արժեքը կարող է լինել նաև ֆունկցիա։ Այդ դեպքում այն կոչվում է մեթոդ։

մեթոդ
const user = {
  name: "Անի",
  age: 14,

  sayHi() {                       // կարճ ձև
    console.log("Բարև։");
  },

  introduce: function() {          // երկար ձև — նույնն է
    console.log("Ես աշակերտ եմ։");
  }
};

user.sayHi();
user.introduce();
Console
Բարև։
Ես աշակերտ եմ։

this — «ես ինքս»

Մեթոդի ներսում this-ը նշանակում է այն օբյեկտը, որի կողքին կետն է կանգնած կանչի պահին։

this
const user = {
  name: "Անի",
  age: 14,

  introduce() {
    console.log(`Ես ${this.name} եմ, ${this.age} տարեկան։`);
  },

  birthday() {
    this.age++;
    return `Շնորհավոր։ Հիմա ${this.age} տարեկան ես։`;
  }
};

user.introduce();
console.log(user.birthday());
user.introduce();
Console
Ես Անի եմ, 14 տարեկան։
Շնորհավոր։ Հիմա 15 տարեկան ես։
Ես Անի եմ, 15 տարեկան։
Կարևոր զգուշացում

Մեթոդների համար մի՛ օգտագործիր arrow ֆունկցիա։ Arrow-ի ներսում this-ը չի ցույց տալիս օբյեկտին։ Այս թեմային կվերադառնանք Դաս 22-ում՝ OOP-ի ժամանակ։

ինչ է լինում arrow-ով
const user = {
  name: "Անի",
  bad: () => console.log(this.name),   // ❌
  good() { console.log(this.name); }      // ✅
};

user.bad();
user.good();
Console
undefined
Անի

05Անցնել օբյեկտով

Օբյեկտի վրա map կամ filter չկա։ Բայց այն կարելի է վերածել զանգվածի։

keys · values · entries
const scores = { math: 18, physics: 14, history: 20 };

console.log(Object.keys(scores));
console.log(Object.values(scores));
console.log(Object.entries(scores));
Console
["math","physics","history"]
[18,14,20]
[["math",18],["physics",14],["history",20]]

Իսկ երբ արդեն զանգված է — Դաս 6-ի բոլոր մեթոդները հասանելի են։

գործնական օգտագործում
const scores = { math: 18, physics: 14, history: 20 };

// միջին գնահատականը
const values = Object.values(scores);
const avg = values.reduce((a, b) => a + b, 0) / values.length;
console.log(avg.toFixed(1));

// գեղեցիկ տպել
Object.entries(scores).forEach(([subject, mark]) => {
  console.log(`${subject}: ${mark}`);
});

// քանի՞ առարկայից է 15-ից բարձր
console.log(values.filter(m => m > 15).length);
Console
17.3
math: 18
physics: 14
history: 20
2

for..in — ուղիղ ձև

for..in
const scores = { math: 18, physics: 14 };

for (const key in scores) {
  console.log(key, "→", scores[key]);
}
Console
math → 18
physics → 14
Մի՛ շփոթիր

for..of — զանգվածների համար, տալիս է արժեքները։
for..in — օբյեկտների համար, տալիս է բանալիները։

06Destructuring

Երբ օբյեկտից պիտի հանես մի քանի հատկություն, ձանձրալի է ամեն անգամ գրել user.։

հին ձև → նոր ձև
const user = { name: "Անի", age: 14, city: "Երևան" };

// հին ձև
const name1 = user.name;
const age1  = user.age;

// destructuring — մեկ տողով
const { name, age, city } = user;

console.log(name, age, city);
Console
Անի 14 Երևան
լրացուցիչ հնարավորություններ
const user = { name: "Անի", age: 14 };

// լռելյայն արժեք, եթե հատկությունը չկա
const { name, city = "Անհայտ" } = user;
console.log(name, city);

// վերանվանել
const { name: userName, age: userAge } = user;
console.log(userName, userAge);

// զանգվածների destructuring — ըստ հերթականության
const colors = ["կարմիր", "կանաչ", "կապույտ"];
const [first, second] = colors;
console.log(first, second);
Console
Անի Անհայտ
Անի 14
կարմիր կանաչ

Ամենակարևոր կիրառությունը՝ ֆունկցիաներում

պարամետրի destructuring
const user = { name: "Անի", age: 14, city: "Երևան" };

// առանց destructuring
function showUser1(u) {
  console.log(`${u.name}, ${u.age}, ${u.city}`);
}

// destructuring-ով — ավելի մաքուր
function showUser2({ name, age, city }) {
  console.log(`${name}, ${age}, ${city}`);
}

showUser1(user);
showUser2(user);
Console
Անի, 14, Երևան
Անի, 14, Երևան

07Պատճենել օբյեկտ

Նույն թակարդն է, ինչ զանգվածների դեպքում (Դաս 5)։ Օբյեկտները նույնպես պահվում են հասցեով։

spread օբյեկտների համար
const user = { name: "Անի", age: 14 };

// ❌ սա պատճեն չէ
const bad = user;
bad.age = 99;
console.log(user.age);

// ✅ իսկական պատճեն
const copy = { ...user };
copy.age = 14;
console.log(user.age, copy.age);
Console
99
99 14
պատճենել և փոխել միանգամից
const user = { name: "Անի", age: 14, city: "Երևան" };

// նոր օբյեկտ՝ փոխված տարիքով
const updated = { ...user, age: 15 };
console.log(updated);

// միացնել երկու օբյեկտ
const settings = { theme: "dark" };
console.log({ ...user, ...settings });
Console
{"name":"Անի","age":15,"city":"Երևան"}
{"name":"Անի","age":14,"city":"Երևան","theme":"dark"}
Հերթականությունը կարևոր է

{ ...user, age: 15 } — նոր արժեքը հետո է, ուստի հաղթում է։ Եթե գրեիր { age: 15, ...user }, user-ի հին տարիքը կվերագրեր նորը։

08Օբյեկտների զանգված

Ահա այն ձևը, որով իրական աշխարհում պահվում են բոլոր տվյալները՝ ապրանքներ, օգտագործողներ, հոդվածներ։ Այստեղ Դաս 6-ը և Դաս 7-ը միանում են։

students.js
const students = [
  { id: 1, name: "Անի",    age: 14, mark: 19, city: "Երևան"  },
  { id: 2, name: "Դավիթ",  age: 15, mark: 11, city: "Գյումրի" },
  { id: 3, name: "Մարիամ", age: 14, mark: 20, city: "Երևան"  },
  { id: 4, name: "Արամ",   age: 16, mark: 8,  city: "Վանաձոր" }
];

// բոլոր անունները
console.log(students.map(s => s.name));

// գերազանցիկները
console.log(students.filter(s => s.mark >= 18).map(s => s.name));

// գտնել ըստ id-ի
console.log(students.find(s => s.id === 3).name);

// միջին գնահատականը
const avg = students.reduce((sum, s) => sum + s.mark, 0) / students.length;
console.log(avg.toFixed(2));

// երևանցիները՝ գնահատականով նվազման կարգով
console.log(
  students
    .filter(s => s.city === "Երևան")
    .sort((a, b) => b.mark - a.mark)
    .map(s => `${s.name} (${s.mark})`)
);

// կա՞ անբավարար
console.log(students.some(s => s.mark < 10));
Console
["Անի","Դավիթ","Մարիամ","Արամ"]
["Անի","Մարիամ"]
Մարիամ
14.50
["Մարիամ (20)","Անի (19)"]

10Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Գիրքը որպես օբյեկտ», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

  1. Ստեղծիր book օբյեկտ՝ վերնագիր, հեղինակ, տարեթիվ, էջերի քանակ, isRead։ Տպիր բոլորը template literal-ով։
  2. Ավելացրու genres զանգված և publisher ներդրված օբյեկտ (անուն, քաղաք)։ Տպիր հրատարակչի քաղաքը։
  3. Ավելացրու getInfo() մեթոդ, որը this-ով վերադարձնում է «[վերնագիր] — [հեղինակ], [տարեթիվ]»։
  4. Ավելացրու markAsRead() մեթոդ, որը isRead-ը դարձնում է true։ Ստուգիր՝ տպելով նախքան և հետո։
  5. Ստեղծիր օբյեկտ առանց price հատկության։ Տպիր book.price ?? "Գինը նշված չէ"։ Ապա փորձիր book.publisher?.country։
  6. Destructuring-ով հանիր վերնագիրը և հեղինակին առանձին փոփոխականների։ Հեղինակին վերանվանիր writer։
  7. Ստեղծիր գնահատականների օբյեկտ (5 առարկա)։ Տպիր՝ բանալիները, արժեքները, միջինը և ամենաբարձր առարկան։
  8. Ստեղծիր 5 ապրանքի զանգված (անուն, գին, կատեգորիա, inStock)։ Գրիր ֆունկցիաներ՝ բոլոր առկա ապրանքները, ընդհանուր արժեքը, ամենաթանկը։
  9. Նույն զանգվածով՝ մեկ շղթայով գտիր որոշակի կատեգորիայի առկա ապրանքները, դասավորիր գնով և վերադարձրու «անուն — գին֏» տողերի զանգված։
  10. Գրիր updateProduct(product, changes) ֆունկցիա, որը վերադարձնում է նոր օբյեկտ՝ { ...product, ...changes }։ Ապացուցիր, որ բնօրինակը չի փոխվել։
