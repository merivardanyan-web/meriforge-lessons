storage.js
function save(key, value) {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch (error) {
    console.warn("Չհաջողվեց պահել՝", error);
  }
}

function load(key, fallback = null) {
  try {
    const raw = localStorage.getItem(key);
    return raw === null ? fallback : JSON.parse(raw);
  } catch (error) {
    console.warn("Փչացած տվյալ՝", error);
    return fallback;
  }
}

// Օգտագործում
save("todos", [{ id: 1, text: "Տնային" }]);

const todos = load("todos", []);       // չկա → դատարկ զանգված
const theme = load("theme", "light");  // չկա → light

console.log(todos.length, theme);

05setInterval — կրկնվող կատարում

կենդանի ժամացույց
const clock = document.querySelector(".clock");
const pad = n => String(n).padStart(2, "0");    // Դաս 8

function updateClock() {
  const now = new Date();
  clock.textContent =
    `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`;
}

updateClock();                    // անմիջապես, առանց սպասելու
setInterval(updateClock, 1000);   // հետո ամեն վայրկյան
Հաճախ մոռացվում է

setInterval-ը առաջին անգամ աշխատում է ինտերվալից հետո, ոչ անմիջապես։ Ուստի ֆունկցիան կանչիր նաև ձեռքով՝ որ առաջին վայրկյանին էջը դատարկ չլինի։

հետհաշվարկ — Դաս 8-ը կենդանանում է
const el = document.querySelector(".countdown");
const target = new Date("2027-01-01");

function tick() {
  const diff = target - new Date();

  if (diff <= 0) {
    el.textContent = "Շնորհավոր Նոր տարի։";
    clearInterval(intervalId);      // կանգնեցնում ենք
    return;
  }

  const days    = Math.floor(diff / 86400000);
  const hours   = Math.floor(diff / 3600000) % 24;
  const minutes = Math.floor(diff / 60000) % 60;
  const seconds = Math.floor(diff / 1000) % 60;

  el.textContent = `${days}օր ${hours}ժ ${minutes}ր ${seconds}վ`;
}

tick();
const intervalId = setInterval(tick, 1000);
Միշտ մաքրիր

Չկանգնեցված setInterval-ը շարունակում է աշխատել անվերջ՝ ուտելով ռեսուրսներ։ Երբ այն այլևս պետք չէ (հետհաշվարկն ավարտվեց, մոդալը փակվեց) — clearInterval։

07Դասարանական · To-do, որ ոչինչ չի մոռանում

Վերցնում ենք Դաս 12-ի to-do-ն և ավելացնում պահպանում։ Ուշադրություն դարձրու՝ որքան քիչ փոփոխություն է պետք։

storage.js
const STORAGE_KEY = "todos";

function save(key, value) {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch (e) {
    console.warn("Չհաջողվեց պահել", e);
  }
}

function load(key, fallback = null) {
  try {
    const raw = localStorage.getItem(key);
    return raw === null ? fallback : JSON.parse(raw);
  } catch (e) {
    return fallback;
  }
}
main.js — փոփոխությունները
// ★ 1. Բեռնում ենք սկզբում
let todos = load(STORAGE_KEY, []);

const form  = document.querySelector(".todo__form");
const input = document.querySelector(".todo__input");
const list  = document.querySelector(".todo__list");
const stats = document.querySelector(".todo__stats");

function createItem({ id, text, done }) {
  return `
    <li class="todo__item ${done ? "is-done" : ""}" data-id="${id}">
      <input type="checkbox" data-action="toggle" ${done ? "checked" : ""}>
      <span class="todo__text">${text}</span>
      <button data-action="edit">✎</button>
      <button data-action="delete">✕</button>
    </li>
  `;
}

// ★ 2. render-ը նաև պահպանում է
function render() {
  list.innerHTML = todos.length === 0
    ? `<li class="todo__empty">Ցուցակը դատարկ է։</li>`
    : todos.map(createItem).join("");

  const done = todos.filter(t => t.done).length;
  stats.textContent = `${done} / ${todos.length} կատարված`;

  save(STORAGE_KEY, todos);     // ← ամբողջ նոր տողը սա է
}

// Մնացած ամեն ինչ նույնն է, ինչ Դաս 12-ում
function addTodo(text) {
  todos.push({ id: Date.now(), text, done: false });
  render();
}

function deleteTodo(id) {
  todos = todos.filter(t => t.id !== id);
  render();
}

function toggleTodo(id) {
  const todo = todos.find(t => t.id === id);
  if (todo) todo.done = !todo.done;
  render();
}

form.addEventListener("submit", (e) => {
  e.preventDefault();
  const text = input.value.trim();
  if (!text) return;
  addTodo(text);
  input.value = "";
  input.focus();
});

list.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-action]");
  if (!btn) return;

  const id = Number(btn.closest(".todo__item").dataset.id);

  switch (btn.dataset.action) {
    case "toggle": toggleTodo(id); break;
    case "delete": deleteTodo(id); break;
  }
});

render();
Ահա ճարտարապետության ուժը

Ամբողջ պահպանումը՝ մեկ տող render()-ի ներսում։ Ինչո՞ւ։ Որովհետև Դաս 12-ում ընդունեցինք կանոնը՝ ամեն փոփոխությունից հետո կանչվում է render()։ Ուստի բավական էր պահպանումը դնել այնտեղ։

Երբ ճարտարապետությունը ճիշտ է, նոր հնարավորությունն ավելացվում է մեկ տողով։ Երբ սխալ է՝ պիտի փոխեիր 10 տեղ։

Բոնուս՝ թեմայի պահպանում

theme.js
const themeBtn = document.querySelector(".theme-toggle");

// Վերականգնում ենք բեռնման ժամանակ
const savedTheme = localStorage.getItem("theme") ?? "light";
document.documentElement.setAttribute("data-theme", savedTheme);

themeBtn.addEventListener("click", () => {
  const current = document.documentElement.getAttribute("data-theme");
  const next = current === "dark" ? "light" : "dark";

  document.documentElement.setAttribute("data-theme", next);
  localStorage.setItem("theme", next);
});

09Տնային աշխատանք

  1. Դաշտ և կոճակ։ Սեղմելիս անունը պահպանիր localStorage-ում։ Էջը թարմացնելիս ցույց տուր «Բարև, [անուն]»։
  2. Ավելացրու «Մոռանալ» կոճակ, որը ջնջում է պահված անունը։
  3. Հաշվիչ, որի արժեքը պահպանվում է էջը թարմացնելուց հետո։
  4. Մուգ/բաց թեմայի փոխարկիչ՝ ընտրության պահպանմամբ։
  5. Գրիր save(key, value) և load(key, fallback) ֆունկցիաները try/catch-ով։ Ստուգիր օբյեկտով, զանգվածով և չգոյություն ունեցող բանալիով։
  6. Կենդանի ժամացույց էջում՝ ժժ:րր:վվ ձևաչափով, padStart-ով։
  7. Հետհաշվարկ մինչև քո ընտրած ամսաթիվը՝ օր, ժամ, րոպե, վայրկյան։ Հասնելիս՝ կանգնեցրու clearInterval-ով։
  8. Ծանուցում (toast), որը հայտնվում է կոճակի սեղմումից և ինքնաբերաբար անհետանում 3 վայրկյան հետո։
  9. «Վեր» կոճակ, որը հայտնվում է 400px սքրոլից հետո և սահուն վերադարձնում վեր։
  10. Վերցրու Դաս 12–13-ի to-do-ն։ Ավելացրու՝ (ա) պահպանում localStorage-ում, (բ) որոնման դաշտ debounce-ով, (գ) «Ջնջել բոլորը» կոճակ confirm-ով։
