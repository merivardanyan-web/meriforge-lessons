src/Http/Response.php — նոր մեթոդ, որ խոստացել էինք Դաս 15-ում
public static function json(array $data, int $status = 200): self
{
    return new self(
        json_encode($data, JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR),
        $status,
        ['Content-Type' => 'application/json; charset=utf-8']
    );
}
Երկու դրոշ

JSON_UNESCAPED_UNICODE-ը թողնում է հայերենը կարդացվող ("Մոնիտոր", ոչ "\u0544...")։ JSON_THROW_ON_ERROR-ը կարևոր է․ առանց դրա կոտրված UTF-8-ի դեպքում json_encode-ը լուռ վերադարձնում է false, և հաճախորդը ստանում է դատարկ պատասխան։

Ինչո՞ւ ծրար, ոչ մերկ զանգված

1. Հաջողությունը միշտ data է, սխալը՝ միշտ error։ JS-ը ստուգում է մեկ բանալի։
2. Մետատվյալները (էջավորում) ունեն տեղ՝ առանց տվյալները խառնելու։
3. Վաղը կարող ես ավելացնել links կամ warnings՝ առանց հին հաճախորդներին կոտրելու։

Սխալների errors բանալիները համընկնում են ֆորմայի դաշտերի անուններին՝ ճիշտ ինչպես Դաս 6-ում։ JS-ը կարող է յուրաքանչյուրը ցույց տալ իր դաշտի կողքին։

Ի՞նչ չպիտի երևա API-ում

Երբեք json_encode($product) ուղիղ

Օբյեկտի public հատկությունները կդառնան JSON։ Այսօր դրանք անվնաս են, բայց վաղը ինչ-որ մեկը կավելացնի costPrice (ինքնարժեք) կամ supplierEmail, և դրանք ավտոմատ կդառնան հանրային։

Այդպես են տեղի ունեցել իրական արտահոսքեր՝ ծրագրավորողը ավելացրել է դաշտ բազայում և չի իմացել, որ API-ն այն հրապարակում է։

03JSON մարմնի ընթերցում

Թակարդ, որ ժամեր է խլում

Երբ fetch-ը ուղարկում է Content-Type: application/json, PHP-ն $_POST-ը թողնում է դատարկ։

$_POST-ը լցվում է միայն application/x-www-form-urlencoded և multipart/form-data ձևաչափերով։ JSON-ը պետք է կարդալ ձեռքով՝ php://input-ից։

Դա կոտրում է Դաս 15-ի Request-ը, որը կարդում է միայն $_POST։ Լուծում ենք մեկ տեղում։

src/Http/Request.php — փոփոխությունները
public function __construct(
    public readonly string $method,
    public readonly string $path,
    private readonly array $query,
    private readonly array $body,
    private readonly array $server = [],     // ⭐ նոր՝ վերնագրերի համար
) {}

public static function fromGlobals(): self
{
    $path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';
    $body = $_POST;

    if (str_contains($_SERVER['CONTENT_TYPE'] ?? '', 'application/json')) {
        $raw = file_get_contents('php://input');

        if ($raw !== '' && $raw !== false) {
            $decoded = json_decode($raw, true);

            // ⭐ Միայն օբյեկտ կամ զանգված։ "5" կամ null-ը վավեր JSON են, բայց մեզ պետք չեն
            if (!is_array($decoded)) {
                throw new HttpException('Սխալ JSON', 400);
            }

            $body = $decoded;
        }
    }

    return new self(
        $_SERVER['REQUEST_METHOD'] ?? 'GET',
        '/' . trim($path, '/'),
        $_GET,
        $body,
        $_SERVER,
    );
}

/** header('X-CSRF-Token') → $_SERVER['HTTP_X_CSRF_TOKEN'] */
public function header(string $name): string
{
    $key = 'HTTP_' . strtoupper(str_replace('-', '_', $name));

    return trim((string) ($this->server[$key] ?? ''));
}

public function expectsJson(): bool
{
    return str_starts_with($this->path, '/api/')
        || str_contains($this->header('Accept'), 'application/json');
}
Արդյունքը

Կոնտրոլերները չեն փոխվում։ $request->int('id') և $request->input('name') աշխատում են նույն կերպ՝ անկախ նրանից՝ տվյալները եկել են HTML ֆորմայից, թե JSON-ից։

Դա Դաս 15-ում Request օբյեկտի ներդրման հատույցն է․ մուտքի աղբյուրը փոխվեց մեկ տեղում։

400 թե՞ 422

400 — հարցումը իմաստ չունի․ JSON-ը կոտրված է, պարտադիր վերնագիրը բացակայում է։
422 — հարցումը հասկանալի է, բայց արժեքները անընդունելի են․ qty: 500։

Տարբերությունը կարևոր է JS-ի համար․ 422-ի դեպքում կա errors և այն կարելի է ցույց տալ դաշտերի կողքին։ 400-ի դեպքում դա ծրագրավորողի սխալ է, ոչ օգտագործողի։

Ամենակարևորը՝ սխալն էլ JSON է

Եթե API-ի սխալի դեպքում սերվերը վերադարձնի HTML էջ, JS-ի response.json()-ը կգցի «Unexpected token <»՝ մի հաղորդագրություն, որ ոչինչ չի ասում իրական պատճառի մասին։

Սա JS բլոկի fetch-ի դասերից հայտնի սխալն է, բայց հիմա մենք ենք սերվերի կողմը։ Պատասխանատվությունը մերն է․ API-ն միշտ պատասխանում է JSON-ով՝ նույնիսկ երբ ամեն ինչ կոտրվել է։

500-ի հաղորդագրությունը

$debug-ը ճիշտ նույն կանոնն է, ինչ Դաս 1, 10 և 15-ում։ Production-ում հաճախորդը տեսնում է «Տեխնիկական խնդիր», իսկ մանրամասները՝ լոգում։ API-ն դրանում բացառություն չէ․ JSON-ի մեջ SQL սխալ արտահոսելը նույնքան վտանգավոր է, որքան էջում։

05API կոնտրոլերներ

src/Controllers/Api/ProductApiController.php
<?php
declare(strict_types=1);

namespace App\Controllers\Api;

use App\Exceptions\NotFoundException;
use App\Http\Request;
use App\Http\Response;
use App\Models\Product;
use App\Repositories\ProductRepository;

class ProductApiController
{
    private const SORTS = ['price_asc', 'price_desc', 'name', 'newest'];
    private const MAX_PER_PAGE = 50;

    public function __construct(
        private readonly ProductRepository $products,
    ) {}

    public function index(Request $request): Response
    {
        $sort    = $request->oneOf('sort', self::SORTS, 'price_asc');
        $page    = max(1, $request->int('page', 1));
        $perPage = min(self::MAX_PER_PAGE, max(1, $request->int('per_page', 12)));   // ⭐

        $items = $this->products->all(null, $sort, $perPage, ($page - 1) * $perPage);
        $total = $this->products->count();

        return Response::json([
            'data' => array_map(fn(Product $p) => $p->toArray(), $items),
            'meta' => [
                'page'      => $page,
                'per_page' => $perPage,
                'total'    => $total,
                'last_page' => (int) ceil($total / $perPage),
            ],
        ]);
    }

    public function show(Request $request, string $slug): Response
    {
        $product = $this->products->findBySlug($slug)
            ?? throw new NotFoundException('Ապրանքը չի գտնվել');

        return Response::json(['data' => $product->toArray()]);
    }
}
Երեք դետալ այս կոդում

1. min(MAX_PER_PAGE, ...) — առանց վերին սահմանի ?per_page=1000000-ը կբեռնի ամբողջ աղյուսակը հիշողություն։ Դա ամենաէժան ծանրաբեռնման հարձակումն է։ Նույն կանոնը, ինչ Դաս 7-ի min($qty, 99)-ում։

2. array_map վերադարձնում է ցուցակ — և հենց այստեղ փակվում է Դաս 3-ի խոստումը։ Եթե փոխարենը օգտագործեինք array_filter, բանալիները կմնային բացերով, և json_encode-ը կվերածեր data-ն օբյեկտի {"0": ..., "2": ...}։ JS-ի .map()-ը կկոտրվեր։ Ահա թե ինչու էինք այդ դասում սովորում array_values։

3. ?? throw — PHP 8-ից throw-ը արտահայտություն է։ Մեկ տողում՝ «գտիր, կամ գցիր 404»։

src/Controllers/Api/CartApiController.php
class CartApiController
{
    public function __construct(
        private readonly ProductRepository $products,
        private readonly Cart $cart,
    ) {}

    public function add(Request $request): Response
    {
        $id  = $request->int('id');
        $qty = $request->int('qty', 1);

        $errors = [];

        if ($qty < 1 || $qty > 99) {
            $errors['qty'] = 'Քանակը՝ 1-ից 99';
        }

        $product = $this->products->find($id);

        if ($product === null) {
            throw new NotFoundException('Ապրանքը չի գտնվել');
        }

        if (!$product->inStock()) {
            $errors['id'] = 'Ապրանքը առկա չէ';
        }

        if ($errors !== []) {
            throw new ValidationException($errors);
        }

        $this->cart->add($id, $qty);

        return Response::json([
            'data' => [
                'count'   => $this->cart->count(),
                'message' => 'Ավելացվեց՝ ' . $product->name,
            ],
        ]);
    }
}

06CSRF API-ում

«API-ն JSON է, ուրեմն CSRF չկա» — սխալ պատկերացում

Եթե API-ն օգտագործում է սեսիայի cookie, բրաուզերը այն կուղարկի ամեն հարցումի հետ, նույնիսկ հարձակվողի էջից։ Դաս 16-ի սցենարն անփոփոխ աշխատում է։

Մի փոքր քողարկում կա․ ուրիշ դոմենից Content-Type: application/json-ով հարցումը պահանջում է preflight (բաժին 08), ուստի հարձակումը ավելի դժվար է։ Բայց հարձակվողը կարող է ուղարկել text/plain տիպով ֆորմա, իսկ եթե սերվերը php://input-ը վերծանում է առանց Content-Type-ը ստուգելու՝ այն կանցնի։

Մեր Request-ը ստուգում է application/json-ը, ինչը օգնում է, բայց պաշտպանությունը չպիտի հիմնվի այդ մանրուքի վրա։ Token-ը մնում է պարտադիր։

public/index.php — Դաս 16-ի CSRF ստուգումը, թարմացված
// Անվտանգ մեթոդները վիճակ չեն փոխում
if (!in_array($request->method, ['GET', 'HEAD', 'OPTIONS'], true)) {

    // Ֆորմայից՝ _token դաշտ։ JS-ից՝ X-CSRF-Token վերնագիր ⭐
    $token = $request->input('_token') ?: $request->header('X-CSRF-Token');

    if (!csrf_verify($token)) {
        throw new HttpException('Ձևաթուղթը ժամկետանց է։ Թարմացրեք էջը։', 419);
    }
}
Ինչու ստուգումը հիմա ընդգրկում է DELETE և PUT

Դաս 16-ում ստուգում էինք միայն POST։ Բայց REST API-ում վիճակը փոխում են նաև PUT, PATCH և DELETE։ Հիմա պայմանն է՝ ամեն ինչ, ինչ GET չէ։

Եվ այդպես մոռացված մեթոդը չի դառնում բաց դուռ։

07Fetch՝ դեպի քո սերվերը

Երկու բլոկի հանդիպման կետը

JS բլոկի դասերում fetch-ը կարդում էր ուրիշների տվյալներ և կառուցում էջեր։ Այսօր այն ուղարկում է պատվեր քո սեփական CartApiController-ին, որը հենց դու գրեցիր։

Եվ առաջին անգամ երկու կողմերն էլ քոնն են՝ սխալը կարող ես փնտրել և՛ Network ներդիրում, և՛ error_log-ում։

public/js/cart.js
const csrf = document.querySelector('meta[name="csrf-token"]')?.content ?? '';

async function addToCart(id, qty = 1) {
    const response = await fetch('/api/cart/items', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'Accept': 'application/json',
            'X-CSRF-Token': csrf,
        },
        body: JSON.stringify({ id, qty }),
    });

    const payload = await response.json();

    // ⭐ fetch-ը 4xx/5xx-ի դեպքում ՉԻ գցում սխալ — պիտի ստուգել ok-ը
    if (!response.ok) {
        const error = new Error(payload.error?.message ?? 'Սխալ');
        error.details = payload.error?.errors ?? {};
        throw error;
    }

    return payload.data;
}

document.addEventListener('submit', async (event) => {
    const form = event.target;

    if (!form.matches('form[data-cart-add]')) {
        return;
    }

    event.preventDefault();

    const button = form.querySelector('button');
    button.disabled = true;

    try {
        const data = await addToCart(Number(form.elements.id.value));

        document.querySelector('[data-cart-count]').textContent = data.count;
        document.querySelector('#cart-status').textContent = data.message;   // ⭐ էկրանընթերցիչի համար

    } catch (error) {
        // Եթե ինչ-որ բան այն չէ՝ թողնում ենք սովորական ֆորմային
        form.submit();

    } finally {
        button.disabled = false;
    }
});

Progressive enhancement

Ֆորման աշխատում է առանց JS-ի՝ սովորական POST և redirect (Դաս 7)։ JS-ը միայն բարելավում է փորձը՝ առանց էջի վերբեռնման։

Եթե JS-ը չբեռնվեց, եթե fetch-ը ձախողվեց, եթե token-ը ժամկետանց է — form.submit()-ը վերադարձնում է սովորական ուղին, որն արդեն ունի իր սխալների էջը։ Օգտագործողը երբեք չի մնում դատարկ կոճակի առաջ։

Երկու ուղին էլ կանչում են նույն Cart ծառայությունը, ուստի կանոնները (min(99)) մեկ տեղում են։

Հասանելիություն՝ role="status"

Երբ կոճակը սեղմվում է և հաշվիչը փոխվում է առանց էջի վերբեռնման, կույր օգտագործողը ոչինչ չի լսում։ Տեսողական փոփոխությունը նրա համար գոյություն չունի։

role="status"-ով տարրի տեքստի փոփոխությունը էկրանընթերցիչը կարդում է ինքնաբերաբար։ Ինչը խոստացել էինք HTML բլոկում․ JS-ով թարմացվող ամեն բան հայտարարվում է։

08CORS

Same-origin policy Էջը՝ https://technomarket.test API՝ https://technomarket.test/api/... ← նույն origin ✅ Origin = սխեմա + դոմեն + պորտ https://technomarket.test:443 նույն http://technomarket.test ուրիշ (սխեմա) https://api.technomarket.test ուրիշ (դոմեն) https://technomarket.test:8000 ուրիշ (պորտ)
Մեր դեպքում CORS-ը պետք չէ

JS-ը և API-ն նույն դոմենում են։ Բրաուզերը ոչինչ չի արգելում։ Հետևաբար CORS վերնագրեր ավելացնելու կարիք չկա։

Ամենահաճախ սխալը՝ սկսնակը ստանում է «CORS error» իր առանձին React հավելվածից (localhost:3000) դեպի API (technomarket.test), «լուծում է» Access-Control-Allow-Origin: *-ով և մոռանում։

Ինչ է իրականում CORS-ը

Դա սերվերի թույլտվությունն է բրաուզերին․ «այս ուրիշ դոմենից եկած JS-ին թույլ է տրվում կարդալ իմ պատասխանը»։

Այն պաշտպանում է օգտագործողին, ոչ սերվերին։ curl-ին CORS-ը չի վերաբերում ընդհանրապես։ Ուստի CORS-ը անվտանգություն չէ API-ի համար՝ այն չի խանգարում որևէ մեկին կանչել քո API-ն ուղղակի։

Խորացված · եթե իսկապես պետք է՝ սպիտակ ցուցակ
$allowed = ['https://app.technomarket.am'];
$origin  = $request->header('Origin');

if (in_array($origin, $allowed, true)) {      // ⭐ Դաս 6-ի օրինաչափությունը
    header("Access-Control-Allow-Origin: {$origin}");
    header('Access-Control-Allow-Headers: Content-Type, X-CSRF-Token');
    header('Access-Control-Allow-Methods: GET, POST, DELETE');
    header('Vary: Origin');
}

// Preflight՝ բրաուզերը նախ հարցնում է OPTIONS-ով
if ($request->method === 'OPTIONS') {
    return new Response('', 204);
}

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «JSON API և fetch», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

Նախագիծ · Առաջադրանք 2

JSON API-ը և fetch-ով «Գնել»-ը փակում են Առաջադրանք 2-ը։

  1. Ավարտիր դասարանականի 5 քայլը և վեց curl փորձարկումը։ Ուղարկիր սքրինշոթներ։
  2. Ավելացրու GET /api/cart՝ վերադարձնում է զամբյուղի տողերը գումարով։ Օգտագործիր findMany-ը (Դաս 14)։
  3. Ավելացրու DELETE /api/cart/items/{id} և ստուգիր, որ առանց token-ի 419 է։
  4. Ցուցադրիր $_POST-ի դատարկության թակարդը՝ var_dump($_POST) JSON հարցումի ժամանակ։
  5. Ցուցադրիր, ինչ կլինի, եթե toArray()-ի փոխարեն օգտագործես json_encode($product)։ Ավելացրու public $costPrice և տես։
  6. Փոխիր array_map-ը array_filter-ով և ցուցադրիր, ինչպես է data-ն դառնում օբյեկտ։
  7. Խորացված․ գրիր /api/products?q=... որոնում և JS ֆայլ՝ debounce-ով, որը թարմացնում է ցուցակը ինչպես JS բլոկի որոնման դասում։
  8. Անջատիր JS-ը DevTools-ում և ստուգիր, որ «Գնել» կոճակը դեռ աշխատում է։
  9. Փորձիր ?per_page=-5 և ?page=abc։ Ի՞նչ է վերադարձնում API-ն։
  10. Խնդրիր AI-ին վերանայել API-ը անվտանգության տեսանկյունից (Դաս 4-ի «վերանայման» ռեժիմով, առանց վերաշարադրելու)։ Ի՞նչ գտավ։

API-ի կանոնները

1. Միշտ JSON — նույնիսկ սխալի դեպքում 2. Բացահայտ ցանկ՝ ինչ ենք տալիս դրսին 3. Վերին սահման ամեն թվային պարամետրի 4. Բիզնես-կանոնները սերվերում, ոչ JS-ում 5. CSRF token ամեն փոփոխող հարցման 6. Ճիշտ կարգավիճակի կոդ (404, 422, 419) 7. CORS-ը անվտանգություն չէ

Ստուգաթերթ

  • API-ն միշտ պատասխանում է JSON-ով, նույնիսկ սխալի դեպքում
  • Ձևաչափը ունի data / error ծրար
  • toArray()-ը բացահայտ ցանկ է
  • per_page-ը սահմանափակված է
  • CSRF token-ը ստուգվում է վերնագրով
  • Ամեն վավերացում կա սերվերում
  • Ֆորման աշխատում է առանց JS-ի
  • JS-ով թարմացվող հաշվիչը հայտարարվում է role="status"-ով
  • Չկա Access-Control-Allow-Origin: * առանց պատճառի

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու $_POST-ը դատարկ է JSON-ի դեպքում
  • Գիտե՞մ 400-ի և 422-ի տարբերությունը
  • Հասկանո՞ւմ եմ, ինչու json_encode($product)-ը վտանգավոր է
  • Կարո՞ղ եմ բացատրել, ինչու CSRF-ը API-ում դեռ կա
  • Գիտե՞մ, ինչու CORS-ը անվտանգություն չէ
Ավարտ՝ սեփական MVC-ն

Երեք դասում (15–17) կառուցեցիր Router, Request, Response, կոնտրոլերներ, մուտքի և CSRF ստուգումներ և API։ Դա ֆրեյմվորքի կմախքն է։

Հաջորդ չորս դասում նույն նախագիծը կփոխարինես Laravel-ով։ Ամեն դաս կլինի հայելի․ «ահա այն, ինչ ձեռքով կառուցեցիր, հիմա՝ արդյունաբերական տարբերակը»։ Դաս 18-ում կտեսնես, որ Route::get('/product/{slug}', [ProductController::class, 'show'])-ը գրեթե ճիշտ քո routes.php-ն է։
