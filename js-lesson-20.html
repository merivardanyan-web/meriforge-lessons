01Ի՞նչ է API-ն

API — հասցե, որին կարող ես հարցում ուղարկել և ստանալ տվյալներ։ Ինչպես ռեստորանի մատուցող․ դու չես մտնում խոհանոց, այլ պատվիրում ես, և քեզ բերում են։

հարցում և պատասխան

04async / await

Խոստումների շղթան արդեն շատ ավելի լավ է callback դժոխքից։ Բայց կա ավելի լավը՝ գրել ասինխրոն կոդ այնպես, ասես այն սինխրոն լիներ։

նույն կոդը՝ երկու ձևով
// .then ձևով
function getUsers() {
  return fetch("https://jsonplaceholder.typicode.com/users")
    .then(response => response.json())
    .then(users => users.map(u => u.name));
}

// async/await ձևով — նույն բանը
async function getUsers() {
  const response = await fetch("https://jsonplaceholder.typicode.com/users");
  const users = await response.json();
  return users.map(u => u.name);
}

Երկրորդ տարբերակը կարդացվում է վերևից ներքև՝ ինչպես սովորական կոդ։ Ոչ մի ներդրում, ոչ մի callback։

ԲառԻնչ է անում
asyncՆշում է ֆունկցիան որպես ասինխրոն։ Այն միշտ վերադարձնում է Promise
await«Սպասիր այս խոստումին և տուր արդյունքը»։ Աշխատում է միայն async ֆունկցիայի ներսում
Կարևոր՝ await-ը էջը չի սառեցնում

await-ը կանգնեցնում է միայն այս ֆունկցիան։ Մնացած ամեն ինչ շարունակում է աշխատել՝ կոճակները, անիմացիաները, մյուս ֆունկցիաները։ Event loop-ը (Դաս 19) շարունակում է իր գործը։

05try / catch / finally

async/await-ի դեպքում .catch()-ի փոխարեն օգտագործում ենք try/catch — նույն ձևը, ինչ սովորական սինխրոն կոդում։

ամբողջական օրինակ
async function getUser(id) {
  try {
    const response = await fetch(`https://jsonplaceholder.typicode.com/users/${id}`);

    if (!response.ok) {
      throw new Error(`Կարգավիճակ ${response.status}`);
    }

    const user = await response.json();
    return user;

  } catch (error) {
    console.error("Չհաջողվեց բեռնել՝", error.message);
    return null;

  } finally {
    console.log("Հարցումն ավարտվեց");
  }
}

const user = await getUser(1);
console.log(user?.name ?? "Օգտագործող չկա");
Console
Հարցումն ավարտվեց

ԲլոկԻնչ է անում
tryԿոդը, որը կարող է սխալ գցել
catch (error)Աշխատում է, եթե սխալ եղավ
finallyԱշխատում է միշտ — և՛ սխալի, և՛ հաջողության դեպքում
Ինչ սխալներ է բռնում catch-ը

Ցանցի խնդիրները, throw-ով նետվածները, JSON-ի փչացած տվյալները, և ցանկացած այլ սխալ try-ի ներսում։ Մեկ բլոկ՝ ամեն ինչի համար։

06Երեք վիճակ ինտերֆեյսում

Ամեն ասինխրոն գործողություն ունի երեք վիճակ, և բոլորն էլ պիտի ցուցադրվեն օգտագործողին։

վիճակների սխեմա
  Սեղմեց կոճակը
       │
       ▼
  ⏳ Բեռնվում է...        ← ցուցիչ, կոճակը disabled

       │
       └──── ❌ Ձախողվեց ──► հաղորդագրություն + «Կրկին փորձել»

js/main.js
const API = "https://jsonplaceholder.typicode.com";

const loadBtn   = document.querySelector(".load-btn");
const statusEl  = document.querySelector(".status");
const container = document.querySelector("#users");

// --- Վիճակների ցուցադրում ---
function showLoading() {
  loadBtn.disabled = true;
  statusEl.textContent = "Բեռնվում է...";
  container.innerHTML = "";
}

function showError(message) {
  statusEl.textContent = "";
  container.innerHTML = `
    <div class="col">
      <div class="wrapper error">
        <p>${message}</p>
        <button class="load-btn">Կրկին փորձել</button>
      </div>
    </div>
  `;
}

function renderUsers(users) {
  statusEl.textContent = `Բեռնվեց ${users.length} օգտագործող`;

  container.innerHTML = users.map(u => `
    <div class="col">
      <div class="wrapper">
        <h3>${u.name}</h3>
        <p>${u.email}</p>
        <p>${u.company.name}</p>
        <p>${u.address.city}</p>
      </div>
    </div>
  `).join("");
}

// --- Բեռնում ---
async function loadUsers() {
  showLoading();

  try {
    const response = await fetch(`${API}/users`);

    if (!response.ok) {
      throw new Error(`Սերվերի սխալ (${response.status})`);
    }

    const users = await response.json();

    if (users.length === 0) {
      showError("Տվյալներ չկան");
      return;
    }

    renderUsers(users);

  } catch (error) {
    showError("Չհաջողվեց բեռնել։ Ստուգեք ինտերնետը։");
    console.error(error);

  } finally {
    loadBtn.disabled = false;
  }
}

// Դելեգացիա — որ «Կրկին փորձել»-ն էլ աշխատի (Դաս 12)
document.addEventListener("click", (e) => {
  if (e.target.closest(".load-btn")) loadUsers();
});

07POST և այլ մեթոդներ

POST — տվյալ ուղարկել
async function createPost(title, body) {
  const response = await fetch(`${API}/posts`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json"
    },
    body: JSON.stringify({ title, body, userId: 1 })
  });

  if (!response.ok) {
    throw new Error(`Կարգավիճակ ${response.status}`);
  }

  return response.json();
}

const created = await createPost("Իմ գրառումը", "Տեքստը");
console.log(created);
Console
{"title":"Իմ գրառումը","body":"Տեքստը","userId":1,"id":101}
ԴաշտԻնչի համար
methodGET (լռելյայն), POST, PUT, PATCH, DELETE
headersԼրացուցիչ տեղեկություն սերվերին
bodyՈւղարկվող տվյալները — միշտ տող
Երկու պարտադիր բան POST-ի համար

1. body-ն պիտի լինի տող → JSON.stringify(...) (Դաս 14)։
2. Content-Type: application/json վերնագիրը՝ որ սերվերը հասկանա, թե ինչ ես ուղարկել։

08Խաղահրապարակ

Ինչպես Դաս 19-ում

Ասինխրոն արդյունքները այս պատուհանում չեն երևա։ Այստեղ փորձարկում ենք տրամաբանությունը՝ կարգավիճակների մշակումը և տվյալների ձևափոխումը։ Իսկական fetch-ը փորձիր F12-ի Console-ում։

Console
սա փորձիր F12-ի Console-ում
const r = await fetch("https://jsonplaceholder.typicode.com/users");
console.log(r.status, r.ok);

const users = await r.json();
console.table(users.map(u => ({ name: u.name, city: u.address.city })));

09Տնային աշխատանք

API՝ https://jsonplaceholder.typicode.com։ Հասանելի ուղիներ՝ /users, /posts, /comments, /todos, /albums։

  1. Բեռնիր /users-ը և տպիր բոլոր անունները Console-ում։ Օգտագործիր .then ձևը։
  2. Կրկնիր նույնը async/await-ով։ Համեմատիր՝ որն է ավելի ընթեռնելի։
  3. Փորձիր բեռնել /users/9999։ Ավելացրու response.ok-ի ստուգում և ցույց տուր ճիշտ սխալ։
  4. Ցուցադրիր օգտագործողներին քարտերով՝ քո .row / .col / .wrapper կառուցվածքով։ Ամեն քարտում՝ անուն, էլ. հասցե, քաղաք, ընկերություն։
  5. Ավելացրու երեք վիճակ՝ բեռնում, հաջողություն, սխալ։ Կոճակը բեռնման ընթացքում պիտի լինի disabled։
  6. Ստուգիր սխալի վիճակը՝ DevTools → Network → Offline։ Համոզվիր, որ «Կրկին փորձել» կոճակն աշխատում է։
  7. Բեռնիր /posts-ը և ցուցադրիր միայն առաջին 10-ը։ Ավելացրու որոնում վերնագրով (Դաս 14-ի debounce-ով)։
  8. Ամեն օգտագործողի քարտին ավելացրու «Գրառումները» կոճակ, որը բեռնում է /posts?userId=X և ցուցադրում մոդալում (Դաս 11)։
  9. Օգտագործիր Promise.all՝ միաժամանակ բեռնելու /users և /posts։ Ցուցադրիր՝ ամեն օգտագործողի կողքին քանի՞ գրառում ունի։
  10. Ֆորմա նոր գրառման համար (վերնագիր + տեքստ)։ Ուղարկիր POST-ով, ցուցադրիր պատասխանը և ավելացրու ցուցակի սկզբում։ Օգտագործիր Դաս 13-ի վալիդացիան։

Ամեն fetch-ի ստուգաթերթ

  • Ցուցադրվո՞ւմ է բեռնման վիճակը
  • Ստուգվո՞ւմ է response.ok-ը
  • Կա՞ try / catch
  • Ցուցադրվո՞ւմ է հասկանալի սխալի հաղորդագրություն
  • Կա՞ «Կրկին փորձել» հնարավորություն
  • Մշակվա՞ծ է դատարկ արդյունքի դեպքը
  • Կոճակը կրկնակի սեղմումից պաշտպանվա՞ծ է

Ինքնաստուգում

  • Գիտե՞մ, ինչու 404-ը չի ընկնում catch-ի մեջ
  • Կարո՞ղ եմ բացատրել, ինչու են երկու .then պետք
  • Հասկանո՞ւմ եմ, որ await-ը էջը չի սառեցնում
  • Կարո՞ղ եմ գրել POST հարցում առանց նայելու
Հաջորդ դասին

API-ների հետ խորը աշխատանք՝ հարցման պարամետրեր, էջավորում, որոնում debounce-ով, AbortController (հին հարցումների չեղարկում) և CORS-ի բացատրություն։
