Զանգվածների մեթոդներ

Այն ամենը, ինչ անցյալ դասին անում էինք ցիկլերով, այսօր կդառնա մեկ տող։ Սա այն դասն է, որից հետո քո կոդը սկսում է նմանվել իսկական ծրագրավորողի կոդի։

01Callback ֆունկցիա

Դաս 4-ում սովորեցինք, որ ֆունկցիան կարելի է դնել փոփոխականի մեջ։ Իսկ եթե ֆունկցիան կարելի է դնել փոփոխականի մեջ, ուրեմն այն կարելի է փոխանցել ուրիշ ֆունկցիային որպես արգումենտ։

ֆունկցիա ֆունկցիայի ներսում
function doTwice(action) {   // action — ֆունկցիա է
  action();
  action();
}

doTwice(function() {
  console.log("Բարև");
});

// նույնը arrow ձևով — ավելի կարճ
doTwice(() => console.log("Ցտեսություն"));
Console
Բարև
Բարև
Ցտեսություն
Ցտեսություն
Callback

Ֆունկցիա, որը փոխանցում ես ուրիշ ֆունկցիային, որպեսզի նա այն կանչի իր ուզած պահին։ Հայերեն կարելի է ասել «հետկանչ»։ Ամբողջ այսօրվա դասը դրա մասին է։

Ինչո՞ւ է սա հեղափոխական

Զանգվածների մեթոդներն ասում են՝ «ես կանցնեմ բոլոր էլեմենտներով, իսկ դու ասա՝ ամեն մեկի հետ ի՞նչ անեմ»։ Ցիկլի «մեխանիկան» գրում է JS-ը, դու գրում ես միայն տրամաբանությունը։

callback-ը միշտ ստանում է 3 բան
const colors = ["կարմիր", "կանաչ", "կապույտ"];

colors.forEach((item, index, array) => {
  console.log(index, item, array.length);
});
Console
0 կարմիր 3
1 կանաչ 3
2 կապույտ 3

Երեքն էլ պարտադիր չեն։ Գրում ես միայն այնքան, որքան պետք է․ ամենից հաճախ՝ միայն առաջինը։

02forEach — անցնել բոլորով

for..of-ի փոխարինիչն է։ Ոչինչ չի վերադարձնում, պարզապես կատարում է գործողությունը ամեն էլեմենտի համար։

հին ձև → նոր ձև
const students = ["Անի", "Դավիթ", "Մարիամ"];

// Դաս 5-ի ձևը
for (let i = 0; i < students.length; i++) {
  console.log(`${i + 1}. ${students[i]}`);
}

// նույնը forEach-ով
students.forEach((student, i) => {
  console.log(`${i + 1}. ${student}`);
});
Console
1. Անի
2. Դավիթ
3. Մարիամ
1. Անի
2. Դավիթ
3. Մարիամ
Կարևոր

forEach-ը միշտ վերադարձնում է undefined։ Այն օգտագործում ենք միայն գործողության համար (տպել, ցուցադրել), ոչ թե արդյունք ստանալու։ Նաև՝ forEach-ում break չի աշխատում։

03map — վերափոխել ամեն էլեմենտ

map-ը վերցնում է զանգվածը, ամեն էլեմենտի հետ ինչ-որ բան անում և վերադարձնում նոր զանգված՝ նույն երկարությամբ։

map
const prices = [1000, 2000, 3000];

// 20% զեղչ բոլորին
const discounted = prices.map(price => price * 0.8);

console.log(discounted);
console.log(prices);        // բնօրինակը անվնաս է
Console
[800, 1600, 2400]
[1000, 2000, 3000]
տիպը կարող է փոխվել
const names = ["անի", "դավիթ", "մարիամ"];

const upper = names.map(n => n.toUpperCase());
console.log(upper);

const lengths = names.map(n => n.length);
console.log(lengths);

const tags = names.map(n => `<li>${n}</li>`);
console.log(tags);
Console
["ԱՆԻ", "ԴԱՎԻԹ", "ՄԱՐԻԱՄ"]
[3, 5, 6]
["<li>անի</li>", "<li>դավիթ</li>", "<li>մարիամ</li>"]
Առաջին հայացքը DOM-ին

Վերջին օրինակը այն է, ինչ կանես Դաս 10-ում․ տվյալների զանգվածից map-ով պատրաստում ես HTML տողեր և դնում էջում։ Հենց այսպես են աշխատում բոլոր ժամանակակից կայքերը։

Ամենահաճախ սխալը

map-ի callback-ը պարտադիր պիտի ունենա return։ Առանց դրա կստանաս undefined-ներով լցված զանգված։

մոռացված return
const nums = [1, 2, 3];

console.log(nums.map(n => { n * 2; }));      // { } կա, return չկա
console.log(nums.map(n => { return n * 2; }));
console.log(nums.map(n => n * 2));           // ամենակարճը
Console
[undefined, undefined, undefined]
[2, 4, 6]
[2, 4, 6]

04filter — ընտրել համապատասխանողները

filter-ի callback-ը պիտի վերադարձնի true կամ false։ Ով true է — մնում է նոր զանգվածում։

filter
const marks = [14, 19, 8, 17, 20, 11];

const good  = marks.filter(m => m >= 14);
const evens = marks.filter(m => m % 2 === 0);

console.log(good);
console.log(evens);
console.log(good.length);
console.log(marks);      // բնօրինակը անվնաս
Console
[14, 19, 17, 20]
[14, 8, 20]
4
[14, 19, 8, 17, 20, 11]
map ընդդեմ filter

map — երկարությունը չի փոխվում, արժեքները՝ փոխվում են։
filter — արժեքները չեն փոխվում, երկարությունը՝ փոխվում է։

որոնում տեքստով
const fruits = ["խնձոր", "բանան", "նարինջ", "տանձ", "նուռ"];
const query = "ն";

const found = fruits.filter(f => f.includes(query));
console.log(found);

const startsWith = fruits.filter(f => f.startsWith(query));
console.log(startsWith);
Console
["բանան", "նարինջ", "տանձ", "նուռ"]
["նարինջ", "նուռ"]

05find և findIndex

filter-ը վերադարձնում է զանգված՝ բոլոր համապատասխանողները։ find-ը վերադարձնում է մեկ էլեմենտ՝ առաջին գտնվածը, և կանգնում է։

find
const marks = [8, 11, 19, 14, 20];

console.log(marks.filter(m => m >= 14));
console.log(marks.find(m => m >= 14));
console.log(marks.findIndex(m => m >= 14));

console.log(marks.find(m => m > 100));       // չգտավ
console.log(marks.findIndex(m => m > 100));
Console
[19, 14, 20]

Ե՞րբ որը

Փնտրում ես մեկ կոնկրետ բան (օգտագործող ըստ id-ի, ապրանք ըստ կոդի) → find։
Փնտրում ես բոլորին, ովքեր համապատասխանում են → filter։

06some և every

Երկուսն էլ վերադարձնում են միայն true կամ false։

some — գոնե մեկը · every — բոլորը
const marks = [14, 19, 8, 17];

console.log(marks.some(m => m < 10));    // կա՞ գոնե մեկ վատ գնահատական
console.log(marks.every(m => m >= 10));  // բոլո՞րն են անցել
console.log(marks.every(m => m > 0));

if (marks.some(m => m < 10)) {
  console.log("Ուշադրություն՝ կա անբավարար գնահատական։");
}
Console
true
false
true
Ուշադրություն՝ կա անբավարար գնահատական։

Հիշիր Դաս 2-ը․ some-ը նման է ||-ին (գոնե մեկը), every-ն՝ &&-ին (բոլորը)։

07sort — դասավորել

Այստեղ մի մեծ թակարդ կա։ Նախ նայենք տեքստերին — այնտեղ ամեն ինչ լավ է։

տեքստերի դասավորում
const names = ["Մարիամ", "Անի", "Դավիթ", "Բարսեղ"];

console.log(names.sort());
console.log(names.reverse());
Console
["Անի", "Բարսեղ", "Դավիթ", "Մարիամ"]
["Մարիամ", "Դավիթ", "Բարսեղ", "Անի"]

Իսկ հիմա՝ թվերը

անսպասելի արդյունք
const nums = [10, 1, 5, 100, 25];

console.log(nums.sort());
Console
[1, 10, 100, 25, 5]   ← ?!
Ինչո՞ւ

Առանց հրահանգի sort-ը թվերը վերածում է տեքստի և դասավորում այբբենական կարգով։ Ինչպես բառարանում՝ «100»-ը գալիս է «25»-ից առաջ, որովհետև «1» < «2»։

ճիշտ ձևը
const nums = [10, 1, 5, 100, 25];

console.log([...nums].sort((a, b) => a - b));   // աճման կարգով
console.log([...nums].sort((a, b) => b - a));   // նվազման կարգով
Console
[1, 5, 10, 25, 100]
[100, 25, 10, 5, 1]

Կանոնը պարզ է․ եթե a - b-ն բացասական է, a-ն գնում է առաջ։ Պարզապես անգիր արա՝ a - b = փոքրից մեծ, b - a = մեծից փոքր։

sort-ը փոխում է բնօրինակը

sort-ը և reverse-ը վիրահատում են ինքը զանգվածը (ինչպես splice-ը)։ Դրա համար վերևում գրեցինք [...nums].sort(...) — նախ պատճենեցինք, հետո դասավորեցինք։ Այս սովորությունը շատ սխալներից կփրկի քեզ։

08reduce — ամեն ինչ մեկ արժեքի բերել

Ամենահզոր և ամենաշատ վախեցնող մեթոդն է։ Բայց իրականում պարզ է․ այն զանգվածը «սեղմում» է մեկ արժեքի՝ գումարի, միջինի, ամենամեծի։

հին ձևը՝ ցիկլով
const prices = [1200, 450, 3000];

let total = 0;               // կուտակիչ
for (const price of prices) {
  total = total + price;      // թարմացնում ենք
}
console.log(total);
Console
4650
նույնը reduce-ով
const prices = [1200, 450, 3000];

const total = prices.reduce((acc, price) => acc + price, 0);
//                            ↑     ↑                    ↑
//                       կուտակիչ  ընթացիկ         սկզբնական արժեք

console.log(total);
Console
4650

արդյունք4650
Հիշիր

acc-ը այն է, ինչ callback-ը վերադարձրեց նախորդ քայլին։ Երկրորդ արգումենտը (0) սկզբնական արժեքն է։ Այն միշտ գրիր — առանց դրա տարօրինակ սխալներ են լինում դատարկ զանգվածների դեպքում։

reduce-ի այլ կիրառություններ
const nums = [4, 9, 2, 7];

// գումար
console.log(nums.reduce((a, b) => a + b, 0));

// արտադրյալ (սկզբնականը 1 է, ոչ 0)
console.log(nums.reduce((a, b) => a * b, 1));

// ամենամեծը
console.log(nums.reduce((max, n) => n > max ? n : max, nums[0]));

// միջինը
console.log(nums.reduce((a, b) => a + b, 0) / nums.length);

09Մեթոդների շղթա

Քանի որ map-ը և filter-ը վերադարձնում են զանգված, դրանց վրա անմիջապես կարելի է կանչել հաջորդ մեթոդը։ Ահա այս դասի գագաթնակետը։

chaining
const prices = [1200, 450, 3000, 800, 2500];

const result = prices
  .filter(p => p >= 1000)      // [1200, 3000, 2500]
  .map(p => p * 0.9)          // [1080, 2700, 2250]
  .reduce((a, b) => a + b, 0); // 6030

console.log(result);
Console
6030

Կարդում ենք ձախից աջ, ինչպես նախադասություն․ «վերցրու գները, թող միայն 1000-ից թանկերը, ամեն մեկին 10% զեղչ արա, հետո գումարիր»։

Ձևավորման կանոն

Ամեն մեթոդ գրիր նոր տողում՝ կետից սկսած։ Երբ շղթան 2-ից երկար է, մեկ տողում գրելն անընթեռնելի է դառնում։

Հերթականությունը

Նախ filter, հետո map — ոչ հակառակը։ Իմաստ չկա վերափոխել այն էլեմենտները, որոնք հետո միևնույն է դեն ես նետելու։

10Դասարանական աշխատանք

Աշխատում ենք ապրանքների իրական ցուցակի հետ։ Ամեն ապրանք՝ առանձին օբյեկտ (դրանք մանրամասն կսովորենք հաջորդ դասին, այսօր պարզապես կարդում ենք p.name, p.price ձևով)։

catalog.js
const products = [
  { name: "Նոութբուք",  price: 450000, inStock: true  },
  { name: "Մկնիկ",      price: 8000,   inStock: true  },
  { name: "Ստեղնաշար",  price: 25000,  inStock: false },
  { name: "Մոնիտոր",    price: 120000, inStock: true  },
  { name: "Ականջակալ",  price: 15000,  inStock: false }
];

// 1. Բոլոր անունները
console.log(products.map(p => p.name));

// 2. Միայն առկա ապրանքները
const available = products.filter(p => p.inStock);
console.log(available.map(p => p.name));

// 3. Առկա ապրանքների ընդհանուր արժեքը
const total = available.reduce((sum, p) => sum + p.price, 0);
console.log(total);

// 4. Գտնել կոնկրետ ապրանք
console.log(products.find(p => p.name === "Մոնիտոր").price);

// 5. Կա՞ արդյոք 500000-ից թանկ ապրանք
console.log(products.some(p => p.price > 500000));

// 6. Էժանից թանկ, միայն անունները
console.log(
  [...products]
    .sort((a, b) => a.price - b.price)
    .map(p => `${p.name} — ${p.price}֏`)
);

12Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Դարակի վիճակագրություն», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

Բոլոր առաջադրանքները կատարիր առանց for ցիկլի — միայն մեթոդներով։

  1. Տրված է թվերի զանգված։ Ստեղծիր նոր զանգված՝ ամեն թիվը քառակուսի բարձրացրած։
  2. Անունների զանգվածից ստացիր դրանց երկարությունների զանգված։
  3. Ֆիլտրիր բոլոր բացասական թվերը։
  4. Ֆիլտրիր 5 տառից երկար բառերը։
  5. Հաշվիր զանգվածի բոլոր թվերի գումարը և միջինը՝ reduce-ով։
  6. Գտիր առաջին թիվը, որը 100-ից մեծ է։ Եթե չկա — տպիր «Չգտնվեց»։
  7. Ստուգիր՝ բոլո՞ր բառերն են սկսվում մեծատառով (every)։
  8. Դասավորիր թվերի զանգվածը նվազման կարգով՝ առանց բնօրինակը փոխելու։ Ապացուցիր՝ տպելով երկուսն էլ։
  9. Օգտագործիր դասարանական products զանգվածը․ գտիր չառկա ապրանքների անունները և դրանց ընդհանուր արժեքը։
  10. Մեկ շղթայով․ վերցրու գնահատականները, թող միայն 14-ից բարձրերը, ամեն մեկին ավելացրու 1, դասավորիր նվազման կարգով և վերածիր տողի՝ ", " բաժանարարով։

ՄեթոդՎերադարձնում էԵ՞րբ օգտագործել
forEachundefinedՊարզապես անել ինչ-որ բան ամեն էլեմենտի հետ

filterնոր զանգված, ավելի կարճԸնտրել համապատասխանողներին
findմեկ էլեմենտ կամ undefinedԳտնել առաջինը
findIndexինդեքս կամ -1Գտնել առաջինի տեղը
sometrue / falseԿա՞ գոնե մեկը
everytrue / falseԲոլո՞րն են այդպիսին
sortնույն զանգվածը ⚠️Դասավորել
reduceմեկ արժեքԳումար, միջին, ամեն ինչ մեկում
Միակ մեթոդը, որ փոխում է բնօրինակը

sort և reverse։ Մնացած բոլորը վերադարձնում են նոր զանգված։ Կասկածի դեպքում գրիր [...arr].sort(...)։

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել՝ ինչ է callback-ը
  • Գիտե՞մ, ինչու map-ը undefined-ներ է վերադարձնում առանց return-ի
  • Հասկանո՞ւմ եմ find-ի և filter-ի տարբերությունը
  • Կարո՞ղ եմ գրել reduce՝ առանց նայելու
  • Հիշո՞ւմ եմ, ինչու [10, 1, 5].sort()-ը սխալ արդյունք է տալիս
Հաջորդ դասին

Օբյեկտներ։ Այսօր ապրանքները արդեն օբյեկտներ էին — հաջորդ դասին կհասկանանք դրանք լիարժեք․ { key: value }, ներդրված օբյեկտներ, destructuring, Object.keys։
