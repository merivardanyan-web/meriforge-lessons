Ցիկլեր

Համակարգիչը ձանձրույթ չգիտի։ Այսօր սովորում ենք, թե ինչպես ստիպել նրան նույն գործողությունը կրկնել 10, 100 կամ միլիոն անգամ՝ ընդամենը երեք տող կոդով։

Նախապայման՝ Դաս 01–02 Բլոկ 1 · Լեզվի հիմքերը Տնային՝ 10 առաջադրանք

  • Ինչո՞ւ են պետք ցիկլերը

  • Ե՞րբ որն օգտագործել
  • break և continue

  • Դասական խնդիրներ՝ բազմապատկման աղյուսակ, բուրգ, պարզ թվեր

01Ինչո՞ւ են պետք ցիկլերը

Պատկերացրու՝ ուսուցիչը խնդրում է տպել 1-ից 5 թվերը։ Առանց ցիկլի կգրես․

Ցիկլ նշանակում է

«Կրկնիր այս կոդը այնքան ժամանակ, քանի դեռ պայմանը ճշմարիտ է»։ Երբ պայմանը դառնում է false — ցիկլը կանգնում է և ծրագիրը շարունակվում է հաջորդ տողից։

02for ցիկլը

Դա ամենահաճախ օգտագործվող ցիկլն է։ Փակագծերում երեք մաս կա, բաժանված ;-ով։

կառուցվածքը
for (սկիզբ; պայման; քայլ) {
  // կրկնվող կոդը
}

// 1. սկիզբ  → աշխատում է ՄԵԿ անգամ, ամենասկզբում
// 2. պայման → ստուգվում է ԱՄԵՆ կրկնությունից ԱՌԱՋ
// 3. քայլ   → աշխատում է ԱՄԵՆ կրկնությունից ՀԵՏՈ

Քայլ առ քայլ՝ ինչ է կատարվում

Վերցնենք for (let i = 1; i <= 3; i++) և հետևենք ամեն քայլին։

Կուտակիչ փոփոխական

Շատ հաճախ մեզ պետք է ցիկլի ընթացքում ինչ-որ բան հավաքել։ Դրա համար ցիկլից դուրս ստեղծում ենք փոփոխական, իսկ ցիկլի ներսում ավելացնում։

1-ից 100 թվերի գումարը
let sum = 0;          // կուտակիչը՝ ցիկլից ԴՈՒՐՍ

for (let i = 1; i <= 100; i++) {
  sum += i;             // ամեն կրկնության ավելացնում ենք
}

console.log(sum);
Console
5050
Ամենահաճախ սխալը

Եթե let sum = 0; գրես ցիկլի ներսում, ամեն կրկնության այն կզրոյացվի և վերջում կստանաս ընդամենը 100։ Կուտակիչը միշտ ցիկլից դուրս է։

03while ցիկլը

for-ն օգտագործում ենք, երբ գիտենք, թե քանի անգամ պիտի կրկնվի։ while-ը՝ երբ չգիտենք, և պարզապես ասում ենք՝ «կրկնիր, քանի դեռ…»։

while
let i = 1;              // 1. սկիզբ

while (i <= 5) {        // 2. պայման
  console.log(i);
  i++;                    // 3. քայլ — ՁԵՌՔՈՎ
}
Console
1
2
3
4
5

Նույն երեք մասերն են, պարզապես ցրված են երեք տեղում։ Հենց դրա համար էլ for-ն ավելի հարմար է հաշվելու համար՝ ամեն ինչ մեկ տողում է։

Անվերջ ցիկլ · վտանգավոր

Եթե մոռանաս i++-ը, պայմանը երբեք չի դառնա false։ Բրաուզերը կսառչի։ Եթե պատահի՝ փակիր ներդիրը (Ctrl + W)։ Ամեն ծրագրավորող սա գոնե մեկ անգամ արել է։

ՄԻ՛ գործարկիր սա
let i = 1;
while (i <= 5) {
  console.log(i);
  // i++ բացակայում է → i-ն միշտ 1 է → պայմանը միշտ true
}
Console

1
1
... անվերջ ...

Երբ while-ն է ավելի ճիշտ

գաղտնաբառի ստուգում
let password = prompt("Մուտքագրիր գաղտնաբառը");

while (password !== "1234") {
  password = prompt("Սխալ է։ Փորձիր նորից");
}

console.log("Բարի գալուստ։");

Այստեղ մենք չգիտենք՝ քանի անգամ կսխալվի օգտագործողը։ Կարող է առաջին անգամից գուշակի, կարող է 20-րդից։ for-ը այստեղ չէր օգնի։

04do..while

Նույն while-ն է, բայց մեկ կարևոր տարբերությամբ․ կոդը կատարվում է գոնե մեկ անգամ, և միայն հետո ստուգվում պայմանը։

համեմատենք
let i = 10;

// while — պայմանը false է, ոչինչ չի տպվում
while (i < 5) {
  console.log("while:", i);
  i++;
}

let j = 10;

// do..while — մեկ անգամ միևնույն է կկատարվի
do {
  console.log("do..while:", j);
  j++;
} while (j < 5);
Console
do..while: 10
Ե՞րբ է պետք

Երբ պիտի նախ հարցնես, հետո ստուգես։ Օրինակ՝ մենյու ցույց տալ, հետո ստուգել՝ օգտագործողն «ելք» ընտրե՞ց։ Իրականում շատ հազվադեպ է օգտագործվում։

05break և continue

break — դուրս թռիր ցիկլից

Երբ գտար այն, ինչ փնտրում էիր, իմաստ չկա շարունակել։

break
for (let i = 1; i <= 10; i++) {
  if (i === 4) {
    console.log("Գտա 4-ը։ Կանգնում եմ։");
    break;
  }
  console.log(i);
}

console.log("Ցիկլն ավարտվեց։");

continue — բաց թող այս կրկնությունը

Ցիկլը շարունակվում է, պարզապես ընթացիկ քայլի մնացած կոդը չի կատարվում։

continue
// տպիր բոլորը, բացի կենտ թվերից
for (let i = 1; i <= 8; i++) {
  if (i % 2 !== 0) {
    continue;   // կենտ է → անցիր հաջորդին
  }
  console.log(i);
}

break = դուրս գալ շենքից։
continue = բաց թողնել այս հարկը և բարձրանալ հաջորդը։

06Ներդրված ցիկլեր

Ցիկլ՝ ցիկլի ներսում։ Պատկերացրու ժամացույցը․ ժամի սլաքը մեկ քայլ է անում, րոպեի սլաքը այդ ընթացքում՝ 60։

ինչպես է աշխատում
for (let i = 1; i <= 3; i++) {          // արտաքին — ժամ
  for (let j = 1; j <= 2; j++) {        // ներքին — րոպե
    console.log(`i = ${i}, j = ${j}`);
  }
}
Console
i = 1, j = 1
i = 1, j = 2
i = 2, j = 1
i = 2, j = 2
i = 3, j = 1
i = 3, j = 2

Կանոնը․ արտաքին ցիկլի յուրաքանչյուր քայլի համար ներքին ցիկլն անցնում է ամբողջությամբ։ 3 × 2 = 6 կրկնություն։

Աստղերի բուրգ

Ահա ներդրված ցիկլերի ամենադասական խնդիրը։

pyramid.js
for (let i = 1; i <= 5; i++) {
  let line = "";                // տողի կուտակիչը

  for (let j = 1; j <= i; j++) {   // ուշադրություն՝ j <= i
    line += "*";
  }

  console.log(line);
}

Գ — Թվի թվանշանների գումարը

digit-sum.js
let number = 4729;
let sum = 0;

while (number > 0) {
  sum += number % 10;              // վերջին թվանշանը
  number = Math.floor(number / 10); // կտրում ենք վերջինը
  console.log(`sum = ${sum}, մնաց = ${number}`);
}

console.log(`Ընդհանուր գումարը՝ ${sum}`);
Console
sum = 9, մնաց = 472
sum = 11, մնաց = 47
sum = 18, մնաց = 4
sum = 22, մնաց = 0
Ընդհանուր գումարը՝ 22

Math.floor()-ը կլորացնում է ներքև․ 472.9 → 472։ Այսպես մենք «կտրում» ենք վերջին թվանշանը։

Դ — Պարզ թիվ է, թե՞ ոչ

Պարզ է այն թիվը, որը բաժանվում է միայն 1-ի և ինքն իրեն։

prime.js
const num = 29;
let isPrime = true;      // ենթադրում ենք՝ պարզ է

if (num < 2) {
  isPrime = false;
}

for (let i = 2; i < num; i++) {
  if (num % i === 0) {
    isPrime = false;
    break;              // գտանք բաժանարար → իմաստ չկա շարունակել
  }
}

console.log(isPrime ? `${num} — պարզ թիվ է` : `${num} — պարզ թիվ չէ`);
Console
29 — պարզ թիվ է
Դրոշակ (flag)

isPrime-ը կոչվում է դրոշակ։ Հնարքը միշտ նույնն է․ ենթադրում ենք լավագույնը (true), և եթե ցիկլում գտնում ենք հակառակ ապացույց՝ իջեցնում ենք դրոշակը (false) և break։

08Խաղահրապարակ

Փոխիր թվերը, ավելացրու break, փորձիր հակառակ ցիկլ։ Ամեն սխալ՝ մեկ նոր բան սովորեցնում է։

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Ընթերցանության պլան», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

  1. Տպիր 1-ից 20 թվերը for-ով, ապա նույնը՝ while-ով։
  2. Տպիր 100-ից 1 թվերը՝ 5-ական քայլով (100, 95, 90…)։
  3. Հաշվիր 1-ից 50 զույգ թվերի գումարը։
  4. Օգտագործողից վերցրու թիվ և տպիր դրա բազմապատկման աղյուսակը 1-ից 10։
  5. Հաշվիր տրված թվի ֆակտորիալը (5! = 1×2×3×4×5 = 120)։
  6. Կառուցիր հակառակ բուրգ՝
    *****

  7. Տպիր 2-ից 50 բոլոր պարզ թվերը (օգտագործիր ներդրված ցիկլ և դրոշակ)։
  8. Տրված է թիվ։ Հաշվիր՝ քանի՞ թվանշան ունի (օրինակ՝ 4729 → 4)։
  9. Կառուցիր 5×5 քառակուսի ■ նշաններից՝ ներդրված ցիկլով։
  10. Խաղ․ ծրագիրը «մտածում» է թիվ (օրինակ՝ 7)։ while-ով հարցրու օգտագործողին, մինչև գուշակի, և հուշիր՝ «շատ է» կամ «քիչ է»։
