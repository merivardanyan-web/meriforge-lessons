Ժառանգում և ինտերֆեյս

Երկու տեսակի ապրանք՝ ֆիզիկական և թվային։ Շատ ընդհանուր բան, մի քանի տարբերություն։ Այսօր սովորում ենք, ինչպես կիսել կոդը — և ամենակարևորը, ե՞րբ չկիսել։

01Խնդիրը

ՏեխնոՄարկետը սկսում է վաճառել նաև ծրագրեր Ֆիզիկական ապրանք Թվային ապրանք ───────────────── ──────────────── id, name, price ✓ id, name, price ✓ discount ✓ discount ✓ stock ✓ stock ✗ անսահման weight ✓ fileSize ✓ առաքում ✓ ներբեռնման հղում ✓ 80% ընդհանուր, 20% տարբեր
Երկու վատ ելք

1. Պատճենել դասը → երկու Product, և ամեն ուղղում պիտի անես երկու տեղում։ Մի օր կմոռանաս։

2. Մեկ դաս բոլորի համար → $product->fileSize-ը ֆիզիկական ապրանքի մոտ null է, $product->weight-ը թվայինի մոտ՝ նույնպես։ Ամեն մեթոդում պիտի ստուգես տեսակը՝ if ($this->type === 'digital')։

Երկրորդը հատկապես նենգ է․ սկզբում այն թվում է պարզ, հետո դառնում է անընթեռնելի պայմանների խառնուրդ։

02Ժառանգում

ընդհանուրը վերևում
class Product
{
    public function __construct(
        public readonly int $id,
        public readonly string $name,
        public readonly int $price,
        public readonly int $discount = 0,
    ) {}

    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }
}

// ---------- Ժառանգ ----------

class PhysicalProduct extends Product
{
    public function __construct(
        int $id,
        string $name,
        int $price,
        int $discount = 0,
        public readonly int $stock = 0,
        public readonly int $weight = 0,
    ) {
        parent::__construct($id, $name, $price, $discount);  // ⭐ պարտադիր
    }

    public function inStock(): bool
    {
        return $this->stock > 0;
    }

    public function shippingCost(): int
    {
        return $this->weight > 5000 ? 3000 : 1000;
    }
}

class DigitalProduct extends Product
{
    public function __construct(
        int $id,
        string $name,
        int $price,
        int $discount = 0,
        public readonly int $fileSize = 0,
    ) {
        parent::__construct($id, $name, $price, $discount);
    }

    public function inStock(): bool
    {
        return true;      // միշտ առկա է
    }

    public function downloadUrl(): string
    {
        return '/download/' . $this->id;
    }
}

03abstract

Ընթացիկ խնդիրը

new Product(1, 'Ինչ-որ բան', 1000) — սա թույլատրված է, բայց անիմաստ։

«Ապրանք ընդհանրապես» գոյություն չունի — ամեն ապրանք կա՛մ ֆիզիկական է, կա՛մ թվային։ Իսկ այդ օբյեկտը չունի inStock(), և կոդն այն կանչելիս կփլվի։

abstract լուծում է երկու խնդիր
abstract class Product
{
    public function __construct(
        public readonly int $id,
        public readonly string $name,
        public readonly int $price,
        public readonly int $discount = 0,
    ) {}

    // Ընդհանուր իրագործում
    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }

    // Պարտադիր, բայց ամեն ժառանգ իր ձևով ⭐
    abstract public function inStock(): bool;
    abstract public function typeLabel(): string;
}

$p = new Product(...);   // ❌ Fatal error — abstract է
Երկու երաշխիք

1. Չես կարող ստեղծել «ապրանք ընդհանրապես»։
2. Ամեն ժառանգ պարտադիր իրագործում է inStock()-ը և typeLabel()-ը — այլապես ֆատալ սխալ դեռ գործարկելուց առաջ։

Երկրորդը կարևոր է․ վաղը երբ ավելացնես SubscriptionProduct, PHP-ն կհիշեցնի, որ մոռացել ես inStock()-ը։ Առանց abstract-ի սխալը կհայտնվեր միայն օգտագործման պահին։

// Ֆունկցիան ընդունում է ամեն ինչ, ինչ առաքվում է
function calculateShipping(array $items): int
{
    $total = 0;

    foreach ($items as $item) {
        if ($item instanceof Shippable) {
            $total += $item->shippingCost();
        }
    }

    return $total;
}

Ահա գաղափարը

foreach-ը չգիտի, թե ինչ տեսակի ապրանք է։ Այն պարզապես կանչում է inStock()։

Ամեն օբյեկտ ինքն է որոշում, ինչպես պատասխանել։ Ֆիզիկականը նայում է պահեստին, թվայինը միշտ ասում է «այո»։

Այլընտրանքը կլիներ՝ if ($p->type === 'physical') { ... } elseif (...) ամեն տեղ։ Եվ ամեն նոր տեսակի հետ պիտի գտնեիր բոլոր այդ if-երը։

Պոլիմորֆիզմը հանում է պայմանները կոդից և տեղափոխում դասերի մեջ։

06Ե՞րբ չժառանգել

Ամենակարևոր բաժինը այս դասում

Ժառանգումը գայթակղիչ է․ այն ակնթարթորեն հեռացնում է կրկնությունը։ Բայց այն նաև ամուր կապում է դասերը իրար։

Ծնողի ամեն փոփոխություն ազդում է բոլոր ժառանգների վրա։ Երբ շղթան դառնում է երեք-չորս մակարդակ, կոդի կարդալը վերածվում է հետաքննության՝ որտեղի՞ց է եկել այս մեթոդը։

Գործնական սահման

Ժառանգման երկուսից ավելի մակարդակը գրեթե միշտ սխալ է։

07Դասարանական

src/Product.php — վերջնական
<?php
declare(strict_types=1);

abstract class Product
{
    public function __construct(
        public readonly int $id,
        public readonly string $name,
        public readonly string $slug,
        public readonly int $price,
        public readonly int $discount = 0,
        public readonly ?string $specs = null,
        public readonly ?string $image = null,
    ) {}

    // ---------- Ֆաբրիկա՝ ընտրում է դասը ----------

    public static function fromRow(array $row): static
    {
        return ($row['type'] ?? 'physical') === 'digital'
            ? DigitalProduct::fromRow($row)
            : PhysicalProduct::fromRow($row);
    }

    // ---------- Ընդհանուր ----------

    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }

    public function hasDiscount(): bool
    {
        return $this->discount > 0;
    }

    public function savings(): int
    {
        return $this->price - $this->finalPrice();
    }

    public function formattedPrice(): string
    {
        return number_format($this->finalPrice(), 0, '.', "\u{00A0}") . "\u{00A0}֏";
    }

    public function url(): string
    {
        return '/product/' . $this->slug;
    }

    public function imageUrl(): string
    {
        return '/images/' . ($this->image ?? 'placeholder.jpg');
    }

    // ---------- Պարտադիր ժառանգներում ----------

    abstract public function inStock(): bool;
    abstract public function typeLabel(): string;
}

class PhysicalProduct extends Product implements Shippable
{
    public function __construct(
        int $id,
        string $name,
        string $slug,
        int $price,
        int $discount = 0,
        ?string $specs = null,
        ?string $image = null,
        public readonly int $stock = 0,
        public readonly int $weightGrams = 0,
    ) {
        parent::__construct($id, $name, $slug, $price, $discount, $specs, $image);
    }

    public static function fromRow(array $row): self
    {
        return new self(
            id:          (int) $row['id'],
            name:        (string) $row['name'],
            slug:        (string) $row['slug'],
            price:       (int) $row['price'],
            discount:    (int) ($row['discount'] ?? 0),
            specs:       $row['specs'] ?? null,
            image:       $row['image'] ?? null,
            stock:       (int) ($row['stock'] ?? 0),
            weightGrams: (int) ($row['weight'] ?? 0),
        );
    }

    public function inStock(): bool
    {
        return $this->stock > 0;
    }

    public function isLowStock(): bool
    {
        return $this->stock > 0 && $this->stock <= 3;
    }

    public function typeLabel(): string
    {
        return 'Ապրանք';
    }

    public function weight(): int
    {
        return $this->weightGrams;
    }

    public function shippingCost(): int
    {
        return $this->weightGrams > 5000 ? 3000 : 1000;
    }
}

  2. Դարձրու Product-ը abstract
  3. Ստեղծիր PhysicalProduct և DigitalProduct
  4. Ստեղծիր Shippable ինտերֆեյսը
  5. Փոխիր Product::fromRow()-ը՝ ընտրի ճիշտ դասը
  6. Ավելացրու թվային ապրանք բազայում
  7. Թարմացրու շաբլոնը instanceof-ով
  8. Գրիր calculateShipping() ֆունկցիան
  9. Փորձիր new Product(...) — տես սխալը
  10. Ջնջիր typeLabel()-ը DigitalProduct-ից — տես, երբ է սխալը հայտնվում
Տասներորդ քայլի իմաստը

Սխալը հայտնվում է ֆայլը բեռնելու պահին, ոչ մեթոդը կանչելիս։

Դա abstract-ի ամբողջ արժեքն է․ սխալը գտնվում է անմիջապես, ոչ երեք ամիս անց, երբ ինչ-որ մեկը բացի թվային ապրանքի էջը։

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Թղթային, աուդիո, էլեկտրոնային», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Կատարիր դասարանականի 10 քայլը։
  2. Ավելացրու երրորդ տեսակ՝ ServiceProduct (վերանորոգում, տեղադրում)։ Ի՞նչ մեթոդներ պիտի իրագործի։
  3. Ցուցադրիր մոռացված parent::__construct()-ի սխալը։
  4. Ստեղծիր Discountable ինտերֆեյսը և իրագործիր այն։
  5. Գրիր ֆունկցիա, որը ընդունում է Shippable և հաշվում ընդհանուր քաշը։
  6. Ցուցադրիր պոլիմորֆիզմը՝ մեկ foreach-ով երեք տարբեր տեսակի ապրանք։
  7. Գրիր նույն կոդը առանց պոլիմորֆիզմի՝ if/elseif-ով։ Համեմատիր տողերի քանակը և ընթեռնելիությունը։
  8. Գտիր քո կոդում տեղ, որտեղ ժառանգումը սխալ որոշում կլիներ։ Բացատրիր, ինչու։
  9. Վերաշարադրիր formattedPrice()-ը կոմպոզիցիայով՝ MoneyFormatter դասով։
  10. Խնդրիր AI-ին քննադատել քո հիերարխիան։ Առաջարկե՞ց կոմպոզիցիա որևէ տեղում։

09Հիշելու թերթիկ

ԳրառումԻնչ է
class A extends BԺառանգում
parent::__construct()Ծնողի կոնստրուկտոր ⭐
protectedԴասում և ժառանգներում
abstract classՉի ստեղծվում ուղիղ
abstract functionՊարտադիր ժառանգներում ⭐
interfaceՊայմանագիր
implements A, BՄի քանիսը միանգամից ⭐
instanceofՏիպի ստուգում
self / staticԱյս դասը / կանչված դասը

Ո՞ր գործիքը

«է» → extends «կարող է» → interface «ունի» → կոմպոզիցիա ⭐ Կիսել կոդ → կոմպոզիցիա, ոչ ժառանգում Պայմանագիր → interface Իրական հիերարխիա → abstract + extends

Ստուգաթերթ

  • Բազային դասը abstract է, եթե ուղիղ չի օգտագործվում
  • Ամեն ժառանգ կանչում է parent::__construct()
  • Ժառանգման մակարդակները երկուսից ավելի չեն
  • Ինտերֆեյսների անունները նկարագրում են ունակություն
  • Կոդ կիսելու համար օգտագործվում է կոմպոզիցիա
  • Շաբլոնում instanceof-ը հազվադեպ է

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել abstract-ի և interface-ի տարբերությունը
  • Գիտե՞մ, ինչու մի քանի ինտերֆեյս հնարավոր է, իսկ մի քանի ծնող՝ ոչ
  • Հասկանո՞ւմ եմ, ի՞նչ է պոլիմորֆիզմը
  • Կարո՞ղ եմ բացատրել, ե՞րբ կոմպոզիցիա, ե՞րբ ժառանգում
  • Գիտե՞մ, ինչու երեք մակարդակը վտանգավոր է
Հաջորդ դասին

Composer, namespace և Repository։ bootstrap.php-ում արդեն 8 require_once կա, և ամեն նոր դասով դրանք կաճեն։

Autoload-ը լուծում է դա մեկ տողով։ Իսկ Repository-ն վերջապես կհանի SQL-ը գլոբալ ֆունկցիաներից և կդառնա այն շերտը, որը Դաս 20-ում Laravel-ի Eloquent-ը կփոխարինի։
