Իրադարձություններ

Էջն արդեն գեղեցիկ է և տվյալները ցուցադրվում են։ Բայց կոճակները դեռ մեռած են։ Այսօր դրանք կենդանանում են՝ սովորում ենք լսել օգտագործողի գործողությունները։

01Ի՞նչ է իրադարձությունը

Իրադարձություն (event) — ամեն ինչ, ինչ տեղի է ունենում էջում․ սեղմում, մկնիկի շարժում, ստեղնի սեղմում, էջի բեռնում, ֆորմայի ուղարկում, սքրոլ։

Բրաուզերն ինքն է հետևում դրանց։ Մեր գործն է ասել՝ «երբ այս բանը պատահի՝ կանչիր այս ֆունկցիան»։ Հիշիր Դաս 6-ի callback-ը — ահա նրա գլխավոր կիրառությունը։

հին ձևը՝ HTML-ում
<button onclick="alert('Բարև')">Սեղմիր</button>
Ինչո՞ւ այսպես չենք գրում

HTML-ը և JS-ը խառնվում են։ Մեկ էլեմենտին կարելի է կախել միայն մեկ գործողություն։ Հեռացնել հնարավոր չէ։ Կոդը դժվար է գտնել։ Օգտագործում ենք միայն addEventListener։

02addEventListener

կառուցվածքը
element.addEventListener("իրադարձություն", ֆունկցիա);
//                        ↑              ↑
//                    ինչ լսել      ինչ անել (callback)
առաջին օրինակը
const btn = document.querySelector(".btn");

btn.addEventListener("click", () => {
  console.log("Կոճակը սեղմեցին։");
});

// կամ առանձին ֆունկցիայով
function handleClick() {
  console.log("Երկրորդ մշակիչը։");
}

btn.addEventListener("click", handleClick);
Console  (մեկ սեղմումից հետո)
Կոճակը սեղմեցին։
Երկրորդ մշակիչը։

Ֆունկցիան փոխանցում ենք առանց փակագծերի։ handleClick ✅ · handleClick() ❌
Փակագծերով դու անմիջապես կանչում ես ֆունկցիան և listener-ին փոխանցում նրա արդյունքը (սովորաբար undefined)։

փակագծերի թակարդը
btn.addEventListener("click", handleClick());  // ❌
// աշխատում է ՀԻՄԱ, մեկ անգամ, սեղմումից անկախ

btn.addEventListener("click", handleClick);    // ✅

// Իսկ եթե պիտի արգումենտ փոխանցես — փաթաթիր arrow-ի մեջ
btn.addEventListener("click", () => deleteItem(42));  // ✅

Հեռացնել մշակիչը

removeEventListener · once
function handleClick() {
  console.log("Սեղմում");
}

btn.addEventListener("click", handleClick);
btn.removeEventListener("click", handleClick);   // նույն ֆունկցիան

// Միայն մեկ անգամ աշխատող մշակիչ
btn.addEventListener("click", () => {
  console.log("Սա միայն մեկ անգամ կաշխատի");
}, { once: true });
Ուշադրություն

removeEventListener-ը աշխատում է միայն այն դեպքում, երբ ֆունկցիան անվանական է։ Անանուն arrow ֆունկցիան հեռացնել հնարավոր չէ — չկա այն անունը, որով հղվես։

03Իրադարձության օբյեկտը

Բրաուզերը callback-ին ինքնաբերաբար փոխանցում է մեկ արգումենտ՝ իրադարձության օբյեկտը։ Այնտեղ ամբողջ տեղեկությունն է կատարվածի մասին։

event օբյեկտ
btn.addEventListener("click", (event) => {
  console.log(event.type);
  console.log(event.target);
  console.log(event.clientX, event.clientY);
});

04preventDefault

Որոշ էլեմենտներ ունեն լռելյայն վարքագիծ․ հղումը տանում է նոր էջ, ֆորման վերաբեռնում է էջը, աջ սեղմումը բացում է մենյու։ Երբեմն մենք ուզում ենք դա կանգնեցնել։

հղումը կանգնեցնել
const link = document.querySelector("a");

link.addEventListener("click", (e) => {
  e.preventDefault();      // էջը չի փոխվի
  console.log("Հղումը սեղմեցին, բայց չենք գնում։");
});
ֆորմայի ուղարկում
const form = document.querySelector("form");

form.addEventListener("submit", (e) => {
  e.preventDefault();      // էջը չի վերաբեռնվի
  console.log("Ֆորման ուղարկվեց JS-ով։");
});
Առանց preventDefault-ի

Ֆորման ուղարկելիս էջը վերաբեռնվում է, և քո ամբողջ JS-ը սկսվում է զրոյից։ Կարծես ոչինչ չի կատարվել։ Դաս 13-ում սա կլինի առաջին տողը՝ ամեն submit մշակիչում։

05Իրադարձությունների տեսակները

ԻրադարձությունԵ՞րբ է տեղի ունենում
clickՍեղմում (նաև հպում հեռախոսին)
dblclickԿրկնակի սեղմում
mouseenter / mouseleaveՄկնիկը մտավ / դուրս եկավ
mousemoveՄկնիկը շարժվում է
contextmenuԱջ սեղմում
keydown / keyupՍտեղնը սեղմեցին / բաց թողեցին
inputԴաշտի արժեքը փոխվեց (ամեն նիշից)
changeԱրժեքը փոխվեց և ֆոկուսը գնաց
submitՖորման ուղարկվեց
focus / blurԴաշտն ակտիվացավ / կորցրեց ֆոկուսը
scrollԷջը սքրոլ արվեց
resizeՊատուհանի չափը փոխվեց
մկնիկի հետ
const card = document.querySelector(".card");

card.addEventListener("mouseenter", () => {
  card.classList.add("is-hovered");
});

card.addEventListener("mouseleave", () => {
  card.classList.remove("is-hovered");
});
Երբ CSS-ը բավական է

Վերևի օրինակը իրականում պիտի լինի պարզ .card:hover CSS-ում։ JS-ը օգտագործիր, երբ պիտի տրամաբանություն լինի (հաշվել, պահել, ուղարկել), ոչ պարզապես գույն փոխել։

06Դասարանական · չորս բաղադրիչ

Այս չորսը կհանդիպեն գրեթե ամեն նախագծում։ Ուշադրություն դարձրու՝ բոլորը կառուցված են նույն սկզբունքով՝ դաս ավելացնել/հանել։

Ա — Բուրգեր մենյու

HTML + CSS
<button class="burger">☰</button>
<nav class="nav">
  <a href="#">Գլխավոր</a>
  <a href="#">Ապրանքներ</a>
</nav>
/* style.css */
.nav { display: none; }
.nav.is-open { display: block; }
burger.js
const burger = document.querySelector(".burger");
const nav    = document.querySelector(".nav");

burger.addEventListener("click", () => {
  nav.classList.toggle("is-open");
  burger.textContent = nav.classList.contains("is-open") ? "✕" : "☰";
});

Չորս տող։ Ամբողջ մենյուն։

Բ — Մոդալ պատուհան

modal.js
const openBtn  = document.querySelector(".open-modal");
const modal    = document.querySelector(".modal");
const closeBtn = document.querySelector(".modal__close");

function openModal() {
  modal.classList.add("is-open");
  document.body.classList.add("no-scroll");
}

function closeModal() {
  modal.classList.remove("is-open");
  document.body.classList.remove("no-scroll");
}

openBtn.addEventListener("click", openModal);
closeBtn.addEventListener("click", closeModal);

// Փակել ֆոնին սեղմելիս
modal.addEventListener("click", (e) => {
  if (e.target === modal) {     // հենց ֆոնը, ոչ ներսի բովանդակությունը
    closeModal();
  }
});

// Փակել Escape-ով
document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && modal.classList.contains("is-open")) {
    closeModal();
  }
});
Ուշադրություն

e.target === modal ստուգումը շատ կարևոր է։ Առանց դրա մոդալը կփակվեր նաև ներսի տեքստին սեղմելիս, որովհետև սեղմումը «բարձրանում է» դեպի վեր։ Ինչո՞ւ — կիմանաս հաջորդ դասին։

Գ — Ակորդեոն

accordion.js
// <div class="accordion__item">
//   <button class="accordion__head">Հարց 1</button>
//   <div class="accordion__body">Պատասխան</div>
// </div>

const heads = document.querySelectorAll(".accordion__head");

heads.forEach(head => {
  head.addEventListener("click", () => {
    const item = head.closest(".accordion__item");   // Դաս 9

    // Փակել մյուսները
    document.querySelectorAll(".accordion__item").forEach(other => {
      if (other !== item) {
        other.classList.remove("is-open");
      }
    });

    // Բացել/փակել ընթացիկը
    item.classList.toggle("is-open");
  });
});

Դ — Թաբեր

tabs.js
// <button class="tab" data-tab="one">Առաջին</button>
// <div class="panel" data-panel="one">...</div>

const tabs   = document.querySelectorAll(".tab");
const panels = document.querySelectorAll(".panel");

tabs.forEach(tab => {
  tab.addEventListener("click", () => {
    const name = tab.dataset.tab;          // Դաս 10

    // Հանել ակտիվությունը բոլորից
    tabs.forEach(t => t.classList.remove("is-active"));
    panels.forEach(p => p.classList.remove("is-active"));

    // Ակտիվացնել ընտրվածը
    tab.classList.add("is-active");
    document
      .querySelector(`.panel[data-panel="${name}"]`)
      .classList.add("is-active");
  });
});
Ընդհանուր օրինաչափությունը

Ուշադրություն դարձրու՝ ակորդեոնում և թաբերում նույն երկու քայլն է․
1. Հանել վիճակը բոլորից։
2. Ավելացնել վիճակը ընտրվածին։
Այս օրինաչափությունը կհանդիպի ամենուր՝ գալերեաներ, ֆիլտրեր, քայլ առ քայլ ֆորմաներ։

Այստեղ ստեղծում ենք կոճակ հիշողության մեջ, կախում մշակիչ և արհեստականորեն «սեղմում» այն click()-ով։ Այսպես կարող ես տեսնել իրադարձության օբյեկտը՝ առանց էջին դիպչելու։

  2. Հաշվիչ․ երկու կոճակ (+ և −) և թիվ էկրանին։ Թիվը 0-ից ցածր չպիտի իջնի։
  3. Մուգ/բաց թեմայի փոխարկիչ՝ document.body.classList.toggle("dark")։
  4. Բուրգեր մենյու։ Ավելացրու՝ մենյուն փակվի նաև ցանկացած հղումի սեղմելիս։
  5. Մոդալ պատուհան՝ բացում, փակում խաչիկով, ֆոնին սեղմելով և Escape-ով։
  6. Ակորդեոն առնվազն 4 հարցով։ Միաժամանակ միայն մեկը պիտի բաց լինի։
  7. Թաբեր՝ 3 ներդիրով և 3 պանելով, data-* ատրիբուտներով։
  8. Դաշտ և պարբերություն։ input իրադարձությամբ պարբերությունը պիտի ցույց տա մուտքագրվածը իրական ժամանակում։
  9. Նույն դաշտի տակ ցույց տուր՝ քանի՞ նիշ է մնացել 100-ից։ 90-ից հետո տեքստը դառնա կարմիր (classList-ով)։
  10. Վերցրու Դաս 10-ի կատալոգը։ Ավելացրու 3 ֆիլտրի կոճակ՝ «Բոլորը», «Առկա», «Էժան»։ Ամեն սեղմումին կանչիր render()-ը ֆիլտրված զանգվածով և ակտիվ կոճակին տուր is-active դասը։
Ուշադրություն 10-րդին

Կնկատես մի խնդիր․ render()-ից հետո քարտերի կոճակների մշակիչները չեն աշխատում։ Սա ճիշտ է և սպասելի։ Բեր այդ խնդիրը հաջորդ դասին — հենց դրանով կսկսենք։
