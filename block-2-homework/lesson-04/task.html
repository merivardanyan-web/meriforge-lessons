function getStatus(read, total) {
    if (read === 0) return "Կարդալու եմ";
    if (read < total) return "Կարդում եմ";
    return "Կարդացել եմ";
}

04Դասի մյուս վարժությունները

Սրանք Դաս 04-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում, մնացածը կատարիր առանձին ֆայլերում։

  1. getSum(a, b) — վերադարձնում է երկու թվի գումարը։
  2. getRectangleArea(w, h) — մակերես։ Եթե պարամետր չուղարկեն, լռելյայն երկուսն էլ 1։
  3. isPositive(n) — վերադարձնում է true կամ false։
  4. maxOfThree(a, b, c) — երեք թվից ամենամեծը։
  5. celsiusToFahrenheit(c) — բանաձևը՝ c * 9 / 5 + 32։
  6. getGrade(mark) — վերադարձնում է «Գերազանց», «Լավ», «Բավարար» կամ «Անբավարար»։ Օգտագործիր return-ը առանց else-ի։ մինի-նախագծում
  7. sumTo(n) — 1-ից n թվերի գումարը (ցիկլով)։
  8. countDigits(n) — քանի՞ թվանշան ունի թիվը։
  9. isPrime(n) — պարզ թիվ է թե ոչ։ Օգտագործիր Դաս 3-ի դրոշակի հնարքը։
  10. Գրիր getTicketPrice(age, isStudent)․ մինչև 6 տարեկան՝ 0, ուսանող՝ 1000, մինչև 65՝ 2500, 65-ից բարձր՝ 1000։ Ստուգիր 5 տարբեր դեպք։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը և դրա կոդը example/js/-ում։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
