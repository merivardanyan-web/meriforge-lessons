js/api.js — buildUrl
export function buildUrl(base, params = {}) {
    const clean = Object.fromEntries(
        Object.entries(params).filter(([, value]) => value !== "" && value !== null && value !== undefined)
    );
    const query = new URLSearchParams(clean).toString();
    return query ? `${base}?${query}` : base;
}

04Դասի մյուս վարժությունները

Սրանք Դաս 21-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում, մնացածը կատարիր առանձին ֆայլերում։

  1. Գրիր buildUrl(path, params) ֆունկցիան։ Ստուգիր 5 տարբեր դեպքով, ներառյալ դատարկ և հայերեն արժեքներով։ մինի-նախագծում
  2. Բեռնիր /posts?_page=1&_limit=10։ Կարդա X-Total-Count վերնագիրը և տպիր էջերի քանակը։
  3. Կառուցիր էջավորում կոճակներով։ Ակտիվ էջը նշիր is-active դասով։ մինի-նախագծում
  4. Ավելացրու «←» և «→» կոճակներ՝ ճիշտ disabled վիճակներով եզրային էջերում։ մինի-նախագծում
  5. Ավելացրու որոնման դաշտ debounce-ով (400 մլվրկ)։ Նոր որոնման դեպքում վերադարձիր առաջին էջ։ մինի-նախագծում
  6. Ավելացրու AbortController։ Ստուգիր՝ DevTools → Network-ում պիտի տեսնես canceled հարցումներ։ մինի-նախագծում
  7. Ավելացրու select՝ էջում 5 / 10 / 20 արդյունք։ Փոփոխելիս վերադարձիր առաջին էջ։ մինի-նախագծում
  8. Կազմակերպիր api.js մոդուլ՝ getPosts, getPost, getUsers, createPost մեթոդներով։ մինի-նախագծում
  9. Ցուցադրիր՝ «Ցուցադրված է 21–30 / 100»։ Օգտագործիր խաղահրապարակի paginate ֆունկցիան։ մինի-նախագծում
  10. Ավելացրու պարզ քեշ․ Map օբյեկտում պահիր արդեն բեռնված էջերը և կրկնակի հարցում մի՛ ուղարկիր։ Ցույց տուր Console-ում՝ «քեշից» թե «սերվերից»։
