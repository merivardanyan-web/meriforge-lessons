01Հարցման պարամետրեր

Սերվերին կարելի է լրացուցիչ պայմաններ ասել՝ ուղիղ հասցեի մեջ, ?-ից հետո։

Ինչ արեց այն

Բացատները դարձրեց +, հայերեն տառերն ու &-ը՝ անվտանգ կոդեր (%D5%B6)։ Սերվերը դրանք հետ կվերծանի ճիշտ։ Երբեք ձեռքով մի՛ կպցրու պարամետրերը։

օգտակար ֆունկցիա
const API = "https://jsonplaceholder.typicode.com";

function buildUrl(path, params = {}) {
  // Հանում ենք դատարկ արժեքները
  const clean = Object.fromEntries(
    Object.entries(params).filter(([, v]) => v !== "" && v != null)
  );

  const query = new URLSearchParams(clean).toString();
  return query ? `${API}${path}?${query}` : `${API}${path}`;
}

console.log(buildUrl("/posts", { userId: 1, _limit: 5 }));
console.log(buildUrl("/posts", { q: "", _limit: 5 }));   // դատարկը դուրս է մնում

02Էջավորում

10 000 գրառում մեկ հարցումով բերելը վատ գաղափար է․ դանդաղ է, ծանր է, և օգտագործողին միևնույն է պետք են առաջին 20-ը։

_page և _limit
async function getPosts(page = 1, limit = 10) {
  const url = buildUrl("/posts", { _page: page, _limit: limit });
  const response = await fetch(url);

  if (!response.ok) throw new Error(`Կարգավիճակ ${response.status}`);

  // Ընդհանուր քանակը գալիս է ՎԵՐՆԱԳՐՈՎ, ոչ մարմնում
  const total = Number(response.headers.get("X-Total-Count"));
  const items = await response.json();

  return {
    items,
    total,
    totalPages: Math.ceil(total / limit)     // Դաս 8
  };
}

const { items, total, totalPages } = await getPosts(2, 10);
console.log(items.length, total, totalPages);

Վերնագրերը

Սերվերը պատասխանի հետ ուղարկում է նաև վերնագրեր՝ լրացուցիչ տեղեկություն։ X-Total-Count-ը այնտեղ է, ոչ տվյալների մեջ։ Առանց դրա չէիր իմանա՝ քանի էջ կա։

Ուշադրություն՝ headers.get()-ը միշտ վերադարձնում է տող, ուստի Number()։

էջավորման կոճակների render
function renderPagination(current, totalPages) {
  if (totalPages <= 1) {
    pagination.innerHTML = "";
    return;
  }

  const pages = Array.from({ length: totalPages }, (_, i) => i + 1);

  pagination.innerHTML = `
    <button data-page="${current - 1}" ${current === 1 ? "disabled" : ""}>←</button>
    ${pages.map(p => `
      <button data-page="${p}" class="${p === current ? "is-active" : ""}">${p}</button>
    `).join("")}
    <button data-page="${current + 1}" ${current === totalPages ? "disabled" : ""}>→</button>
  `;
}

// Դելեգացիա (Դաս 12)
pagination.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-page]");
  if (!btn || btn.disabled) return;

  state.page = Number(btn.dataset.page);
  load();
});
Array.from

Array.from({ length: 5 }, (_, i) => i + 1) → [1, 2, 3, 4, 5]։ Հարմար ձև՝ թվերի հաջորդականություն ստանալու։ _-ը նշանակում է «այս արգումենտը պետք չէ»։

03Մրցավազքի խնդիրը

searchInput.addEventListener("input", debounce(async () => {
  const query = searchInput.value.trim();

}, 300));

04AbortController

Լուծումը՝ նոր հարցում ուղարկելուց առաջ չեղարկել հինը։

ինչպես է աշխատում
const controller = new AbortController();

fetch(url, { signal: controller.signal })
  .then(r => r.json())

  .catch(err => {
    if (err.name === "AbortError") {
      console.log("Հարցումը չեղարկվեց");
    }
  });

// Չեղարկում
controller.abort();
✅ ճիշտ որոնում
let controller = null;

async function search(query) {
  // Չեղարկում ենք նախորդը, եթե դեռ ընթացքում է
  if (controller) controller.abort();

  controller = new AbortController();

  try {
    const url = buildUrl("/posts", { q: query, _limit: 10 });
    const response = await fetch(url, { signal: controller.signal });

    if (!response.ok) throw new Error(`Կարգավիճակ ${response.status}`);

    const data = await response.json();
    render(data);

  } catch (error) {
    // Չեղարկումը սխալ չէ — լուռ անտեսում ենք
    if (error.name === "AbortError") return;

    showError("Որոնումը չհաջողվեց");
  }
}
Երկու պաշտպանության շերտ

debounce (Դաս 14) — նվազեցնում է հարցումների քանակը։ Առանց դրա 9 տառ = 9 հարցում։
AbortController — երաշխավորում է, որ կցուցադրվի վերջին հարցման արդյունքը։

Երկուսն էլ պետք են։ Դրանք տարբեր խնդիրներ են լուծում։

բոնուս՝ ժամանակի սահմանափակում
async function fetchWithTimeout(url, ms = 5000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), ms);

  try {
    return await fetch(url, { signal: controller.signal });
  } finally {
    clearTimeout(timer);
  }
}

05CORS

Վաղ թե ուշ կտեսնես այս սխալը։ Այն շփոթեցնում է բոլորին։

դասական սխալ
fetch("https://some-api.com/data");
Console
Access to fetch at 'https://some-api.com/data' from origin
'http://localhost:5500' has been blocked by CORS policy

Ինչո՞ւ է սա գոյություն ունի

Պատկերացրու՝ մուտք ես գործել բանկի կայք։ Հետո բացում ես վնասակար կայք։ Եթե սահմանափակում չլիներ, նրա JS-ը կարող էր հարցում ուղարկել բանկին քո անունից՝ քո cookie-ներով։

Դրա համար բրաուզերն ասում է․ «Դու site-a.com-ից ես։ Ուզում ես դիմել site-b.com-ին։ Թող site-b-ն ինքը թույլ տա»։

թույլտվությունը գալիս է վերնագրով
# Սերվերի պատասխանի վերնագրերում
Access-Control-Allow-Origin: *                        # բոլորին
Access-Control-Allow-Origin: https://mysite.am        # միայն մեկին
Ամենակարևորը

CORS-ը սերվերի կողմն է։ Դու JS-ով այն շրջանցել չես կարող — ոչ մի հնարք չկա։ Եթե API-ն թույլ չի տալիս, ուրեմն այն նախատեսված չէ բրաուզերից ուղիղ օգտագործելու համար։

ԻրավիճակԼուծում
Հրապարակային API (jsonplaceholder)Ամեն ինչ արդեն աշխատում է
Քո սեփական սերվերըԱվելացրու վերնագիրը — PHP-ի բլոկում կսովորենք
Ուրիշի API՝ առանց CORS-իՀարցումն արա սերվերից, ոչ բրաուզերից
API բանալի պահանջող APIԲանալին երբեք չդնես ֆրոնտենդում
Հիշիր Դաս 18-ը

Նույն CORS-ի պատճառով մոդուլները չեն աշխատում file://-ով։ Բրաուզերի համար ամեն տեղական ֆայլ առանձին «ծագում» է։ Live Server-ը լուծում է հենց այդ խնդիրը։

06API շերտ

Երբ հարցումները շատանում են, դրանք պիտի հավաքվեն մեկ տեղում։ Դա կոչվում է API շերտ։

js/api.js
const BASE_URL = "https://jsonplaceholder.typicode.com";

function buildUrl(path, params = {}) {
  const clean = Object.fromEntries(
    Object.entries(params).filter(([, v]) => v !== "" && v != null)
  );
  const query = new URLSearchParams(clean).toString();
  return query ? `${BASE_URL}${path}?${query}` : `${BASE_URL}${path}`;
}

// Ընդհանուր ֆունկցիա — բոլոր հարցումները անցնում են այստեղով
async function request(path, { params, signal, ...options } = {}) {
  const response = await fetch(buildUrl(path, params), { signal, ...options });

  if (!response.ok) {
    throw new Error(`${response.status} ${response.statusText}`);
  }

  return {
    data: await response.json(),
    total: Number(response.headers.get("X-Total-Count")) || 0
  };
}

// Հրապարակային մեթոդներ — պարզ և ընթեռնելի
export const api = {
  getPosts(params, signal) {
    return request("/posts", { params, signal });
  },

  getPost(id, signal) {
    return request(`/posts/${id}`, { signal });
  },

  getUsers(signal) {
    return request("/users", { signal });
  },

  createPost(post) {
    return request("/posts", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(post)
    });
  }
};
օգտագործում — մաքուր և կարճ

Վաղը API-ի հասցեն կփոխվի — կուղղես մեկ տողում։ Կպահանջվի ավելացնել authorization վերնագիր — մեկ տեղում։ Կուզես գրանցել բոլոր սխալները — մեկ տեղում։

Առանց API շերտի այս փոփոխությունները կպահանջեին 30 ֆայլ բացել։

07Դասարանական · որոնման հավելված

js/main.js
import { api } from "./api.js";
import { debounce } from "./utils.js";

const state = {
  query: "",
  page: 1,
  limit: 10,
  total: 0
};

let controller = null;

const searchInput = document.querySelector(".search");
const container   = document.querySelector("#results");
const pagination  = document.querySelector(".pagination");
const statusEl    = document.querySelector(".status");

// ---------- Բեռնում ----------
async function load() {
  if (controller) controller.abort();
  controller = new AbortController();

  showLoading();

  try {
    const { data, total } = await api.getPosts({
      q: state.query,
      _page: state.page,
      _limit: state.limit
    }, controller.signal);

    state.total = total;
    renderResults(data);
    renderPagination();

  } catch (error) {
    if (error.name === "AbortError") return;   // չեղարկումը սխալ չէ
    showError("Չհաջողվեց բեռնել արդյունքները");
  }
}

// ---------- Ցուցադրում ----------
function showLoading() {
  statusEl.textContent = "Բեռնվում է...";
}

function showError(message) {
  statusEl.textContent = "";
  pagination.innerHTML = "";
  container.innerHTML = `
    <div class="col"><div class="wrapper error">
      <p>${message}</p>
      <button data-action="retry">Կրկին փորձել</button>
    </div></div>
  `;
}

function renderResults(posts) {
  if (posts.length === 0) {
    statusEl.textContent = "";
    container.innerHTML = `<div class="col"><div class="wrapper empty">
      <p>Ոչինչ չգտնվեց «${state.query}» հարցումով</p>
    </div></div>`;
    return;
  }

  statusEl.textContent = `Գտնվեց ${state.total} արդյունք`;

  container.innerHTML = posts.map(p => `
    <div class="col">
      <div class="wrapper">
        <h3>${p.title}</h3>
        <p>${p.body.slice(0, 100)}...</p>
      </div>
    </div>
  `).join("");
}

function renderPagination() {
  const totalPages = Math.ceil(state.total / state.limit);

  if (totalPages <= 1) {
    pagination.innerHTML = "";
    return;
  }

  const pages = Array.from({ length: totalPages }, (_, i) => i + 1);

  pagination.innerHTML = pages.map(p => `
    <button data-page="${p}" class="${p === state.page ? "is-active" : ""}">${p}</button>
  `).join("");
}

  1. Գրիր buildUrl(path, params) ֆունկցիան։ Ստուգիր 5 տարբեր դեպքով, ներառյալ դատարկ և հայերեն արժեքներով։

  6. Ավելացրու AbortController։ Ստուգիր՝ DevTools → Network-ում պիտի տեսնես canceled հարցումներ։
  7. Ավելացրու select՝ էջում 5 / 10 / 20 արդյունք։ Փոփոխելիս վերադարձիր առաջին էջ։
  8. Կազմակերպիր api.js մոդուլ՝ getPosts, getPost, getUsers, createPost մեթոդներով։
  9. Ցուցադրիր՝ «Ցուցադրված է 21–30 / 100»։ Օգտագործիր խաղահրապարակի paginate ֆունկցիան։
