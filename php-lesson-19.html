<?php

namespace App\Services;

class Cart
{
    private const MAX_QTY = 99;

    public function all(): array
    {
        return session('cart', []);
    }

    public function add(int $id, int $qty = 1): void
    {
        if ($qty < 1) {
            return;
        }

        $items = $this->all();
        $items[$id] = min(($items[$id] ?? 0) + $qty, self::MAX_QTY);

        session(['cart' => $items]);
    }

    public function remove(int $id): void
    {
        $items = $this->all();
        unset($items[$id]);

        session(['cart' => $items]);
    }

    public function count(): int
    {
        return array_sum($this->all());
    }

<?php

namespace App\Http\Controllers;

use App\Services\Cart;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class CartController extends Controller
{
    public function __construct(
        private readonly Cart $cart,       // ⭐ ոչ մի գրանցում, ոչ մի match
    ) {}

    public function store(Request $request)
    {
        $id  = $request->integer('id');
        $qty = $request->integer('qty', 1);

        $product = DB::table('products')
            ->where('id', $id)
            ->where('is_active', 1)
            ->first();

        // Բիզնես-կանոնները՝ սերվերում (Դաս 17) ⭐
        abort_if($product === null, 404, 'Ապրանքը չի գտնվել');
        abort_if($qty < 1 || $qty > 99, 422, 'Քանակը՝ 1-ից 99');
        abort_if($product->stock < 1, 422, 'Ապրանքը առկա չէ');

        $this->cart->add($id, $qty);

        // Մեկ կոնտրոլեր — երկու ձևաչափ (Դաս 17-ի expectsJson)
        if ($request->expectsJson()) {
            return response()->json([
                'data' => [
                    'count'   => $this->cart->count(),
                    'message' => 'Ավելացվեց՝ ' . $product->name,
                ],
            ]);
        }

        return back()->with('success', 'Ավելացվեց զամբյուղ');   // ⭐ Դաս 7-ի flash()
    }

    public function destroy(int $id)
    {
        $this->cart->remove($id);

        return back()->with('success', 'Հանվեց զամբյուղից');
    }

HTML ֆորման կարող է ուղարկել միայն GET կամ POST։ <form method="put">-ը բրաուզերը պարզապես կվերածի GET-ի։
