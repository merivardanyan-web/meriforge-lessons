app/Http/Requests/ProductRequest.php — Դաս 11-ի validateProduct()-ը
<?php

namespace App\Http\Requests;

use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Validation\Rule;

class ProductRequest extends FormRequest
{
    /** Ո՞վ կարող է ուղարկել այս ֆորման — բաժին 08 */
    public function authorize(): bool
    {
        return $this->user()?->can('manage-products') ?? false;
    }

    /** Checkbox-ը չի ուղարկվում, երբ նշված չէ — Դաս 6-ի տնայինը ⭐ */
    protected function prepareForValidation(): void
    {
        $this->merge(['is_active' => $this->boolean('is_active')]);
    }

    public function rules(): array
    {
        return [
            'name'        => ['required', 'string', 'max:200'],
            'category_id' => ['nullable', 'integer', Rule::exists('categories', 'id')],
            'specs'       => ['nullable', 'string', 'max:255'],
            'description' => ['nullable', 'string', 'max:5000'],
            'price'       => ['required', 'integer', 'min:1', 'max:99999999'],
            'discount'    => ['required', 'integer', 'between:0,90'],
            'stock'       => ['required', 'integer', 'min:0', 'max:100000'],
            'type'        => ['required', Rule::in(['physical', 'digital'])],
            'is_active'   => ['boolean'],
        ];
    }

    /** Դաշտերի հայերեն անունները սխալների համար */
    public function attributes(): array
    {
        return [
            'name'        => 'անվանումը',
            'category_id' => 'կատեգորիան',
            'price'       => 'գինը',
            'discount'    => 'զեղչը',
            'stock'       => 'քանակը',
        ];
    }
}
Ինչո՞ւ առանձին դաս

Դաս 15-ի կանոնը՝ «եթե կոնտրոլերը 40 տողից երկար է, այնտեղ ընկել է մի բան, որի տեղը այնտեղ չէ»։ Ինը կանոն կոնտրոլերում՝ դա հենց այդ դեպքն է։

Form Request-ը վալիդացիան տեղափոխում է առանձին ֆայլ։ Կոնտրոլերը ստանում է ProductRequest $request, և երբ մեթոդը սկսում է կատարվել, վալիդացիան արդեն անցել է։ Autowiring-ը (Դաս 19) դա անում է ինքնաբերաբար։

Եվ նույն դասը օգտագործվում է store-ում և update-ում՝ Դաս 11-ի «մեկ ֆորմա երկու գործի համար» գաղափարը վալիդացիայի մակարդակում։

authorize()-ը լռելյայն false է

make:request-ը գեներացնում է authorize(), որը վերադարձնում է false։ Եթե այն չփոխես, ամեն ուղարկում կավարտվի 403 Forbidden-ով, և սկզբում պարզ չի լինի ինչու։

Գայթակղություն կլինի գրել return true; և շարունակել։ Մի՛ արա։ Այդ մեթոդը հենց այն տեղն է, որտեղ ասում ես՝ ո՞վ կարող է կատարել գործողությունը։ Մենք այն կապում ենք Gate-ի հետ բաժին 08-ում։

public static function uniqueSlugFor(string $name, ?int $exceptId = null): string
{
    $base = slugify($name) ?: 'product';   // Դաս 11-ի հայերեն քարտեզը app/helpers.php-ում
    $slug = $base;
    $i = 1;

    while (
        static::withTrashed()                           // ⭐ Դաս 20-ի թակարդը
            ->where('slug', $slug)
            ->when($exceptId, fn ($q) => $q->whereKeyNot($exceptId))   // ⭐ Դաս 11-ի $exceptId-ը
            ->exists()
    ) {
        $slug = $base . '-' . (++$i);
    }

    return $slug;
}

06Մուտք՝ Fortify-ով

Ինչո՞ւ ոչ starter kit — ազնիվ բացատրություն

Laravel 13-ի պաշտոնական starter kit-երը չորսն են՝ React, Vue, Svelte և Livewire։ Դրանք նոր նախագծի կաղապարներ են, որ ընտրվում են laravel new-ի ժամանակ, և ամենը կառուցված են Tailwind-ի վրա։ Մեր նախագիծը արդեն գոյություն ունի և ունի իր սեփական CSS-ը։

Կարևոր փաստ՝ բոլոր չորս kit-երը մուտքի համար օգտագործում են Laravel Fortify՝ «գլուխ չունեցող» (headless) փաթեթ, որը տալիս է ուղիները, կոնտրոլերները և տրամաբանությունը, բայց ոչ մի շաբլոն։

Ուստի մենք տեղադրում ենք Fortify-ը ուղիղ և գրում ենք մեր շաբլոնները մեր CSS-ով։ Դա այն է, ինչ kit-երը անում են ներսում։ Դաս 19-ի էջավորման և Vite-ի որոշման նույն սկզբունքն է՝ CSS-ը մերն է։ Laravel-ի հին Breeze kit-ը, որ շատ ձեռնարկներում կհանդիպես, Laravel 12-ից այլևս պաշտոնական փաստաթղթերում չէ։

Ահա այն։ Ստուգիր php artisan route:list --path=admin -v-ով․ ամեն տողի կողքին կտեսնես auth և can:manage-products։ Դա Դաս 18-ի «անվտանգության քարտեզն» է։

08Gate-եր՝ ո՞վ ինչ կարող է

ԿոդՆշանակությունՈ՞վ է որոշում
401 / redirect /loginՉգիտենք, թե ով եսauth — նույնականացում
403Գիտենք ով ես, բայց չես կարողGate — լիազորում ⭐
Երկու տարբեր հարց

Նույնականացում (authentication) — «ո՞վ ես դու»։ Fortify-ը և auth-ը։
Լիազորում (authorization) — «ի՞նչ կարող ես անել»։ Gate-երը։

Դաս 17-ի կոդերի աղյուսակում 401-ը և 403-ը առանձին տողեր էին հենց այս պատճառով։ Հարցազրույցում այս տարբերությունը հաճախ է հարցվում։

app/Providers/AppServiceProvider.php
use App\Models\User;
use Illuminate\Support\Facades\Gate;

public function boot(): void
{
    // Մենեջերը և ադմինը՝ ստեղծել և խմբագրել
    Gate::define('manage-products', fn (User $user) =>
        in_array($user->role, ['admin', 'manager'], true)
    );

    // Միայն ադմինը՝ ջնջել
    Gate::define('delete-products', fn (User $user) =>
        $user->role === 'admin'
    );

    // ... shouldBeStrict, View::composer
}
երեք տեղ, որտեղ օգտագործվում է
// 1. Ուղու վրա — բաժին 07
Route::middleware(['auth', 'can:manage-products'])->...

// 2. Կոնտրոլերում — կոնկրետ գործողության համար ⭐
public function destroy(Product $product)
{
    Gate::authorize('delete-products');      // չի՞ կարող → 403

    $product->delete();
    // ...
}

// 3. Շաբլոնում — միայն ցուցադրման համար
@can('delete-products')

        <button class="btn btn--danger btn--sm">Ջնջել</button>
    </form>
@endcan
@can-ը պաշտպանություն չէ

Եթե @can('delete-products')-ը թաքցնում է կոճակը, բայց destroy()-ում չկա Gate::authorize, մենեջերը կարող է ջնջել ապրանքը curl-ով կամ DevTools-ում ֆորմա ավելացնելով։

Սա Դաս 6-ի required-ի, Դաս 11-ի confirm()-ի և Դաս 17-ի «թաքցրած կոճակի» նույն դասն է՝ չորրորդ անգամ։ Ինտերֆեյսը թաքցնում է, սերվերը արգելում է։ Միշտ երկուսն էլ, և երկրորդը՝ պարտադիր։

Gate-ը կարդում է role-ը սերվերում

$user-ը բեռնվում է բազայից սեսիայի ID-ով։ role-ը գալիս է բազայից, ոչ հարցումից։ Դա Դաս 7-ի կանոնն է՝ «Cookie-ում երբեք "ես ադմին եմ"։ Սեսիայում պահվում է հղումը, ոչ տվյալները»։

Իսկ բազայում role-ը կարող է փոխվել միայն ուղիղ վերագրմամբ, որովհետև Դաս 20-ում այն $fillable-ից դուրս թողեցինք։ Երեք դասի որոշումները միասին են կազմում այս պաշտպանությունը։

09Դասարանական

  1. Գործարկիր php artisan lang:publish, ստեղծիր lang/hy/, սահմանիր APP_LOCALE=hy և ստեղծիր ProductRequest-ը
  2. Ավելացրու Product::uniqueSlugFor()-ը, իրագործիր store և update մեթոդները validated()-ով
  3. Թարմացրու admin/products/form-ը՝ old(), @error, aria-invalid
  4. Տեղադրիր Fortify-ը, սահմանիր home և redirects.register, ստեղծիր auth/login.blade.php-ը, auth/register.blade.php-ը (Դաս 16-ի ֆորման Blade-ով) և ելքի ֆորման header-ում
  5. Ավելացրու երկու Gate-ը, փաթեթավորիր ադմին ուղիները ['auth', 'can:manage-products']-ով և destroy-ում ավելացրու Gate::authorize
Վեց պարտադիր փորձարկում

1. Ադմին ֆորմայում ուղարկիր դատարկ անուն և discount=150։ Սխալները հայերեն են, լրացրած մյուս դաշտերը մնացե՞լ են։
2. curl-ով ուղարկիր category_id=99999՝ Դաս 11-ի նույն փորձարկումը։ Rule::exists-ը բռնե՞ց։
3. Ելք կատարիր և բացիր /admin/products/1/edit։ Մուտքից հետո վերադարձա՞ր հենց այդ էջը։
4. Սխալ գաղտնաբառ վեց անգամ։ Ապա ջնջիր cookie-ները և կրկնիր։ Արգելափակումը մնա՞ց։ ⭐
5. Գրանցվիր նոր հասցեով՝ ֆորմային DevTools-ով ավելացնելով role=admin։ Հայտնվեցի՞ր գլխավոր էջում, role-ը customer է, իսկ /admin/products-ը տալի՞ս է 403։ ⭐
6. Ստեղծիր մենեջեր tinker-ում (role-ը ուղիղ վերագրմամբ), մուտք գործիր որպես նա։ «Ջնջել» կոճակը չկա։ Ապա ուղարկիր DELETE հարցումը DevTools-ով կամ curl-ով՝ ճիշտ token-ով։ Պիտի լինի 403։ ⭐

10Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Վալիդացիա, Fortify, Gate-եր», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր դասարանականի 5 քայլը և վեց փորձարկումը։ Ուղարկիր սքրինշոթներ։
  2. Վերաշարադրիր Դաս 6-ի կոնտակտների ֆորման Laravel-ում՝ Form Request-ով, old()-ով և @error-ով։ Համեմատիր տողերի քանակը։
  3. Թարգմանիր lang/hy/validation.php-ը AI-ի օգնությամբ և ստուգիր, որ բոլոր տեղապահները (:attribute, :max…) անփոփոխ են։
  4. Թարգմանիր lang/hy/auth.php-ը։ Ստուգիր, որ սխալ գաղտնաբառի և անգոյ հասցեի հաղորդագրությունները նույնն են (Դաս 16)։
  5. Ավելացրու նկարի վերբեռնում ադմին ֆորմայում․ 'image' => ['nullable', 'image', 'mimes:jpg,png,webp', 'max:2048'], $request->file('image')->store('products', 'public'), php artisan storage:link։ Համեմատիր Դաս 7-ի վեց քայլի հետ։
  6. Փոխարինիր CartController-ի abort_if(..., 422)-ը $request->validate()-ով և տպիր payload.errors-ը cart.js-ում։
  7. Խորացված․ ստեղծիր ProductPolicy (php artisan make:policy ProductPolicy --model=Product) և տեղափոխիր երկու Gate-երը նրա մեջ։ Ի՞նչ փոխվեց կոնտրոլերում։

Ստուգաթերթ

  • Ամեն ֆորմա վավերացվում է սերվերում
  • Ամեն տեքստային դաշտ ունի max
  • Ընտրովի արժեքները ստուգվում են Rule::in կամ Rule::exists-ով
  • Մոդելներին փոխանցվում է validated(), ոչ all()
  • Սխալից հետո լրացրածը մնում է, գաղտնաբառը՝ ոչ
  • Սխալները հայերեն են և հասանելի են էկրանընթերցիչին
  • Գրանցումը անջատված է
  • Բոլոր ադմին ուղիները auth խմբում են
  • Ամեն վտանգավոր գործողություն ունի Gate::authorize
  • authorize()-ը Form Request-ում return true չէ
  • Ելքը POST է

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչ է կատարվում վալիդացիայի ձախողման դեպքում ֆորմայի և JSON-ի համար
  • Գիտե՞մ, ինչու գաղտնաբառը չի վերադառնում old()-ով
  • Հասկանո՞ւմ եմ, ինչու ենք օգտագործում Fortify, ոչ starter kit
  • Կարո՞ղ եմ բացատրել նույնականացման և լիազորման տարբերությունը
  • Գիտե՞մ, ինչու @can-ը բավարար չէ
  • Կարո՞ղ եմ նշել, թե Fortify-ը Դաս 16-ի ո՞ր չորս բանն է փոխարինում
Laravel-ի չորս դասը՝ ավարտված

Դաս 18–21-ում Laravel-ով վերակառուցեցիր այն, ինչ ձեռքով գրել էիր Դաս 5–17-ում․ ուղիներ, շաբլոններ, կոմպոնենտներ, մոդելներ, միգրացիաներ, վալիդացիա, մուտք և իրավունքներ։

Ինչպես ասացինք Դաս 18-ում, սա ծանոթություն է, ոչ տիրապետում։ Բայց հիմա, երբ բացես Laravel-ի որևէ նախագիծ, ոչ մի ֆայլ անծանոթ չի լինի, և կիմանաս՝ ինչու է այն այդպես գրված։

Հաջորդ դասին սկսվում է կարիերայի փուլը։ ՏեխնոՄարկետը դառնում է պորտֆոլիո՝ deploy, README, GitHub պրոֆիլ, դեմո-հղում և LinkedIn։ Նախագիծը, որ կառուցեցիր 72 դասում, պետք է ոչ միայն աշխատի, այլև երևա։
