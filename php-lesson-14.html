02Public թղթապանակ

Անվտանգության հիմնական քայլ

Հիմա .env-ը, src/-ը և vendor/-ը գտնվում են վեբ-արմատում։ Եթե Apache-ի կարգավորումը սխալ լինի, կամ .htaccess-ը չաշխատի, technomarket.test/.env-ը կվերադարձնի գաղտնաբառը տեքստով։

04Router

src/Http/Router.php
<?php
declare(strict_types=1);

namespace App\Http;

class Router
{
    /** @var array<string, array<string, callable>> */
    private array $routes = [];

    public function get(string $path, callable $handler): void
    {
        $this->add('GET', $path, $handler);
    }

    public function post(string $path, callable $handler): void
    {
        $this->add('POST', $path, $handler);
    }

    private function add(string $method, string $path, callable $handler): void
    {
        $this->routes[$method][$path] = $handler;
    }

    public function dispatch(string $method, string $uri): void
    {
        // 1. Մաքրում ենք հասցեն
        $path = parse_url($uri, PHP_URL_PATH) ?? '/';
        $path = '/' . trim($path, '/');

        // 2. Ուղիղ համընկնում
        if (isset($this->routes[$method][$path])) {
            ($this->routes[$method][$path])();
            return;
        }

        // 3. Դինամիկ հատվածներով
        foreach ($this->routes[$method] ?? [] as $route => $handler) {

            $params = $this->match($route, $path);

            if ($params !== null) {
                $handler(...$params);
                return;
            }
        }

        // 4. Ուղին կա՞ ուրիշ մեթոդով
        foreach ($this->routes as $m => $paths) {
            if ($m === $method) {
                continue;
            }

            foreach (array_keys($paths) as $route) {
                if ($route === $path || $this->match($route, $path) !== null) {
                    $this->abort(405, 'Մեթոդը թույլատրված չէ');
                }
            }
        }

        $this->abort(404, 'Էջը չի գտնվել');
    }

    /** Վերադարձնում է պարամետրերը կամ null */
    private function match(string $route, string $path): ?array
    {
        if (!str_contains($route, '{')) {
            return null;
        }

        // /product/{slug} → #^/product/([^/]+)$#
        $pattern = preg_replace(
            '#\{[a-z_]+\}#',
            '([^/]+)',
            preg_quote($route, '#')
        );

        // preg_quote-ը փախցնում է { } — վերականգնում ենք
        $pattern = str_replace(['\{', '\}'], ['{', '}'], $pattern);
        $pattern = preg_replace('#\{[a-z_]+\}#', '([^/]+)', $pattern);

        if (!preg_match('#^' . $pattern . '$#u', $path, $m)) {
            return null;
        }

        array_shift($m);        // առաջինը ամբողջ համընկնումն է

        return array_map('urldecode', $m);
    }

    public function abort(int $code, string $message): never
    {
        http_response_code($code);

        echo render('layout', [
            'title'   => $code . ' — ' . $message,
            'content' => render('pages/error', compact('code', 'message')),
        ]);

        exit;
    }
}
Ինչո՞ւ 405-ի ստուգումը

Երբ ուղին կա, բայց մեթոդը սխալ է, 404-ը շփոթեցնող է — ծրագրավորողը կմտածի, որ ուղին չկա և կփնտրի տառասխալ։

405 Method Not Allowed-ը ասում է ուղիղ․ «ուղին ճիշտ է, բայց GET-ի փոխարեն պետք է POST»։

Դա այն մանրուքն է, որ ժամեր է խնայում ախտորոշման ժամանակ։ Laravel-ը նույնպես տալիս է 405 այս դեպքում։

never վերադարձի տիպը

PHP 8.1-ից։ Նշանակում է՝ ֆունկցիան երբեք չի վերադառնում — կա՛մ exit, կա՛մ բացառություն։

Օգտակար է, որովհետև IDE-ն և ստատիկ անալիզատորները հասկանում են, որ abort()-ից հետո կոդ չի կատարվի, և չեն զգուշացնում «անհասանելի կոդի» մասին։

06Կոնտրոլերներ

src/controllers.php — առայժմ ֆունկցիաներով
<?php
declare(strict_types=1);

use App\Container;

function homePage(Container $app): void
{
    $sort    = $_GET['sort'] ?? 'price_asc';
    $page    = max(1, (int) ($_GET['page'] ?? 1));
    $perPage = 12;

    $products = $app->products()->findAll(null, $sort, $perPage, ($page-1)*$perPage);
    $total    = $app->products()->count();

    echo render('layout', [
        'title'   => 'ՏեխնոՄարկետ — տեխնիկա Երևանում',
        'content' => render('pages/home', compact('products', 'total', 'sort', 'page', 'perPage')),
    ]);
}

function productPage(Container $app, string $slug): void
{
    $product = $app->products()->findBySlug($slug);

    if ($product === null) {
        abort(404, 'Ապրանքը չի գտնվել');      // ⭐
    }

    echo render('layout', [
        'title'   => $product->name . ' — ՏեխնոՄարկետ',
        'content' => render('pages/product', compact('product')),
    ]);
}
404-ը կարևոր է, ոչ միայն գեղեցկության համար

Առանց abort(404)-ի՝ /product/գոյություն-չունեցող-ը կվերադարձներ դատարկ էջ 200 կոդով։

Google-ը դա կինդեքսավորեր որպես վավեր էջ։ Հազարավոր դատարկ էջեր ինդեքսում — դասական SEO խնդիր, որը կոչվում է soft 404։

Իսկ մոնիտորինգի համակարգերը չէին տեսնի, որ ինչ-որ բան կոտրվել է — 200-ը «ամեն ինչ կարգին է» է նշանակում։

07Հասցեների գեներացում

Խնդիրը, որ ի հայտ կգա շուտով

Շաբլոններում գրում ես /product/<?= $p->slug ?>։ Երեսուն տեղում։

Վաղը որոշում ես, որ հասցեն պիտի լինի /products/{slug} (հոգնակի)։ Հիմա պիտի գտնես բոլոր երեսուն տեղը։

Եվ ամենավատը՝ ինչ-որ մեկը կմնա անփոփոխ, և կստացվի կոտրված հղում, որը կհայտնաբերվի ամիսներ անց։

src/helpers.php
function url(string $path, array $query = []): string
{
    $url = '/' . ltrim($path, '/');

    if ($query !== []) {
        $url .= '?' . http_build_query($query);   // ⭐ ինքն է փախցնում
    }

    return $url;
}

function abort(int $code, string $message = ''): never
{
    http_response_code($code);

    echo render('layout', [
        'title'   => (string) $code,
        'content' => render('pages/error', compact('code', 'message')),
    ]);

    exit;
}

function redirect(string $path, int $code = 303): never
{
    header('Location: ' . url($path), true, $code);
    exit;
}

function isCurrent(string $path): bool
{
    $current = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH) ?? '/';

    return '/' . trim($current, '/') === '/' . trim($path, '/');
}

aria-current-ը վերադառնում է

HTML Դաս 18-ում սովորեցինք, որ ընթացիկ էջը նշվում է aria-current="page"-ով, և CSS-ը ոճավորում է այն առանց լրացուցիչ դասի։

Այն ժամանակ այն գրում էինք ձեռքով ամեն էջում։ Հիմա սերվերն ինքն է որոշում։

CSS-ը՝ .menu a[aria-current="page"] { ... } — չի փոխվել ընդհանրապես։

http_build_query-ն ապահով է

Այն ինքն է կոդավորում պարամետրերը՝ բացատները, հայերենը, հատուկ նիշերը։

Ձեռքով գրելը՝ '?q=' . $query — վտանգավոր է․ եթե $query-ում լինի & կամ #, հասցեն կկոտրվի, իսկ չակերտի դեպքում կստանաս XSS ատրիբուտում (Դաս 5)։

Հարցումի ուղին

Բրաուզեր │ GET /product/laptop-pro-14 ▼ Apache + .htaccess │ ֆայլ չկա → public/index.php ▼ bootstrap.php │ .env, session, PDO, Container ▼ Router::dispatch() │ համեմատում է ուղիների հետ ▼ Կոնտրոլեր │ Repository → Product ▼ render() → layout → echo

Կարգավիճակի կոդեր

200 OK 303 See Other → PRG (Դաս 6) 404 Not Found → ուղին չկա 405 Method Not Allowed → ուղին կա, մեթոդը՝ ոչ 500 Server Error → բացառություն

Ստուգաթերթ

  • Վեբ-արմատը public/-ն է
  • /.env-ը հասանելի չէ բրաուզերից
  • Ստատիկ ֆայլերը բացվում են ուղիղ
  • Չգտնված ուղին տալիս է 404, ոչ 200
  • Սխալ մեթոդը տալիս է 405
  • Հասցեները գեներացվում են url()-ով
  • Ուղիների ցուցակը մեկ ֆայլում է
  • Հին էջերի ֆայլերը ջնջված են

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու մեկ մուտք
  • Գիտե՞մ, ինչ է անում առաջին RewriteCond-ը
  • Հասկանո՞ւմ եմ, ինչու public/-ը անվտանգության հարց է
  • Կարո՞ղ եմ բացատրել 404-ի և 405-ի տարբերությունը
  • Գիտե՞մ, ինչու soft 404-ը վնասակար է
Հաջորդ դասին

Controller և View — մինի-ֆրեյմվորքի հավաքում։ Կոնտրոլերները ֆունկցիաներից կդառնան դասեր, կհայտնվեն Request և Response օբյեկտները, իսկ render()-ը՝ View դաս։

Դրանից հետո քո նախագիծը կունենա ճիշտ նույն կառուցվածքը, ինչ Laravel-ը — և Դաս 18-ում կտեսնես, որ ֆրեյմվորքը զարմանք չէ, այլ նույն գաղափարների հասուն տարբերակը։
