01Composer

02Namespace

Խնդիրը, որ լուծում է

Երկու դաս նույն անունով չեն կարող գոյություն ունենալ։

Քո Product-ը և գրադարանի Product-ը → ֆատալ սխալ։ Քո Cart-ը և վճարային համակարգի Cart-ը → նույնը։

Namespace-ը տալիս է դասին լրիվ անուն՝ App\Models\Product, և բախումն անհնար է դառնում։

PSR-4 — ուղին և անվանատարածքը համընկնում են composer.json՝ "App\\" : "src/" src/Models/Product.php → App\Models\Product src/Models/PhysicalProduct.php → App\Models\PhysicalProduct src/Repositories/ ProductRepository.php → App\Repositories\ProductRepository src/Support/Database.php → App\Support\Database

Ամենահաճախ սխալը namespace-ի հետ

Namespace-ի ներսում new PDO()-ն փնտրվում է որպես App\Repositories\PDO — և չի գտնվում։

Երկու ելք՝ use PDO; վերևում, կամ new \PDO() շեղագծով։

Ինչպես է աշխատում

Երբ PHP-ն հանդիպում է անհայտ դասի, կանչում է autoloader-ը։ Composer-ը վերցնում է լրիվ անունը, փոխարինում է App\-ը src/-ով, \-ը՝ /-ով, ավելացնում .php և բեռնում ֆայլը։

04Repository

Ընթացիկ խնդիրը

products.php-ում գլոբալ ֆունկցիաներ են՝ getProducts(), getProduct(), createProduct()։

Ամեն մեկը կանչում է db()։ Ամեն մեկը գլոբալ է։ Փորձարկելը՝ անհնար։ Փոխարինելը՝ նույնպես։

Իսկ գլխավորը՝ SQL-ը ցրված է, և չկա մեկ տեղ, որտեղ նայելով կհասկանաս, ինչպես է նախագիծը աշխատում տվյալների հետ։

Շերտերի բաժանում Կոնտրոլեր → ի՞նչ պիտի կատարվի │ Repository → ինչպե՞ս ստանալ/պահել տվյալները │ PDO → բազա │ Model → ի՞նչ է ապրանքը և ինչ կարող է անել
src/Repositories/ProductRepository.php
<?php
declare(strict_types=1);

namespace App\Repositories;

use App\Models\Product;
use PDO;

class ProductRepository
{
    private const SORT_MAP = [
        'price_asc'  => 'final_price ASC',
        'price_desc' => 'final_price DESC',
        'name'       => 'p.name ASC',
        'newest'     => 'p.created_at DESC',
    ];

    private const SELECT_FIELDS = "
        p.id, p.name, p.slug, p.specs, p.image,
        p.price, p.discount, p.stock, p.weight, p.type,
        c.name AS category_name,
        ROUND(p.price * (100 - p.discount) / 100) AS final_price
    ";

    public function __construct(private PDO $pdo) {}

    // ---------- Read ----------

    /** @return Product[] */
    public function all(
        ?int $categoryId = null,
        string $sort = 'price_asc',
        int $limit = 12,
        int $offset = 0
    ): array {

        $order = self::SORT_MAP[$sort] ?? self::SORT_MAP['price_asc'];

        $where = ['p.is_active = 1'];

        if ($categoryId !== null) {
            $where[] = 'p.category_id = :category';
        }

        $sql = "SELECT " . self::SELECT_FIELDS . "
                FROM products p
                LEFT JOIN categories c ON c.id = p.category_id
                WHERE " . implode(' AND ', $where) . "
                ORDER BY {$order}
                LIMIT :limit OFFSET :offset";

        $stmt = $this->pdo->prepare($sql);

        if ($categoryId !== null) {
            $stmt->bindValue(':category', $categoryId, PDO::PARAM_INT);
        }

        $stmt->bindValue(':limit',  $limit,  PDO::PARAM_INT);
        $stmt->bindValue(':offset', $offset, PDO::PARAM_INT);
        $stmt->execute();

        return array_map(
            fn(array $row): Product => Product::fromRow($row),
            $stmt->fetchAll()
        );
    }

    public function find(int $id): ?Product
    {
        $stmt = $this->pdo->prepare(
            "SELECT " . self::SELECT_FIELDS . "
             FROM products p
             LEFT JOIN categories c ON c.id = p.category_id
             WHERE p.id = :id AND p.is_active = 1"
        );

        $stmt->execute(['id' => $id]);
        $row = $stmt->fetch();

        return $row ? Product::fromRow($row) : null;
    }

    public function findBySlug(string $slug): ?Product
    {
        $stmt = $this->pdo->prepare(
            "SELECT " . self::SELECT_FIELDS . "
             FROM products p
             LEFT JOIN categories c ON c.id = p.category_id
             WHERE p.slug = :slug AND p.is_active = 1"
        );

        $stmt->execute(['slug' => $slug]);
        $row = $stmt->fetch();

        return $row ? Product::fromRow($row) : null;
    }

    /** @param int[] $ids  @return Product[] */
    public function findMany(array $ids): array
    {
        if ($ids === []) {
            return [];
        }

        $in = implode(',', array_fill(0, count($ids), '?'));

        $stmt = $this->pdo->prepare(
            "SELECT " . self::SELECT_FIELDS . "
             FROM products p
             LEFT JOIN categories c ON c.id = p.category_id
             WHERE p.id IN ($in)"
        );

        $stmt->execute(array_values($ids));

        return array_map(
            fn(array $row): Product => Product::fromRow($row),
            $stmt->fetchAll()
        );
    }

    public function count(?int $categoryId = null): int
    {
        $sql = "SELECT COUNT(*) FROM products WHERE is_active = 1";
        $params = [];

        if ($categoryId !== null) {
            $sql .= " AND category_id = :category";
            $params['category'] = $categoryId;
        }

        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);

        return (int) $stmt->fetchColumn();
    }

    // ---------- Write ----------

    public function create(array $data): int
    {
        // ... Դաս 11-ի INSERT-ը
        return (int) $this->pdo->lastInsertId();
    }

    public function update(int $id, array $data): bool
    {
        // ... Դաս 11-ի UPDATE-ը
    }

    public function archive(int $id): bool
    {
        $stmt = $this->pdo->prepare(
            "UPDATE products SET is_active = 0 WHERE id = :id LIMIT 1"
        );

        return $stmt->execute(['id' => $id]);
    }
}
Ի՞նչ ձեռք բերեցինք

1. Ամբողջ SQL-ը մեկ ֆայլում է։ Ուզում ես տեսնել, ինչպես է նախագիծը դիմում ապրանքներին — բացիր ProductRepository-ն։

2. SELECT_FIELDS-ը հաստատուն է։ Նոր դաշտ ավելացնելիս փոխում ես մեկ տեղում, ոչ չորսում։

3. PDO-ն գալիս է կոնստրուկտորով, ոչ գլոբալ db()-ից → կարելի է փոխարինել փորձարկումների համար։

4. Անունները կարճացան՝ $products->all(), $products->find(5) — համատեքստն արդեն օբյեկտում է։

Ինչո՞ւ Repository-ն չի վերադարձնում զանգվածներ

Ամեն մեթոդ վերադարձնում է Product օբյեկտ կամ Product[]։

Դա նշանակում է, որ բազայի կառուցվածքը մնում է Repository-ի ներսում։ Կոնտրոլերը և շաբլոնները չգիտեն, որ դաշտը կոչվում է category_name, ոչ categoryName։

Վաղը կարող ես վերանվանել աղյուսակի դաշտը կամ ամբողջովին փոխել SQL-ը՝ և ոչ մի շաբլոն չի կոտրվի։ Փոխվում է միայն fromRow()-ը։

Կապ Դաս 3-ի հետ

Հիշո՞ւմ ես․ «ֆունկցիաները չեն տեսնում արտաքին փոփոխականները, և դա լավ է — կարդալով ստորագրությունը գիտես, ինչից է կախված»։

Կախվածությունների ներարկումը նույն սկզբունքն է դասերի մակարդակում։ __construct(PDO $pdo)-ն բացահայտ ասում է․ «ինձ բազա է պետք»։

Իսկ db()-ի ներսում կանչելը հենց այն թաքնված գլոբալ վիճակն է, որից Դաս 3-ում խուսափում էինք։

src/Support/Database.php
<?php
declare(strict_types=1);

namespace App\Support;

use PDO;
use PDOException;
use RuntimeException;

class Database
{
    public static function connect(array $config): PDO
    {
        $dsn = sprintf(
            'mysql:host=%s;dbname=%s;charset=%s',
            $config['host'],
            $config['name'],
            $config['charset'] ?? 'utf8mb4'
        );

        try {
            return new PDO($dsn, $config['user'], $config['password'], [
                PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_EMULATE_PREPARES   => false,
            ]);
        } catch (PDOException $e) {
            error_log('DB connection failed: ' . $e->getMessage());

            throw new RuntimeException('Տեխնիկական խնդիր։', 500, $e);
        }
    }
}

Երբ Դաս 19-ում գրես public function index(ProductRepository $products), Laravel-ը ինքը կստեղծի ProductRepository-ն և կփոխանցի PDO-ն դրան։

Դու հիմա ձեռքով անում ես այն, ինչ ֆրեյմվորքը կանի ինքնաբերաբար։ Հենց դրա համար է այս հերթականությունը՝ մեխանիզմը, հետո գործիքը։

Երբ դասը «չի գտնվում»

1. namespace-ը համընկնո՞ւմ է թղթապանակի ուղուն, ներառյալ ռեգիստրը

3. Կանչե՞լ ես composer dump-autoload նոր թղթապանակ ավելացնելուց հետո
4. Կա՞ use ներկառուցված դասերի համար՝ PDO, DateTime

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Composer, namespace, Repository», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր դասարանականի 5 քայլը։
  2. Ցուցադրիր use PDO;-ի բացակայության սխալը։
  3. Փոխիր թղթապանակի ռեգիստրը (models/ փոքրատառով) և տես, աշխատո՞ւմ է Windows-ում։ Բացատրիր, ինչու Linux-ում չի աշխատի։
  4. Ստեղծիր OrderRepository տրանզակցիայով (Դաս 11)։
  5. Տեղափոխիր Validator-ը դասի՝ App\Services\Validator։
  6. Ցուցադրիր, ինչու SELECT_FIELDS հաստատունը օգտակար է՝ ավելացրու նոր դաշտ։
  7. Փոխարինիր PDO-ն Repository-ի կոնստրուկտորում փորձնական օբյեկտով և ցուցադրիր, որ դա հնարավոր է։
  8. Գտիր packagist-ում փաթեթ նկարների չափափոխման համար։ Գնահատիր չորս չափանիշով։ Արժե՞ օգտագործել, թե՞ գրել ինքդ։
  9. Համեմատիր հին և նոր index.php-ը։ Քանի՞ տող, քանի՞ գլոբալ ֆունկցիա։
  10. Խնդրիր AI-ին վերանայել քո կառուցվածքը։ Առաջարկե՞ց այլ բաժանում։ Համաձա՞յն ես։

09Հիշելու թերթիկ

Հրաման / գրառումԻնչ է անում
composer initՍտեղծել composer.json
composer require x/yՏեղադրել փաթեթ
composer installՎերականգնել lock-ից
composer dump-autoloadԹարմացնել autoloader-ը ⭐
namespace App\Models;Առաջին հրահանգը ⭐
use App\Models\Product;Ներմուծում
use PDO;Ներկառուցված դասերի համար ⭐
\PDOԱյլընտրանք՝ շեղագծով

PSR-4

"App\\": "src/" App\Models\Product → src/Models/Product.php App\Repositories\OrderRepository → src/Repositories/OrderRepository.php Ուղին և անվանատարածքը՝ ճշգրիտ համընկնում, ներառյալ ռեգիստրը։

Շերտերը

Model → ի՞նչ է օբյեկտը, ի՞նչ կարող է անել Repository → ինչպե՞ս ստանալ և պահել Service → գործարար տրամաբանություն (Cart) Support → տեխնիկական օգնականներ Controller → կապում է ամեն ինչ

Ստուգաթերթ

  • composer.json-ում կա PSR-4
  • vendor/, .env, config.php — .gitignore-ում
  • composer.lock-ը պահոցում է
  • Կա .env.example
  • Մեկ require ամբողջ նախագծում
  • Ամբողջ SQL-ը Repository-ներում է
  • Կախվածությունները փոխանցվում են կոնստրուկտորով
  • Repository-ները վերադարձնում են օբյեկտներ, ոչ զանգվածներ

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչպես է աշխատում autoload-ը
  • Գիտե՞մ, ինչու use PDO; է պետք
  • Հասկանո՞ւմ եմ, ինչու vendor/-ը չի գնում Git
  • Կարո՞ղ եմ բացատրել, ի՞նչ խնդիր է լուծում Repository-ն
  • Գիտե՞մ, ինչու կախվածությունը փոխանցվում է, ոչ ստեղծվում ներսում
Հաջորդ դասին

Մինի-MVC՝ Router, Controller, View։
