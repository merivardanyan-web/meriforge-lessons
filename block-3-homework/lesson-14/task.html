public function find(int $id): ?Book
{
    $stmt = $this->pdo->prepare('SELECT ' . self::SELECT_FIELDS . ' ' . self::FROM . ' WHERE b.id = :id AND b.is_active = 1');
    $stmt->execute(['id' => $id]);
    $row = $stmt->fetch();

    return $row === false ? null : Book::fromRow($row);
}

04Դասի տնային աշխատանքը

Սրանք Դաս 14-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ավարտիր դասարանականի 5 քայլը։ մինի-նախագծում
  2. Ցուցադրիր use PDO;-ի բացակայության սխալը։ մինի-նախագծում
  3. Փոխիր թղթապանակի ռեգիստրը (models/ փոքրատառով) և տես, աշխատո՞ւմ է Windows-ում։ Բացատրիր, ինչու Linux-ում չի աշխատի։
  4. Ստեղծիր OrderRepository տրանզակցիայով (Դաս 11)։ մինի-նախագծում
  5. Տեղափոխիր Validator-ը դասի՝ App\Services\Validator։ մինի-նախագծում
  6. Ցուցադրիր, ինչու SELECT_FIELDS հաստատունը օգտակար է՝ ավելացրու նոր դաշտ։ մինի-նախագծում
  7. Փոխարինիր PDO-ն Repository-ի կոնստրուկտորում փորձնական օբյեկտով և ցուցադրիր, որ դա հնարավոր է։ մինի-նախագծում
  8. Գտիր packagist-ում փաթեթ նկարների չափափոխման համար։ Գնահատիր չորս չափանիշով։ Արժե՞ օգտագործել, թե՞ գրել ինքդ։
  9. Համեմատիր հին և նոր index.php-ը։ Քանի՞ տող, քանի՞ գլոբալ ֆունկցիա։ մինի-նախագծում
  10. Խնդրիր AI-ին վերանայել քո կառուցվածքը։ Առաջարկե՞ց այլ բաժանում։ Համաձա՞յն ես։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Օրինակում lab-… ֆայլերը փորձնական են․ դրանք հաջորդ դասի օրինակում այլևս չկան։ Գործարկելուց առաջ ստեղծիր բազան example/sql/-ի ֆայլերով և .env-ը (composer install)։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
