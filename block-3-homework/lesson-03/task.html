function filterByStatus(array $books, string $status): array
{
    $result = array_filter($books, fn(array $b): bool => bookStatus($b) === $status);

    return array_values($result);        // ⭐ վերահամարակալում՝ 0, 1, 2...
}

04Դասի տնային աշխատանքը

Սրանք Դաս 03-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Կատարիր դասարանականի 8 քայլը։
  2. Ավելացրու data.php-ում category դաշտ և գրիր filterByCategory()։ մինի-նախագծում
  3. Գրիր findProduct(array $products, int $id): ?array։ Ի՞նչ է վերադարձնում, երբ չի գտնում։ մինի-նախագծում
  4. Ցուցադրիր, ինչ է լինում array_filter-ից հետո առանց array_values-ի։ Օգտագործիր var_dump։ մինի-նախագծում
  5. Համեմատիր json_encode-ի արդյունքը array_values-ով և առանց դրա։
  6. Փորձարկիր strict_types-ը՝ միացրած և անջատած։ Ի՞նչ տարբերություն։ մինի-նախագծում
  7. Ցուցադրիր տեսանելիության տիրույթի խնդիրը և ուղղիր երեք ձևով։
  8. Ջնջիր __DIR__-ը require-ից և կանչիր էջը ենթաթղթապանակից։ Նկարագրիր սխալը։ մինի-նախագծում
  9. Գրիր groupByCategory(), որը վերադարձնում է զանգված, որտեղ բանալին կատեգորիան է։ մինի-նախագծում
  10. Հաշվիր միջին գինը, ամենաթանկը և ամենաէժանը՝ երեք ֆունկցիայով տիպերի հայտարարմամբ։ մինի-նախագծում

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
