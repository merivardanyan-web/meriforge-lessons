02Քայլեր

Քայլ 1 · public/

  • css/ և images/՝ public/-ի մեջ
  • public/index.php՝ միակ PHP ֆայլը, որ բացվում է բրաուզերից
  • Laragon-ում Document Root-ը՝ public
public/index.php
$router = new Router();
require __DIR__ . '/../routes.php';

try {
    $response = $router->dispatch(Request::fromGlobals());
} catch (HttpException $e) {
    $response = Response::html(page('pages/error', ['code' => $e->status, 'message' => $e->getMessage()], (string) $e->status), $e->status);
} catch (Throwable $e) {
    error_log((string) $e);                       // ⭐ մանրամասները՝ լոգում, ոչ էկրանին
    $message = $debug ? $e->getMessage() : 'Տեխնիկական խնդիր։ Փորձիր ավելի ուշ։';
    $response = Response::html(page('pages/error', ['code' => 500, 'message' => $message], '500'), 500);
}

$response->send();

Քայլ 2 · Router

  • get(), post(), dispatch()
  • /book/{slug} → #^/book/([^/]+)$#u, urldecode՝ հայերեն slug-ի համար (տնային 5)
  • Ուղին կա, մեթոդը՝ ոչ → 405 (տնային 7)
src/Http/Router.php — match()
private function match(string $route, string $path): ?array
{
    // {slug}-ը՝ «ցանկացած նիշ, բացի /»։ #-ը՝ սահմանազատիչ, որ /-ը չփախցնենք
    $pattern = '#^' . preg_replace('#\{[a-z_]+\}#', '([^/]+)', $route) . '$#u';

    if (!preg_match($pattern, $path, $matches)) {
        return null;
    }

    array_shift($matches);                         // [0]-ն ամբողջ համընկնումն է

    return array_map('urldecode', $matches);       // հայերեն slug՝ %D5%BD... → ս (տնային 5)
}

04Դասի տնային աշխատանքը

Սրանք Դաս 15-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ավարտիր դասարանականը և հինգ փորձարկումը։ Ուղարկիր /.env-ի պատասխանի սքրինշոթը։ մինի-նախագծում
  2. Ավելացրու /category/{slug} ուղին և ProductController::category() մեթոդը։ մինի-նախագծում
  3. Ավելացրու /search ուղին ?q= պարամետրով։ մինի-նախագծում
  4. Ավելացրու aria-current="page" մենյուում isCurrent()-ով (HTML Դաս 18)։ մինի-նախագծում
  5. Փորձարկիր հայերեն slug՝ /product/նոութբուք։ Աշխատո՞ւմ է urldecode-ը։ մինի-նախագծում
  6. Հաշվիր՝ քանի տող է ProductController-ը։ Կա՞ դրանում SQL կամ HTML։ մինի-նախագծում
  7. Խորացված․ ավելացրու 405-ի ստուգումը Router-ում։ մինի-նախագծում
  8. Խնդրիր AI-ին վերանայել Router-ը անվտանգության տեսանկյունից։ Ի՞նչ գտավ։ Համաձա՞յն ես։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Օրինակում lab-… ֆայլերը փորձնական են․ դրանք հաջորդ դասի օրինակում այլևս չկան։ Գործարկելուց առաջ ստեղծիր բազան example/sql/-ի ֆայլերով և .env-ը (composer install)։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
