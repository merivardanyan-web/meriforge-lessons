02Քայլեր

Քայլ 1 · validate.php

  • validateSuggestion(array $data): array՝ վերադարձնում է ['դաշտ' => 'սխալ']
  • Ժանրը՝ GENRES ցուցակից, in_array(..., true)-ով (տնային 7)
validate.php — հեռախոս և ժանր
// --- Հեռախոս՝ ոչ պարտադիր, բայց եթե կա՝ ճիշտ ձևաչափով (տնային, կետ 5) ---
$phone = trim($data['phone'] ?? '');
if ($phone !== '' && !preg_match('/^\+374\s?\d{2}\s?\d{6}$/', $phone)) {
    $errors['phone'] = 'Հեռախոսը՝ +374 XX XXXXXX ձևաչափով';
}

// --- Գիրք ---
$title = trim($data['title'] ?? '');
if ($title === '') {
    $errors['title'] = 'Նշիր գրքի վերնագիրը';
} elseif (mb_strlen($title) > 120) {
    $errors['title'] = 'Վերնագիրը՝ առավելագույնը 120 նիշ';
}

$author = trim($data['author'] ?? '');
if (mb_strlen($author) > 80) {
    $errors['author'] = 'Հեղինակը՝ առավելագույնը 80 նիշ';     // ⭐ վերին սահման ամեն դաշտում (կետ 9)
}

// --- Ժանր՝ ցուցակից, խիստ համեմատությամբ (կետ 7) ---
if (!in_array($data['genre'] ?? '', GENRES, true)) {
    $errors['genre'] = 'Ընտրիր ժանրը ցուցակից';
}

Քայլ 3 · PRG

suggest.php
if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    $errors = validateSuggestion($_POST);

    if ($errors === []) {
        // Այստեղ կուղարկեինք նամակ Անիին կամ կպահեինք բազայում (Դաս 10–11)
        $_SESSION['flash'] = 'sent';

        header('Location: suggest.php', true, 303);
        exit;                      // ⭐ առանց սրա կոդը կշարունակեր կատարվել
    }

    $old = $_POST;

04Դասի տնային աշխատանքը

Սրանք Դաս 06-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Կատարիր դասարանականի 10 քայլը։ մինի-նախագծում
  2. Ուղարկիր curl-ի հրամանը և պատասխանը՝ սքրինշոթով։
  3. Ցուցադրիր, ինչ է լինում, երբ exit-ը բացակայում է header()-ից հետո։ մինի-նախագծում
  4. Դիտմամբ դիր բացատ <?php-ից առաջ և բացատրիր սխալը։
  5. Ավելացրու հեռախոսի դաշտ preg_match-ով վալիդացիայով։ մինի-նախագծում
  6. Ավելացրու checkbox՝ «Համաձայն եմ պայմաններին»։ Ինչպե՞ս ես ստուգում այն PHP-ում։ մինի-նախագծում
  7. Ցուցադրիր, ինչու in_array-ի երրորդ արգումենտը կարևոր է։ Բեր օրինակ, որտեղ առանց դրա ստուգումն անցնում է սխալմամբ։ մինի-նախագծում
  8. Ցուցադրիր filter_var-ի 0-ի թակարդը՝ !-ով և === false-ով։ մինի-նախագծում
  9. Ավելացրու ամեն դաշտի վերին սահման։ Փորձարկիր 5000 նիշանոց տեքստով։ մինի-նախագծում
  10. Ստուգիր ֆորման միայն ստեղնաշարով և էկրանընթերցիչով (NVDA կամ VoiceOver)։ Սխալները կարդացվո՞ւմ են։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
