app/Http/Controllers/BookController.php — shelf()
private function shelf(Request $request, ?object $genre)
{
    $sort = $request->query('sort');
    $sort = is_string($sort) && isset(self::SORTS[$sort]) ? $sort : 'progress';
    [, $column, $direction] = self::SORTS[$sort];

    $books = $this->query()
        ->when($genre, fn ($q) => $q->where('books.genre_id', $genre->id))
        ->orderBy($column, $direction)
        ->orderBy('books.id')
        ->paginate(6)
        ->withQueryString();                         // ?sort=-ը մնում է երկրորդ էջում

    return view('books.index', [
        'books'  => $books,
        'genre'  => $genre,
        'genres' => DB::table('genres')->orderBy('name')->get(),
        'sort'   => $sort,
        'sorts'  => self::SORTS,
    ]);
}

04Դասի տնային աշխատանքը

Սրանք Դաս 18-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ավարտիր դասարանականի 5 քայլը և չորս փորձարկումը։ Ուղարկիր սքրինշոթներ։ մինի-նախագծում
  2. Բացիր public/index.php-ը և գրիր, թե ամեն տողը քո Դաս 15-ի front controller-ի որ տողին է համապատասխանում։
  3. Ավելացրու /category/{slug} ուղին անունով և կոնտրոլերի մեթոդով։ մինի-նախագծում
  4. Փոխիր product.show-ի ուղին /products/{slug}-ի։ Քանի՞ ֆայլ պիտի փոխեիր։ Համեմատիր Դաս 15-ի հետ։
  5. Հանիր SORTS սպիտակ ցուցակը և օգտագործիր orderBy($request->query('sort'))։ Ի՞նչ է լինում ?sort=xyz-ով։ Վերադարձրու ցուցակը։ մինի-նախագծում
  6. Ցուցադրիր {{ }}-ի և {!! !!}-ի տարբերությունը վտանգավոր անվանով ապրանքի վրա։ մինի-նախագծում
  7. Խնդրիր AI-ին գրել Laravel-ի կոնտրոլեր։ Ստուգիր՝ օգտագործո՞ւմ է հին կառուցվածքը (Kernel.php)։ Կա՞ orderBy առանց սպիտակ ցուցակի։
