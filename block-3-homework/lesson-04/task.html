functions.php — progress()-ը ուղղումից հետո
/**
 * Դաս 04-ի վերանայումից հետո․
 *  - pages = 0 → DivisionByZeroError → վերադարձնում ենք 0
 *  - read > pages (տառասխալ ադմինում) → 120% → սահմանափակում ենք 100-ով
 *  - read < 0 → -10% → 0
 */
function progress(array $book): int
{
    if ($book['pages'] <= 0) {
        return 0;
    }

    $percent = (int) ($book['read'] * 100 / $book['pages']);

    return max(0, min(100, $percent));
}

04Դասի տնային աշխատանքը

Սրանք Դաս 04-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ստեղծիր քո context.txt-ը՝ նախագծի նկարագրություն, ստեկ, կանոններ, CSS դասեր, արգելքներ։ մինի-նախագծում
  2. Փորձարկիր այն երեք տարբեր առաջադրանքի վրա։ Բարելավիր ըստ արդյունքների։ մինի-նախագծում
  3. Կատարիր երեք դասարանական վարժանքը և գրիր արդյունքները։ մինի-նախագծում
  4. Խնդրիր բացատրել PDO-ն։ Ապա ստուգիր բացատրությունը php.net-ում։ Կա՞ր անճշտություն։
  5. Խնդրիր գեներացնել կոդ, որը կապվում է բազային։ Մի՛ օգտագործիր այն։ Փոխարենը՝ գտիր 3 խնդիր դրանում։
  6. Վերցրու քո functions.php-ը և անցիր վերանայման ռեժիմով։ Ուղղիր ամեն խնդիր ինքդ։ մինի-նախագծում
  7. Կատարիր ինքնաստուգումը՝ փակիր ամեն ինչ և գրիր sortProducts()-ը զրոյից։
  8. Գրիր ցուցակ՝ ի՞նչ 5 բան երբեք չես դնի չաթում։ մինի-նախագծում
  9. Գտիր օրինակ, երբ AI-ն քեզ սխալ պատասխան տվեց։ Նկարագրիր, ինչպես նկատեցիր։ մինի-նախագծում
  10. Գրիր կարճ տեքստ՝ «Ինչպե՞ս եմ օգտագործելու AI-ն մնացած 20 դասում»։ Դա քո աշխատանքային պայմանագիրն է ինքդ քեզ հետ։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
