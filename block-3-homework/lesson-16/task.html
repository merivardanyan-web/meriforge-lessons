src/Services/Auth.php — attempt()
public function attempt(string $email, string $password): bool
{
    $user = $this->users->findByEmail($email);

    // ⭐ Ստուգում ենք ՄԻՇՏ, նույնիսկ երբ անդամը չկա․ թե չէ պատասխանի արագությունը
    // կհուշեր, որ այդ հասցեն գոյություն չունի (timing attack, տնային 5)
    $hash = $user?->passwordHash ?? '$2y$10$' . str_repeat('x', 53);

    if (!password_verify($password, $hash) || $user === null) {
        return false;
    }

    if (password_needs_rehash($hash, PASSWORD_DEFAULT)) {
        $this->users->updatePassword($user->id, password_hash($password, PASSWORD_DEFAULT));
    }

    $this->login($user);

    return true;
}

03Ստուգաթերթ

  • Սխալ գաղտնաբառ և անհայտ հասցե՝ նույն հաղորդագրությունը
  • Մուտքից հետո PHPSESSID-ը փոխվեց
  • Արամը /admin/books-ում ստանում է 403, Լիլիթը չի տեսնում «Թաքցնել»
  • POST առանց _token-ի → 419
  • Գրանցված անդամի դերը՝ member, ինչ էլ ուղարկվի ֆորմայով

Հաճախ հանդիպող սխալներ

ՍխալԻնչպես ուղղել
Ամեն ֆորմա տալիս է 419Ֆորմայում չկա <?= csrf_field() ?>, կամ սեսիան չի պահվում
Մուտքից հետո դեռ «Մուտք» է գրվածHeader-ը կարդում է auth()-ը, իսկ login()-ը չի գրել $_SESSION['user_id']
password_verify-ը միշտ false էՍյունակը VARCHAR(60)-ից կարճ է, և հեշը կտրվել է․ VARCHAR(255)
Editor-ը կարողանում է թաքցնել curl-ովԿոճակը թաքցնելը պաշտպանություն չէ․ ստուգումը՝ կոնտրոլերում

04Դասի տնային աշխատանքը

Սրանք Դաս 16-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ավարտիր դասարանականի 6 քայլը և յոթ փորձարկումը։ մինի-նախագծում
  2. Ուղարկիր սքրինշոթներ CSRF հարձակման՝ պաշտպանությամբ և առանց դրա։ մինի-նախագծում
  3. Ցուցադրիր, որ նույն գաղտնաբառը տալիս է տարբեր հեշեր։ Բացատրիր, ինչու ===-ը այստեղ չի աշխատի։ մինի-նախագծում
  4. Չափիր password_hash()-ի ժամանակը microtime()-ով։ Համեմատիր md5()-ի հետ։ մինի-նախագծում
  5. Խորացված․ ցուցադրիր timing attack-ի հնարավորությունը՝ չափիր մուտքի ժամանակը գոյություն ունեցող և չգոյություն ունեցող հասցեով, կեղծ հեշի ստուգումով և առանց դրա։
  6. Ցուցադրիր session_regenerate_id()-ի աշխատանքը՝ համեմատիր PHPSESSID-ն մուտքից առաջ և հետո։ մինի-նախագծում
  7. Ավելացրու երրորդ ստուգում․ /admin/delete-ը թույլատրված է միայն role = 'admin'-ին ($auth->isAdmin()), մյուսներին՝ 403։ մինի-նախագծում
  8. Ավելացրու ադմին էջ /manage/orders հասցեով (առանց /admin-ի)։ Պաշտպանվա՞ծ է։ Ինչ կանոն պետք է պահել, որ սա չպատահի։ մինի-նախագծում
  9. Ավելացրու «Բարև, [անուն]» և ելքի կոճակ header-ում։ մինի-նախագծում
  10. DevTools-ով ավելացրու գրանցման ֆորմային <input name="role" value="admin"> և ուղարկիր։ HeidiSQL-ում ստուգիր նոր օգտագործողի դերը։ Բացատրիր, ինչու այն չփոխվեց։ մինի-նախագծում
  11. Ավելացրու գրանցմանը փորձերի սահմանափակում LoginThrottle-ի օրինակով՝ առավելագույնը 3 նոր հաշիվ մեկ սեսիայից ժամում։ Ինչու՞ է սա ուսումնական, ոչ իրական պաշտպանություն (բաժին 05)։ մինի-նախագծում
  12. Խնդրիր AI-ին վերանայել Auth-ը։ Ի՞նչ գտավ։ Համաձա՞յն ես։
