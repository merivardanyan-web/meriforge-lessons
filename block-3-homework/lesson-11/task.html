books.php — logReading()
function logReading(int $bookId, int $pages, string $date, ?string $note): void
{
    $pdo = db();
    $pdo->beginTransaction();

    try {
        $stmt = $pdo->prepare('
            INSERT INTO reading_sessions (book_id, pages, read_on, note)
            VALUES (:book, :pages, :read_on, :note)
        ');
        $stmt->execute(['book' => $bookId, 'pages' => $pages, 'read_on' => $date, 'note' => $note]);

        // Պայմանը WHERE-ում․ եթե մեկ ուրիշ ներդիրից արդեն ավելացրել ես էջեր, չենք անցնի pages-ից
        $stmt = $pdo->prepare('
            UPDATE books
            SET read_pages = read_pages + :pages
            WHERE id = :id AND read_pages + :pages2 <= pages
        ');
        $stmt->execute(['pages' => $pages, 'pages2' => $pages, 'id' => $bookId]);

        if ($stmt->rowCount() === 0) {
            throw new RuntimeException('Էջերը շատ են․ գիրքն այդքան էջ չունի');
        }

        $pdo->commit();
    } catch (Throwable $e) {
        $pdo->rollBack();          // INSERT-ը նույնպես չեղարկվում է
        throw $e;                  // կանչողը կորոշի՝ ինչ ցույց տալ
    }
}

04Դասի տնային աշխատանքը

Սրանք Դաս 11-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Կատարիր դասարանականի 10 քայլը և չորս փորձարկումը։ մինի-նախագծում
  2. Ավելացրու նկարի վերբեռնում ադմին ֆորմայում (Դաս 7-ի uploadImage())։ մինի-նախագծում
  3. Իրագործիր createOrder()-ը տրանզակցիայով։ մինի-նախագծում
  4. Փորձարկիր տրանզակցիան․ դիտմամբ գցիր բացառություն երկրորդ INSERT-ից հետո։ Պատվերը մնա՞ց բազայում։ մինի-նախագծում
  5. Փոխիր աղյուսակի շարժիչը MyISAM-ի և կրկնիր փորձը։ Ի՞նչ փոխվեց։ մինի-նախագծում
  6. Ցուցադրիր UPDATE-ը առանց WHERE-ի փորձնական աղյուսակի վրա։ մինի-նախագծում
  7. Ցուցադրիր rowCount()-ի թակարդը՝ պահպանիր առանց փոփոխության։ մինի-նախագծում
  8. Ավելացրու ապրանքների որոնում ադմին ցուցակում։ մինի-նախագծում
  9. Ավելացրու զանգվածային գործողություն՝ նշել մի քանիսը և արխիվացնել միանգամից։ մինի-նախագծում
  10. Խնդրիր AI-ին վերանայել CRUD ֆունկցիաները։ Ի՞նչ եզրային դեպքեր գտավ, որ բաց էիր թողել։

05Համեմատիր օրինակի հետ

Նախ ստուգաթերթը, հետո օրինակը

Երբ բաժին 03-ի բոլոր կետերը նշված են, բացիր օրինակը։ Օրինակում lab-… ֆայլերը փորձնական են․ դրանք հաջորդ դասի օրինակում այլևս չկան։ Գործարկելուց առաջ ստեղծիր բազան example/sql/-ի ֆայլերով և config.php-ը։ Գտիր մեկ բան, որ դու արել ես այլ կերպ, և մտածիր՝ որն է ավելի հասկանալի և ինչու։ Քո լուծումը կարող է տարբերվել և լինել ճիշտ։
