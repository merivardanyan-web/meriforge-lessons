functions.php — safeUrl()
/** Միայն http, https, mailto, tel և ներքին ուղիներ */
function safeUrl(?string $url, string $fallback = '#'): string
{
    $url = trim($url ?? '');

    if ($url === '') {
        return $fallback;
    }

    if (str_starts_with($url, '/') && !str_starts_with($url, '//')) {
        return $url;
    }

    $scheme = strtolower((string) parse_url($url, PHP_URL_SCHEME));

    return in_array($scheme, ['http', 'https', 'mailto', 'tel'], true) ? $url : $fallback;
}

04Դասի տնային աշխատանքը

Սրանք Դաս 05-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Կատարիր դասարանականի 8 քայլը։
  2. Ցուցադրիր XSS-ը երկու սքրինշոթով՝ առանց e()-ի և դրանով։ մինի-նախագծում
  3. Փորձարկիր հինգ տարբեր payload՝ <script>, <img onerror>, <svg onload>, javascript: հղումում, ' onfocus=' ատրիբուտում։ Որո՞նք են աշխատում առանց e()-ի։ մինի-նախագծում
  4. Ցուցադրիր ատրիբուտի խնդիրը՝ չակերտներով և առանց դրանց, e()-ի հետ միասին։ մինի-նախագծում
  5. Ստեղծիր երրորդ էջ՝ about.php, նույն layout-ով։ Քանի՞ տող կրկնեցիր՝ համեմատած Դաս 3-ի partials/ մոտեցման հետ։ մինի-նախագծում
  6. Ավելացրու $title-ը ամեն էջի համար։ մինի-նախագծում
  7. Բացատրիր սեփական բառերով, ինչու փախցնում ենք ելքի, ոչ մուտքի ժամանակ։
  8. Ցուցադրիր, ինչ է լինում, երբ e()-ն կիրառում ես երկու անգամ նույն տողի վրա։ մինի-նախագծում
  9. Օգտագործիր json_encode-ը <script>-ի ներսում հայերեն տեքստի հետ։ Համեմատիր JSON_UNESCAPED_UNICODE-ով և առանց դրա։
  10. Խնդրիր AI-ին վերանայել քո շաբլոնները և գտնել բաց թողնված e()-երը։ Ուղղիր ինքդ։
