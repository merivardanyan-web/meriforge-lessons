/** Ավելացվել է վերջին 30 օրում */
public function isNew(): bool
{
    // strtotime()՝ «2026-10-14 10:00:00» → վայրկյաններ 1970-ից, ինչպես time()-ը
    return strtotime($this->createdAt) > time() - 30 * 24 * 60 * 60;
}

/** Երկար վերնագիրը՝ 40 նիշից կտրած, քարտի համար */
public function shortTitle(int $max = 40): string
{
    // mb_substr()՝ substr()-ի հայերեն-անվտանգ տարբերակը, ինչպես mb_strlen()-ը (Դաս 02)
    return mb_strlen($this->title) > $max ? mb_substr($this->title, 0, $max - 1) . '…' : $this->title;
}

classes/ReadingList.php — setPerDay()
/** Տնային 7 (Cart::setQty-ի նման)․ օրական 1–200 էջ, միայն ցուցակում եղած գրքի համար */
public function setPerDay(int $id, int $pages): bool
{
    if (!$this->has($id) || $pages < 1 || $pages > 200) {
        return false;
    }

    $this->storage[$id] = $pages;

    return true;
}

04Դասի տնային աշխատանքը

Սրանք Դաս 12-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Կատարիր դասարանականի 10 քայլը։ մինի-նախագծում
  2. Ավելացրու Product-ին երեք նոր մեթոդ՝ discountLabel(), isNew() (30 օրից պակաս), shortName() (40 նիշից կտրած)։ մինի-նախագծում
  3. Ստեղծիր Category դասը fromRow()-ով։ մինի-նախագծում
  4. Ցուցադրիր տարբերությունը՝ $p['prise'] և $p->prise։ Ի՞նչ է տալիս ամեն մեկը։ մինի-նախագծում
  5. Ջնջիր readonly-ն և ցուցադրիր, ինչու է այն օգտակար՝ փոխելով գինը շաբլոնում։ մինի-նախագծում
  6. Ցուցադրիր, ինչ կլինի, եթե Cart-ի կոնստրուկտորից հանես &-ը։ մինի-նախագծում
  7. Ավելացրու Cart::setQty() մեթոդ վալիդացիայով։ մինի-նախագծում
  8. Համեմատիր հին cart-functions.php-ը և նոր Cart դասը։ Քանի՞ տող, քանի՞ գլոբալ ֆունկցիա։
  9. Գրիր CartItem դաս, որը պահում է Product-ը և քանակը, և ունի sum() մեթոդ։ մինի-նախագծում
  10. Խնդրիր AI-ին վերանայել Product-ը։ Ի՞նչ մեթոդ առաջարկեց, որ չէիր մտածել։
