src/Controllers/Api/BookApiController.php — index()
public function index(Request $request): Response
{
    $filters = BookRepository::filtersFrom($request->allQuery());
    $sort    = $request->query('sort');
    $sort    = isset(BookRepository::SORTS[$sort]) ? $sort : 'progress';
    $page    = max(1, $request->int('page', 1));                                   // ?page=abc → 1 (տնային 9)
    $perPage = min(self::MAX_PER_PAGE, max(1, $request->int('per_page', 6)));      // ?per_page=-5 → 1, 1000 → 50

    $total = $this->books->count($filters);
    $found = $this->books->search($filters, $sort, $perPage, ($page - 1) * $perPage);

    return Response::json([
        // array_map-ը պահում է 0, 1, 2 բանալիները → JSON զանգված [...]։
        // array_filter-ից հետո բանալիները կարող են լինել 0, 2 → JSON օբյեկտ {"0":..,"2":..} (տնային 6)
        'data' => array_map(fn(Book $b): array => $b->toArray(), $found),
        'meta' => [
            'page'      => $page,
            'per_page'  => $perPage,
            'total'     => $total,
            'last_page' => max(1, (int) (($total + $perPage - 1) / $perPage)),
        ],
    ]);
}

public/js/api.js — request()
async function request(url, options = {}) {
    const response = await fetch(url, {
        ...options,
        headers: {
            "Accept": "application/json",
            "X-CSRF-Token": csrf,                 // ⭐ առանց սրա POST-ը և DELETE-ը կստանան 419
            ...(options.body ? { "Content-Type": "application/json" } : {}),
        },
    });

    const payload = await response.json();

    // fetch-ը 4xx/5xx-ի դեպքում ՉԻ գցում սխալ․ ok-ը ստուգում ենք ինքներս
    if (!response.ok) {
        throw new ApiError(payload.error?.message ?? "Սխալ", response.status);
    }

    return payload;
}

04Դասի տնային աշխատանքը

Սրանք Դաս 17-ի տնային աշխատանքի կետերն են։ «մինի-նախագծում» նշվածները արդեն կատարել ես վերևի քայլերում՝ Անիի ակումբում։ Մնացածը կատարիր ՏեխնոՄարկետում, ինչպես գրված է դասում։

  1. Ավարտիր դասարանականի 5 քայլը և վեց curl փորձարկումը։ Ուղարկիր սքրինշոթներ։ մինի-նախագծում
  2. Ավելացրու GET /api/cart՝ վերադարձնում է զամբյուղի տողերը գումարով։ Օգտագործիր findMany-ը (Դաս 14)։ մինի-նախագծում
  3. Ավելացրու DELETE /api/cart/items/{id} և ստուգիր, որ առանց token-ի 419 է։ մինի-նախագծում
  4. Ցուցադրիր $_POST-ի դատարկության թակարդը՝ var_dump($_POST) JSON հարցումի ժամանակ։ մինի-նախագծում
  5. Ցուցադրիր, ինչ կլինի, եթե toArray()-ի փոխարեն օգտագործես json_encode($product)։ Ավելացրու public $costPrice և տես։ մինի-նախագծում
  6. Փոխիր array_map-ը array_filter-ով և ցուցադրիր, ինչպես է data-ն դառնում օբյեկտ։ մինի-նախագծում
  7. Խորացված․ գրիր /api/products?q=... որոնում և JS ֆայլ՝ debounce-ով, որը թարմացնում է ցուցակը ինչպես JS բլոկի որոնման դասում։ մինի-նախագծում
  8. Անջատիր JS-ը DevTools-ում և ստուգիր, որ «Գնել» կոճակը դեռ աշխատում է։ մինի-նախագծում
  9. Փորձիր ?per_page=-5 և ?page=abc։ Ի՞նչ է վերադարձնում API-ն։ մինի-նախագծում
  10. Խնդրիր AI-ին վերանայել API-ը անվտանգության տեսանկյունից (Դաս 4-ի «վերանայման» ռեժիմով, առանց վերաշարադրելու)։ Ի՞նչ գտավ։
