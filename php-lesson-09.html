PDO և SQL injection

Այսօր միացնում ենք PHP-ն բազային։ Եվ սովորում ենք ամենակարևոր անվտանգության դասը PHP-ում՝ ինչու տողերի միացումով հարցում կառուցելը վտանգավոր է։

01Հարձակումը

❌ երբեք այսպես
$id = $_GET['id'];

$sql = "SELECT * FROM products WHERE id = $id";
Նորմալ հարցում՝ ?id=5 SELECT * FROM products WHERE id = 5 Հարձակում՝ ?id=5 OR 1=1 SELECT * FROM products WHERE id = 5 OR 1=1 ───── միշտ ճիշտ է → վերադարձնում է ԲՈԼՈՐ ապրանքները Ավելի վատ՝ ?id=5; DROP TABLE products-- SELECT * FROM products WHERE id = 5; DROP TABLE products-- ────────────────── երկրորդ հրաման
Ինչո՞ւ է սա տեղի ունենում

Բազան ստանում է մեկ տող և ինքն է որոշում, թե որտեղ է հրամանը, որտեղ՝ տվյալը։

Երբ դու միացնում ես տողերը, օգտագործողի տեքստը դառնում է հրամանի մաս։ Բազան չգիտի, որ OR 1=1-ը եկել է դրսից — նրա համար դա քո գրած հարցումն է։

Ուղիղ նույն տրամաբանությունը, ինչ XSS-ում (Դաս 5)․ տվյալը խառնվում է կոդի հետ։ Տարբերությունը միայն այն է, թե ով է կատարում՝ բրաուզերը կամ բազան։

Ի՞նչ կարող է անել հարձակվողը
Կարդալ բոլոր տվյալները՝ օգտագործողներ, պատվերներ, գաղտնաբառերի հեշեր
Փոխել գները և պատվերները
Ջնջել աղյուսակները
Մուտք գործել առանց գաղտնաբառի (Դաս 16)
Որոշ կոնֆիգուրացիաներում՝ կարդալ ֆայլեր սերվերից
դասական մուտքի շրջանցում
// Խոցելի կոդ
$sql = "SELECT * FROM users
        WHERE email = '$email' AND password = '$pass'";

// Հարձակվողը մուտքագրում է էլ. հասցեի դաշտում՝
//   ' OR '1'='1'--
// Ստացվում է՝

SELECT * FROM users
WHERE email = '' OR '1'='1'--' AND password = '...

// -- -ը մեկնաբանություն է SQL-ում։
// Գաղտնաբառի ստուգումն ընդհանրապես չի կատարվում։
Փախցնելը լուծում չէ

Կա գայթակղություն՝ «պարզապես մաքրեմ չակերտները»։ addslashes, str_replace, ռեգուլյար արտահայտություններ — բոլորն էլ շրջանցելի են։

Բազմաբայթ կոդավորումներ, թվային համատեքստեր, ներդրված հարցումներ — միշտ գտնվում է դեպք, որ բաց ես թողել։

Միակ հուսալի լուծումը՝ տվյալը երբեք չդառնա հարցման տեքստի մաս։ Հենց դա են անում prepared statements-ը։

02Prepared statements

Երկու առանձին ուղարկում 1. Հարցման ԿԱՌՈՒՑՎԱԾՔԸ SELECT * FROM products WHERE id = ? → բազան վերլուծում է և հիշում պլանը → այստեղ արդեն որոշված է, որ ? -ը ՏՎՅԱԼ է 2. ՏՎՅԱԼՆԵՐԸ [5] կամ ["5 OR 1=1"] → գնում են առանձին ալիքով → բազան դրանք ՉԻ վերլուծում որպես SQL Արդյունքում "5 OR 1=1"-ը փնտրվում է որպես id-ի ԱՐԺԵՔ, և ոչինչ չի գտնվում։
✅ ճիշտ ձևը
// Դիրքային պարամետրեր
$stmt = $pdo->prepare("SELECT id, name, price FROM products WHERE id = ?");
$stmt->execute([$id]);

// Անվանված պարամետրեր — ընթեռնելի է ⭐
$stmt = $pdo->prepare(
    "SELECT id, name, price
     FROM products
     WHERE is_active = 1
       AND price BETWEEN :min AND :max
     ORDER BY price
     LIMIT :limit"
);

$stmt->execute([
    'min'   => $min,
    'max'   => $max,
    'limit' => $limit,
]);

$products = $stmt->fetchAll();
Ուշադրություն՝ ոչ մի չակերտ

WHERE name = :name — առանց չակերտների, նույնիսկ տողերի համար։

Չակերտները ավելացնում է PDO-ն։ Եթե գրես ':name', դա կդառնա բառացի տեքստ :name, ոչ պարամետր։

Սա հաճախակի սխալ է առաջին օրերին։

db.php
<?php
declare(strict_types=1);

function db(): PDO
{
    static $pdo = null;      // մեկ միացում մեկ հարցումի ընթացքում ⭐

    if ($pdo !== null) {
        return $pdo;
    }

    $config = require __DIR__ . '/config.php';
    $c = $config['db'];

    $dsn = "mysql:host={$c['host']};dbname={$c['name']};charset={$c['charset']}";

    try {
        $pdo = new PDO($dsn, $c['user'], $c['password'], [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ]);
    } catch (PDOException $e) {
        // ⚠️ Օգտագործողին երբեք չցուցադրել $e->getMessage()
        error_log('DB connection failed: ' . $e->getMessage());
        http_response_code(500);
        exit('Տեխնիկական խնդիր։ Փորձեք ավելի ուշ։');
    }

    return $pdo;
}
ԿարգավորումԻնչու
ERRMODE_EXCEPTIONՍխալները գցում են բացառություն, ոչ լուռ false ⭐
FETCH_ASSOCԱսոցիատիվ զանգված, ոչ կրկնակի
EMULATE_PREPARES = falseԻրական prepared statements ⭐
charset=utf8mb4 DSN-ումՀայերենի համար պարտադիր
EMULATE_PREPARES-ը կարևոր է

Լռելյայն PDO-ն ձևացնում է․ ինքն է փախցնում պարամետրերը և ուղարկում մեկ տող։ Դա արագ է, բայց դա արդեն ոչ թե երկու ալիք է, այլ փախցնում — իսկ փախցնումը շրջանցելի է որոշ կոդավորումներում։

false-ով միացնում ենք իրական երկքայլ մեխանիզմը՝ բազան ինքն է վերլուծում կառուցվածքը։

Կողմնակի օգուտ՝ LIMIT :limit-ը սկսում է աշխատել ճիշտ որպես թիվ, ոչ տող։

Սխալի հաղորդագրությունը օգտագործողին

$e->getMessage()-ը պարունակում է հոսթը, բազայի անունը, երբեմն՝ օգտագործողի անունը։ Դա նվեր է հարձակվողի համար։

Կանոն՝ մանրամասները՝ լոգում, օգտագործողին՝ ընդհանուր հաղորդագրություն։ Նույն սկզբունքը, ինչ Դաս 1-ի display_errors-ի դեպքում։

04Արդյունքի ընթերցում

չորս ձև
// Բոլոր տողերը — զանգված ասոցիատիվ զանգվածներից
$products = $stmt->fetchAll();
// [ ['id'=>1, 'name'=>'...'], ['id'=>2, ...] ]

// Մեկ տող կամ false
$product = $stmt->fetch();

// Մեկ արժեք
$count = (int) $stmt->fetchColumn();

// Ցիկլով — մեծ արդյունքների համար, հիշողությունը չի լցվում
while ($row = $stmt->fetch()) {
    // ...
}
Ահա այն, ինչ խոստացել էի Դաս 3-ում

fetchAll()-ը վերադարձնում է ճիշտ նույն կառուցվածքը, ինչ data.php-ի $products-ը՝ զանգված ասոցիատիվ զանգվածներից։

Դա նշանակում է, որ քո foreach-ը շաբլոնում ընդհանրապես չի փոխվի։ Փոխվում է միայն այն, թե որտեղից են գալիս տվյալները։

Հենց դրա համար Դաս 3-ում զանգվածները գրում էինք ձեռքով այդ ձևով։

Տիպերի թակարդ

MySQL-ից DECIMAL-ը գալիս է որպես տող՝ "450000.00", ոչ թիվ։

strict_types-ով finalPrice(int $price)-ը կգցի TypeError։

Երկու լուծում՝ բացահայտ փոխարկում կոդում՝ (int) $row['price'], կամ SQL-ում՝ CAST(price AS UNSIGNED) AS price։

Առաջինը նախընտրելի է — ավելի պարզ է և տեսանելի։

05Ի՞նչ չի կարելի պարամետրացնել

Կարևոր սահմանափակում

Պարամետրերը փոխարինում են միայն արժեքները։ Աղյուսակների անունները, դաշտերի անունները, ASC/DESC-ը՝ ոչ։

ORDER BY :field — չի աշխատի։ Բազան կտեսնի ORDER BY 'price'՝ տողային հաստատունով, և տեսակավորում չի լինի։

❌ խոցելի
$sort = $_GET['sort'];
$sql = "SELECT * FROM products ORDER BY $sort";

// ?sort=price; DROP TABLE products--
✅ սպիտակ ցուցակ — Դաս 6-ի օրինաչափությունը
$sortMap = [
    'price_asc'  => 'price ASC',
    'price_desc' => 'price DESC',
    'name'       => 'name ASC',
    'newest'     => 'created_at DESC',
];

$key = $_GET['sort'] ?? 'price_asc';
$order = $sortMap[$key] ?? $sortMap['price_asc'];

$sql = "SELECT id, name, price FROM products
        WHERE is_active = 1
        ORDER BY {$order}";
Ինչո՞ւ սա անվտանգ է

Հարցման մեջ ընկնում է միայն այն, ինչ դու ինքդ ես գրել զանգվածում։ Օգտագործողի տեքստը օգտագործվում է որպես բանալի, ոչ որպես արժեք։

Անհայտ բանալին ինքնաբերաբար դառնում է լռելյայն արժեք։ Ոչ մի ստուգում չի կարող բաց թողնվել։

Հիշո՞ւմ ես Դաս 6-ի in_array($x, $allowed, true) օրինաչափությունը։ Ահա այն նույն գաղափարը՝ ավելի հզոր տեսքով։

IN (...) — դինամիկ քանակ պարամետրերի
$ids = [1, 5, 12];

// Գեներացնում ենք ? ? ? ըստ քանակի
$in = implode(',', array_fill(0, count($ids), '?'));

$stmt = db()->prepare("SELECT id, name FROM products WHERE id IN ($in)");
$stmt->execute($ids);
Սա բացառություն չէ

Հարցման մեջ ընկնում է միայն ?,?,? — ոչ մի օգտագործողի տվյալ։ Քանակը գալիս է count()-ից, որը միշտ թիվ է։

Իսկ արժեքներն անցնում են սովորական ձևով execute()-ով։ Կանոնը պահպանված է։

06Հարցումների ֆայլ

data.php-ը փոխարինում ենք products.php-ով՝ նույն ինտերֆեյսով։

products.php
<?php
declare(strict_types=1);

const SORT_MAP = [
    'price_asc'  => 'final_price ASC',
    'price_desc' => 'final_price DESC',
    'name'       => 'p.name ASC',
    'newest'     => 'p.created_at DESC',
];

function getProducts(
    ?int $categoryId = null,
    string $sort = 'price_asc',
    int $limit = 12,
    int $offset = 0
): array {

    $order = SORT_MAP[$sort] ?? SORT_MAP['price_asc'];

    $where  = ['p.is_active = 1'];
    $params = [];

    if ($categoryId !== null) {
        $where[] = 'p.category_id = :category';
        $params['category'] = $categoryId;
    }

    $sql = "
        SELECT
            p.id, p.name, p.slug, p.specs,
            p.price, p.discount, p.image, p.stock,
            c.name AS category_name,
            ROUND(p.price * (100 - p.discount) / 100) AS final_price
        FROM products p
        LEFT JOIN categories c ON c.id = p.category_id
        WHERE " . implode(' AND ', $where) . "
        ORDER BY {$order}
        LIMIT :limit OFFSET :offset
    ";

    $stmt = db()->prepare($sql);

    foreach ($params as $k => $v) {
        $stmt->bindValue(":{$k}", $v, PDO::PARAM_INT);
    }

    $stmt->bindValue(':limit',  $limit,  PDO::PARAM_INT);
    $stmt->bindValue(':offset', $offset, PDO::PARAM_INT);

    $stmt->execute();

    return $stmt->fetchAll();
}

function getProduct(int $id): ?array
{
    $stmt = db()->prepare("
        SELECT p.*, c.name AS category_name,
               ROUND(p.price * (100 - p.discount) / 100) AS final_price
        FROM products p
        LEFT JOIN categories c ON c.id = p.category_id
        WHERE p.id = :id AND p.is_active = 1
    ");

    $stmt->execute(['id' => $id]);

    return $stmt->fetch() ?: null;
}

function countProducts(?int $categoryId = null): int
{
    $sql = "SELECT COUNT(*) FROM products WHERE is_active = 1";
    $params = [];

    if ($categoryId !== null) {
        $sql .= " AND category_id = :category";
        $params['category'] = $categoryId;
    }

    $stmt = db()->prepare($sql);
    $stmt->execute($params);

    return (int) $stmt->fetchColumn();
}

function getCategories(): array
{
    return db()
        ->query("SELECT id, name, slug FROM categories ORDER BY name")
        ->fetchAll();
}
Ինչո՞ւ bindValue, ոչ execute([...])

execute()-ը բոլոր պարամետրերը փոխանցում է որպես տողեր։

LIMIT '12'-ը շարահյուսական սխալ է MySQL-ում — LIMIT-ը թիվ է պահանջում։

bindValue(':limit', $limit, PDO::PARAM_INT)-ը բացահայտ նշում է տիպը։ Դա անհրաժեշտ է միայն LIMIT-ի և OFFSET-ի համար — մնացած տեղերում execute()-ը լիովին բավարար է։

Դինամիկ WHERE-ի օրինաչափությունը

$where զանգվածը և implode(' AND ', $where)-ը թույլ են տալիս ավելացնել պայմաններ ըստ անհրաժեշտության։

Կարևորը՝ զանգվածում ընկնում են միայն մեր գրած տողերը, իսկ արժեքները՝ $params-ում։ Օգտագործողի տվյալը երբեք չի դառնում SQL-ի մաս։

Այս օրինաչափությունը կաշխատի ցանկացած քանակի ֆիլտրի հետ՝ գնի միջակայք, որոնում, կատեգորիա, առկայություն։

07Դասարանական

index.php — վերջնական տեսք
<?php
declare(strict_types=1);

require_once __DIR__ . '/bootstrap.php';

// Պարամետրեր URL-ից
$sort = $_GET['sort'] ?? 'price_asc';
$page = max(1, (int) ($_GET['page'] ?? 1));
$perPage = 12;

$categoryId = filter_var($_GET['category'] ?? null, FILTER_VALIDATE_INT) ?: null;

// Տվյալներ բազայից
$products = getProducts($categoryId, $sort, $perPage, ($page - 1) * $perPage);
$total      = countProducts($categoryId);
$categories = getCategories();

$content = render('pages/home', compact(
    'products', 'categories', 'total', 'sort', 'page', 'perPage', 'categoryId'
));

echo render('layout', [
    'title'   => 'ՏեխնոՄարկետ — տեխնիկա Երևանում',
    'content' => $content,
]);

08Տնային աշխատանք

  1. Կատարիր դասարանականի 10 քայլը։
  2. Ցուցադրիր SQL injection-ը փորձնական խոցելի ֆայլում՝ ?id=1 OR 1=1։ Ուղարկիր սքրինշոթ։ Ապա ուղղիր prepared statement-ով։
  3. Ցուցադրիր մուտքի շրջանցումը ' OR '1'='1'---ով նույն փորձնական ֆայլում։
  4. Փորձիր ORDER BY :field։ Ի՞նչ եղավ և ինչու։
  5. Ցուցադրիր LIMIT-ի խնդիրը execute([...])-ով և bindValue-ով։
  6. Ջնջիր ERRMODE_EXCEPTION-ը և կատարիր սխալ հարցում։ Ի՞նչ ես տեսնում։ Ինչո՞ւ դա վտանգավոր է։
  7. Ցուցադրիր DECIMAL-ի տիպի թակարդը strict_types-ով։
  8. Ավելացրու որոնում անվանով՝ LIKE :q։ Ուշադրություն՝ %-ը որտե՞ղ պիտի լինի։
  9. Ավելացրու գնի միջակայքի ֆիլտր դինամիկ WHERE-ի օրինաչափությամբ։
  10. Խնդրիր AI-ին վերանայել products.php-ը անվտանգության տեսանկյունից։ Ի՞նչ գտավ։

Երեք կանոն

1. Օգտագործողի ամեն արժեք → պարամետր 2. Դաշտերի անուններ և ուղղություն → սպիտակ ցուցակ 3. Գաղտնաբառերը → .gitignore

Ի՞նչ կարելի է պարամետրացնել

✅ WHERE id = :id ✅ WHERE name LIKE :q ✅ VALUES (:name, :price) ✅ LIMIT :limit (bindValue-ով) ❌ ORDER BY :field → սպիտակ ցուցակ ❌ FROM :table → սպիտակ ցուցակ ❌ ORDER BY price :dir → սպիտակ ցուցակ

Ստուգաթերթ

  • Ոչ մի փոփոխական հարցման տողում
  • Դաշտերի անունները՝ սպիտակ ցուցակից
  • ERRMODE_EXCEPTION միացված է
  • EMULATE_PREPARES անջատված է
  • Սխալի մանրամասները չեն ցուցադրվում օգտագործողին
  • config.php-ը .gitignore-ում է
  • Կա config.example.php
  • Շաբլոնները չեն փոխվել

  • Կարո՞ղ եմ բացատրել, ինչու փախցնելը լուծում չէ
  • Հասկանո՞ւմ եմ, ինչպես են աշխատում երկու ալիքները
