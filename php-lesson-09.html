01Խնդիրը

Ենթադրենք կատեգորիան պահում ենք ուղիղ products-ում՝ տեքստով․

-- Դաս 8-ի աղյուսակին ավելացնում ենք սյունակ և կապ
ALTER TABLE products
    ADD COLUMN category_id INT UNSIGNED NULL AFTER id,
    ADD CONSTRAINT fk_products_category
        FOREIGN KEY (category_id) REFERENCES categories(id)
        ON DELETE SET NULL;

Սա Դաս 8-ի NOT NULL-ի նման պաշտպանություն է, բայց կապերի համար։ Դաս 11-ում ադմին ֆորման կստուգի նույնը PHP-ում, որպեսզի օգտագործողը տեսնի հասկանալի հաղորդագրություն, ոչ թե բազայի սխալ։

Երբ կատեգորիան ջնջվում էԻ՞նչ է կատարվում ապրանքների հետ
ON DELETE SET NULL ⭐Ապրանքները մնում են, category_id-ն դառնում է NULL
ON DELETE RESTRICT (լռելյայն)Ջնջումը արգելվում է, քանի դեռ կան ապրանքներ
ON DELETE CASCADEԱպրանքները նույնպես ջնջվում են ⚠️
Ինչու SET NULL

Կատեգորիան ջնջելը չպետք է ջնջի 50 ապրանք։ CASCADE-ը հարմար է այնտեղ, որտեղ «զավակը» առանց «ծնողի» իմաստ չունի․ օրինակ՝ պատվերի տողերը առանց պատվերի։ Ապրանքը առանց կատեգորիայի դեռ ապրանք է։

03Պատվերներ

categories products ┌────────────────┐ ┌──────────────────────┐ │ id PK │◀───┐ │ id PK │ │ name │ └───│ category_id FK │ │ slug │ │ name, slug, price … │ └────────────────┘ └──────────────────────┘ ▲ orders │ ┌────────────────┐ │ │ id PK │ │ │ customer_name │ │ │ email, phone │ │ │ total │ │ │ status │ │ │ created_at │ │ └────────────────┘ │ ▲ │ │ order_items │ │ ┌─────────────────┐ │ └────────│ order_id FK │ │ │ product_id FK │─┘ │ quantity │ │ price │ ← գինը գնման պահին ⭐ └─────────────────┘ Մեկ պատվեր → շատ տողեր։ Մեկ ապրանք → շատ պատվերներում։
orders և order_items
CREATE TABLE orders (
    id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    email         VARCHAR(190) NOT NULL,
    phone         VARCHAR(30)  NOT NULL,
    total         DECIMAL(10,2) NOT NULL,
    status        ENUM('new', 'paid', 'shipped', 'cancelled') NOT NULL DEFAULT 'new',
    created_at    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE order_items (
    id         INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_id   INT UNSIGNED NOT NULL,
    product_id INT UNSIGNED NOT NULL,
    quantity   INT UNSIGNED NOT NULL,
    price      DECIMAL(10,2) NOT NULL,

    FOREIGN KEY (order_id)   REFERENCES orders(id)   ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE RESTRICT
) ENGINE=InnoDB;

Ինչու order_items-ում կրկնում ենք գինը

Թվում է ավելորդ է, որովհետև գինն արդեն կա products-ում։ Բայց գները փոխվում են։ Եթե պատվերի էջը գինը վերցնի products-ից, վաղը, երբ գինը բարձրանա, հին պատվերները ցույց կտան նոր գինը, և գումարները չեն համընկնի։

order_items.price-ը գինն է գնման պահին։ Դա պատմական փաստ է։ Կանոն․ պատմությունը սառեցվում է, ընթացիկ վիճակը՝ ոչ։ Սա բաժին 01-ի «մեկ փաստ՝ մեկ տեղում» կանոնի բացառությունը չէ․ դրանք երկու տարբեր փաստեր են՝ «ինչ արժե հիմա» և «ինչով գնվեց»։

04JOIN

products-ում կա միայն category_id = 2։ Կատեգորիայի անունը ցույց տալու համար պետք է կարդալ երկու աղյուսակը միասին։

ապրանքները կատեգորիայի անունով
SELECT
    p.name,
    p.price,
    c.name AS category_name
FROM products p
LEFT JOIN categories c ON c.id = p.category_id
WHERE p.is_active = 1
ORDER BY p.price;
ՄասՆշանակություն
products pԿարճ անուն (alias)․ p.name-ը նշանակում է «ապրանքի անունը»
c.name AS category_nameԵրկու աղյուսակն էլ ունեն name։ AS-ը վերանվանում է արդյունքում
ON c.id = p.category_idԻնչպես են տողերը զուգակցվում ⭐
INNER JOIN LEFT JOIN products categories products categories Air 13 ●──────● Նոութբուքեր Air 13 ●──────● Նոութբուքեր Մոնիտոր ●──────● Մոնիտորներ Մոնիտոր ●──────● Մոնիտորներ Տեսախցիկ ● ✗ Տեսախցիկ ●────── NULL Միայն զույգերը Բոլոր ապրանքները, Տեսախցիկը անհետացավ նույնիսկ առանց կատեգորիայի
Փորձիր երկուսն էլ

Կատարիր հարցումը LEFT JOIN-ով, ապա փոխիր INNER JOIN-ի։ Վեբ-տեսախցիկը, որը կատեգորիա չունի, կանհետանա երկրորդ արդյունքից։

Կանոն․ INNER JOIN, երբ կապը պարտադիր է, LEFT JOIN, երբ կարող է բացակայել։ Մեր category_id-ն կարող է լինել NULL, ուստի կատալոգում՝ LEFT JOIN։ Դասական սխալ է գրել INNER JOIN և հետո չհասկանալ, թե ինչու մի քանի ապրանք չեն երևում կայքում։

երեք աղյուսակ — պատվերի բովանդակությունը
SELECT
    o.customer_name,
    p.name                   AS product_name,
    oi.quantity,
    oi.price,
    oi.quantity * oi.price   AS line_total
FROM orders o
INNER JOIN order_items oi ON oi.order_id = o.id
INNER JOIN products p     ON p.id = oi.product_id
WHERE o.id = 1;
Այստեղ INNER-ը ճիշտ է

Պատվերի տողը միշտ ունի պատվեր և ապրանք (NOT NULL + RESTRICT), ուստի «անզույգ» տողեր չեն կարող լինել։ Ընտրությունը կախված է սխեմայից, ոչ սովորությունից։

05Հաշվել

հինգ ֆունկցիա
SELECT
    COUNT(*)     AS total,            -- քանի՞ ապրանք
    AVG(price)   AS avg_price,        -- միջին գին
    MIN(price)   AS cheapest,
    MAX(price)   AS most_expensive,
    SUM(stock)   AS total_stock       -- պահեստում ընդհանուր
FROM products
WHERE is_active = 1;
Դաս 3-ի count() և array_sum()

Դաս 3-ում նույնը անում էինք PHP-ում՝ count($products) և array_sum(array_column($products, 'stock'))։ Արդյունքը մեկ տող է, ոչ ցուցակ։

GROUP BY — հաշվել ըստ խմբերի

ապրանքների քանակը ամեն կատեգորիայում
SELECT
    c.name,
    COUNT(p.id)        AS product_count,
    ROUND(AVG(p.price)) AS avg_price
FROM categories c
LEFT JOIN products p ON p.category_id = c.id
GROUP BY c.id, c.name
ORDER BY product_count DESC;
GROUP BY-ը աշխատում է այսպես 1. Միացնում է տողերը (JOIN) 2. Բաժանում է կույտերի ըստ c.id-ի Նոութբուքեր [Air 13, Gamer 16] Մոնիտորներ [27 4K, 24] Սմարթֆոններ [X 128, X 256, Պլանշետ] Աքսեսուարներ [Ստեղնաշար, Մկնիկ, …] 3. Ամեն կույտից մեկ տող՝ COUNT(), AVG() Արդյունքում 4 տող, ոչ թե 12
HAVING — ֆիլտրել խմբերը
-- Կատեգորիաներ, որտեղ 2-ից ավելի ապրանք կա
SELECT c.name, COUNT(p.id) AS product_count
FROM categories c
LEFT JOIN products p ON p.category_id = c.id
GROUP BY c.id, c.name
HAVING product_count > 2;
WHERE թե HAVING

WHERE-ը ֆիլտրում է տողերը խմբավորումից առաջ։
HAVING-ը ֆիլտրում է խմբերը խմբավորումից հետո։

WHERE COUNT(p.id) > 2-ը սխալ է, որովհետև WHERE-ի պահին կույտերը դեռ չկան, և հաշվելու բան չկա։

Խորացված · ON թե WHERE LEFT JOIN-ում

Եթե ուզում ես հաշվել միայն ակտիվ ապրանքները, բայց ցույց տալ բոլոր կատեգորիաները (նաև դատարկները՝ 0-ով), պայմանը դիր ON-ում․

LEFT JOIN products p ON p.category_id = c.id AND p.is_active = 1

Եթե այն դնես WHERE p.is_active = 1, դատարկ կատեգորիաները կանհետանան, որովհետև նրանց p.is_active-ը NULL է, իսկ NULL = 1-ը ճիշտ չէ (Դաս 8)։ Փորձիր երկուսը և համեմատիր։

06Ինդեքսներ՝ ծանոթություն

Առանց ինդեքսի Ինդեքսով Կարդալ 1-ին տողը Բացել գրքի «բովանդակությունը» Կարդալ 2-րդ տողը Գտնել էջի համարը ... Գնալ ուղիղ այնտեղ Կարդալ 100 000-րդը ~100 000 քայլ ~17 քայլ
ստեղծել և ստուգել
CREATE INDEX idx_price ON products(price);

-- Բազան օգտագործու՞մ է այն
EXPLAIN SELECT name FROM products WHERE price < 100000;
-- key սյունակում՝ idx_price → այո
Որտեղ արդեն ինդեքս կա

PRIMARY KEY, UNIQUE (slug) և FOREIGN KEY սյունակները ինդեքս ունեն ինքնաբերաբար։ Ուստի WHERE slug = '...'-ը և JOIN ... ON c.id = p.category_id-ը արդեն արագ են։

12 ապրանքի դեպքում տարբերությունը չես տեսնի։ Ինդեքսները կարևոր են դառնում հազարավոր տողերի դեպքում։ Կանոն՝ ինդեքս դիր այն սյունակներին, որոնցով հաճախ WHERE կամ ORDER BY ես անում, ոչ «ամեն դեպքում» ամեն սյունակին․ ամեն ինդեքս դանդաղեցնում է INSERT-ը և UPDATE-ը։

07Դասարանական

  1. Ստեղծիր categories-ը, ավելացրու կապը և բաշխիր ապրանքները
  2. Փորձիր category_id = 99 սխալը
  3. Ստեղծիր orders և order_items, ավելացրու փորձնական պատվերը
  4. Համեմատիր LEFT և INNER JOIN-ը վեբ-տեսախցիկի վրա
  5. Գրիր 8 հարցումը ներքևում
8 հարցում
-- 1. Ակտիվ ապրանքները կատեգորիայի անունով
-- 2. Միայն «Մոնիտորներ» կատեգորիայի ապրանքները (JOIN + WHERE c.slug)
-- 3. Ապրանքները, որոնք կատեգորիա չունեն
-- 4. Պատվեր 1-ի բովանդակությունը՝ ապրանքների անուններով
-- 5. Պատվեր 1-ի ընդհանուր գումարը SUM(quantity * price)-ով
-- 6. Ապրանքների քանակը ամեն կատեգորիայում
-- 7. Ամենաթանկ ապրանքը ամեն կատեգորիայում (MAX)
-- 8. Կատեգորիաներ, որտեղ միջին գինը 100 000-ից բարձր է (HAVING)
5-րդ հարցումը ստուգում է պատվերը

Եթե SUM-ը հավասար չէ orders.total-ին, ինչ-որ բան սխալ է։ Նման ստուգումները իրական խանութներում կատարվում են կանոնավոր։ Դաս 11-ում պատվերը կստեղծենք PHP-ով, և գումարը կհաշվենք այնպես, որ այն միշտ համընկնի։

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Ժանրեր և ընթերցանության օրեր», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր 8 հարցումը։ Ուղարկիր SQL-ը և արդյունքների սքրինշոթները։
  2. Ավելացրու երկրորդ պատվեր երեք ապրանքով և գրիր հարցում, որը ցույց է տալիս բոլոր պատվերները՝ ամեն մեկի տողերի քանակով և գումարով։
  3. Փորձիր ջնջել ապրանք, որը կա պատվերում։ Ի՞նչ սխալ ստացար։ Ինչու է դա լավ։
  4. Ջնջիր կատեգորիա և ստուգիր, թե ինչ եղավ նրա ապրանքների հետ։ Վերադարձրու կատեգորիան։
  5. Համեմատիր ON ... AND p.is_active = 1 և WHERE p.is_active = 1 տարբերակները բաժին 05-ի խորացված նշումից։ Բացատրիր տարբերությունը քո բառերով։
  6. Գրիր հարցում․ «ամենավաճառված 3 ապրանքը»՝ SUM(oi.quantity)-ով։
  7. Պահիր բոլոր CREATE TABLE հրամանները մեկ schema.sql ֆայլում։ Ջնջիր բազան, կատարիր ֆայլը և ստուգիր, որ ամեն ինչ վերականգնվեց։ Այս ֆայլը պետք կլինի հաջորդ դասին։
  8. Տուր AI-ին քո schema.sql-ը և խնդրիր վերանայել այն։ Ի՞նչ առաջարկեց։ Համաձա՞յն ես։ Ինչու։

Հարցման լրիվ կարգը

SELECT ← ի՞նչ սյունակներ FROM ← որտեղից JOIN ← ինչի՞ հետ միացնել WHERE ← ո՞ր տողերը GROUP BY ← ինչպե՞ս խմբավորել HAVING ← ո՞ր խմբերը ORDER BY ← ինչպե՞ս դասավորել LIMIT ← քանի՞սը

Ստուգաթերթ

  • Կատեգորիան պահվում է մեկ անգամ, ապրանքը պահում է category_id
  • Կապերը ունեն FOREIGN KEY
  • ON DELETE-ը ընտրված է գիտակցաբար
  • order_items-ը պահում է գինը գնման պահին
  • Կատալոգում՝ LEFT JOIN
  • schema.sql-ը վերականգնում է ամբողջ բազան

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու կատեգորիան առանձին աղյուսակ է
  • Գիտե՞մ INNER-ի և LEFT JOIN-ի տարբերությունը օրինակով
  • Կարո՞ղ եմ բացատրել, ինչու order_items-ում կրկնվում է գինը
  • Գիտե՞մ WHERE-ի և HAVING-ի տարբերությունը
  • Հասկանո՞ւմ եմ, ինչ է անում GROUP BY-ը
Հաջորդ դասին

SQL-ը պատրաստ է։ Դաս 10՝ PDO և SQL injection, անվտանգության երրորդ դասը։ Կմիացնենք PHP-ն բազային, կտեսնենք, ինչու տողերի միացումով հարցում կառուցելը ամենավտանգավոր բանն է, որ կարելի է անել, և data.php-ը վերջապես կանհետանա։
