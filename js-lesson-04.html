Ֆունկցիաներ

Մինչ այժմ կոդը գրում էինք վերևից ներքև՝ մեկ երկար ցուցակով։ Այսօր սովորում ենք այն փաթեթավորել՝ մեկ անգամ գրիր, հազար անգամ կանչիր։ Սա ամենակարևոր դասն է առաջին բլոկում։

Նախապայման՝ Դաս 01–03 Բլոկ 1 · Լեզվի հիմքերը Տնային՝ 10 առաջադրանք

00Ի՞նչ ենք սովորելու այսօր

  • Ինչո՞ւ են պետք ֆունկցիաները
  • Ֆունկցիա ստեղծել և կանչել
  • Պարամետրեր և արգումենտներ
  • return — արժեք վերադարձնելը
  • Լռելյայն արժեքներ պարամետրերի համար
  • Տեսանելիության տիրույթ (scope)՝ գլոբալ և լոկալ
  • Ֆունկցիա-արտահայտություն և arrow ֆունկցիաներ
  • Ֆունկցիա՝ ֆունկցիայի ներսում

01Ինչո՞ւ են պետք ֆունկցիաները

Ենթադրենք՝ ծրագրում երեք տեղ պիտի հաշվես ուղղանկյան մակերեսը։

վատ եղանակ
const area1 = 5 * 3;
console.log(`Մակերեսը՝ ${area1} սմ²`);

const area2 = 10 * 7;
console.log(`Մակերեսը՝ ${area2} սմ²`);

const area3 = 2 * 8;
console.log(`Մակերեսը՝ ${area3} սմ²`);

Նույն կոդը երեք անգամ։ Իսկ եթե վաղը որոշես, որ «սմ²»-ի փոխարեն պիտի գրի «քառ. սմ»։ Պիտի փոխես երեք տեղում։ Իսկ եթե 50 տեղ լինե՞ր։ Մի տեղ անպայման կմոռանաս։

ֆունկցիայով
function showArea(width, height) {
  const area = width * height;
  console.log(`Մակերեսը՝ ${area} սմ²`);
}

showArea(5, 3);
showArea(10, 7);
showArea(2, 8);
Console
Մակերեսը՝ 15 սմ²
Մակերեսը՝ 70 սմ²
Մակերեսը՝ 16 սմ²

Հիմա տեքստը փոխելու համար բավական է ուղղել մեկ տեղում։

DRY

Ծրագրավորման գլխավոր սկզբունքներից է՝ Don't Repeat Yourself («մի՛ կրկնիր ինքդ քեզ»)։ Երբ նույն կոդը գրում ես երկրորդ անգամ — կանգ առ և մտածիր ֆունկցիայի մասին։

Ֆունկցիան նման է խոհանոցային մեքենայի

  • Ներս ես դնում միրգը (պարամետրեր)
  • Ներսում ինչ-որ բան է կատարվում (կոդը)
  • Դուրս է գալիս հյութը (return)

Կարևորը՝ քեզ պետք չէ իմանալ, թե ինչպես է այն աշխատում ներսում։ Դու պարզապես օգտվում ես։

04return — արժեք վերադարձնելը

Մինչ այժմ ֆունկցիաները միայն տպում էին։ Բայց շատ ավելի հաճախ մեզ պետք է, որ ֆունկցիան արդյունք տա, որպեսզի հետո այն օգտագործենք։

console.log vs return
// Տարբերակ 1 — միայն տպում է
function printSum(a, b) {
  console.log(a + b);
}

// Տարբերակ 2 — վերադարձնում է արժեքը
function getSum(a, b) {
  return a + b;
}

printSum(3, 4);
const result = getSum(3, 4);

console.log(result);
console.log(result * 10);          // արդյունքի հետ կարող ենք աշխատել
console.log(getSum(1, getSum(2, 3)));  // նույնիսկ իրար մեջ
Console
7
7
70
6
Շատ կարևոր

console.log()-ը ցույց է տալիս մարդուն։ return-ը արժեքը տալիս է ծրագրին։ Եթե ֆունկցիան միայն տպում է, նրա արդյունքը հետո չես կարող օգտագործել։

ինչ է լինում առանց return-ի

05Տեսանելիության տիրույթ (scope)

Ֆունկցիայի ներսում ստեղծված փոփոխականը գոյություն ունի միայն այնտեղ։ Դրսից նրան հասնել չի լինի։

լոկալ փոփոխական
function calculate() {
  const secret = 42;    // լոկալ — ապրում է միայն այստեղ
  console.log(secret);
}

calculate();
console.log(secret);       // ← փորձում ենք դրսից

Կանոն

Ֆունկցիան պիտի ստանա ամեն ինչ պարամետրերով և տա արդյունքը return-ով։ Գլոբալ փոփոխականներից կախված ֆունկցիաները դժվար է փորձարկել և վերաօգտագործել։

06Ֆունկցիա գրելու երեք ձև

1. Function declaration — հայտարարում

declaration
sayHi();   // աշխատում է ՆՈՒՅՆԻՍԿ մինչև հայտարարումը

function sayHi() {
  console.log("Բարև");
}
Console
Բարև

JS-ը ֆայլը կարդալուց առաջ «հավաքում» է բոլոր function-ները և բարձրացնում վերև։ Այս երևույթը կոչվում է hoisting։

2. Function expression — արտահայտություն

expression
const sayHi = function() {
  console.log("Բարև");
};

sayHi();

Ֆունկցիան պարզապես դրվում է փոփոխականի մեջ՝ ինչպես թիվ կամ տեքստ։ Այստեղ hoisting չկա՝ մինչև հայտարարումը կանչելը սխալ կտա։

3. Arrow function — սլաքով ✅

Ժամանակակից և ամենակարճ ձևը։ Հաջորդ դասերին այս ձևը կդառնա հիմնականը։

քայլ առ քայլ կարճացնում ենք
// 1. սովորական
const double1 = function(n) {
  return n * 2;
};

// 2. function բառը փոխարինում ենք => սլաքով
const double2 = (n) => {
  return n * 2;
};

// 3. մեկ պարամետր → փակագծերն ավելորդ են
const double3 = n => {
  return n * 2;
};

// 4. մեկ տող → {} և return-ն էլ ավելորդ են
const double4 = n => n * 2;

console.log(double1(5), double2(5), double3(5), double4(5));
Console
10 10 10 10
Ուշադրություն

Երբ ձևավոր փակագծերը բացակայում են, return-ը ենթադրվում է ինքնաբերաբար։ Իսկ երբ { } գրում ես — return-ը պարտադիր է։ Սա շատ հաճախ շփոթում են։

դասական սխալ

07Ֆունկցիա ֆունկցիայի ներսում

Ֆունկցիաները կարող են կանչել միմյանց։ Այսպես մեծ խնդիրը բաժանում ենք փոքր, հասկանալի կտորների։

զամբյուղի հաշվարկ
const getSubtotal = (price, qty) => price * qty;

const getDiscount = subtotal => {
  if (subtotal >= 50000) return 0.2;
  if (subtotal >= 20000) return 0.1;
  return 0;
};

function getTotal(price, qty) {
  const subtotal = getSubtotal(price, qty);
  const discount = getDiscount(subtotal);
  return subtotal - subtotal * discount;
}

console.log(getTotal(3000, 2));
console.log(getTotal(3000, 10));
console.log(getTotal(3000, 20));

08Դասարանական աշխատանք

Կառուցում ենք մեր առաջին ֆունկցիաների գրադարանը՝ ֆայլ, որտեղ ամեն ֆունկցիա մեկ փոքր գործ է անում։

utils.js
// --- Թվեր ---
const isEven = n => n % 2 === 0;

const max = (a, b) => a > b ? a : b;

function factorial(n) {
  let result = 1;
  for (let i = 2; i <= n; i++) {
    result *= i;
  }
  return result;
}

// --- Տեքստ ---
function reverseString(text) {
  let reversed = "";
  for (let i = text.length - 1; i >= 0; i--) {
    reversed += text[i];
  }
  return reversed;
}

// --- Երկրաչափություն ---
const circleArea = r => Math.PI * r ** 2;

// --- Ստուգում ---
console.log(isEven(14));
console.log(max(17, 42));
console.log(factorial(5));
console.log(reverseString("JavaScript"));
console.log(circleArea(3).toFixed(2));

text[i] — i-երորդ տառը (հաշվարկը սկսվում է 0-ից)։

10Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Ֆունկցիաների գրադարան», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

Ամեն ֆունկցիա գրիր և՛ function ձևով, և՛ arrow ձևով։ Ամեն մեկը ստուգիր առնվազն 3 տարբեր արգումենտով։

  1. getSum(a, b) — վերադարձնում է երկու թվի գումարը։
  2. getRectangleArea(w, h) — մակերես։ Եթե պարամետր չուղարկեն, լռելյայն երկուսն էլ 1։
  3. isPositive(n) — վերադարձնում է true կամ false։
  4. maxOfThree(a, b, c) — երեք թվից ամենամեծը։
  5. celsiusToFahrenheit(c) — բանաձևը՝ c * 9 / 5 + 32։
  6. getGrade(mark) — վերադարձնում է «Գերազանց», «Լավ», «Բավարար» կամ «Անբավարար»։ Օգտագործիր return-ը առանց else-ի։
  7. sumTo(n) — 1-ից n թվերի գումարը (ցիկլով)։
  8. countDigits(n) — քանի՞ թվանշան ունի թիվը։
  9. isPrime(n) — պարզ թիվ է թե ոչ։ Օգտագործիր Դաս 3-ի դրոշակի հնարքը։
  10. Գրիր getTicketPrice(age, isStudent)․ մինչև 6 տարեկան՝ 0, ուսանող՝ 1000, մինչև 65՝ 2500, 65-ից բարձր՝ 1000։ Ստուգիր 5 տարբեր դեպք։

11Հիշելու թերթիկ

ԳրառումԻնչ է անում
function name() { }Հայտարարում (hoisting կա)
const f = function() { };Ֆունկցիա-արտահայտություն
const f = () => { };Arrow ֆունկցիա
const f = n => n * 2;Կարճ arrow — return-ը ենթադրվում է
name();Կանչ։ Առանց () չի աշխատում
function f(a, b)a, b — պարամետրեր
f(3, 5)3, 5 — արգումենտներ
function f(x = 10)Լռելյայն արժեք
return x;Վերադարձնում է արժեքը և կանգնեցնում ֆունկցիան
առանց return-իՖունկցիան վերադարձնում է undefined
լոկալ փոփոխականԱպրում է միայն ֆունկցիայի ներսում

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել return-ի և console.log-ի տարբերությունը
  • Գիտե՞մ՝ ինչու n => { n * 2 }-ը տալիս է undefined
  • Հասկանո՞ւմ եմ, թե ինչու ֆունկցիայի ներսի փոփոխականը դրսում չի երևում
  • Կարո՞ղ եմ նույն ֆունկցիան գրել երեք ձևով
