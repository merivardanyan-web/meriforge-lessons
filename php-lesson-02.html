02Տողեր

Այստեղ PHP-ն ունի բան, որ JS-ում չկա՝ չակերտի տեսակը փոխում է վարքագիծը։

մեկ vs կրկնակի չակերտ ⭐
$name = "Անի";

// Կրկնակի — փոփոխականները ՀԱՇՎԱՐԿՎՈՒՄ են
echo "Բարև, $name!";      // Բարև, Անի!

// Մեկ — ամեն ինչ բառացի
echo 'Բարև, $name!';      // Բարև, $name!

// Ձևավոր փակագծերը՝ երբ սահմանը պարզ չէ
echo "Գին՝ {$price}֏";
echo "Ապրանք՝ {$product['name']}";   // զանգվածի համար պարտադիր
Ի՞նչ ընտրել

Մեկ չակերտ — երբ փոփոխական չկա։ Չնչին, բայց ավելի արագ է և ավելի ազնիվ ընթերցողի համար․ անմիջապես երևում է, որ ինտերպոլացիա չկա։

Կրկնակի — երբ ներսում փոփոխական է։

Իսկ HTML-ի ներսում լավագույնը՝ ընդհանրապես չօգտագործել echo, այլ դուրս գալ PHP-ից։ Այդ մասին՝ բաժին 07-ում։

օգտակար ֆունկցիաներ
strlen($s);                 // երկարություն ԲԱՅՏԵՐՈՎ ⚠️
mb_strlen($s);              // երկարություն ՆԻՇԵՐՈՎ ✅

mb_strtoupper($s);
mb_strtolower($s);
trim($s);                   // հանում է եզրերի բացատները
str_replace("ա", "բ", $s);
str_contains($s, "նոութ");   // PHP 8+
explode(",", $s);            // տող → զանգված (JS՝ split)
implode(", ", $arr);        // զանգված → տող (JS՝ join)
number_format(450000, 0, ".", " ");   // 450 000
mb_ նախածանցը հայերենի համար պարտադիր է

strlen("Անի") վերադարձնում է 6, ոչ 3։ Հայերեն ամեն նիշը UTF-8-ում երկու բայտ է։

Նույնը՝ strtoupper-ը հայերեն տառերը չի փոխի, substr-ը կկտրի բառը մեջտեղից և կստացվի կոտրված նիշ։

Կանոն՝ եթե տողը կարող է հայերեն լինել — օգտագործիր mb_ տարբերակը։ Մեր դասընթացում դա գրեթե միշտ այդպես է։

03== և === — վտանգավոր բաժինը

Ուշադրություն

JS-ում ==-ը վատ սովորություն է։ PHP-ում այն անվտանգության խոցելիություն է։ Այս բաժինը արժե կարդալ երկու անգամ։

համեմատիր JS-ի հետ
// Երկուսում էլ նույնը
"25" == 25          // true  — տիպերը փոխարկվում են
"25" === 25         // false — տիպերն էլ են համեմատվում ✅

// PHP 8-ում ուղղվեց մեծ խնդիր
"abc" == 0          // PHP 7՝ true (!)  ·  PHP 8՝ false

// Բայց սա դեռ գործում է
"1" == "01"          // true
"10" == "1e1"        // true — երկուսն էլ թիվ են
0 == ""              // false PHP 8-ում
null == false        // true
0 == false           // true
[] == false          // true
ինչու է սա անվտանգության հարց
// ❌ Իրական խոցելիություն
if ($userToken == $dbToken) {
    // Եթե երկու token-ն էլ նման են թվերի,
    // դրանք համեմատվում են ՈՐՊԵՍ ԹԻՎ։
    // "0e123" == "0e456" → true, քանի որ երկուսն էլ 0 են։
}

// ✅ Ճիշտ
if ($userToken === $dbToken) { ... }

// ✅ Գաղտնաբառերի և token-ների համար՝ էլ ավելի լավ
if (hash_equals($dbToken, $userToken)) { ... }
Կանոն ամբողջ դասընթացի համար

Միշտ ===։ Բացառություններ չկան։

Եթե տիպերը տարբեր են և դա նորմալ է, բացահայտ փոխարկիր՝ (int) $_GET['id'] === $product['id']։

Այդպես կոդը ասում է, թե ինչ է անում, և չի հենվում PHP-ի թաքնված կանոնների վրա։ Դաս 16-ում, երբ գրենք նույնականացումը, սա կդառնա կրիտիկական։

04Դատարկություն

PHP-ում «դատարկ»-ը երեք տարբեր բան է նշանակում, և դրանք շփոթելը հաճախակի սխալ է։

Ի՞նչ է ստուգումԵ՞րբ
isset($x)Գոյություն ունի՞ և null չէ՞$_POST, զանգվածի բանալի ⭐
empty($x)Բացակայո՞ւմ է կամ «դատարկ» էՖորմայի դաշտ
is_null($x)Ուղիղ null է՞Բազայից եկած արժեք
array_key_exists()Բանալին կա՞, նույնիսկ null-ովԵրբ null-ը վավեր արժեք է
empty-ի թակարդը
// Այս ամենի համար empty() վերադարձնում է true
empty("");        // true
empty("0");       // true  ⚠️ տողային զրո
empty(0);         // true  ⚠️
empty(0.0);       // true
empty([]);        // true
empty(null);      // true
empty(false);     // true

// ❌ Իրական սխալ ֆորմայում
if (empty($_POST['quantity'])) {
    // Օգտագործողը գրեց 0 → համարվում է «չի լրացրել»
}

// ✅ Ճիշտ
if (!isset($_POST['quantity']) || $_POST['quantity'] === '') { ... }
երկու շատ օգտակար օպերատոր
// ?? — null coalescing (նույնը, ինչ JS-ում)
$page = $_GET['page'] ?? 1;
// Եթե 'page' չկա → 1։ Առանց սխալի։ ⭐

// ?-> — nullsafe (PHP 8+)
$city = $user?->getAddress()?->city;
// Եթե $user-ը null է → $city = null, ոչ սխալ

// Կարճ ternary
$title = $product['name'] ?: "Անանուն";
??-ը կդառնա քո ամենաշատ օգտագործվող օպերատորը

Դաս 6-ից սկսած ամեն ֆորմայի դաշտ կկարդաս այսպես․

$name = trim($_POST['name'] ?? '');

Առանց ??-ի ամեն բացակայող դաշտ սխալ է առաջացնում։ Սա ոչ թե ոճ է, այլ անհրաժեշտություն։

05Պայմաններ

ծանոթ է, բացի elseif-ից
if ($price > 100000) {
    echo "Թանկ";
} elseif ($price > 50000) {   // միասին, ոչ else if
    echo "Միջին";
} else {
    echo "Մատչելի";
}

// Տրամաբանական օպերատորներ — նույնը
$a && $b;   $a || $b;   !$a;

match — PHP 8-ի նոր գործիքը

switch-ի ավելի լավ տարբերակը
// switch — բազմաբառ, պահանջում է break
switch ($status) {
    case 'new':      $label = 'Նոր';      break;
    case 'paid':     $label = 'Վճարված';  break;
    default:         $label = 'Անհայտ';
}

// match — կարճ, վերադարձնում է արժեք, խիստ === համեմատում ⭐
$label = match ($status) {
    'new'      => 'Նոր',
    'paid'     => 'Վճարված',
    'shipped'  => 'Ճանապարհին',
    default   => 'Անհայտ',
};

// Մի քանի արժեք մեկ ճյուղում
$class = match ($status) {
    'paid', 'shipped' => 'badge--success',
    'cancelled'      => 'badge--danger',
    default          => 'badge',
};
Ինչու match

1. Օգտագործում է ===, ոչ == — ինքնաբերաբար ավելի անվտանգ։
2. Չկա break → չկա մոռացված break-ի սխալ։
3. Վերադարձնում է արժեք → հարմար է վերագրման համար։
4. Առանց default-ի և չհամընկնելու դեպքում գցում է սխալ, ոչ լուռ անցնում։

Օգտագործիր match-ը լռելյայն։ switch-ը պետք է միայն այն դեպքում, երբ մեկ ճյուղում մի քանի հրահանգ է։

foreach — PHP-ի գլխավոր ցիկլը

երկու ձև

    echo $product['name'];

// Ինդեքսի հետ

    echo ($i + 1) . ". " . $product['name'];
}
foreach-ի հղումային թակարդ

Սա PHP-ի ամենահայտնի «WTF»-երից է․

foreach ($arr as &$v) { ... }
foreach ($arr as $v) { ... } ← այստեղ վերջին տարրը կփչանա

Առաջին ցիկլից հետո $v-ն մնում է որպես հղում վերջին տարրին։ Երկրորդ ցիկլը գրում է հենց այդ հղումի մեջ։

Լուծում՝ unset($v); հղումային foreach-ից անմիջապես հետո։ Կամ ավելի պարզ՝ ընդհանրապես չօգտագործել & և վերադարձնել նոր զանգված։

Հենց դա էր խոստումը Դաս 24-ում․ «JS-ը և PHP-ն փոխում են դասերը և բովանդակությունը, ոչ ոճերը»։ Հիմա տեսնում ես այն գործողության մեջ։

Կարևոր զգուշացում

Այս օրինակում <?= $p['name'] ?>-ը դեռ վտանգավոր է։ Եթե ապրանքի անվան մեջ լինի <script>, այն կկատարվի բրաուզերում։

Դա XSS-ն է, և Դաս 5-ը ամբողջովին դրա մասին է։ Այնտեղ կսովորենք htmlspecialchars-ը և կվերադառնանք այս կոդին՝ ուղղելու։

Հիմա հիշիր միայն՝ բազայից եկած տվյալը երբեք ուղիղ չի դրվում էջում։

09Տնային աշխատանք

  1. Կատարիր դասարանականի բոլոր 8 քայլը։
  2. Փորձարկիր 6 համեմատություն ==-ով և ===-ով։ Կազմիր աղյուսակ արդյունքներով։
  3. Ցուցադրիր, թե ինչու "0e123" == "0e456"-ը վտանգավոր է։ Բացատրիր սեփական բառերով։
  4. Փորձարկիր strlen-ը և mb_strlen-ը հայերեն բառի վրա։ Բացատրիր տարբերությունը։
  5. Կազմիր աղյուսակ՝ isset, empty, is_null 6 տարբեր արժեքի վրա։
  6. Գտիր empty-ի թակարդի իրական օրինակ քո ֆորմայում և ուղղիր։
  7. Վերաշարադրիր որևէ switch match-ով։ Հաշվիր՝ քանի՞ տող կրճատվեց։
  8. Ցուցադրիր foreach-ի հղումային թակարդը կոդով և բացատրիր, ինչու է վերջին տարրը փչանում։
  9. Վերաշարադրիր մեկ բաժին echo-ով, ապա այլընտրանքային շարահյուսությամբ։ Ո՞րն է ընթեռնելի։
  10. Օգտագործիր ??-ը առնվազն երեք տեղում։

Երեք վտանգավոր տեղը

1. == → միշտ գրիր === 2. empty("0") → true, ուստի զգույշ ֆորմաներում 3. strlen("Անի") → 6, ոչ 3։ Օգտագործիր mb_

<?php foreach ($items as $i): ?> <li><?= $i ?></li> <?php endforeach; ?>

Ստուգաթերթ

  • Ամենուր ===, ոչ ==
  • Հայերեն տողերի համար mb_
  • Ֆորմայի դաշտերը կարդացվում են ??-ով
  • HTML-ը գրված է շաբլոնի շարահյուսությամբ, ոչ echo-ով
  • Ապրանքները գալիս են զանգվածից
  • Նոր ապրանք ավելացնելիս HTML-ը չի փոխվում

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու ==-ը անվտանգության խնդիր է
  • Գիտե՞մ, ե՞րբ isset, ե՞րբ empty
  • Հասկանո՞ւմ եմ մեկ և կրկնակի չակերտի տարբերությունը
  • Գիտե՞մ, ինչու match-ը լավն է switch-ից
  • Կարո՞ղ եմ բացատրել, ինչու mb_strlen
Հաջորդ դասին

Զանգվածներ և ֆունկցիաներ։ Ասոցիատիվ զանգվածները PHP-ի սիրտն են — բազայի ամեն տող դրանք են։ Կտեսնենք նաև array_map, array_filter, usort — ուղիղ համապատասխանություններ JS Դաս 6-ի հետ։

Ֆունկցիաներում կսովորենք տիպերի հայտարարումը — բան, որ JS-ում չկար։
