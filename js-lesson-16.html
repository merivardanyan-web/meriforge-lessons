Զամբյուղ, ռեֆակտորինգ, դեբագ

Կատալոգը պատրաստ է։ Այսօր ավելացնում ենք զամբյուղ, սովորում ենք կոդը կարգի բերել և վերջապես ծանոթանում ենք ծրագրավորողի ամենակարևոր գործիքին՝ դեբագերին։

Նախապայման՝ Դաս 15-ի աշխատող կատալոգ Բլոկ 2-ի եզրափակիչ Օր 2-ից 2

00Այսօրվա պլանը

  • Զամբյուղի վիճակի ճիշտ մոդելը
  • Ավելացում, քանակի փոփոխում, ջնջում
  • Պահպանում localStorage-ում
  • Զամբյուղի մոդալ պատուհան
  • Կոդի ռեֆակտորինգ և ֆայլերի բաժանում

  • Կոդի վերանայման ստուգաթերթ

01Զամբյուղի մոդելը

Առաջին հարցը՝ ի՞նչ ենք պահում զամբյուղում։ Երկու մոտեցում կա, և ընտրությունը կարևոր է։

❌ վատ մոտեցում
const cart = [
  { id: 2, name: "Անլար մկնիկ", price: 8000, category: "accessory", qty: 2 },
  { id: 4, name: "Մոնիտոր 27",  price: 120000, category: "monitor", qty: 1 }
];

Ի՞նչն է վատ։ Տվյալները կրկնվում են։ Եթե վաղը մկնիկի գինը փոխես data.js-ում, զամբյուղում կմնա հին գինը։ Իսկ եթե ապրանքը ջնջես — զամբյուղում կմնա գոյություն չունեցող ապրանք։

✅ ճիշտ մոտեցում
let cart = [
  { id: 2, qty: 2 },
  { id: 4, qty: 1 }
];

Պահում ենք միայն այն, ինչ իսկապես պատկանում է զամբյուղին՝ ի՞նչ և քանի՞ հատ։ Մնացած տեղեկությունը վերցնում ենք PRODUCTS-ից՝ ամեն անգամ թարմ։

js/cart.js — տվյալները միացնելը
// Զամբյուղի տողերը՝ ամբողջական տեղեկությամբ
function getCartItems() {
  return cart
    .map(item => {
      const product = PRODUCTS.find(p => p.id === item.id);
      if (!product) return null;              // ապրանքը ջնջվել է

      return {
        ...product,                            // Դաս 7
        qty: item.qty,
        subtotal: product.price * item.qty
      };
    })
    .filter(Boolean);                          // հանում է null-երը
}

function getCartTotal() {
  return getCartItems().reduce((sum, i) => sum + i.subtotal, 0);
}

function getCartCount() {
  return cart.reduce((sum, i) => sum + i.qty, 0);
}
filter(Boolean)

Հնարք։ Boolean-ը ֆունկցիա է, որը վերադարձնում է true truthy արժեքների համար (Դաս 2)։ Ուստի filter(Boolean)-ը հանում է բոլոր null, undefined, 0, "" արժեքները։ Կարճ և ընթեռնելի։

02Զամբյուղի գործողությունները

js/cart.js
const CART_KEY = "catalog:cart";

let cart = load(CART_KEY, []);          // Դաս 14

function addToCart(id) {
  const existing = cart.find(i => i.id === id);

  if (existing) {
    existing.qty++;
  } else {
    cart.push({ id, qty: 1 });
  }

  renderCart();
}

function removeFromCart(id) {
  cart = cart.filter(i => i.id !== id);
  renderCart();
}

function changeQty(id, delta) {
  const item = cart.find(i => i.id === id);
  if (!item) return;

  item.qty += delta;

  if (item.qty <= 0) {
    removeFromCart(id);      // այն ինքն է կանչում renderCart-ը
    return;
  }

  renderCart();
}

function clearCart() {
  if (cart.length === 0) return;
  if (!confirm("Դատարկե՞լ զամբյուղը")) return;

  cart = [];
  renderCart();
}
Ուշադրություն

cart = cart.filter(...) գրելու համար cart-ը պիտի լինի let, ոչ const։ filter-ը վերադարձնում է նոր զանգված (Դաս 6), ուստի պիտի վերագրենք։

03Ստուգակետ 1 · Render և պահպանում

HTML — զամբյուղի մոդալ

<button class="cart-btn">
  Զամբյուղ <span class="cart-count">0</span>
</button>

<!-- Մոդալ -->
<div class="modal">
  <div class="modal__box">
    <button class="modal__close">✕</button>

    <div class="cart-list"></div>

js/cart.js — renderCart
const cartList  = document.querySelector(".cart-list");
const cartTotal = document.querySelector(".cart-total");
const cartCount = document.querySelector(".cart-count");

function createCartRow({ id, name, price, qty, subtotal }) {
  return `
    <div class="cart-row" data-id="${id}">
      <span class="cart-row__name">${name}</span>
      <span class="cart-row__price">${formatPrice(price)}</span>

      <div class="qty">
        <button data-action="minus">−</button>
        <span>${qty}</span>
        <button data-action="plus">+</button>
      </div>

      <span class="cart-row__subtotal">${formatPrice(subtotal)}</span>
      <button data-action="remove">✕</button>
    </div>
  `;
}

function renderCart() {
  const items = getCartItems();

  cartList.innerHTML = items.length === 0
    ? `<p class="empty">Զամբյուղը դատարկ է։</p>`
    : items.map(createCartRow).join("");

  cartTotal.textContent = `Ընդամենը՝ ${formatPrice(getCartTotal())}`;

  const count = getCartCount();
  cartCount.textContent = count;
  cartCount.classList.toggle("is-empty", count === 0);   // Դաս 14

  save(CART_KEY, cart);          // ← ամբողջ պահպանումը՝ մեկ տող
}
Կրկին նույն սկզբունքը

Ինչպես Դաս 14-ի to-do-ում՝ պահպանումը մեկ տող է render-ի ներսում։ Ամեն գործողություն կանչում է renderCart(), ուստի ոչինչ չի կարող կորչել։ Կարիք չկա ամեն ֆունկցիայում հիշել save-ի մասին։

04Ստուգակետ 2 · Իրադարձություններ

js/cart.js — մշակիչներ
const modal   = document.querySelector(".modal");
const cartBtn = document.querySelector(".cart-btn");

// ---------- Մոդալի բացում/փակում (Դաս 11) ----------
const openCart  = () => modal.classList.add("is-open");
const closeCart = () => modal.classList.remove("is-open");

cartBtn.addEventListener("click", openCart);

modal.addEventListener("click", (e) => {
  if (e.target === modal) closeCart();          // ֆոնին սեղմեցին
  if (e.target.closest(".modal__close")) closeCart();
});

document.addEventListener("keydown", (e) => {
  if (e.key === "Escape") closeCart();
});

// ---------- Զամբյուղի ներսի գործողությունները (Դաս 12) ----------
modal.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-action]");
  if (!btn) return;

  if (btn.dataset.action === "clear") {
    clearCart();
    return;
  }

  const row = btn.closest(".cart-row");
  if (!row) return;

  const id = Number(row.dataset.id);

  switch (btn.dataset.action) {
    case "plus":   changeQty(id, 1);  break;
    case "minus":  changeQty(id, -1); break;
    case "remove": removeFromCart(id); break;
  }
});

// ---------- Կատալոգի «Ավելացնել» կոճակը ----------
// main.js-ի արդեն գոյություն ունեցող մշակիչում փոխարինում ենք console.log-ը
if (btn.dataset.action === "add") {
  addToCart(Number(btn.dataset.id));
  showToast("Ավելացվեց զամբյուղում");     // Դաս 14
}

// ---------- Գործարկում ----------
renderCart();   // վերականգնում է պահված զամբյուղը
Ուշադրություն

modal-ի վրա երկու listener կա՝ մեկը փակելու, մյուսը՝ գործողությունների։ Դա նորմալ է — addEventListener-ը թույլ է տալիս քանի ուզես (Դաս 11)։ Կարելի էր նաև միավորել մեկում։

05Ստուգակետ 3 · Փորձարկում

  • «Ավելացնել» սեղմելիս հաշվիչը մեծանում է
  • Նույն ապրանքը կրկին ավելացնելիս քանակը դառնում է 2, նոր տող չի ավելանում
  • Զամբյուղի կոճակը բացում է մոդալը
  • Մոդալը փակվում է խաչիկով, ֆոնին սեղմելով և Escape-ով
  • Ներսի բովանդակությանը սեղմելիս մոդալը չի փակվում
  • «+» և «−» փոխում են քանակը և ենթագումարը
  • Քանակը 0 դառնալիս տողը ջնջվում է
  • «✕»-ը ջնջում է տողը
  • Ընդհանուր գումարը ճիշտ է հաշվվում
  • «Դատարկել»-ը հարցնում է հաստատում
  • Դատարկ զամբյուղում երևում է հաղորդագրություն
  • Էջը թարմացնելուց հետո զամբյուղը մնում է տեղում
  • Ֆիլտրերը փոխելիս զամբյուղը չի կորչում
  • Սպառված ապրանքի կոճակը չի աշխատում

06Ռեֆակտորինգ

Ռեֆակտորինգ — կոդի կառուցվածքի բարելավում՝ առանց վարքագիծը փոխելու։ Ծրագիրն աշխատում է նույնը, բայց կոդն ավելի հեշտ է կարդալ և փոխել։

1. Ֆայլերի բաժանում

վերջնական կառուցվածք
catalog/
├── index.html
├── css/
│   └── style.css
└── js/
    ├── data.js       ← PRODUCTS, CATEGORIES
    ├── storage.js    ← save, load
    ├── utils.js      ← formatPrice, debounce, showToast
    ├── catalog.js    ← state, getVisibleProducts, createCard, render
    ├── cart.js       ← cart, addToCart, renderCart ...
    └── main.js       ← միայն մշակիչներ և գործարկում
Ֆայլերի բաժանման կանոն

Ֆայլը պիտի ունենա մեկ պատասխանատվություն։ Եթե չես կարողանում ֆայլի անվանումը տալ մեկ բառով — ուրեմն այն շատ բան է անում։

2. Կրկնվող կոդի հեռացում

debounce-ը՝ ֆունկցիա
// ❌ Դաս 15-ում debounce-ը գրված էր ուղիղ մշակիչում
let searchTimer;
searchInput.addEventListener("input", () => {
  clearTimeout(searchTimer);
  searchTimer = setTimeout(() => { ... }, 300);
});

// ✅ js/utils.js — վերաօգտագործելի ֆունկցիա
function debounce(fn, delay = 300) {
  let timer;

  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };
}

// Օգտագործում
searchInput.addEventListener("input", debounce(() => {
  state.search = searchInput.value.trim();
  render();
}, 300));
Ի՞նչ է կատարվում այստեղ

debounce-ը ֆունկցիա է, որը վերադարձնում է ուրիշ ֆունկցիա։ Առաջին անգամ ես տեսնում այսպիսի բան։ Դաս 23-ում կհասկանանք ամբողջությամբ (փակույթներ, closures)։ Հիմա բավական է իմանալ՝ այն աշխատում է։

3. Կախարդական թվերի հեռացում

անուն տուր ամեն բանի
// ❌
if (window.scrollY > 400) { ... }
setTimeout(fn, 300);

// ✅
const SCROLL_THRESHOLD = 400;
const SEARCH_DELAY = 300;

if (window.scrollY > SCROLL_THRESHOLD) { ... }
setTimeout(fn, SEARCH_DELAY);

4. Ինքնաստուգման ցուցակ

  • Ամեն ֆունկցիա անում է մեկ գործ
  • Ֆունկցիաների անունները սկսվում են բայով՝ getCartTotal, renderCart
  • Չկա կրկնվող կոդ (եթե երեք անգամ գրել ես նույնը — հանիր ֆունկցիա)
  • Չկան կախարդական թվեր
  • Չկան var-եր, միայն const և let
  • Չկան մոռացված console.log-եր
  • Օգտագործողի տվյալները չեն դրվում innerHTML-ի մեջ առանց մտածելու
  • Ամեն մշակիչ փոխում է վիճակը և կանչում render, ոչ ուղիղ DOM

07Դեբագ · DevTools

console.log-ը լավ է, բայց կա շատ ավելի հզոր գործիք։

Breakpoint — կանգնեցնել ծրագիրը

  1. DevTools → ներդիր Sources
  2. Ձախից գտիր քո main.js-ը
  3. Սեղմիր տողի համարին — կհայտնվի կապույտ նշան
  4. Կատարիր գործողությունը էջում — ծրագիրը կկանգնի հենց այդ տողում

Կանգնած վիճակում կարող ես տեսնել բոլոր փոփոխականների արժեքները այդ պահին։ Դա շատ ավելի արագ է, քան տասը console.log գրելը։

09Նախագծի հանձնում

Նախագիծ · Առաջադրանք 1

Զամբյուղը ՏեխնոՄարկետում՝ Առաջադրանք 1, քայլեր 6–8։ ՏեխնոՄարկետում զամբյուղը առանձին էջ է, իսկ մոդալը ցույց է տալիս «Ավելացվեց զամբյուղ»։

Սա Բլոկ 2-ի նախագծի առաջին փուլն է։ Այն կշարունակվի Դաս 17–23-ում, իսկ Դաս 24-ում կներկայացնես ամբողջը և կբացատրես քո կոդը։

Պարտադիր

  1. Աշխատող կատալոգ՝ ֆիլտր, որոնում, տեսակավորում, «միայն առկա»
  2. Աշխատող զամբյուղ՝ ավելացում, քանակ, ջնջում, դատարկում
  3. Զամբյուղը պահպանվում է էջը թարմացնելուց հետո
  4. Զամբյուղի մոդալ՝ երեք եղանակով փակվող
  5. Դատարկ վիճակների մշակում՝ և՛ կատալոգում, և՛ զամբյուղում
  6. Կոդը բաժանված է առնվազն 4 ֆայլի
  7. Ոչ մի var, ոչ մի մոռացված console.log, ոչ մի կախարդական թիվ

Լրացուցիչ (ընտրովի)

  1. Ֆիլտրերի վիճակի պահպանում localStorage-ում
  2. «Հավանածներ» ցուցակ՝ սրտիկի կոճակով, առանձին պահոցում
  3. Ապրանքի մանրամասների մոդալ
  4. Toast ծանուցումներ բոլոր գործողությունների համար
  5. Զեղչ՝ 200 000-ից բարձր պատվերի դեպքում 10%
