3. Իրադարձության մշակիչում

4. Arrow ֆունկցիա — չունի սեփական this

class Product {
  static currency = "֏";
  static count = 0;

#private — իսկապես փակ

վանդակագիծը՝ պաշտպանություն
class BankAccount {
  #balance = 0;              // փակ դաշտ

  constructor(owner) {
    this.owner = owner;
  }

  deposit(amount) {
    if (amount <= 0) throw new Error("Գումարը պիտի դրական լինի");
    this.#balance += amount;
    return this.#balance;
  }

  withdraw(amount) {
    if (amount > this.#balance) throw new Error("Բավարար միջոց չկա");
    this.#balance -= amount;
    return this.#balance;
  }

  get balance() {
    return this.#balance;
  }
}

const acc = new BankAccount("Անի");

acc.deposit(50000);
console.log(acc.balance);

acc.#balance = 1000000;   // փորձում ենք խաբել

class Product {
  constructor(name, price) {

  constructor(name, price, fileSize) {

Չչարաշահել

Ժառանգումը հզոր է, բայց խորը շղթաները (A → B → C → D) դժվար են հասկանալի։ Ժամանակակից մոտեցումը՝ կոմպոզիցիա ժառանգման փոխարեն՝ ֆունկցիոնալությունը հավաքել առանձին կտորներից։ Ժառանգումն օգտագործիր, երբ իսկապես «X-ը Y-ի տեսակ է»։

07Դասարանական · Զամբյուղը որպես դաս

Վերցնում ենք Դաս 16-ի զամբյուղը և վերածում դասի։ Համեմատիր՝ ինչ է փոխվում։

js/Cart.js
import { save, load } from "./storage.js";

export class Cart {
  #items = [];                    // փակ — դրսից ոչ ոք չի դիպչի
  #storageKey;
  #products;

  constructor(products, storageKey = "cart") {
    this.#products = products;
    this.#storageKey = storageKey;
    this.#items = load(storageKey, []);
  }

  // ---------- Կարդալու համար ----------
  get items() {
    return this.#items
      .map(item => {
        const product = this.#products.find(p => p.id === item.id);
        if (!product) return null;
        return { ...product, qty: item.qty, subtotal: product.price * item.qty };
      })
      .filter(Boolean);
  }

  get total() {
    return this.items.reduce((sum, i) => sum + i.subtotal, 0);
  }

  get count() {
    return this.#items.reduce((sum, i) => sum + i.qty, 0);
  }

  get isEmpty() {
    return this.#items.length === 0;
  }

  // ---------- Փոփոխելու համար ----------
  add(id) {
    const existing = this.#items.find(i => i.id === id);

    if (existing) {
      existing.qty++;
    } else {
      this.#items.push({ id, qty: 1 });
    }

    this.#save();
    return this;                 // թույլ է տալիս շղթա գրել
  }

  remove(id) {
    this.#items = this.#items.filter(i => i.id !== id);
    this.#save();
    return this;
  }

  changeQty(id, delta) {
    const item = this.#items.find(i => i.id === id);
    if (!item) return this;

    item.qty += delta;
    if (item.qty <= 0) return this.remove(id);

    this.#save();
    return this;
  }

  clear() {
    this.#items = [];
    this.#save();
    return this;
  }

  // ---------- Ներքին ----------
  #save() {
    save(this.#storageKey, this.#items);
  }
}
