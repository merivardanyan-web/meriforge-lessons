Փուչիկացում և դելեգացիա

Անցյալ դասի վերջին տնայինը չաշխատեց՝ render-ից հետո կոճակները մեռան։ Այսօր հասկանում ենք՝ ինչու, և սովորում ենք այն հնարքը, որով մեկ listener-ը կարող է սպասարկել հազար կոճակ։

Նախապայման՝ Դաս 09–11 Բլոկ 2 · DOM և իրադարձություններ Դասարանում՝ ամբողջական to-do

innerHTML = ...-ը ջնջում է հին էլեմենտները և ստեղծում բոլորովին նորերը։ Հին կոճակների վրա կախված մշակիչները ջնջվեցին նրանց հետ միասին։ Նոր կոճակները տեսքով նույնն են, բայց դրանք ուրիշ օբյեկտներ են՝ առանց մշակիչների։

Կարելի է ամեն render-ից հետո նորից կախել բոլոր մշակիչները։ Բայց դա անարդյունավետ է․ 100 ապրանք = 100 listener, և ամեն ֆիլտրից հետո՝ նորից 100։ Կա շատ ավելի լավ լուծում։

03stopPropagation

Փուչիկացումը կարելի է կանգնեցնել։

կանգնեցնում ենք
document.querySelector(".btn").addEventListener("click", (e) => {
  e.stopPropagation();
  console.log("1. btn — և վերջ");
});
Console  (սեղմեցինք կոճակին)
1. btn — և վերջ
Զգուշությամբ

stopPropagation-ը հաճախ սկսնակների «համընդհանուր դեղամիջոցն» է։ Բայց այն կարող է կոտրել այլ ֆունկցիոնալություն՝ մոդալի փակումը, վերլուծական սկրիպտները, դելեգացիան։ Առաջին լուծումը չպիտի լինի stopPropagation։ Նախ մտածիր՝ կարո՞ղ ես խնդիրը լուծել e.target-ի ստուգումով։

04Դելեգացիա

Գաղափարը՝ listener-ը կախում ենք ծնողի վրա, որը միշտ կա էջում։ Երբ սեղմում են երեխաներից որևէ մեկին, իրադարձությունը բարձրանում է ծնողի մոտ, և մենք e.target-ով պարզում ենք՝ ում սեղմեցին։

հին ձև — 100 listener
document.querySelectorAll(".btn").forEach(btn => {
  btn.addEventListener("click", () => { ... });
});
// ❌ չի աշխատում նոր էլեմենտների վրա
// ❌ ամեն render-ից հետո պիտի կրկնել

const container = document.querySelector("#products");

container.addEventListener("click", (e) => {
  const btn = e.target.closest(".btn");

  if (!btn) return;              // սեղմեցին ուրիշ տեղ — անտեսում ենք

05Դասարանական · To-do ցուցակ

Ամբողջական հավելված՝ ավելացում, նշում, խմբագրում, ջնջում։ Ամեն ինչ՝ մեկ click listener-ով։

main.js — տվյալներ և render
let todos = [
  { id: 1, text: "Տնային գրել", done: false },
  { id: 2, text: "Հաց գնել",    done: true  }
];

const form  = document.querySelector(".todo__form");
const input = document.querySelector(".todo__input");
const list  = document.querySelector(".todo__list");
const stats = document.querySelector(".todo__stats");

function createItem({ id, text, done }) {
  return `
    <li class="todo__item ${done ? "is-done" : ""}" data-id="${id}">
      <input type="checkbox" data-action="toggle" ${done ? "checked" : ""}>
      <span class="todo__text">${text}</span>
      <button data-action="edit">✎</button>
      <button data-action="delete">✕</button>
    </li>
  `;
}

function render() {
  if (todos.length === 0) {
    list.innerHTML = `<li class="todo__empty">Ցուցակը դատարկ է։</li>`;
  } else {
    list.innerHTML = todos.map(createItem).join("");
  }

  const done = todos.filter(t => t.done).length;
  stats.textContent = `${done} / ${todos.length} կատարված`;
}
main.js — գործողություններ
function addTodo(text) {
  todos.push({
    id: Date.now(),        // պարզ ձև՝ եզակի id ստանալու
    text,                    // կարճ գրելաձև՝ text: text
    done: false
  });
  render();
}

function deleteTodo(id) {
  todos = todos.filter(t => t.id !== id);   // Դաս 6
  render();
}

function toggleTodo(id) {
  const todo = todos.find(t => t.id === id);
  if (todo) todo.done = !todo.done;
  render();
}

function editTodo(id) {
  const todo = todos.find(t => t.id === id);
  if (!todo) return;

  const newText = prompt("Խմբագրել գործը՝", todo.text);
  if (newText === null) return;          // սեղմեցին Cancel

  const trimmed = newText.trim();          // Դաս 8
  if (trimmed === "") return;

  todo.text = trimmed;
  render();
}
main.js — մեկ listener բոլորի համար
// Ֆորմա
form.addEventListener("submit", (e) => {
  e.preventDefault();                      // Դաս 11

  const text = input.value.trim();
  if (text === "") return;

  addTodo(text);
  input.value = "";
  input.focus();
});

// ★ Դելեգացիա — ամբողջ ցուցակի համար մեկ մշակիչ
list.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-action]");
  if (!btn) return;

  const item = btn.closest(".todo__item");
  const id = Number(item.dataset.id);       // dataset-ը միշտ տող է

  switch (btn.dataset.action) {
    case "toggle": toggleTodo(id); break;
    case "edit":   editTodo(id);   break;
    case "delete": deleteTodo(id); break;
  }
});

render();
Ուշադրություն ճարտարապետությանը

Կա մեկ ճշմարտության աղբյուր՝ todos զանգվածը։ Ամեն գործողություն փոխում է զանգվածը և կանչում render()-ը։ DOM-ը երբեք ուղիղ չենք խմբագրում։

Սա հենց այն սկզբունքն է, որի վրա կառուցված են React-ը, Vue-ն և բոլոր ժամանակակից ֆրեյմվորքերը՝ տվյալները փոխվում են → ինտերֆեյսը վերագծվում է։

Կարևոր մանրուք

item.dataset.id-ն տող է, իսկ todo.id-ն՝ թիվ։ Առանց Number()-ի t.id === id համեմատությունը միշտ false կլիներ (Դաս 2՝ ===)։ Սա ամենահաճախ հանդիպող սխալն է այս խնդրում։

07Տնային աշխատանք

  1. Կառուցիր ներդրված երեք բլոկ և ամեն մեկին կախիր մշակիչ։ Սեղմիր ամենաներքինին և գրիր մեկնաբանության մեջ՝ ինչ հերթականությամբ աշխատեցին։
  2. Նույն կոդում ավելացրու stopPropagation միջինին։ Նկարագրիր՝ ինչ փոխվեց։
  3. Ցուցակ 5 էլեմենտով։ Մեկ դելեգացված listener-ով՝ սեղմած էլեմենտին ավելացրու is-selected դաս, մյուսներից հանիր։
  4. Նույն ցուցակին ավելացրու «Ավելացնել» կոճակ։ Համոզվիր, որ նոր էլեմենտների վրա էլ ընտրությունն աշխատում է։
  5. Ամբողջական to-do ցուցակ դասարանական օրինակի հիման վրա։ Գրիր զրոյից՝ առանց նայելու, հետո համեմատիր։
  6. To-do-ին ավելացրու ֆիլտրի կոճակներ՝ «Բոլորը», «Ակտիվ», «Կատարված»։ Ֆիլտրը պահիր փոփոխականում և հաշվի առ render()-ում։
  7. To-do-ին ավելացրու «Ջնջել կատարվածները» կոճակ։
  8. Ավելացրու հաստատում ջնջելուց առաջ՝ confirm("Վստա՞հ եք")։
  9. Վերցրու Դաս 11-ի 10-րդ տնայինը (կատալոգի ֆիլտրերը) և վերաշարադրիր դելեգացիայով։ Համոզվիր, որ «Գնել» կոճակներն այժմ աշխատում են ֆիլտրից հետո։
  10. Կատալոգին ավելացրու զամբյուղ․ առանձին զանգված, «Գնել» սեղմելիս ապրանքն ավելանում է, էկրանին ցուցադրվում է քանակը և ընդհանուր գումարը։
