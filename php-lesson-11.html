function createProduct(array $data): int
{
    $stmt = db()->prepare("
        INSERT INTO products
            (category_id, name, slug, specs, description,
             price, discount, stock, image, is_active)
        VALUES
            (:category_id, :name, :slug, :specs, :description,

        'slug'        => $data['slug'],

        'stock'       => $data['stock'],

}

function slugify(string $text): string
{
    // Հայերենը → լատինատառ
    $map = [
        'ա'=>'a', 'բ'=>'b', 'գ'=>'g', 'դ'=>'d', 'ե'=>'e', 'զ'=>'z',
        'է'=>'e', 'ը'=>'y', 'թ'=>'t', 'ժ'=>'zh', 'ի'=>'i', 'լ'=>'l',
        'խ'=>'kh', 'ծ'=>'ts', 'կ'=>'k', 'հ'=>'h', 'ձ'=>'dz', 'ղ'=>'gh',
        'ճ'=>'ch', 'մ'=>'m', 'յ'=>'y', 'ն'=>'n', 'շ'=>'sh', 'ո'=>'o',
        'չ'=>'ch', 'պ'=>'p', 'ջ'=>'j', 'ռ'=>'r', 'ս'=>'s', 'վ'=>'v',
        'տ'=>'t', 'ր'=>'r', 'ց'=>'ts', 'ւ'=>'w', 'փ'=>'p', 'ք'=>'q',
        'օ'=>'o', 'ֆ'=>'f', 'և'=>'ev',
        'ու'=>'u',   // ⭐ երկու տառ՝ մեկ ձայն։ strtr-ը նախ փորձում է ամենաերկար բանալին
    ];

    $text = mb_strtolower(trim($text));
    $text = strtr($text, $map);
    $text = preg_replace('/[^a-z0-9]+/', '-', $text);

    return trim($text, '-') ?: 'item';
}

/** Ապահովում է եզակիությունը */
function uniqueSlug(string $base, ?int $exceptId = null): string
{
    $slug = $base;
    $i = 1;

    while (true) {
        $sql = "SELECT COUNT(*) FROM products WHERE slug = :slug";
        $params = ['slug' => $slug];

        if ($exceptId !== null) {
            $sql .= " AND id != :id";
            $params['id'] = $exceptId;
        }

        $stmt = db()->prepare($sql);
        $stmt->execute($params);

        if ((int) $stmt->fetchColumn() === 0) {
            return $slug;
        }

        $slug = $base . '-' . (++$i);
    }
}

rowCount()-ի թակարդը

$stmt->rowCount()-ը վերադարձնում է փոփոխված տողերի քանակը, ոչ գտնվածների։

Եթե օգտագործողը սեղմեց «Պահպանել», բայց ոչինչ չփոխեց — rowCount()-ը կլինի 0, թեև ամեն ինչ կարգին է։

Ուստի «հաջողվե՞ց» հարցի պատասխանը execute()-ի արդյունքն է, ոչ rowCount()-ը։

05Տրանզակցիաներ

Պատվերի ստեղծում՝ երեք գործողություն 1. INSERT INTO orders → order_id 2. INSERT INTO order_items → ամեն ապրանքի համար 3. UPDATE products SET stock → պահեստի նվազում Ի՞նչ կլինի, եթե 2-րդից հետո սերվերն ընկնի։ → Պատվերը կա, բայց դատարկ է → Պահեստը չի նվազել → Տվյալները անհամապատասխան են
ամեն ինչ կամ ոչինչ
function createOrder(array $customer, array $items): int
{
    $pdo = db();
    $pdo->beginTransaction();

    try {
        // 1. Պատվերը
        $total = array_sum(array_column($items, 'sum'));

        $stmt = $pdo->prepare("
            INSERT INTO orders (customer_name, email, phone, total, status)
            VALUES (:name, :email, :phone, :total, 'new')
        ");

        $stmt->execute([
            'name'  => $customer['name'],
            'email' => $customer['email'],
            'phone' => $customer['phone'],
            'total' => $total,
        ]);

        $orderId = (int) $pdo->lastInsertId();

        // 2. Պատվերի տողերը
        $itemStmt = $pdo->prepare("
            INSERT INTO order_items (order_id, product_id, quantity, price)
            VALUES (:order_id, :product_id, :quantity, :price)
        ");

        $stockStmt = $pdo->prepare("
            UPDATE products SET stock = stock - :qty
            WHERE id = :id AND stock >= :qty
        ");

        foreach ($items as $item) {
            $itemStmt->execute([
                'order_id'   => $orderId,
                'product_id' => $item['product']['id'],
                'quantity'   => $item['qty'],
                'price'      => $item['product']['final_price'],
            ]);

            // 3. Պահեստ
            $stockStmt->execute([
                'id'  => $item['product']['id'],
                'qty' => $item['qty'],
            ]);

            if ($stockStmt->rowCount() === 0) {
                throw new RuntimeException(
                    'Ապրանքը սպառվել է՝ ' . $item['product']['name']
                );
            }
        }

        $pdo->commit();

        return $orderId;

    } catch (Throwable $e) {
        $pdo->rollBack();      // ⭐ ամեն ինչ հետ է գլորվում
        throw $e;
    }
}
Երկու նրբություն այս կոդում

1. WHERE id = :id AND stock >= :qty — պահեստի ստուգումը կատարվում է հենց հրամանի մեջ։ Առանձին SELECT-ով ստուգելը վտանգավոր է․ երկու օգտագործող կարող են միաժամանակ տեսնել «մնացել է 1» և երկուսն էլ գնել։

2. Այստեղ rowCount() === 0-ը ճիշտ ցուցիչ է — եթե ոչ մի տող չփոխվեց, նշանակում է պայմանը չբավարարվեց, այսինքն՝ պահեստը բավարար չէր։

Համեմատիր բաժին 03-ի հետ, որտեղ rowCount()-ը սխալ ցուցիչ էր։ Համատեքստը որոշում է։

Տրանզակցիաները միայն InnoDB-ում

MyISAM շարժիչում beginTransaction()-ը լուռ ոչինչ չի անի։ Կոդը կաշխատի, բայց պաշտպանություն չի լինի։

Դա պատճառներից մեկն է, որ Դաս 9-ում աղյուսակները ստեղծեցինք ENGINE=InnoDB-ով։

function validateProduct(array $data): array
{
    $errors = [];

    $name = trim($data['name'] ?? '');
    if ($name === '') {
        $errors['name'] = 'Անվանումը պարտադիր է';
    } elseif (mb_strlen($name) > 200) {
        $errors['name'] = 'Առավելագույնը 200 նիշ';
    }

    $price = filter_var($data['price'] ?? '', FILTER_VALIDATE_INT, [
        'options' => ['min_range' => 1, 'max_range' => 99999999]
    ]);
    if ($price === false) {
        $errors['price'] = 'Գինը պիտի լինի դրական թիվ';
    }

    $discount = filter_var($data['discount'] ?? '0', FILTER_VALIDATE_INT, [
        'options' => ['min_range' => 0, 'max_range' => 90]
    ]);
    if ($discount === false) {
        $errors['discount'] = 'Զեղչը՝ 0-ից 90';
    }

    $stock = filter_var($data['stock'] ?? '0', FILTER_VALIDATE_INT, [
        'options' => ['min_range' => 0]
    ]);
    if ($stock === false) {
        $errors['stock'] = 'Քանակը չի կարող բացասական լինել';
    }

    // Կատեգորիան՝ իրապես գոյություն ունեցողներից ⭐
    $catId = $data['category_id'] ?? '';
    if ($catId !== '') {
        $ids = array_column(getCategories(), 'id');
        if (!in_array((int) $catId, array_map('intval', $ids), true)) {
            $errors['category_id'] = 'Անհայտ կատեգորիա';
        }
    }

    return $errors;
}

<select>-ը ցույց է տալիս միայն վավեր տարբերակներ։ Բայց Դաս 6-ի դասը հիշո՞ւմ ես — curl-ը <select> չի տեսնում։

Առանց ստուգման՝ category_id=99999 կխախտի արտաքին բանալին և կստանաս ֆատալ սխալ։ Կամ, եթե բանալի չլիներ, բազայում կհայտնվեր անհամապատասխան տվյալ։

Կանոն՝ ամեն ընտրովի արժեք ստուգվում է աղբյուրի դեմ, ոչ միայն ձևաչափի։

Ինչո՞ւ մեկ ֆայլ ստեղծելու և խմբագրելու համար

Ֆորման նույնն է։ Վալիդացիան նույնն է։ Տարբերությունը երկու տողում է՝ createProduct թե updateProduct։

Երկու առանձին ֆայլ նշանակում է 90% կրկնվող կոդ և երկու տեղ, որտեղ պիտի ուղղես ամեն փոփոխություն։

Laravel-ում այս երկուսը սովորաբար բաժանվում են (store և update), բայց ֆորման մնում է մեկ շաբլոն։ Դաս 21-ում կտեսնես տարբերությունը։

getProductForAdmin — առանձին ֆունկցիա
/** Ադմինի համար՝ ԱՌԱՆՑ is_active ֆիլտրի */
function getProductForAdmin(int $id): ?array
{
    $stmt = db()->prepare("SELECT * FROM products WHERE id = :id");
    $stmt->execute(['id' => $id]);

    return $stmt->fetch() ?: null;
}

/** Ադմինի ցուցակ՝ նաև անջատվածները */
function getAllProducts(): array
{
    return db()->query("
        SELECT p.id, p.name, p.price, p.discount,
               p.stock, p.is_active,
               c.name AS category_name
        FROM products p
        LEFT JOIN categories c ON c.id = p.category_id
        ORDER BY p.created_at DESC
    ")->fetchAll();
}
Ինչո՞ւ առանձին ֆունկցիա ադմինի համար

getProduct()-ը ունի AND is_active = 1։ Դա ճիշտ է հանրային էջի համար։

Բայց ադմինը պիտի կարողանա խմբագրել անջատված ապրանքը — հակառակ դեպքում այն կդառնա անհասանելի ընդմիշտ։

Գայթակղություն կա ավելացնել պարամետր՝ getProduct($id, $includeInactive = true)։ Դա վտանգավոր է․ մի օր ինչ-որ մեկը կփոխանցի true հանրային էջում։ Առանձին ֆունկցիան ավելի ապահով է։

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Ադմին և «Կարդացի այսօր»», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

Նախագիծ · Առաջադրանք 1

Պատվերը տրանզակցիայով և ադմին CRUD-ը փակում են Առաջադրանք 1-ը։ Անցիր ընդունման ստուգաթերթով, հետո բացիր օրինակը։

  1. Կատարիր դասարանականի 10 քայլը և չորս փորձարկումը։
  2. Ավելացրու նկարի վերբեռնում ադմին ֆորմայում (Դաս 7-ի uploadImage())։
  3. Իրագործիր createOrder()-ը տրանզակցիայով։
  4. Փորձարկիր տրանզակցիան․ դիտմամբ գցիր բացառություն երկրորդ INSERT-ից հետո։ Պատվերը մնա՞ց բազայում։
  5. Փոխիր աղյուսակի շարժիչը MyISAM-ի և կրկնիր փորձը։ Ի՞նչ փոխվեց։
  6. Ցուցադրիր UPDATE-ը առանց WHERE-ի փորձնական աղյուսակի վրա։
  7. Ցուցադրիր rowCount()-ի թակարդը՝ պահպանիր առանց փոփոխության։
  8. Ավելացրու ապրանքների որոնում ադմին ցուցակում։
  9. Ավելացրու զանգվածային գործողություն՝ նշել մի քանիսը և արխիվացնել միանգամից։
  10. Խնդրիր AI-ին վերանայել CRUD ֆունկցիաները։ Ի՞նչ եզրային դեպքեր գտավ, որ բաց էիր թողել։

09Հիշելու թերթիկ

ԳրառումԻնչ է անում
INSERT INTO ... VALUESՍտեղծել
lastInsertId()Նոր ID ⭐
UPDATE ... SET ... WHEREWHERE-ը պարտադիր ⭐
DELETE FROM ... WHEREՋնջել
SET is_active = 0Փափուկ ջնջում ⭐
beginTransaction()Սկսել
commit()Հաստատել
rollBack()Հետ գլորել ⭐
rowCount()Փոփոխված տողեր, ոչ գտնված

CRUD-ի կոնտրոլերի օրինաչափությունը

1. Կարդալ պարամետրերը 2. Եթե POST չէ → ցուցադրել ֆորման 3. Վալիդացիա 4. Եթե սխալ կա → $old = $_POST → ֆորմա 5. Գործողություն բազայում 6. flash() 7. header() + exit

Ստուգաթերթ

  • Բոլոր փոփոխող գործողությունները POST են
  • Ամեն UPDATE և DELETE ունի WHERE
  • Ընտրովի արժեքները ստուգվում են աղբյուրի դեմ
  • Slug-ը եզակի է և չի աճում խմբագրելիս
  • Կապված գործողությունները տրանզակցիայում են
  • Օգտագործվում է փափուկ ջնջում
  • Ադմինի հարցումները չեն ֆիլտրում is_active-ով
  • Հաջողությունից հետո՝ redirect

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու ջնջումը POST է
  • Գիտե՞մ, ինչու փափուկ ջնջում
  • Հասկանո՞ւմ եմ, ե՞րբ է պետք տրանզակցիա
  • Կարո՞ղ եմ բացատրել rowCount()-ի երկակի բնույթը
  • Գիտե՞մ, ինչու ադմինի համար առանձին հարցումներ են
Հաջորդ դասին

OOP՝ դասեր և օբյեկտներ։ products.php-ը արդեն 200 տող է և կաճի։ Ամեն ֆունկցիա սկսվում է db()-ով, ամեն մեկը կրկնում է նույն կառուցվածքը։

Դասերը թույլ կտան այս ամենը հավաքել մեկ տեղում՝ վիճակի հետ միասին։ Եվ վերջապես կլուծվի Դաս 3-ի խնդիրը՝ ֆունկցիաների գլոբալ վիճակի բացակայությունը։
