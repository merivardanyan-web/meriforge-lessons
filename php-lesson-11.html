class Product
{
    public string $name;
    public int $price;
    public int $discount;

    public function __construct(string $name, int $price, int $discount = 0)
    {
        $this->name     = $name;
        $this->price    = $price;
        $this->discount = $discount;
    }

    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }
}

// Օգտագործում

class Cart
{
    private array $items = [];      // ⭐ ներսից միայն

    public function add(int $id, int $qty = 1): void
    {
        if ($qty < 1) {
            return;
        }

        $current = $this->items[$id] ?? 0;
        $this->items[$id] = min($current + $qty, 99);
    }

    public function all(): array
    {
        return $this->items;
    }
}

$cart = new Cart();
$cart->add(5, 3);        // ✅ սահմանները պահպանվում են
$cart->items[5] = 99999; // ❌ Fatal error — private է
Ինկապսուլյացիա — մեկ նախադասությամբ

private-ը երաշխավորում է, որ կանոնները չեն շրջանցվի։

Դաս 7-ում cartAdd()-ում գրեցինք min($qty, 99)։ Բայց ոչինչ չէր խանգարում ուրիշ տեղում գրել $_SESSION['cart'][5] = 99999 և շրջանցել այն։

Հիմա միակ մուտքը add()-ն է։ Կանոնը մեկ տեղում է և չի կարող շրջանցվել — ոչ պատահաբար, ոչ դիտմամբ։

Կանոն՝ հատկությունները լռելյայն private։ public դարձրու միայն այն, ինչին իսկապես պետք է դրսից դիմել։

05Product որպես տիպ

src/Product.php
<?php
declare(strict_types=1);

class Product
{
    public function __construct(
        public readonly int $id,
        public readonly string $name,
        public readonly string $slug,
        public readonly int $price,
        public readonly int $discount = 0,
        public readonly int $stock = 0,
        public readonly ?string $specs = null,
        public readonly ?string $image = null,
        public readonly ?string $categoryName = null,
    ) {}

    // ---------- Ֆաբրիկա ----------

    public static function fromRow(array $row): self
    {
        return new self(
            id:           (int) $row['id'],
            name:         (string) $row['name'],
            slug:         (string) $row['slug'],
            price:        (int) $row['price'],
            discount:     (int) ($row['discount'] ?? 0),
            stock:        (int) ($row['stock'] ?? 0),
            specs:        $row['specs'] ?? null,
            image:        $row['image'] ?? null,
            categoryName: $row['category_name'] ?? null,
        );
    }

    // ---------- Տրամաբանություն ----------

    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }

    public function hasDiscount(): bool
    {
        return $this->discount > 0;
    }

    public function inStock(): bool
    {
        return $this->stock > 0;
    }

    public function isLowStock(): bool
    {
        return $this->stock > 0 && $this->stock <= 3;
    }

    public function savings(): int
    {
        return $this->price - $this->finalPrice();
    }

    // ---------- Ներկայացում ----------

    public function formattedPrice(): string
    {
        return number_format($this->finalPrice(), 0, '.', "\u{00A0}") . "\u{00A0}֏";
    }

    public function url(): string
    {
        return '/product/' . $this->slug;
    }

    public function imageUrl(): string
    {
        return '/images/' . ($this->image ?? 'placeholder.jpg');
    }
}

readonly — PHP 8.1

Հատկությունը կարելի է վերագրել միայն մեկ անգամ՝ կոնստրուկտորում։ Դրանից հետո՝ ֆատալ սխալ։

Ինչո՞ւ դա լավ է․ բազայից բեռնված ապրանքը պատկեր է տվյալների։ Եթե ինչ-որ տեղ պատահաբար փոխես $product->price-ը, բազայում ոչինչ չի փոխվի, բայց էջում կերևա այլ գին։ Դա գտնելը շատ դժվար է։

readonly-ն դարձնում է այդ սխալն անհնար։ Գինը փոխելու համար կա ProductRepository::update() — բացահայտ և տեսանելի ճանապարհ։

static և self

public static function fromRow() — մեթոդ, որը կանչվում է դասի վրա, ոչ օբյեկտի՝ Product::fromRow($row)։

Այն օբյեկտ դեռ չունի, ուստի $this չկա։ Փոխարենը՝ self, որը նշանակում է «այս դասը»։

Այս օրինաչափությունը կոչվում է ֆաբրիկա մեթոդ․ բազայի տողից ստանում ենք օբյեկտ, և բոլոր փոխարկումները հավաքված են մեկ տեղում։ Հիշո՞ւմ ես Դաս 9-ի DECIMAL-ի թակարդը։ Հիմա (int)-ն այստեղ է՝ մեկ անգամ, ոչ ամեն օգտագործման տեղում։

class Cart
{
    private const MAX_QTY = 99;

    public function __construct(private array &$storage) {}

    public function add(int $id, int $qty = 1): void
    {
        if ($qty < 1) {
            return;
        }

        $current = $this->storage[$id] ?? 0;
        $this->storage[$id] = min($current + $qty, self::MAX_QTY);
    }

    public function remove(int $id): void
    {
        unset($this->storage[$id]);
    }

    public function count(): int
    {
        return array_sum($this->storage);
    }

    public function isEmpty(): bool
    {
        return $this->storage === [];
    }

    public function ids(): array
    {
        return array_map('intval', array_keys($this->storage));
    }

    public function qtyOf(int $id): int
    {
        return $this->storage[$id] ?? 0;
    }
}

$cart = new Cart($_SESSION['cart']);
&$storage — հղում

Ամպերսանդը նշանակում է, որ Cart-ը աշխատում է ուղիղ $_SESSION['cart']-ի հետ, ոչ պատճենի։

Առանց դրա $cart->add(5)-ը կփոխեր միայն ներքին պատճենը, և սեսիայում ոչինչ չէր պահվի։

Դա Դաս 3-ի հղումային foreach-ի նույն մեխանիզմն է — բայց այստեղ այն ճիշտ օգտագործում է, որովհետև հենց դա է նպատակը։

Դաս 15-ում կտեսնենք ավելի մաքուր տարբերակ՝ առանձին SessionStorage դասով։

  1. Ստեղծիր src/ թղթապանակը
  2. Գրիր src/Product.php-ը
  3. Միացրու այն bootstrap.php-ում require_once-ով
  4. Փոխիր getProducts-ը և getProduct-ը՝ վերադարձնեն օբյեկտներ
  5. Թարմացրու շաբլոնները
  6. Գրիր src/Cart.php-ը
  7. Փոխարինիր cart.php-ի ֆունկցիաները
  8. Ավելացրու isLowStock()-ի ցուցադրումը քարտում
  9. Փորձիր փոխել $product->price-ը — տես readonly-ի սխալը
  10. Փորձիր $cart->storage[5] = 999 — տես private-ի սխալը

08Տնային աշխատանք

  1. Կատարիր դասարանականի 10 քայլը։
  2. Ավելացրու Product-ին երեք նոր մեթոդ՝ discountLabel(), isNew() (30 օրից պակաս), shortName() (40 նիշից կտրած)։
  3. Ստեղծիր Category դասը fromRow()-ով։
  4. Ցուցադրիր տարբերությունը՝ $p['prise'] և $p->prise։ Ի՞նչ է տալիս ամեն մեկը։
  5. Ջնջիր readonly-ն և ցուցադրիր, ինչու է այն օգտակար՝ փոխելով գինը շաբլոնում։
  6. Ցուցադրիր, ինչ կլինի, եթե Cart-ի կոնստրուկտորից հանես &-ը։
  7. Ավելացրու Cart::setQty() մեթոդ վալիդացիայով։
  8. Համեմատիր հին cart.php-ը և նոր Cart դասը։ Քանի՞ տող, քանի՞ գլոբալ ֆունկցիա։
  9. Գրիր CartItem դաս, որը պահում է Product-ը և քանակը, և ունի sum() մեթոդ։
  10. Խնդրիր AI-ին վերանայել Product-ը։ Ի՞նչ մեթոդ առաջարկեց, որ չէիր մտածել։

Ստուգաթերթ

  • Հատկությունները լռելյայն private կամ readonly են
  • Կա fromRow() ֆաբրիկա
  • Բոլոր տիպերի փոխարկումները fromRow()-ում են
  • Տրամաբանությունը մեթոդներում է, ոչ շաբլոնում
  • Մեթոդների անունները նկարագրում են իմաստը, ոչ իրագործումը
  • e()-ն դեռ կիրառվում է
  • Ամեն դաս առանձին ֆայլում է

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել դասի և օբյեկտի տարբերությունը
  • Գիտե՞մ, ինչու private-ը կարևոր է
  • Հասկանո՞ւմ եմ, ի՞նչ է անում readonly-ն
  • Կարո՞ղ եմ բացատրել, ինչո՞ւ static-ում չկա $this
  • Գիտե՞մ, ինչու ֆաբրիկա մեթոդը օգտակար է
Հաջորդ դասին

Ժառանգում, interface և abstract։ Կունենանք երկու տեսակի ապրանք՝ ֆիզիկական և թվային, և կտեսնենք, ինչպես կիսել ընդհանուր կոդը՝ առանց կրկնության։

Կսովորենք նաև ամենակարևոր հարցի պատասխանը՝ ե՞րբ ժառանգում, ե՞րբ ինտերֆեյս, ե՞րբ պարզապես առանձին դաս։
