01Ինչպես է աշխատում սեսիան

Առաջին այցելություն Բրաուզեր ──── GET / ────────────▶ Սերվեր │ session_start() Ստեղծում է ID՝ a3f9k2... Գլխավոր միտքը

Տվյալները մնում են սերվերում։ Բրաուզերը կրում է միայն բանալին։

Դա կարևոր տարբերություն է cookie-ից, որտեղ տվյալները ֆիզիկապես գտնվում են օգտագործողի մոտ և կարող են խմբագրվել։

Ուստի սեսիայում կարելի է պահել վստահելի տվյալ՝ «այս օգտագործողը մուտք է գործել որպես ադմին»։ Cookie-ում նման բան պահելը կլիներ խոցելիություն։

հիմնական օգտագործում
session_start();          // ամեն ֆայլում, ելքից ԱՌԱՋ ⭐

$_SESSION['user_id'] = 42;
$_SESSION['cart'] = [];

echo $_SESSION['user_id'] ?? 'հյուր';

unset($_SESSION['cart']);   // ջնջել մեկ բանալի

// Ամբողջովին ավարտել (Դաս 16՝ ելք)
$_SESSION = [];
session_destroy();
session_start()-ը ելքից առաջ

Այն ուղարկում է Set-Cookie վերնագիր։ Իսկ վերնագրերը չեն կարող գնալ բովանդակությունից հետո (Դաս 6)։

Ուստի՝ ֆայլի ամենասկզբում, declare(strict_types=1)-ից անմիջապես հետո։ Ոչ մի բացատ, ոչ մի HTML դրանից առաջ։

Անվտանգ կարգավորումներ

bootstrap.php — մեկ անգամ ամբողջ նախագծի համար
session_start([
    'cookie_httponly' => true,    // JS-ը չի կարդում ⭐
    'cookie_samesite' => 'Lax',   // CSRF-ի դեմ (Դաս 16)
    'cookie_secure'   => false,   // true — HTTPS-ի դեպքում
    'use_strict_mode' => true,    // չի ընդունում կողմնակի ID ⭐
]);
ԿարգավորումԻնչից է պաշտպանում
cookie_httponlyXSS-ը չի կարող գողանալ սեսիան ⭐
cookie_samesiteCSRF՝ ուրիշ կայքից հարցում
cookie_secureԳաղտնալսում HTTP-ով
use_strict_modeSession fixation
httponly-ը կապում է Դաս 5-ը այսօրվա հետ

Հիշո՞ւմ ես XSS-ի հարձակումը՝ document.cookie-ի գողությունը։

cookie_httponly-ով JavaScript-ը ընդհանրապես չի տեսնում PHPSESSID-ն։ Նույնիսկ եթե XSS-ը հաջողվի, սեսիան չի գողացվի։

Դա շերտավոր պաշտպանության օրինակ է․ e()-ն կանխում է XSS-ը, httponly-ն սահմանափակում է վնասը, եթե e()-ն ինչ-որ տեղ բաց թողնվի։ Իրական համակարգերում միշտ պիտի լինի երկրորդ շերտ։

Cookie-ն դրվում է միայն հաջորդ հարցումից

Դա տրամաբանական է․ cookie-ն ուղարկվում է բրաուզերին պատասխանի հետ, և վերադառնում է միայն հաջորդ հարցումով։

03Flash-հաղորդագրություններ

Դաս 6-ում օգտագործեցինք առանց բացատրության։ Հիմա դարձնենք ֆունկցիա։

functions.php
function flash(string $key, ?string $message = null): ?string
{
    // Գրելու ռեժիմ
    if ($message !== null) {
        $_SESSION['_flash'][$key] = $message;
        return null;
    }

    // Կարդալու ռեժիմ — կարդում ենք ԵՎ ջնջում
    $value = $_SESSION['_flash'][$key] ?? null;
    unset($_SESSION['_flash'][$key]);

    return $value;
}
օգտագործում
// Կոնտրոլերում՝ redirect-ից առաջ
flash('success', 'Հաղորդագրությունն ուղարկված է');
header('Location: /contact.php', true, 303);
exit;

// Շաբլոնում
<?php if ($msg = flash('success')): ?>
    <div class="alert alert--success" role="status">
        <?= e($msg) ?>
    </div>
<?php endif; ?>
«Կարդալ և ջնջել» — ահա ամբողջ գաղափարը

Flash-հաղորդագրությունն ապրում է ուղիղ մեկ ցուցադրում։

Առանց unset-ի «Հաղորդագրությունն ուղարկված է» գրությունը կերևար ամեն էջում մինչև սեսիայի ավարտը։

Laravel-ում կա ճիշտ նույն մեխանիզմը՝ session()->flash()։ Դաս 21-ում կտեսնես, որ սկզբունքը նույնն է։

04Զամբյուղ

cart-functions.php — տրամաբանությունը
<?php
declare(strict_types=1);

/** Զամբյուղը՝ [id => քանակ] */
function cart(): array
{
    return $_SESSION['cart'] ?? [];
}

function cartAdd(int $id, int $qty = 1): void
{
    if ($qty < 1) {
        return;
    }

    $current = $_SESSION['cart'][$id] ?? 0;
    $_SESSION['cart'][$id] = min($current + $qty, 99);   // վերին սահման ⭐
}

function cartRemove(int $id): void
{
    unset($_SESSION['cart'][$id]);
}

function cartCount(): int
{
    return array_sum(cart());
}

/** Զամբյուղը + ապրանքների տվյալները */
function cartItems(array $products): array
{
    $items = [];

    foreach (cart() as $id => $qty) {
        $product = findProduct($products, (int) $id);

        if ($product === null) {
            continue;          // ապրանքը ջնջվել է ⭐
        }

        $items[] = [
            'product' => $product,
            'qty'     => $qty,
            'sum'     => finalPrice($product) * $qty,
        ];
    }

    return $items;
}

function cartTotal(array $products): int
{
    return array_sum(array_column(cartItems($products), 'sum'));
}
Ինչո՞ւ պահում ենք միայն ID-ն և քանակը

Գայթակղիչ է սեսիայում պահել ամբողջ ապրանքը՝ անունը, գինը, նկարը։ Դա սխալ է։

Եթե գինը փոխվի, օգտագործողի զամբյուղում կմնա հին գինը՝ երբեմն շաբաթներ։ Իսկ եթե օգտագործողը հասնի սեսիայի տվյալներին (կամ դրանք տեղափոխվեն cookie), գինը կդառնա խմբագրելի։

Կանոն՝ սեսիայում պահվում է հղումը, ոչ տվյալները։ Իրական արժեքները միշտ վերցվում են աղբյուրից։

Երկու պաշտպանություն այս կոդում

min($current + $qty, 99) — առանց դրա հարձակվողը կուղարկի qty=999999999, և գումարը կհոսի (integer overflow) կամ էջը կդանդաղի։

if ($product === null) continue; — ապրանքը կարող էր ջնջվել այն բանից հետո, երբ ընկել է զամբյուղ։ Առանց ստուգման՝ ֆատալ սխալ։

Երկուսն էլ եզրային դեպքեր են, որոնց մասին Դաս 4-ում խոսեցինք․ ամեն կոդ փորձարկվում է դատարկով, բացասականով, շատ մեծով։

cart-add.php — կոնտրոլեր
<?php
declare(strict_types=1);

require_once __DIR__ . '/bootstrap.php';

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    header('Location: /', true, 303);
    exit;
}

$id = filter_var($_POST['id'] ?? '', FILTER_VALIDATE_INT);

if ($id === false || findProduct($products, $id) === null) {
    flash('error', 'Ապրանքը չի գտնվել');
} else {
    cartAdd($id);
    flash('success', 'Ավելացվեց զամբյուղ');
}

// Վերադառնում ենք այնտեղ, որտեղից եկել ենք
$back = $_POST['back'] ?? '/';

// ⚠️ Միայն ներքին ուղիներ — բաց վերահղման դեմ
if (!str_starts_with($back, '/') || str_starts_with($back, '//')) {
    $back = '/';
}

header('Location: ' . $back, true, 303);
exit;
Բաց վերահղում — քիչ հայտնի, բայց իրական խոցելիություն

Եթե $back-ը վերցնես առանց ստուգման, հարձակվողը կարող է ուղարկել back=https://evil.com։

Օգտագործողը սեղմում է կոճակը քո կայքում և հայտնվում է կեղծ կայքում, որը նման է քոնին և խնդրում է գաղտնաբառ։ Վստահությունը՝ քո դոմենից։

Կանոն՝ վերահղում միայն ներքին ուղիներով։ //-ի ստուգումը կարևոր է — //evil.com-ը արտաքին հասցե է, թեև սկսվում է շեղագծով։

name-ը և type-ը գալիս են client-ից։

Հարձակվողը ուղարկում է shell.php ֆայլ, բայց նշում type: image/jpeg։ Եթե ստուգես միայն type-ը և պահես բնօրինակ անունով — սերվերում հայտնվում է կատարվող PHP ֆայլ։

Դա ամենալուրջ խոցելիություններից է․ հարձակվողը ստանում է հրամանների կատարում սերվերում։

        return [null, 'Ֆայլը 2 ՄԲ-ից մեծ է'];
    }

    // 3. ԻՐԱԿԱՆ տիպը՝ բովանդակությունից, ոչ ատրիբուտից ⭐
    $finfo = new finfo(FILEINFO_MIME_TYPE);
    $mime  = $finfo->file($file['tmp_name']);

    $allowed = [
        'image/jpeg' => 'jpg',
        'image/png'  => 'png',
        'image/webp' => 'webp',
    ];

    if (!isset($allowed[$mime])) {
        return [null, 'Թույլատրվում են միայն JPG, PNG, WebP'];
    }

    // 4. Լրացուցիչ ստուգում՝ իսկապե՞ս նկար է
    if (getimagesize($file['tmp_name']) === false) {
        return [null, 'Ֆայլը նկար չէ'];
    }

    // 5. ՄԵՆՔ ենք տալիս անունը — ոչ օգտագործողը ⭐
    $name = bin2hex(random_bytes(8)) . '.' . $allowed[$mime];
    $path = rtrim($dir, '/') . '/' . $name;

    // 6. Տեղափոխում
    if (!move_uploaded_file($file['tmp_name'], $path)) {
        return [null, 'Չհաջողվեց պահպանել'];
    }

    return [$name, null];
}

Երկու բան, որ պիտի անես սերվերում

1. Վերբեռնումների թղթապանակում PHP-ի կատարումն արգելիր։
uploads/.htaccess-ում՝ php_flag engine off
Դա վերջին շերտն է — եթե ինչ-որ կերպ վնասակար ֆայլ ընկնի այնտեղ, այն չի կատարվի։

2. Իդեալական դեպքում՝ ֆայլերը պահիր web-արմատից դուրս և տուր առանձին սկրիպտով։ Այդ դեպքում ուղիղ հասցեով դրանց ընդհանրապես չեն հասնի։

Ինչո՞ւ random_bytes, ոչ բնօրինակ անունը

· ../../index.php — ուղու շրջանցում
· shell.php — կատարվող կոդ
· Հայերեն և բացատներ — խնդիրներ URL-ում
· Նույն անունով երկու ֆայլ — վերագրում

Պատահական անունը լուծում է չորսն էլ միանգամից։ Իսկ բնօրինակ անունը, եթե պետք է, պահիր բազայում, ոչ ֆայլային համակարգում։

07Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Կարդալու ցուցակ, թեմա, շապիկ», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

Նախագիծ · Առաջադրանք 1

Զամբյուղը, flash-ը և նկարի վերբեռնումը ՏեխնոՄարկետում՝ Առաջադրանք 1, քայլեր։

  1. Կատարիր դասարանականի 10 քայլը։
  2. Բացիր DevTools → Application → Cookies և գտիր PHPSESSID-ը։ Ուղարկիր սքրինշոթ։
  3. Փորձիր կարդալ document.cookie-ն Console-ում httponly-ով և առանց դրա։ Նկարագրիր տարբերությունը։
  4. Գտիր սեսիայի ֆայլը սերվերում (Laragon → tmp) և բացիր այն։ Ի՞նչ ես տեսնում։
  5. Ավելացրու քանակի փոփոխում զամբյուղում (+/−)։
  6. Ցուցադրիր, ինչ կլինի, եթե սեսիայում պահես գինը և հետո փոխես այն data.php-ում։
  7. Ավելացրու թեմայի պահպանում cookie-ով (Դաս 19-ի HTML բլոկից)։
  8. Կառուցիր ֆայլի վերբեռնման ֆորմա uploadImage()-ով։
  9. Փորձիր վերբեռնել .php ֆայլ՝ վերանվանելով .jpg։ Բռնվե՞ց finfo-ով։
  10. Ցուցադրիր բաց վերահղման խնդիրը՝ ուղարկելով back=https://example.com ստուգումով և առանց դրա։

Ի՞նչ որտեղ պահել

Սեսիա → մուտք, զամբյուղ, իրավունքներ, flash Cookie → թեմա, լեզու, նախընտրություններ Բազա → ամեն ինչ, ինչ պիտի ապրի երկար Երբեք cookie-ում՝ դեր, գին, զեղչ, իրավունք

Վերբեռնման վեց քայլ

1. error === UPLOAD_ERR_OK 2. size սահման 3. finfo → իրական MIME 4. getimagesize → իսկապե՞ս նկար 5. Սեփական պատահական անուն 6. move_uploaded_file + .htaccess՝ php_flag engine off

Ստուգաթերթ

  • session_start()-ը կարգավորումներով է
  • cookie_httponly միացված է
  • Սեսիայում պահվում են ID-ներ, ոչ տվյալներ
  • Զամբյուղում կա քանակի վերին սահման
  • Ջնջված ապրանքը չի կոտրում զամբյուղը
  • Վերահղումները միայն ներքին ուղիներով են
  • Ֆայլերի տիպը ստուգվում է finfo-ով
  • Ֆայլերի անունները գեներացվում են սերվերում
  • Վերբեռնումների թղթապանակում PHP-ն անջատված է

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչպես է սեսիան կապվում cookie-ի հետ
  • Գիտե՞մ, ինչու httponly-ն կապված է XSS-ի հետ
  • Հասկանո՞ւմ եմ, ինչու սեսիայում պահում ենք միայն ID-ն
  • Գիտե՞մ, ինչու $_FILES['type']-ին չի կարելի վստահել
  • Կարո՞ղ եմ բացատրել բաց վերահղումը
Հաջորդ դասին

MySQL։ Զանգվածները data.php-ում վերջապես կփոխարինվեն իրական բազայով։

Բազաներ և SQL՝ երկու դաս։ Դաս 8-ում՝ մեկ աղյուսակ և չորս հրաման (INSERT, SELECT, UPDATE, DELETE), Դաս 9-ում՝ կապեր և JOIN։ Կաշխատենք HeidiSQL-ում՝ նախքան PHP-ից դիմելը։
