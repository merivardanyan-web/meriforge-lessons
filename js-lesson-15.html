Ապրանքների կատալոգ

Տասնչորս դաս սովորեցինք առանձին գործիքներ։ Այսօր և հաջորդ դասին դրանք միանում են մեկ իրական նախագծում՝ կատալոգ ֆիլտրերով, տեսակավորմամբ, որոնմամբ և զամբյուղով։

Օգտագործում ենք՝ Դաս 04–14 Բլոկ 2 · Նախագիծ Օր 1-ից 2

Դասը կառուցված է ստուգակետերով։ Ամեն քայլից հետո կանգ առ, բացիր բրաուզերը և համոզվիր, որ աշխատում է։ Մի՛ գրիր ամբողջ կոդը միանգամից — այդպես սխալ գտնելն անհնար կդառնա։

const formatPrice = n => n.toLocaleString("hy-AM") + " ֏";   // Դաս 8

function createCard(product) {
  const { id, name, price, rating, inStock } = product;      // Դաս 7

  return `
    <div class="col">
      <div class="wrapper ${inStock ? "" : "is-disabled"}">
        <h3 class="card__title">${name}</h3>
        <p class="card__rating">★ ${rating}</p>
        <p class="card__price">${formatPrice(price)}</p>
        <span class="badge ${inStock ? "" : "is-out"}">
          ${inStock ? "Առկա է" : "Սպառված"}
        </span>
        <button class="btn" data-action="add" data-id="${id}"
                ${inStock ? "" : "disabled"}>
          Ավելացնել
        </button>
      </div>
    </div>
  `;
}

function render() {
  const visible = getVisibleProducts();

  if (visible.length === 0) {
    container.innerHTML = `
      <div class="col">
        <div class="wrapper empty">
          <p>Ապրանք չգտնվեց։</p>
          <button class="btn" data-action="reset">Մաքրել ֆիլտրերը</button>
        </div>
      </div>
    `;
  } else {
    container.innerHTML = visible.map(createCard).join("");   // Դաս 10
  }

  renderStats(visible);
}

function renderStats(visible) {
  const total = visible.reduce((sum, p) => sum + p.price, 0);   // Դաս 6
  statsEl.textContent =
    `Ցուցադրված է ${visible.length} / ${PRODUCTS.length} ապրանք · ընդհանուր ${formatPrice(total)}`;
}

Ժամանակավորապես գրիր function getVisibleProducts() { return PRODUCTS; } և կանչիր render()։ Բացիր բրաուզերը։ 8 քարտ պիտի երևա։ Եթե չեն երևում — ստուգիր defer-ը և Console-ի սխալները։ Հետո միայն շարունակիր։

06Ստուգակետ 2 · Ֆիլտր և տեսակավորում

Ահա նախագծի սիրտը։ Մեկ ֆունկցիա, որը state-ից պատրաստում է ցուցադրվող ցուցակը։

js/main.js — pipeline
function getVisibleProducts() {
  let result = [...PRODUCTS];        // պատճեն — բնօրինակը չենք դիպչում (Դաս 5)

  // 1. Կատեգորիա
  if (state.category !== "all") {
    result = result.filter(p => p.category === state.category);
  }

  // 2. Միայն առկա
  if (state.onlyStock) {
    result = result.filter(p => p.inStock);
  }

  // 3. Որոնում
  if (state.search !== "") {
    const query = state.search.toLowerCase();      // Դաս 8
    result = result.filter(p => p.name.toLowerCase().includes(query));
  }

  // 4. Տեսակավորում
  switch (state.sort) {                            // Դաս 2
    case "price-asc":
      result.sort((a, b) => a.price - b.price);
      break;
    case "price-desc":
      result.sort((a, b) => b.price - a.price);
      break;
    case "rating":
      result.sort((a, b) => b.rating - a.rating);
      break;
    case "name":
      result.sort((a, b) => a.name.localeCompare(b.name, "hy"));
      break;
  }

  return result;
}
Ինչո՞ւ է սա գեղեցիկ

Ֆիլտրերն իրար չեն խանգարում։ Ամեն մեկը վերցնում է նախորդի արդյունքը և նեղացնում այն։ Կարող ես միաժամանակ ընտրել «Աքսեսուարներ» + «Միայն առկա» + որոնել «USB» + տեսակավորել գնով։ Ամեն ինչ կաշխատի, և դու ոչինչ լրացուցիչ չես գրել։

localeCompare

Հայերեն տեքստերի ճիշտ այբբենական տեսակավորման համար։ Սովորական a.name > b.name-ը հայերենի հետ սխալ արդյունք կտա։

Ուշադրություն

result.sort(...)-ը փոխում է result-ը, բայց դա անվտանգ է, որովհետև result-ը արդեն պատճեն է (առաջին տողում [...PRODUCTS])։ Առանց այդ պատճենի սկզբնական ցուցակը կփչանար ընդմիշտ (Դաս 6)։

// ---------- Ֆիլտրի կոճակներ (դելեգացիա) ----------

sortSelect.addEventListener("change", () => {
  state.sort = sortSelect.value;
  render();

// ---------- Միայն առկա ----------
stockCheck.addEventListener("change", () => {

// ---------- Որոնում debounce-ով (Դաս 14) ----------
let searchTimer;

searchInput.addEventListener("input", () => {
  clearTimeout(searchTimer);

  searchTimer = setTimeout(() => {
    state.search = searchInput.value.trim();
    render();
  }, 300);
});

// ---------- Քարտերի կոճակներ (դելեգացիա) ----------
container.addEventListener("click", (e) => {
  const btn = e.target.closest("[data-action]");
  if (!btn) return;

  if (btn.dataset.action === "reset") {
    resetFilters();
    return;
  }

  if (btn.dataset.action === "add") {
    const id = Number(btn.dataset.id);          // dataset-ը տող է
    const product = PRODUCTS.find(p => p.id === id);
    console.log("Զամբյուղ՝", product.name);    // վաղը կդառնա իսկական
  }
});

// ---------- Վերականգնում ----------
function resetFilters() {
  state.category  = "all";
  state.search    = "";
  state.sort      = "default";
  state.onlyStock = false;

  searchInput.value  = "";
  sortSelect.value   = "default";
  stockCheck.checked = false;

  filtersBox.querySelectorAll(".filter")
    .forEach(f => f.classList.toggle("is-active", f.dataset.category === "all"));

  render();
}

// ---------- Գործարկում ----------
render();

  • «Միայն առկա»-ն թաքցնում է սպառվածները
  • «Գինը՝ էժանից» ճիշտ է դասավորում (8000 → 450000)
  • «Անունով» դասավորումը հայերեն այբուբենով է
  • Որոնումը աշխատում է, և միայն գրելուց 0.3 վրկ հետո
  • Որոնումը կարգ չի զգում՝ «նոութ» և «ՆՈՒԹ» — երկուսն էլ գտնում են
  • Կատեգորիա + առկա + որոնում + տեսակավորում միասին աշխատում են
  • Անհնար համադրության դեպքում երևում է «Ապրանք չգտնվեց» + «Մաքրել ֆիլտրերը» կոճակը
  • «Մաքրել ֆիլտրերը» կոճակը վերականգնում է ամեն ինչ, ներառյալ դաշտերի տեսքը
  • Սպառված ապրանքի կոճակը disabled է
  • Առկա ապրանքի կոճակը Console-ում տպում է անունը
  • Վիճակագրության տողը թարմացվում է ամեն փոփոխությունից
Եթե ինչ-որ բան չի աշխատում

1. Բացիր Console-ը — կա՞ կարմիր սխալ։
2. console.log(state) մշակիչի ներսում — վիճակը փոխվո՞ւմ է։
3. console.log(getVisibleProducts()) — ցուցակը ճի՞շտ է։
4. Եթե վիճակը ճիշտ է, իսկ էջը՝ ոչ — խնդիրը render()-ում է։

10Տնային աշխատանք

Ավարտիր այսօրվա մասը և ավելացրու հետևյալը։ Հաջորդ դասին կսկսենք զամբյուղից՝ պատրաստի կատալոգի վրա։

  1. Ավարտիր բոլոր չորս ստուգակետերը։ Ամեն կետը 08-րդ բաժնի ցուցակից պիտի աշխատի։
  2. Ձևավորիր քարտերը CSS-ով՝ քո .row / .col / .wrapper մեթոդաբանությամբ։ JS-ը մի՛ դիպչիր։
  3. Ավելացրու առնվազն 4 նոր ապրանք data.js-ում, ներառյալ նոր կատեգորիա։ Ավելացրու համապատասխան ֆիլտրի կոճակ։
  4. Ավելացրու «Գինը՝ մինչև» դաշտ (type="number")։ Ավելացրու state.maxPrice և համապատասխան ֆիլտր pipeline-ում։
  5. Որոնումը թող փնտրի նաև կատեգորիայի անվան մեջ (օգտագործիր CATEGORIES օբյեկտը)։
  6. Վիճակագրության տողում ավելացրու միջին գինը և ամենաթանկ ապրանքի անունը։
  7. Ավելացրու «Ապրանքների քանակը էջում» select՝ 4 / 8 / բոլորը։ Օգտագործիր slice։
  8. Ամեն քարտի կոճակի կողքին ավելացրու «Մանրամասն» կոճակ, որը alert-ով ցույց է տալիս ամբողջ տեղեկությունը։ Օգտագործիր նույն դելեգացված listener-ը՝ data-action="details"-ով։
  9. Պահպանիր ֆիլտրերի վիճակը localStorage-ում։ Էջը թարմացնելիս ֆիլտրերը պիտի վերականգնվեն (և՛ state-ը, և՛ դաշտերի տեսքը)։
  10. Ավելացրու URL-ի աջակցություն՝ ?category=laptop։ Հուշում՝ new URLSearchParams(location.search).get("category")։ Սա լրացուցիչ բարդության առաջադրանք է։
