function checkPassword(password) {
  const errors = [];

  if (password.length < 8) {
    errors.push("Առնվազն 8 նիշ");
  }
  if (password === password.toLowerCase()) {
    errors.push("Առնվազն մեկ մեծատառ");
  }
  if (!"0123456789".split("").some(d => password.includes(d))) {
    errors.push("Առնվազն մեկ թիվ");
  }
  if (password.includes(" ")) {
    errors.push("Բացատներ չեն թույլատրվում");
  }

  if (errors.length === 0) {
    return "✓ Գաղտնաբառը հուսալի է";
  }
  return "✗ Խնդիրներ՝ " + errors.join(", ");
}

Բ — Տարիքի հաշվիչ

age.js
function getAge(birthDateString) {
  const birth = new Date(birthDateString);
  const today = new Date();

  let age = today.getFullYear() - birth.getFullYear();

  // եթե ծննդյան օրը դեռ չի եկել այս տարի — հանում ենք 1
  const monthDiff = today.getMonth() - birth.getMonth();
  if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birth.getDate())) {
    age--;
  }

  return age;
}

Գ — Հետհաշվարկ

countdown.js
function countdown(targetDate) {
  const target = new Date(targetDate);
  const now = new Date();
  let diff = target - now;

  if (diff <= 0) {
    return "Ժամանակն սպառվել է։";
  }

  const days    = Math.floor(diff / (1000 * 60 * 60 * 24));
  const hours   = Math.floor(diff / (1000 * 60 * 60)) % 24;
  const minutes = Math.floor(diff / (1000 * 60)) % 60;

  return `${days} օր, ${hours} ժամ, ${minutes} րոպե`;
}

console.log(countdown("2027-01-01"));
console.log(countdown("2020-01-01"));
Console
144 օր, 9 ժամ, 27 րոպե
Ժամանակն սպառվել է։
Կամուրջ դեպի Դաս 14

%-ը այստեղ շատ խելացի է աշխատում․ Math.floor(diff / ժամ) % 24 տալիս է «ժամերը՝ բացի ամբողջ օրերից»։ Դաս 14-ում այս ֆունկցիան կկապենք setInterval-ի հետ, և հետհաշվարկը կսկսի կենդանի աշխատել էջում։

  2. countVowels(str) — քանի՞ ձայնավոր կա բառում (ա, ե, է, ը, ի, ո, ու, օ)։
  3. isPalindrome(str) — բառը նույնն է կարդացվում երկու կողմից (անտեսիր բացատներն ու ռեգիստրը)։
  4. maskEmail(email) — "anna@mail.com" → "a***@mail.com"։
  5. formatPrice(n) — 4500000 → "4 500 000 ֏"։ Օգտագործիր toLocaleString։
  6. Օգտագործողից վերցրու երկու թիվ prompt-ով։ Ստուգիր Number.isNaN-ով, և եթե սխալ է — տպիր «Սխալ մուտք»։
  7. getRandomColor() — վերադարձնում է պատահական hex գույն ("#a3f2c1")։ Հուշում՝ "0123456789abcdef" և ցիկլ 6 անգամ։
  8. Նմանակիր զառախաղ․ 1000 անգամ նետիր զառը և հաշվիր՝ քանի՞ անգամ ընկավ 6։ Արդյունքը պիտի մոտ լինի 166-ի։
  9. getDaysUntil(dateString) — քանի՞ օր է մնացել մինչև տրված ամսաթիվը։ Անցյալ ամսաթվի դեպքում՝ «Այդ օրն արդեն անցել է»։
  10. formatDateTime(date) — վերադարձնում է "09.08.2026, 14:05"։ Օգտագործիր padStart։

09Հիշելու թերթիկ

Տողեր

ՄեթոդԻնչ է անում
str.lengthՆիշերի քանակը
str[0] / str.at(-1)Առաջին / վերջին տառը
str.trim()Հանում է եզրերի բացատները
str.toUpperCase()Մեծատառ
str.includes(x)Պարունակո՞ւմ է
str.startsWith(x)Սկսվո՞ւմ է դրանով
str.indexOf(x)Դիրքը կամ -1
str.slice(a, b)Կտրում է կտոր
str.replaceAll(a, b)Փոխարինում է բոլորը
str.repeat(n)Կրկնում է
str.padStart(n, "0")Լրացնում է ձախից
str.split(",")Տող → զանգված

Թվեր և Math

ԳրառումԻնչ է անում
Number("42")Տող → թիվ (խիստ)
parseInt("42px")Վերցնում է սկզբի ամբողջ թիվը
n.toFixed(2)2 նիշ ստորակետից հետո (տող է)
Number.isNaN(x)Ստուգում է NaN-ը
Math.round / floor / ceilԿլորացում
Math.abs(-7)Մոդուլ
Math.sqrt(x)Քառակուսի արմատ
Math.random()Պատահական 0–1
Math.max(...arr)Ամենամեծը

Date

ԳրառումԻնչ է վերադարձնում
new Date()Հենց հիմա
new Date("2026-08-09")Կոնկրետ ամսաթիվ
getFullYear()Տարին
getMonth()Ամիսը ⚠️ 0–11
getDate()Ամսվա օրը (1–31)
getDay()Շաբաթվա օրը (0 = կիրակի)
getHours() / getMinutes()Ժամ / րոպե
getTime()Միլիվայրկյաններ 1970-ից
toLocaleDateString("hy-AM")Տեղային ձևաչափ
