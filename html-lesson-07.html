Գույն և տիպոգրաֆիա

Կայքի 95%-ը տեքստ է։ Այսօր սովորում ենք գույների ձևաչափերը, տառատեսակների միացումը, միավորները և ամենակարևորը՝ ինչպես ստուգել, որ տեքստն իսկապես ընթեռնելի է։

Նախապայման՝ Դաս 06 Բլոկ 1 · CSS հիմունքներ Կոնտրաստի ստուգում

00Ի՞նչ ենք սովորելու այսօր

  • Գույնի չորս ձևաչափ
  • Կոնտրաստի հարաբերակցություն և դրա ստուգումը
  • Տառատեսակների միացում, հայերենի աջակցություն
  • Միավորներ՝ px, rem, em, %
  • Տեքստի հատկություններ
  • Տիպոգրաֆիկ սանդղակ

01Գույնի ձևաչափեր

չորս ձև՝ նույն գույնը
/* 1. Անուն — 140 գույն ունի անուն */
color: navy;

/* 2. HEX — ամենատարածվածը */
color: #003459;
color: #036;        /* կարճ ձև, երբ զույգերը կրկնվում են */

/* 3. RGB — կարմիր, կանաչ, կապույտ (0–255) */
color: rgb(0, 52, 89);
color: rgba(0, 52, 89, 0.5);   /* 50% թափանցիկ */

/* 4. HSL — երանգ, հագեցվածություն, պայծառություն */
color: hsl(205, 100%, 17%);
HEX-ի կառուցվածքը # 00 34 59 └┬┘ └┬┘ └┬┘ R G B կարմիր կանաչ կապույտ ամեն զույգը՝ 00-ից FF (0–255) #000000 = սև #FFFFFF = սպիտակ #FF0000 = կարմիր #00FF00 = կանաչ
ՁևաչափԵ՞րբ օգտագործել
ԱնունԱրագ փորձարկում։ Իրական նախագծերում՝ հազվադեպ
HEXՀիմնական ձևը։ Դիզայներները տալիս են հենց այս ձևաչափով
RGBAԵրբ պետք է թափանցիկություն
HSLԵրբ պետք է նույն գույնի ավելի բաց/մուգ տարբերակ
HSL-ի առավելությունը

hsl(205, 100%, 17%) — երանգը 205 է։ Ուզո՞ւմ ես ավելի բաց տարբերակ։ Փոխիր միայն երրորդ թիվը՝ hsl(205, 100%, 40%)։ HEX-ով դա անելու համար պիտի հաշվարկես երեք զույգ։

Դրա համար դիզայն-համակարգերում հաճախ օգտագործում են HSL։

Թափանցիկություն

rgba vs opacity
/* rgba — թափանցիկ է ՄԻԱՅՆ ֆոնը */
.overlay {
    background-color: rgba(0, 0, 0, 0.6);
    color: white;      /* տեքստը մնում է լիարժեք */
}

/* opacity — թափանցիկ է ԱՄԲՈՂՋ էլեմենտը */
.faded {
    opacity: 0.6;       /* և ֆոնը, և տեքստը, և երեխաները */
}
Հաճախ շփոթում են

Եթե ուզում ես մուգ շերտ նկարի վրա, բայց պայծառ տեքստ դրա վրա — օգտագործիր rgba։ opacity-ն կթուլացնի նաև տեքստը։

02Կոնտրաստ

Ամենակարևոր բաժինը այս դասում։ Գեղեցիկ գունային համադրությունը կարող է լինել անընթեռնելի։

❌ իրական սխալներ
/* Մուգ ֆոն + սև տեքստ */
.bg-dark {
    background-color: #253237;
    color: black;          /* կոնտրաստ 1.3:1 — չի երևում */
}

/* Բաց ֆոն + սպիտակ տեքստ */
.bg-light {
    background-color: #f5f5f5;
    color: white;          /* կոնտրաստ 1.1:1 — անտեսանելի */
}

/* Մոխրագույն «նուրբ» տեքստ */
.muted {
    color: #aaa;            /* սպիտակի վրա 2.3:1 — շատ թույլ */
}

Կանոնները

ՏեքստՆվազագույն կոնտրաստ
Սովորական (մինչև 24px)4.5:1
Խոշոր (24px+ կամ 19px+ հաստ)3:1
Կոճակների եզրագծեր, պատկերակներ3:1
Ինչպես ստուգել DevTools-ով

1. Աջ սեղմում տեքստի վրա → Inspect
2. Styles վահանակում գտիր color-ը
3. Սեղմիր գույնի քառակուսու վրա
4. Կբացվի գունապնակը՝ Contrast ratio տողով

Եթե կողքին ✓✓ է — ամեն ինչ լավ է։ Եթե ⚠ — ուղղիր։ Այնտեղ կա նաև գիծ, որը ցույց է տալիս՝ որ երանգից սկսած է կոնտրաստը բավարար դառնում։

✅ ուղղված տարբերակը
.bg-dark {
    background-color: #253237;
    color: #ffffff;        /* 12.6:1 ✓ */
}

.bg-light {
    background-color: #f5f5f5;
    color: #1a1a1a;        /* 15.8:1 ✓ */
}

.muted {
    color: #6b6b6b;          /* 5.3:1 ✓ */
}
Ոսկե կանոն

Ամեն ֆոնի դասը գրիր իր տեքստի գույնի հետ միասին։

Երբ .bg-dark-ը սահմանում է միայն ֆոնը, ինչ-որ պահի այն կհայտնվի սև տեքստի կողքին և ոչ ոք չի նկատի, մինչև որ օգտագործողը բողոքի։

Ո՞ւմ համար

Աշխարհում մոտ 300 միլիոն մարդ ունի գունային տեսողության խանգարում։ Բացի այդ՝ արևի տակ հեռախոս նայողները, հին էկրաններով մարդիկ, տարեցները։ Լավ կոնտրաստը հատուկ խմբի համար չէ — այն բոլորի համար է։

03Տառատեսակներ

font-family — պահեստային շղթա
body {
    font-family: "Noto Sans Armenian", "Segoe UI", system-ui, sans-serif;
}

Բրաուզերը փորձում է ձախից աջ։ Եթե առաջինը չկա՝ անցնում է հաջորդին։ Վերջում պիտի լինի ընդհանուր ընտանիք՝ sans-serif, serif կամ monospace։

ԸնտանիքԲնութագիրԵ՞րբ
sans-serifԱռանց ծայրագծերիԻնտերֆեյս, հիմնական տեքստ
serifԾայրագծերովՎերնագրեր, երկար հոդվածներ
monospaceՀավասար լայնությունԿոդ

Google Fonts — հայերենով

head-ում
<link rel="preconnect" href="https://fonts.googleapis.com">
<link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
<link href="https://fonts.googleapis.com/css2?family=Noto+Sans+Armenian:wght@400;500;700&display=swap"
      rel="stylesheet">
Ոչ բոլոր տառատեսակներն ունեն հայերեն

Google Fonts-ի հազարավոր տառատեսակներից միայն մի քանի տասնյակն ունեն հայկական այբուբեն։ Եթե ընտրես տառատեսակ առանց հայերենի, բրաուզերը կանցնի պահեստայինին, և էջում կլինեն երկու տարբեր տառատեսակ — լատինատառը մեկը, հայերենը՝ մյուսը։

Google Fonts-ում ֆիլտրիր՝ Language → Armenian։

ՏառատեսակՈճ
Noto Sans ArmenianՉեզոք, ունիվերսալ
Noto Serif ArmenianԾայրագծերով, վերնագրերի համար
Arian AMUԴասական հայկական
MardotoԺամանակակից
Երկու խորհուրդ

1. Միացրու միայն այն հաստությունները, որ իսկապես օգտագործում ես։ wght@400;700 — երկուսը բավական է սովորաբար։ Ամեն ավելորդ հաստություն դանդաղեցնում է էջը։
2. display=swap-ը պարտադիր է — առանց դրա տեքստն անտեսանելի է, մինչև տառատեսակը բեռնվի։

04Միավորներ

ՄիավորԻնչի՞ նկատմամբԵ՞րբ օգտագործել
pxԲացարձակԵզրագծեր, ստվերներ, փոքր բացակներ
remhtml-ի տառաչափիՏառաչափեր, բացակներ
emԸնթացիկ էլեմենտի տառաչափիՆերսի բացակներ կոմպոնենտում
%ԾնողիԼայնություններ
vw / vhԷկրանի լայնության/բարձրությանԱմբողջ էկրանով բլոկներ
rem — ինչպես է աշխատում
/* Բրաուզերի լռելյայն՝ html { font-size: 16px } */

.title  { font-size: 2rem;    }   /* 32px */
.text   { font-size: 1rem;    }   /* 16px */
.small  { font-size: 0.875rem; } /* 14px */
Ինչո՞ւ rem, ոչ px

Օգտագործողը կարող է բրաուզերի կարգավորումներում մեծացնել լռելյայն տառաչափը — դա անում են վատ տեսողությամբ մարդիկ։

Եթե գրել ես font-size: 16px, տեքստը չի մեծանա։ Եթե 1rem — կմեծանա։ Մեկ որոշում, որը շատ մարդկանց համար բան է փոխում։

em — կուտակվում է
.card       { font-size: 20px; }
.card .title { font-size: 2em;  }   /* 40px — 20-ի կրկնապատիկ */

/* Թակարդ՝ ներդրվածության դեպքում */
ul { font-size: 0.9em; }
/* ul > ul     = 0.81em */
/* ul > ul > ul = 0.73em — տեքստը փոքրանում է անվերջ */
Պարզ կանոն

rem — տառաչափերի և բացակների համար։
em — երբ չափը պիտի կապված լինի հենց այս էլեմենտի տեքստի հետ (օրինակ՝ կոճակի padding-ը)։
px — եզրագծերի և այն բաների համար, որոնք չպիտի մասշտաբավորվեն։

05Տեքստի հատկություններ

հիմնականները
.text {
    font-size: 1rem;
    font-weight: 400;           /* 100–900 */
    font-style: italic;
    line-height: 1.5;           /* առանց միավորի ✅ */
    letter-spacing: 0.02em;
    text-align: left;
    text-transform: uppercase;
    text-decoration: underline;
}
ՀատկությունԱրժեքներ
font-weight400 սովորական, 700 հաստ։ Կամ normal/bold
line-heightՏողերի միջև հեռավորությունը
letter-spacingՏառերի միջև
text-alignleft, center, right, justify
text-transformuppercase, lowercase, capitalize
text-decorationnone, underline, line-through
line-height առանց միավորի

line-height: 1.5 ✅ — 1.5 × ընթացիկ տառաչափը։ Աշխատում է ճիշտ ամեն մակարդակում։
line-height: 24px ❌ — ֆիքսված։ Երբ ներսի տեքստը 32px է, տողերը կկպչեն իրար։

Հարմարավետ արժեքներ՝ 1.5–1.7 հիմնական տեքստի համար, 1.1–1.3 խոշոր վերնագրերի։

Երկու բան, որ վնասում են ընթեռնելիությանը

1. text-align: justify — ստեղծում է անհավասար բացատներ բառերի միջև, հատկապես նեղ սյուներում։ Խուսափիր վեբում։
2. text-transform: uppercase երկար տեքստի համար — ՄԵԾԱՏԱՌ ՏԵՔՍՏԸ ԿԱՐԴԱԼՆ ԱՎԵԼԻ ԴԱՆԴԱՂ Է, ՈՐՈՎՀԵՏԵՎ ԲՈԼՈՐ ՏԱՌԵՐԸ ՆՈՒՅՆ ԲԱՐՁՐՈՒԹՅԱՆ ԵՆ։ Կարճ պիտակների համար՝ լավ է։

Տողի օպտիմալ երկարությունը

Հարմարավետ կարդալու համար՝ 50–75 նիշ տողում։ CSS-ում դա հեշտ է սահմանել՝ max-width: 65ch։ ch միավորը «0» նիշի լայնությունն է։

06Տիպոգրաֆիկ սանդղակ

Չափերը չեն ընտրվում պատահական։ Դրանք կազմում են սանդղակ՝ որոշակի հարաբերակցությամբ։

css/main.css
/* ---------- Գույներ ---------- */
:root {
    --dark:      #253237;
    --secondary: #003459;
    --light:     #9db4c0;
    --bg:        #f5f5f5;
    --text:      #1a1a1a;
    --muted:     #6b6b6b;
    --on-dark:   #ffffff;
}

/* ---------- Բազա ---------- */
body {
    font-family: "Noto Sans Armenian", system-ui, sans-serif;
    font-size: 1rem;
    line-height: 1.6;
    color: var(--text);
    background-color: var(--bg);
}

/* ---------- Վերնագրերի սանդղակ ---------- */
h1, .title-1 { font-size: 3.75rem; font-weight: 700; line-height: 1.1; }
h2, .title-2 { font-size: 2.5rem;  font-weight: 700; line-height: 1.2; }
h3, .title-3 { font-size: 1.875rem;font-weight: 700; line-height: 1.3; }
h4, .title-4 { font-size: 1.25rem; font-weight: 700; line-height: 1.4; }

/* ---------- Օժանդակ ---------- */
.text-muted { color: var(--muted); }
.text-small { font-size: 0.875rem; }
Առաջին անգամ՝ CSS փոփոխականներ

--dark: #253237 — սա փոփոխական է։ Օգտագործվում է var(--dark)-ով։

Առավելությունը՝ գույնը փոխելու համար ուղղում ես մեկ տեղում։ Մանրամասն կսովորենք 19-րդ դասին, բայց սկսում ենք օգտագործել հիմա — որ սովորություն դառնա։

Ինչո՞ւ 3.75rem, ոչ 60px

60 ÷ 16 = 3.75։ Նույն արդյունքն է, բայց մասշտաբավորվում է օգտագործողի կարգավորումների հետ։

Հարմար հաշվարկ՝ բաժանիր 16-ի։ 20px → 1.25rem, 24px → 1.5rem, 32px → 2rem, 40px → 2.5rem։

07Խաղահրապարակ

Փոխիր գույները և ստուգիր կոնտրաստը
Փորձիր

1. .bg-secondary-ի color-ը դարձրու black։ Ընթեռնե՞լի է։
2. Համեմատիր .overlay-ը և .faded-ը։
3. Ջնջիր max-width: 65ch-ը և ընդլայնիր պատուհանը։
4. Փոխիր line-height-ը 1.6-ից 1-ի։

08Դասարանական

  1. Ընտրիր հայերենով տառատեսակ Google Fonts-ից (Language → Armenian)
  2. Միացրու այն՝ միայն 400 և 700 հաստությամբ
  3. main.css-ում ստեղծիր :root բլոկը 6–7 գույնով
  4. Գրիր body-ի բազային ոճերը
  5. Ստեղծիր վերնագրերի սանդղակը rem-ով
  6. Ստեղծիր .bg-* դասերը՝ ամեն մեկը իր տեքստի գույնով
  7. Ստուգիր բոլոր համադրությունները DevTools-ի կոնտրաստի ցուցիչով
global.css — այսօրվա ավելացումը
/* ---------- Ֆորմայի տարրերը ժառանգում են տառատեսակը ---------- */
input, textarea, select, button {
    font: inherit;
    color: inherit;
}
Ինչո՞ւ է սա պետք

Ֆորմայի տարրերը չեն ժառանգում տառատեսակը body-ից — դա բրաուզերի հին որոշում է։ Առանց այս երկու տողի քո գեղեցիկ հայկական տառատեսակը կլինի ամենուր, բացի դաշտերից և կոճակներից, որտեղ կմնա բրաուզերի լռելյայնը։

Փորձիր ինքդ․ ջնջիր այս կանոնը և նայիր ֆորմային։

09Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Գույն և տիպոգրաֆիա», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

  1. Գրիր նույն գույնը չորս ձևաչափով՝ անուն, HEX, RGB, HSL։ Ստուգիր, որ արդյունքը նույնն է։
  2. Ցուցադրիր rgba-ի և opacity-ի տարբերությունը՝ երկու նույնական բլոկով։
  3. Ստեղծիր 5 գունային համադրություն։ Ամեն մեկի կոնտրաստը ստուգիր DevTools-ով և գրիր արժեքը մեկնաբանության մեջ։
  4. Գտիր դիտավորյալ վատ համադրություն (2:1-ից պակաս) և ուղղիր այն։ Ցույց տուր՝ առաջ և հետո։
  5. Միացրու հայերենով տառատեսակ։ Փորձիր նաև առանց հայերենի տառատեսակ և նկարագրիր՝ ինչ եղավ։
  6. Ստեղծիր վերնագրերի սանդղակ rem-ով՝ h1-ից h4։ Ցույց տուր հաշվարկը մեկնաբանություններում։
  7. Փորձիր բրաուզերի կարգավորումներում մեծացնել լռելյայն տառաչափը։ Համեմատիր px-ով և rem-ով գրված տեքստերի պահվածքը։
  8. Ցուցադրիր em-ի կուտակումը՝ երեք մակարդակ ներդրված ցուցակով, որտեղ font-size: 0.9em։
  9. Կիրառիր max-width: 65ch հոդվածի տեքստին։ Համեմատիր առանց դրա տարբերակի հետ լայն էկրանին։
  10. Ոճավորիր Դաս 5-ի սեմանտիկ էջը՝ ամբողջական տիպոգրաֆիայով և գույներով։ Բոլոր կոնտրաստները պիտի անցնեն 4.5:1 շեմը։

Ստուգաթերթ

  • Ամեն .bg-* դաս ունի իր color-ը
  • Բոլոր կոնտրաստները ≥ 4.5:1
  • Տառաչափերը rem-ով են
  • line-height-ը առանց միավորի է
  • Տառատեսակն ունի հայերեն
  • Միացված են միայն անհրաժեշտ հաստությունները
  • Կա display=swap
  • Երկար տեքստն ունի max-width

  • Կարո՞ղ եմ բացատրել rgba-ի և opacity-ի տարբերությունը
  • Գիտե՞մ, ինչպես ստուգել կոնտրաստը

  • Գիտե՞մ, ինչու line-height-ը առանց միավորի
  • Կարո՞ղ եմ բացատրել em-ի կուտակումը
