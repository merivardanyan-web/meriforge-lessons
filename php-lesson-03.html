01Երկու տեսակի զանգված

իրականում մեկն է
// Ինդեքսավորված — բանալիները 0, 1, 2...
$colors = ["կարմիր", "կապույտ", "կանաչ"];
echo $colors[0];        // կարմիր

// Ասոցիատիվ — բանալիները տողեր են
$product = [
    'name'  => 'Նոութբուք Pro 14',
    'price' => 450000,
    'stock' => 12,
];
echo $product['name'];
Կարևոր տարբերություն JS-ից

JS-ում զանգվածը և օբյեկտը երկու տարբեր բան են՝ [1,2,3] և {a: 1}։

PHP-ում դա մեկ տիպ է — array։ Ինդեքսավորվածը պարզապես ասոցիատիվ է, որտեղ բանալիները թվեր են։

Ուստի PHP-ի զանգվածը միաժամանակ JS-ի Array-ը և Object-ն է։

հիմնական գործողություններ
$colors[] = "դեղին";              // ավելացնել վերջում (JS՝ push)
$product['brand'] = "TechPro";   // նոր բանալի

count($colors);                    // քանակը (JS՝ .length)
in_array("կապույտ", $colors);      // կա՞ արժեքը
array_key_exists('price', $product);
array_keys($product);               // ['name','price','stock']
array_values($product);
array_search("կանաչ", $colors);    // ինդեքսը կամ false

unset($colors[1]);                // ջնջել
$colors = array_values($colors);   // վերահամարակալել ⭐
unset-ը բացեր է թողնում

unset($colors[1])-ից հետո զանգվածն ունի բանալիներ 0, 2, 3 — երկուսը բացակայում է։

foreach-ը կաշխատի նորմալ, բայց for ($i=0; $i<count(...); $i++)-ը կկոտրվի, և JSON-ի փոխարկելիս զանգվածը կդառնա օբյեկտ։

Ուստի ջնջելուց հետո՝ array_values()։ Դա Դաս 17-ի API-ում կրիտիկական կդառնա։

02Ներդրված զանգվածներ

Այս կառուցվածքը կհանդիպի ամեն օր մինչև դասընթացի վերջ։

զանգված զանգվածներից
$products = [
    ['id' => 1, 'name' => 'Նոութբուք', 'price' => 450000],
    ['id' => 2, 'name' => 'Մոնիտոր',  'price' => 120000],
    ['id' => 3, 'name' => 'Ստեղնաշար', 'price' => 25000],
];

echo $products[0]['name'];    // Նոութբուք

foreach ($products as $p) {
    echo $p['name'] . " — " . $p['price'];
}
Ինչո՞ւ սա այդքան կարևոր է

MySQL-ի ամեն հարցում վերադարձնում է հենց այս ձևը։ Դաս 9-ում $stmt->fetchAll()-ը կտա ճիշտ նույն կառուցվածքը՝ զանգված ասոցիատիվ զանգվածներից։

Այսինքն՝ քո foreach-ը այսօր և Դաս 9-ի foreach-ը կլինեն նույնական։ Փոխվում է միայն տվյալների աղբյուրը։

Դա նաև պատճառն է, որ այսօր զանգվածները գրում ենք ձեռքով — որ վաղը բազան զարմանք չլինի։

03Զանգվածների ֆունկցիաներ

JS Դաս 6-ի ուղիղ համապատասխանությունները — բայց ուշադրություն արգումենտների կարգին։

JavaScriptPHP
arr.map(fn)array_map(fn, $arr) ⚠️ հակառակ կարգ
arr.filter(fn)array_filter($arr, fn)
arr.reduce(fn, 0)array_reduce($arr, fn, 0)
arr.sort(fn)usort($arr, fn)
arr.find(fn)Չկա — array_filter + reset
arr.includes(x)in_array($x, $arr)
arr.slice()array_slice($arr, 0, 3)
Արգումենտների կարգը անհետևողական է

array_map(fn, $arr) — ֆունկցիան առաջ
array_filter($arr, fn) — զանգվածն առաջ

Դա PHP-ի հայտնի թերությունն է՝ ստանդարտ գրադարանը կառուցվել է տարիների ընթացքում առանց միասնական կանոնի։

Չկա տրամաբանություն, որ հիշես — խմբագրիչի ավտոլրացումը այստեղ լավագույն ընկերն է։

գործնականում
// Ֆիլտր՝ միայն մատչելի ապրանքները
$cheap = array_filter($products, fn($p) => $p['price'] < 100000);

// Միայն անունները
$names = array_map(fn($p) => $p['name'], $products);

// Ընդհանուր գումարը
$total = array_reduce($products, fn($sum, $p) => $sum + $p['price'], 0);

// fn() => ... կարճ ձևը PHP 7.4+ է, նման JS-ի arrow-ին
// Հին ձևը՝
$names = array_map(function ($p) {
    return $p['name'];
}, $products);
array_filter-ը պահպանում է բանալիները ⭐

Ֆիլտրից հետո զանգվածն ունի բանալիներ 0, 2, 5 — բացերով։

foreach-ի համար խնդիր չէ։ Բայց json_encode()-ը այդպիսի զանգվածը կվերածի օբյեկտի, ոչ ցուցակի — և JS-ի կողմում .map()-ը կկոտրվի։

Կանոն՝ array_filter-ից հետո միշտ array_values(), եթե արդյունքը գնում է JSON կամ ինդեքսավորված ցիկլ։

տեսակավորում
// Ըստ գնի՝ աճման կարգով
usort($products, fn($a, $b) => $a['price'] <=> $b['price']);

// Նվազման կարգով — պարզապես փոխիր տեղերով
usort($products, fn($a, $b) => $b['price'] <=> $a['price']);

// Ըստ անվան
usort($products, fn($a, $b) => strcmp($a['name'], $b['name']));
<=> — «տիեզերանավը»

Վերադարձնում է -1, 0 կամ 1՝ կախված համեմատությունից։

Առանց դրա պիտի գրեիր՝ $a['price'] < $b['price'] ? -1 : ($a['price'] > $b['price'] ? 1 : 0)։

Նաև ուշադրություն՝ usort-ը փոխում է բուն զանգվածը և չի վերադարձնում նորը։ Դա JS-ի .sort()-ի պես է, ոչ .map()-ի։

04Ֆունկցիաներ

հիմնականը ծանոթ է
function formatPrice($price) {
    return number_format($price, 0, ".", " ") . " ֏";
}

echo formatPrice(450000);    // 450 000 ֏

Տիպերի հայտարարում — JS-ում սա չկար

նույն ֆունկցիան, ամրագրված
declare(strict_types=1);   // ֆայլի ամենասկզբում ⭐

function formatPrice(int $price): string
{
    return number_format($price, 0, ".", " ") . " ֏";
}

formatPrice(450000);     // ✅ 450 000 ֏
formatPrice("շատ");      // ❌ TypeError — անմիջապես, հստակ սխալ
Ինչո՞ւ սա արժեքավոր է

JS-ում formatPrice("շատ") կվերադարձներ NaN կամ "շատ ֏", և սխալը կհայտնվեր ինչ-որ տեղ էջում՝ առանց բացատրության։

PHP-ն strict_types-ով անմիջապես ասում է․ «այս ֆունկցիան int է ուզում, դու տվեցիր string, տողը՝ 42»։

Դա TypeScript-ի գաղափարն է՝ ներկառուցված լեզվի մեջ։ Դիր declare(strict_types=1) ամեն նոր ֆայլում։

տիպերի տեսականին
function f(int $a, string $b, float $c, bool $d, array $e): void {}

// Կարող է լինել null
function findProduct(int $id): ?array { ... }

// Լռելյայն արժեք
function getProducts(int $limit = 10): array { ... }

// Անվանված արգումենտներ (PHP 8+)
createUser(name: "Անի", isAdmin: true);

Տեսանելիության տիրույթ

այստեղ PHP-ն խիստ տարբերվում է JS-ից ⚠️
$siteName = "ՏեխնոՄարկետ";

function showTitle() {
    echo $siteName;   // ❌ Warning: Undefined variable
}
Ֆունկցիաները չեն տեսնում արտաքին փոփոխականները

JS-ում ֆունկցիան տեսնում է ամեն ինչ իր վերևում (closure)։ PHP-ում՝ ոչ։

Ֆունկցիայի ներսում աշխարհն ամբողջովին փակ է — կան միայն պարամետրերը։

Դա սկզբում անհարմար է թվում, բայց իրականում լավ է․ ֆունկցիան չի կարող գաղտնի կախված լինել գլոբալ վիճակից։ Կարդալով ստորագրությունը՝ գիտես, ինչից է կախված։

երեք ելք
// ✅ 1. Փոխանցել պարամետրով — ՃԻՇՏ ձևը
function showTitle(string $name): void {
    echo $name;
}
showTitle($siteName);

// ⚠️ 2. global — աշխատում է, բայց խուսափիր
function showTitle2(): void {
    global $siteName;
    echo $siteName;
}

// ✅ 3. Հաստատուն — կոնֆիգուրացիայի համար
const SITE_NAME = "ՏեխնոՄարկետ";
function showTitle3(): void {
    echo SITE_NAME;   // հաստատունները գլոբալ են
}
Ինչո՞ւ խուսափել global-ից

Ֆունկցիան դառնում է կախված արտաքին վիճակից, որը ստորագրության մեջ չի երևում։ Փորձարկելը դժվարանում է, տեղափոխելը՝ անհնար։

Դաս 11-ում կտեսնենք ավելի լավ լուծում՝ դասեր, որտեղ վիճակը պահվում է օբյեկտի ներսում։ Մինչ այդ՝ պարամետրեր և հաստատուններ։

05include և require

Առաջին քայլը դեպի կոդի կազմակերպում։ Դաս 5-ում սա կդառնա շաբլոնների համակարգ։

ՁևՖայլը չկաԵ՞րբ
includeWarning, շարունակում էՈչ կրիտիկական մաս
requireFatal error, կանգնում էԿրիտիկական ⭐
include_onceWarningԿրկնությունից պաշտպանված
require_onceFatal errorՖունկցիաներ, դասեր ⭐
գործնականում
// Կոնֆիգ և ֆունկցիաներ — առանց դրանց էջն անիմաստ է
require_once 'config.php';
require_once 'functions.php';

// Շաբլոնի մասեր
require 'partials/header.php';
require 'partials/footer.php';
Ինչո՞ւ _once ֆունկցիաների համար

Եթե functions.php-ը միանա երկու անգամ, PHP-ն կփորձի հայտարարել նույն ֆունկցիան կրկին → Fatal error: Cannot redeclare։

Իսկ շաբլոնի մասերի համար _once պետք չէ — երբեմն նույն բլոկը պիտի կրկնվի էջում մի քանի անգամ։

ուղիների խնդիրը ⭐
// ❌ Հարաբերական ուղին կախված է նրանից,
//    ՈՐՏԵՂԻՑ է կանչվել ֆայլը
require 'partials/header.php';

// ✅ Բացարձակ ուղի ընթացիկ ֆայլից
require __DIR__ . '/partials/header.php';
Սա ժամեր է խլում, երբ չգիտես

__DIR__-ը այն թղթապանակն է, որտեղ գտնվում է ընթացիկ ֆայլը։

Առանց դրա՝ index.php-ից require 'partials/header.php'-ն աշխատում է։ Բայց երբ admin/products.php-ից կանչում ես նույն functions.php-ը, որն իր հերթին ինչ-որ բան է միացնում — ուղին հաշվվում է admin/-ից, և ամեն ինչ փլվում է։

Դաս 14-ում, երբ ամեն հարցում կանցնի մեկ index.php-ով, __DIR__-ը կդառնա պարտադիր։ Սովորիր հիմա։

Ռեգիստրի հիշեցում

Windows-ում require 'Functions.php'-ը կաշխատի, նույնիսկ եթե ֆայլը functions.php է։ Հրապարակելուց հետո Linux սերվերում՝ ոչ։

06Դասարանական

Ապրանքների ֆիլտր և տեսակավորում — առանց բազայի, զուտ զանգվածներով։

functions.php
<?php
declare(strict_types=1);

function formatPrice(int $price): string
{
    return number_format($price, 0, ".", "&nbsp;") . "&nbsp;֏";
}

function finalPrice(array $product): int
{
    $discount = $product['discount'] ?? 0;
    return (int) round($product['price'] * (100 - $discount) / 100);
}

function filterByMaxPrice(array $products, int $max): array
{
    $result = array_filter(
        $products,
        fn(array $p): bool => finalPrice($p) <= $max
    );

    return array_values($result);   // ⭐ վերահամարակալում
}

function sortProducts(array $products, string $by = 'price_asc'): array
{
    match ($by) {
        'price_asc'  => usort($products, fn($a, $b) => finalPrice($a) <=> finalPrice($b)),
        'price_desc' => usort($products, fn($a, $b) => finalPrice($b) <=> finalPrice($a)),
        'name'       => usort($products, fn($a, $b) => strcmp($a['name'], $b['name'])),
        default      => null,
    };

    return $products;
}

function totalValue(array $products): int
{
    return array_reduce(
        $products,
        fn(int $sum, array $p): int => $sum + finalPrice($p),
        0
    );
}

Header-ը և footer-ը հիմա մեկ տեղում են։ Երբ ավելացնես երկրորդ էջ (Դաս 6-ում՝ կոնտակտների ֆորմա), դրանք չես կրկնօրինակի։

Իսկ երբ պիտի փոխես մենյուն, կփոխես մեկ ֆայլում, ոչ քսանում։ Դա նույն սկզբունքն է, ինչ global.css-ի դեպքում՝ մեկ աղբյուր ճշմարտության։

07Տնային աշխատանք

  1. Կատարիր դասարանականի 8 քայլը։
  2. Ավելացրու data.php-ում category դաշտ և գրիր filterByCategory()։
  3. Գրիր findProduct(array $products, int $id): ?array։ Ի՞նչ է վերադարձնում, երբ չի գտնում։
  4. Ցուցադրիր, ինչ է լինում array_filter-ից հետո առանց array_values-ի։ Օգտագործիր var_dump։
  5. Համեմատիր json_encode-ի արդյունքը array_values-ով և առանց դրա։
  6. Փորձարկիր strict_types-ը՝ միացրած և անջատած։ Ի՞նչ տարբերություն։
  7. Ցուցադրիր տեսանելիության տիրույթի խնդիրը և ուղղիր երեք ձևով։
  8. Ջնջիր __DIR__-ը require-ից և կանչիր էջը ենթաթղթապանակից։ Նկարագրիր սխալը։
  9. Գրիր groupByCategory(), որը վերադարձնում է զանգված, որտեղ բանալին կատեգորիան է։
  10. Հաշվիր միջին գինը, ամենաթանկը և ամենաէժանը՝ երեք ֆունկցիայով տիպերի հայտարարմամբ։

08Հիշելու թերթիկ

ԳրառումԻնչ է անում
$arr[] = $xԱվելացնել վերջում
count($arr)Քանակ
array_map(fn, $arr)Ֆունկցիան առաջ ⚠️
array_filter($arr, fn)Զանգվածն առաջ ⚠️
array_values($arr)Վերահամարակալում ⭐
usort($arr, fn)Փոխում է բուն զանգվածը
<=>Համեմատություն տեսակավորման համար
declare(strict_types=1)Խիստ տիպեր ⭐
function f(int $a): stringՏիպերի հայտարարում
?arrayԿարող է լինել null
require_once __DIR__ . '/x.php'Ապահով միացում ⭐

Երեք թակարդ

1. array_filter → բանալիները մնում են բացերով → array_values() JSON-ից և ցիկլերից առաջ 2. Ֆունկցիաները չեն տեսնում արտաքին փոփոխականները → փոխանցիր պարամետրով 3. require առանց __DIR__-ի → կոտրվում է, երբ կանչում ես այլ թղթապանակից

Ստուգաթերթ

  • Ամեն ֆայլ սկսվում է declare(strict_types=1)-ով
  • Բոլոր ֆունկցիաներն ունեն տիպերի հայտարարում
  • require-ը միշտ __DIR__-ով է
  • Ֆայլերի անունները փոքրատառ են
  • Չկա global
  • Header-ը և footer-ը առանձին ֆայլերում են
  • array_filter-ից հետո կա array_values

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու PHP-ում զանգվածը և օբյեկտը մեկ տիպ են
  • Գիտե՞մ, ինչու array_values-ը կարևոր է
  • Հասկանո՞ւմ եմ, ինչու ֆունկցիաները չեն տեսնում գլոբալ փոփոխականները
  • Կարո՞ղ եմ բացատրել strict_types-ի արժեքը
  • Գիտե՞մ, ինչու __DIR__
