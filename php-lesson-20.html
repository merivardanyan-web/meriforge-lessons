Ինչ է իրականում Eloquent-ը

Eloquent-ը Active Record օրինաչափություն է․ մոդելը միաժամանակ տվյալն է (Դաս 12-ի Product) և բազայի հետ աշխատելու միջոցը (Դաս 14-ի Repository)։

Մենք Դաս 14-ում դրանք դիտավորյալ բաժանեցինք։ Eloquent-ը դրանք կրկին միավորում է՝ հարմարության համար։ Դա փոխզիջում է, ոչ ճշմարտություն․ որոշ մեծ նախագծեր Laravel-ում էլ պահում են Repository շերտը։ Սովորական նախագծերի համար Active Record-ը բավական է։

02Միգրացիաներ

schema.sql-ի խնդիրը

Դաս 8–9-ում սխեման ստեղծեցիր HeidiSQL-ում։ Դաս 13-ում ավելացրիր type և weight, Դաս 16-ում՝ users։ Ամեն անգամ ձեռքով։

Հիմա պատկերացրու, որ թիմում երեք ծրագրավորող կա և մեկ սերվեր։ Ո՞վ գիտի, որ սերվերի բազայում արդեն կա weight, իսկ Անիի մեքենայում՝ ոչ։ Ոչ ոք։ Սխալը կհայտնվի production-ում։

Միգրացիան սխեմայի փոփոխությունն է՝ գրված կոդով և պահված Git-ում։ Ով քաշում է նախագիծը, գործարկում է մեկ հրաման և ստանում ճիշտ նույն բազան։

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\SoftDeletes;

class Product extends Model
{
    use HasFactory, SoftDeletes;

    /** Mass assignment-ի սպիտակ ցուցակ — բաժին 07 */
    protected $fillable = [
        'category_id', 'name', 'slug', 'specs', 'description',
        'price', 'discount', 'stock', 'type', 'weight', 'image', 'is_active',
    ];

    /** Դաս 12-ի fromRow()-ի (int)-երը ⭐ */
    protected function casts(): array
    {
        return [
            'price'     => 'integer',
            'discount'  => 'integer',
            'stock'     => 'integer',
            'weight'    => 'integer',
            'is_active' => 'boolean',
        ];
    }

    // ---------- Կապեր ----------

    public function category(): BelongsTo
    {
        return $this->belongsTo(Category::class);
    }

    // ---------- Scope-ներ ----------

    public function scopeActive(Builder $query): void
    {
        $query->where('is_active', true);
    }

    // ---------- Դաս 12-ի մեթոդները ----------

    public function finalPrice(): int
    {
        return (int) round($this->price * (100 - $this->discount) / 100);
    }

    public function hasDiscount(): bool
    {
        return $this->discount > 0;
    }

    public function inStock(): bool
    {
        return $this->type === 'digital' || $this->stock > 0;
    }

    public function isLowStock(): bool
    {
        return $this->type === 'physical' && $this->stock > 0 && $this->stock <= 3;
    }

    public function formattedPrice(): string
    {
        return number_format($this->finalPrice(), 0, '.', "\u{00A0}") . "\u{00A0}֏";
    }

    public function imageUrl(): string
    {
        return asset('images/' . ($this->image ?? 'placeholder.jpg'));
    }
}

Էջավորված Collection-ից resource-ը վերադարձնում է {"data": [...], "links": {...}, "meta": {...}}։ data և meta՝ ճիշտ այն ծրարը, որ ձեռքով կառուցեցինք Դաս 17-ում։

whenLoaded('category')-ը ավելացնում է կատեգորիան միայն եթե այն բեռնված է with()-ով։ Այդպես resource-ը ինքը չի առաջացնում N+1 հարցումներ։

route('product.show', $this->resource)-ը փոխանցում է մոդելը, և Laravel-ը ինքն է վերցնում slug-ը՝ ուղու {product:slug}-ից։

11Դասարանական

  1. Ստեղծիր technomarket_laravel բազան, թարմացրու .env-ը և ավելացրու role-ը users միգրացիայում
  2. Գրիր categories, products, orders, order_items միգրացիաները և գործարկիր php artisan migrate
  3. Ստեղծիր չորս մոդելը կապերով, տեղափոխիր Դաս 12-ի մեթոդները Product, ավելացրու scopeActive և shouldBeStrict()
  4. Գրիր ProductFactory-ն և DatabaseSeeder-ը, գործարկիր php artisan migrate:fresh --seed
  5. Փոխարինիր բոլոր DB::table-երը Eloquent-ով, միացրու route model binding-ը և իրագործիր destroy-ը
Հինգ պարտադիր փորձարկում

1. Հանիր with('category')-ը և բացիր գլխավոր էջը։ Պիտի տեսնես LazyLoadingViolationException։ Վերադարձրու։
2. tinker-ում՝ Product::first()->prise։ Խիստ ռեժիմով և առանց դրա։
3. tinker-ում՝ User::create(['name'=>'X','email'=>'x@x.test','password'=>'x','role'=>'admin'])։ Ի՞նչ եղավ role-ի հետ։ ⭐
4. Ադմինում անջատիր ապրանքը (is_active = false) և բացիր դրա հանրային հասցեն → 404։ Ապա հանիր abort_unless-ը և կրկնիր։
5. Ջնջիր ապրանքը և tinker-ում ստեղծիր նորը նույն slug-ով։ Կարդա սխալը։ Ապա ստուգիր Product::withTrashed()->where('slug', ...)->exists()։

Երրորդը՝ բլոկի կարևորագույն փորձերից

Խիստ ռեժիմում կստանաս սխալ role-ի մասին։ Առանց դրա օգտագործողը կստեղծվի որպես manager՝ role-ը լուռ կանտեսվի։

Երկու դեպքում էլ հարձակումը չանցավ։ Իսկ հիմա փորձիր նույնը՝ role-ը ավելացնելով $fillable-ում։ Այդ մեկ բառը տարբերությունն է օգտագործողի և ադմինի միջև։ Ջնջիր այն անմիջապես։

12Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Eloquent և միգրացիաներ», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր դասարանականի 5 քայլը և հինգ փորձարկումը։ Ուղարկիր սքրինշոթներ։
  2. Փոխիր <x-product-card>-ը՝ formattedPrice(), hasDiscount(), imageUrl()։
  3. Ստեղծիր նոր միգրացիա add_brand_to_products_table։ Գործարկիր, ապա migrate:rollback։ Ի՞նչ արեց down()-ը։
  4. Ավելացրու /category/{category:slug} էջ $category->products()->active()->paginate(12)-ով։
  5. Ցուցադրիր N+1-ը թվերով․ DB::enableQueryLog(), ապա count(DB::getQueryLog())՝ with()-ով և առանց դրա։
  6. Ավելացրու ադմինում «Ջնջվածներ» էջ onlyTrashed()-ով և «Վերականգնել» կոճակ restore()-ով։
  7. Ստեղծիր /api/products ուղի ProductResource-ով և համեմատիր պատասխանը Դաս 17-ի հետ։
  8. Խնդրիր AI-ին գրել Eloquent հարցում կատեգորիաներով և ապրանքներով։ Ստուգիր՝ կա՞ with(), թե՞ N+1։ Կա՞ $guarded = []։

Ստուգաթերթ

  • Ամբողջ սխեման միգրացիաներում է, ոչ ձեռքով
  • Կատարված միգրացիաները չեն խմբագրվում
  • Ամեն մոդել ունի $fillable, ոչ $guarded = []
  • role-ը $fillable-ում չէ
  • Գները integer են, ոչ float
  • shouldBeStrict()-ը միացված է զարգացման ժամանակ
  • Կապերը բեռնվում են with()-ով
  • Ապրանքները ջնջվում են փափուկ
  • Հին պատվերները ցույց են տալիս գինը գնման պահին
  • Հանրային show-ը ստուգում է is_active
  • API-ն վերադարձնում է Resource, ոչ մոդել

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու միգրացիաները լավ են schema.sql-ից
  • Գիտե՞մ, ինչու չի կարելի խմբագրել կատարված միգրացիան
  • Կարո՞ղ եմ բացատրել N+1-ը և դրա լուծումը
  • Հասկանո՞ւմ եմ $fillable-ի և $guarded-ի տարբերությունը և ինչու սպիտակ ցուցակ
  • Գիտե՞մ is_active-ի և deleted_at-ի տարբերությունը
  • Կարո՞ղ եմ նշել երկու բան, որ կորցրինք Դաս 12–13-ից Eloquent-ին անցնելիս
Հաջորդ դասին

Վալիդացիա և auth — Laravel-ի վերջին դասը և Դաս 6-ի ու 16-ի հայելին։

validateProduct()-ը կդառնա $request->validate([...]) կամ Form Request դաս, $old-ը՝ old(), սխալները դաշտերի կողքին՝ @error։ Ադմինի store-ը և update-ը վերջապես կաշխատեն $request->validated()-ով։
