Գները երբեք FLOAT չեն

FLOAT-ը և DOUBLE-ը պահում են մոտավոր արժեք։ 0.1 + 0.2-ը այնտեղ 0.3 չէ։

Հազար պատվերի վրա կուտակվում է սխալ, և հաշվապահությունը չի համընկնում։ Դա ոչ թե տեսական խնդիր է, այլ դասական սխալ, որ գտնում են աուդիտի ժամանակ։

DECIMAL(10,2) — ճշգրիտ, 10 նիշ ընդհանուր, 2-ը՝ ստորակետից հետո։

Այլընտրանք՝ գումարը պահել INT-ով, դրամով (ոչ լումայով)։ Հայկական դրամի դեպքում դա հաճախ ամենապարզն է։

Կոդավորում

Բազան և աղյուսակները՝ utf8mb4, համեմատումը՝ utf8mb4_unicode_ci։

Հին utf8-ը MySQL-ում իրական UTF-8 չէ — այն չի պահում էմոջի և որոշ նիշեր։ Անունը շփոթեցնող է, բայց utf8mb4-ն է ճիշտը։

03ՏեխնոՄարկետի սխեման

schema.sql
CREATE DATABASE technomarket
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE technomarket;

CREATE TABLE categories (
    id    INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name  VARCHAR(100) NOT NULL,
    slug  VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB;

CREATE TABLE products (
    id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    category_id INT UNSIGNED NULL,

    name        VARCHAR(200) NOT NULL,
    slug        VARCHAR(200) NOT NULL UNIQUE,
    specs       VARCHAR(255) NULL,
    description TEXT NULL,

    price       DECIMAL(10,2) NOT NULL,
    discount    TINYINT UNSIGNED NOT NULL DEFAULT 0,
    stock       INT UNSIGNED NOT NULL DEFAULT 0,

    image       VARCHAR(255) NULL,
    is_active   TINYINT(1) NOT NULL DEFAULT 1,
    created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_products_category
        FOREIGN KEY (category_id) REFERENCES categories(id)
        ON DELETE SET NULL,

    INDEX idx_active_price (is_active, price),
    INDEX idx_category (category_id)
) ENGINE=InnoDB;
ՈրոշումԻնչու
slug դաշտՄարդկային հասցեներ՝ /product/laptop-pro-14 (Դաս 14)
is_activeԹաքցնել առանց ջնջելու ⭐
discount առանձինՀին գինը մնում է, զեղչը՝ ժամանակավոր
created_atՏեսակավորում «նոր»-ով, վերլուծություն
ON DELETE SET NULLԿատեգորիան ջնջելիս ապրանքները մնում են
ENGINE=InnoDBԱրտաքին բանալիներ և տրանզակցիաներ
Ինչո՞ւ order_items-ում կրկնում ենք գինը

Թվում է ավելորդություն — գինն արդեն կա products-ում։

Բայց գները փոխվում են։ Եթե պատվերի էջում գինը վերցնես products-ից, ապա վաղը, երբ գինը բարձրանա, հին պատվերները կցուցադրեն նոր գինը։ Հաշվապահությունը կփլվի։

order_items.price-ը գինն է գնման պահին։ Դա պատմական փաստ է, ոչ ընթացիկ արժեք։

Սա տվյալների մոդելավորման կարևոր սկզբունք է․ պատմությունը սառեցվում է, ընթացիկ վիճակը՝ ոչ։

04SELECT

հիմունքներ
-- Բոլորը (միայն ուսումնական նպատակով)
SELECT * FROM products;

-- ✅ Հստակ դաշտեր
SELECT id, name, price FROM products;

-- Պայման
SELECT id, name, price
FROM products
WHERE is_active = 1 AND price < 200000;

-- Տեսակավորում և սահման
SELECT id, name, price
FROM products
WHERE is_active = 1
ORDER BY price ASC
LIMIT 12 OFFSET 0;
Ինչո՞ւ ոչ SELECT *

1. Փոխանցվում է ավելորդ տվյալ — description-ը ցուցակում պետք չէ
2. Երբ ավելացնես նոր դաշտ, կոդը լուռ կսկսի քաշել այն
3. Ինդեքսը հաճախ չի կարող ծածկել հարցումը

Կատալոգի էջում description-ը 5 ԿԲ է × 12 ապրանք = 60 ԿԲ ավելորդ ամեն հարցումով։

WHERE-ի հնարավորությունները
WHERE price BETWEEN 50000 AND 200000
WHERE category_id IN (1, 3, 5)
WHERE name LIKE '%նոութբուք%'
WHERE description IS NULL
WHERE discount > 0 AND stock > 0
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
NULL-ը հատուկ է

WHERE description = NULL — երբեք ոչինչ չի վերադարձնի։

NULL-ը «անհայտ» է, ոչ արժեք։ Անհայտը հավասար չէ անհայտին։

Ճիշտ ձևը՝ IS NULL և IS NOT NULL։ Սա ամենահաճախ սխալներից է սկսնակների մոտ։

հաշվարկվող դաշտեր
SELECT
    id,
    name,
    price,
    discount,
    ROUND(price * (100 - discount) / 100) AS final_price
FROM products
WHERE is_active = 1
ORDER BY final_price ASC;
Ուշադրություն՝ սա փոխարինում է PHP-ի ֆունկցիային

Դաս 3-ում գրեցինք finalPrice() PHP-ում և տեսակավորեցինք usort-ով։

Հիմա բազան կարող է անել երկուսն էլ՝ ավելի արագ և առանց բոլոր տողերը հիշողություն բերելու։

Ընդհանուր կանոն՝ ֆիլտրում և տեսակավորում՝ բազայում, ձևաչափում՝ PHP-ում։

05JOIN

ապրանքները կատեգորիայի անունով
SELECT
    p.id,
    p.name,
    p.price,
    c.name AS category_name
FROM products p
LEFT JOIN categories c ON c.id = p.category_id
WHERE p.is_active = 1
ORDER BY p.price;
INNER JOIN LEFT JOIN products categories products categories ●─────────● ●─────────● ●─────────● ●─────────● ● ✗ ●─────── NULL Միայն զույգերը Բոլոր ապրանքները, նույնիսկ առանց կատեգորիայի
Ե՞րբ որը

INNER JOIN — երբ կապը պարտադիր է։ Առանց կատեգորիայի ապրանքը պարզապես չի ցուցադրվի։

LEFT JOIN — երբ կապը ցանկալի է։ Ապրանքը ցուցադրվում է, կատեգորիան կլինի NULL։

Մեր սխեմայում category_id-ն NULL կարող է լինել (ON DELETE SET NULL) → ուստի LEFT JOIN։

Դասական սխալ՝ INNER JOIN և հետո զարմանք, թե ինչու ապրանքների կեսը անհետացավ։

երեք աղյուսակ — պատվերի բովանդակությունը
SELECT
    o.id            AS order_id,
    o.customer_name,
    p.name          AS product_name,
    oi.quantity,
    oi.price,
    oi.quantity * oi.price AS line_total
FROM orders o
INNER JOIN order_items oi ON oi.order_id = o.id
INNER JOIN products p    ON p.id = oi.product_id
WHERE o.id = 42;

06Ագրեգացիա

հինգ ֆունկցիա
SELECT
    COUNT(*)     AS total,
    AVG(price)   AS avg_price,
    MIN(price)   AS cheapest,
    MAX(price)   AS most_expensive,
    SUM(stock)   AS total_stock
FROM products
WHERE is_active = 1;
GROUP BY — ապրանքների քանակը կատեգորիաներով
SELECT
    c.name,
    COUNT(p.id) AS product_count,
    ROUND(AVG(p.price)) AS avg_price
FROM categories c
LEFT JOIN products p ON p.category_id = c.id AND p.is_active = 1
GROUP BY c.id, c.name
HAVING product_count > 0
ORDER BY product_count DESC;
WHERE vs HAVING

WHERE — ֆիլտրում է տողերը խմբավորումից առաջ
HAVING — ֆիլտրում է խմբերը խմբավորումից հետո

COUNT()-ով չես կարող ֆիլտրել WHERE-ում, որովհետև այն դեռ գոյություն չունի։

Ուշադրություն նաև՝ պայմանը p.is_active = 1 դրված է ON-ում, ոչ WHERE-ում։ WHERE-ում այն կչեղարկեր LEFT JOIN-ը՝ դատարկ կատեգորիաները կանհետանային։

07Ինդեքսներ

Առանց ինդեքսի Ինդեքսով Կարդալ 1-ին տողը Բացել «բովանդակությունը» Կարդալ 2-րդ տողը Գտնել էջը ... Գնալ ուղիղ այնտեղ Կարդալ 100 000-րդը ~100 000 գործողություն ~17 գործողություն
ստեղծել և ստուգել
CREATE INDEX idx_price ON products(price);

-- Կոմպոզիտային — կարգը կարևոր է ⭐
CREATE INDEX idx_active_price ON products(is_active, price);

-- Ստուգում՝ օգտագործվու՞մ է
EXPLAIN SELECT id, name FROM products
WHERE is_active = 1 ORDER BY price;
ԻնդեքսավորելՉինդեքսավորել
Դաշտեր WHERE-ումՓոքր աղյուսակներ
Դաշտեր JOIN-ումԴաշտեր, որ հազվադեպ են որոնվում
Դաշտեր ORDER BY-ումՀաճախ փոփոխվող դաշտեր
UNIQUE դաշտեր՝ slugԱմեն դաշտ «ամեն դեպքում»
Ինդեքսը անվճար չէ

Ամեն ինդեքս դանդաղեցնում է INSERT-ը, UPDATE-ը և DELETE-ը, և զբաղեցնում է տեղ։

Ամեն դաշտի վրա ինդեքս դնելը վնասակար է։ Դիր այնտեղ, որտեղ իսկապես որոնում ես — և ստուգիր EXPLAIN-ով, որ այն օգտագործվում է։

Կոմպոզիտային ինդեքսում կարգը կարևոր է․ (is_active, price)-ը աշխատում է WHERE is_active = 1 ORDER BY price-ի համար, բայց ոչ միայն ORDER BY price-ի։

-- 8. Միջին գինը կատեգորիաներով
-- 9. Կատեգորիաներ, որտեղ 3-ից ավելի ապրանք կա (HAVING)
-- 10. Վերջին 5 ավելացված ապրանքը
Ինչո՞ւ այսօր առանց PHP-ի

SQL-ը առանձին լեզու է, ոչ PHP-ի մաս։ Երբ այն սովորում ես PHP-ի տողերի ներսում, երկու բան միանգամից է լինում՝ և SQL, և PDO — և հեշտ է շփոթել, թե որ սխալն որտեղից է։

phpMyAdmin-ում սխալը անմիջապես երևում է հստակ հաղորդագրությամբ։ Երբ վաղը հարցումը տեղափոխես PHP, արդեն կիմանաս, որ SQL-ը ճիշտ է։

Դա նույն սկզբունքն է, ինչ HTML բլոկի Դաս 13-ում՝ մեկ բան միանգամից։

09Տնային աշխատանք

  1. Ստեղծիր բազան և լրացրու 12 ապրանքով և 4 կատեգորիայով։
  2. Գրիր 10 հարցումը։ Ուղարկիր SQL-ը և արդյունքների սքրինշոթերը։
  3. Ցուցադրիր WHERE description = NULL-ի և IS NULL-ի տարբերությունը։
  4. Համեմատիր INNER JOIN-ը և LEFT JOIN-ը՝ ապրանք առանց կատեգորիայի ավելացնելով։
  5. Տեղափոխիր p.is_active = 1-ը ON-ից WHERE։ Ի՞նչ փոխվեց GROUP BY-ի արդյունքում։
  6. Ստեղծիր orders և order_items աղյուսակները և գրիր հարցում պատվերի բովանդակության համար։
  7. Ցուցադրիր, ինչու FLOAT-ը գների համար վատ է․ ստեղծիր փորձնական աղյուսակ և գումարիր 0.1 + 0.2։
  8. Կատարիր EXPLAIN ինդեքսով և առանց դրա։ Ուղարկիր երկու սքրինշոթ։
  9. Ավելացրու ապրանք հայերեն անունով և էմոջիով։ Ստուգիր, որ utf8mb4-ը աշխատում է։
  10. Խնդրիր AI-ին վերանայել քո սխեման։ Ի՞նչ առաջարկեց։ Համաձա՞յն ես։ Ինչու։

Հարցման կարգը

SELECT ← ի՞նչ դաշտեր FROM ← որտեղից JOIN ← ինչի՞ հետ կապել WHERE ← ո՞ր տողերը GROUP BY ← ինչպե՞ս խմբավորել HAVING ← ո՞ր խմբերը ORDER BY ← ինչպե՞ս տեսակավորել LIMIT ← քանի՞սը

Սխեմայի կանոններ

· Գները՝ DECIMAL, երբեք FLOAT · utf8mb4, ոչ utf8 · ENGINE=InnoDB · is_active՝ ջնջելու փոխարեն · Պատմական տվյալները սառեցվում են (order_items.price) · Ինդեքս այնտեղ, որտեղ WHERE / JOIN / ORDER BY

Ստուգաթերթ

  • Բազան utf8mb4 է
  • Ամեն աղյուսակ ունի PRIMARY KEY
  • Գները DECIMAL են
  • Կան արտաքին բանալիներ
  • Կան ինդեքսներ որոնվող դաշտերի վրա
  • EXPLAIN-ը ցույց է տալիս ինդեքսի օգտագործումը
  • Հայերենը և էմոջին ճիշտ են պահվում

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու գները FLOAT չեն
  • Գիտե՞մ INNER-ի և LEFT JOIN-ի տարբերությունը
  • Հասկանո՞ւմ եմ, ինչու = NULL-ը չի աշխատում
  • Կարո՞ղ եմ բացատրել, ինչու order_items-ում կրկնվում է գինը
  • Գիտե՞մ WHERE-ի և HAVING-ի տարբերությունը
Հաջորդ դասին

PDO և SQL injection — անվտանգության երրորդ դասը։ Կմիացնենք PHP-ն բազային և կտեսնենք, ինչու տողերի միացումով հարցում կառուցելը ամենավտանգավոր բանն է, որ կարելի է անել։

Այսօրվա հարցումները կտեղափոխվեն կոդ — և data.php-ը վերջապես կանհետանա։
