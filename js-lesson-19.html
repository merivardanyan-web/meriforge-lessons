01Մեկ հերթ

JavaScript-ը միաթել (single-threaded) լեզու է։ Դա նշանակում է՝ նա կարող է կատարել ուղիղ մեկ հրաման միաժամանակ։ Ոչ երկու։

սովորական՝ սինխրոն կոդ

console.log("2. Ցերեկ");
console.log("3. Երեկո");

02Event loop

JS-ը մենակ չէ։ Բրաուզերն ունի լրացուցիչ մեխանիզմներ, որոնք աշխատում են զուգահեռ։

մեխանիզմը
   ┌──────────────┐
   │  Call Stack  │  ← JS-ի միակ հերթը
   │  (կույտ)     │
   └──────┬───────┘
          │  «սա երկար է, վերցրու»
          ▼
   ┌──────────────┐
   │  Web APIs    │  ← բրաուզերի մասը (ոչ JS)
   │  ժամաչափեր   │     աշխատում է զուգահեռ
   │  ցանց, DOM   │
   └──────┬───────┘
          │  «պատրաստ է»
          ▼
   ┌──────────────┐
   │    Queue     │  ← հերթ՝ սպասող callback-ներ
   └──────┬───────┘
          │
     Event Loop  ← «Կույտը դատա՞րկ է։ Ուրեմն վերցնում եմ հերթից»
          │
          └──────► Call Stack

Ահա ամբողջ գաղտնիքը։ Երբ գրում ես setTimeout(fn, 2000)․

  1. JS-ը չի սպասում։ Նա հանձնում է ժամաչափը բրաուզերին և անմիջապես գնում առաջ։
  2. Բրաուզերը հաշվում է 2 վայրկյանը զուգահեռ։
  3. Ժամանակը լրանալիս callback-ը գնում է հերթ։
  4. Event loop-ը սպասում է, մինչև Call Stack-ը դատարկվի, և միայն այդ ժամանակ վերցնում է հերթից։

03Callback-ներ

Դաս 6-ից գիտենք՝ callback-ը ֆունկցիա է, որ փոխանցում ենք ուրիշին։ Ասինխրոն աշխարհում դա նշանակում է՝ «երբ պատրաստ լինի, կանչիր սա»։

04Callback դժոխք

Խնդիրը սկսվում է, երբ գործողությունները պիտի կատարվեն հերթականությամբ․ նախ վերցնել օգտագործողին, հետո նրա պատվերները, հետո ամեն պատվերի ապրանքները։

💀 callback hell
getUser(1, (err, user) => {
  if (err) return console.error(err);

  getOrders(user.id, (err, orders) => {
    if (err) return console.error(err);

    getProducts(orders[0].id, (err, products) => {
      if (err) return console.error(err);

      getReviews(products[0].id, (err, reviews) => {
        if (err) return console.error(err);

        console.log(reviews);
      });
    });
  });
});
Ինչո՞ւ է սա վատ

Կոդը գնում է աջ, ոչ ներքև։ Սխալի մշակումը կրկնվում է ամեն մակարդակում։ Ամեն նոր քայլ ավելացնում է մեկ մակարդակ։ Կարդալ, փոփոխել կամ փորձարկել գրեթե անհնար է։ Այս ձևը ստացել է իր անունը՝ «callback դժոխք» կամ «մահվան բուրգ»։

Հենց այս խնդիրը լուծելու համար 2015-ին JS-ում հայտնվեցին խոստումները։

05Promise — խոստում

Պատվիրում ես սուրճ։ Բարիստան անմիջապես տալիս է քեզ չեկ։ Չեկը սուրճ չէ — դա խոստում է, որ սուրճը կլինի։ Դու գնում ես նստելու, ոչ թե կանգնում վաճառասեղանի մոտ։

Հետո երկու բան կարող է լինել․ քեզ տալիս են սուրճը (հաջողություն), կամ ասում են՝ «կաթը վերջացավ» (ձախողում)։

ՎիճակՆշանակություն
pendingՍպասում է — դեռ հայտնի չէ
fulfilledԿատարվեց — կա արդյունք
rejectedՁախողվեց — կա սխալ
Կարևոր

Վիճակը փոխվում է ուղիղ մեկ անգամ՝ pending-ից դեպի fulfilled կամ rejected։ Հետ ճանապարհ չկա։ Խոստումը «նորից չի սպասում»։

ստեղծել խոստում
const promise = new Promise((resolve, reject) => {
  // Այս կոդն աշխատում է ԱՆՄԻՋԱՊԵՍ
  console.log("Պատրաստում ենք սուրճը...");

  setTimeout(() => {
    const success = true;

    if (success) {
      resolve("☕ Ձեր սուրճը");      // հաջողություն
    } else {
      reject("Կաթը վերջացավ");       // ձախողում
    }
  }, 2000);
});

console.log(promise);
Console
Պատրաստում ենք սուրճը...
Promise { <pending> }
resolve և reject

Դրանք ֆունկցիաներ են, որոնք JS-ն ինքն է տալիս քեզ։ Անունները սովորույթ են — կարող էիր գրել ok և fail։ resolve-ը կանչելով ասում ես՝ «պատրաստ է, ահա արդյունքը»։

06then · catch · finally

արդյունքի ստացում

  .catch(error => {
    console.error("Սխալ՝", error);

    console.log("Ամեն դեպքում կատարվում է");

07Շղթա — դժոխքի ելքը

Ահա գլխավոր հնարքը․ .then()-ը վերադարձնում է նոր խոստում։ Ուստի դրանք կարելի է շարել իրար հետևից՝ առանց ներդրման։

նույն խնդիրը՝ խոստումներով
getUser(1)
  .then(user => getOrders(user.id))
  .then(orders => getProducts(orders[0].id))
  .then(products => getReviews(products[0].id))
  .then(reviews => console.log(reviews))
  .catch(error => console.error("Ինչ-որ տեղ սխալ եղավ՝", error));

Համեմատիր Բաժին 04-ի բուրգի հետ։ Կոդը գնում է ներքև, ոչ աջ։ Իսկ սխալի մշակումը՝ մեկ տեղում ամբողջ շղթայի համար։

08Ֆունկցիա, որ վերադարձնում է խոստում

callback → Promise
// Հին ձև՝ callback-ով
function getUserOld(id, callback) {
  setTimeout(() => callback(null, { id, name: "Անի" }), 1000);
}

// Նոր ձև՝ Promise-ով
function getUser(id) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id <= 0) {
        reject(new Error("Սխալ id"));
        return;
      }
      resolve({ id, name: "Անի" });
    }, 1000);
  });
}

getUser(1)
  .then(user => console.log(user.name))
  .catch(err => console.error(err.message));

09Promise.all և ընկերները

Երբեմն մի քանի հարցում իրար կախված չեն։ Հերթով սպասելն այդ դեպքում ժամանակի վատնում է։

հերթով — 3 վայրկյան
getUser(1)                                  // 1 վրկ
  .then(() => getProducts())               // + 1 վրկ
  .then(() => getCategories())             // + 1 վրկ
  .then(() => console.log("Պատրաստ"));
զուգահեռ — 1 վայրկյան ✅
Promise.all([
  getUser(1),
  getProducts(),
  getCategories()
])
  .then(([user, products, categories]) => {   // destructuring — Դաս 7
    console.log(user.name);
    console.log(products.length);
    console.log(categories.length);
  })
  .catch(error => console.error("Մեկը ձախողվեց՝", error));
ՄեթոդԵ՞րբ է ավարտվումԵթե մեկը ձախողվի
Promise.allԵրբ բոլորը հաջողվենԱմբողջը ձախողվում է
Promise.allSettledԵրբ բոլորն ավարտվենՈչինչ — տալիս է բոլորի արդյունքը
Promise.raceԵրբ առաջինն ավարտվիԿախված է՝ ով առաջինն էր
Promise.anyԵրբ առաջինը հաջողվիԱնտեսում է ձախողվածները
allSettled — երբ մասնակի արդյունքն էլ է լավ
Promise.allSettled([
  Promise.resolve("Ա"),
  Promise.reject(new Error("Բ ձախողվեց")),
  Promise.resolve("Գ")
]).then(results => {
  results.forEach(r => {
    if (r.status === "fulfilled") {
      console.log("✓", r.value);
    } else {
      console.log("✗", r.reason.message);
    }
  });
});

Ե՞րբ որը

Էջի բեռնման համար բոլոր տվյալները պարտադիր են → Promise.all։
Ցուցադրում ես այն, ինչ ստացվեց, մնացածի փոխարեն՝ սխալի հաղորդագրություն → Promise.allSettled։

10Խաղահրապարակ

Կարդա սա ուշադիր

Այս խաղահրապարակը ցույց է տալիս միայն սինխրոն տողերը։ Ասինխրոն արդյունքները չեն երևա, որովհետև դրանք գալիս են ավելի ուշ՝ արդեն այն բանից հետո, երբ պատուհանն արդեն նկարվել է։

Եվ դա հենց այսօրվա դասն է։ Իսկական արդյունքը տես բրաուզերի Console-ում՝ F12։

Console

11Տնային աշխատանք

  1. Կանխագուշակիր հերթականությունը, հետո ստուգիր․ console.log("A"), setTimeout(...,0) «B»-ով, Promise.resolve().then() «C»-ով, console.log("D")։ Բացատրիր արդյունքը մեկնաբանության մեջ։
  2. Գրիր delay(ms) ֆունկցիան։ Օգտագործիր այն 3 վայրկյանանոց հետհաշվարկի համար՝ 3, 2, 1, «Մեկնարկ»։
  3. Գրիր getRandomNumber(), որը 1 վայրկյան հետո resolve է անում պատահական թիվ։ 50% դեպքերում՝ reject։ Մշակիր երկու դեպքն էլ։
  4. Գրիր getUser(id), որը վերադարձնում է Promise։ Անվավեր id-ի դեպքում՝ reject(new Error(...))։
  5. Ավելացրու getOrders(userId) և կապիր շղթայով՝ նախ օգտագործող, հետո նրա պատվերները։
  6. Կրկնիր նույն շղթան, բայց մոռացիր return-ը մեկ .then-ում։ Գրիր՝ ինչ ստացվեց և ինչու։
  7. Օգտագործիր Promise.all՝ երեք «հարցում» զուգահեռ։ Չափիր ժամանակը console.time-ով և համեմատիր հերթականի հետ։
  8. Կրկնիր նույնը Promise.allSettled-ով, որտեղ մեկը ձախողվում է։ Ցուցադրիր՝ որոնք հաջողվեցին և որը՝ ոչ։
  9. Կառուցիր էջ՝ կոճակով, բեռնման ցուցիչով և արդյունքի բլոկով։ Սեղմելիս ցուցիչը հայտնվում է, 2 վրկ հետո՝ արդյունքը։ Օգտագործիր .finally()։
  10. Նույն էջում 30% հավանականությամբ նմանակիր սխալ և ցույց տուր հաղորդագրություն՝ «Կրկին փորձել» կոճակով։

12Հիշելու թերթիկ

ԳրառումԻնչ է անում
new Promise((res, rej) => {})Ստեղծում է խոստում
resolve(value)Հաջողություն
reject(new Error(...))Ձախողում
.then(v => {})Հաջողության մշակիչ
.catch(e => {})Սխալի մշակիչ
.finally(() => {})Միշտ
Promise.resolve(v)Պատրաստի հաջողված
Promise.all([...])Բոլորը զուգահեռ, բոլորը պարտադիր
Promise.allSettled([...])Բոլորը, անկախ արդյունքից
Promise.race([...])Առաջին ավարտվածը

Երեք կանոն, որ պիտի հիշես

  1. Ամբողջ սինխրոն կոդն առաջինն է։ Ասինխրոնը՝ միշտ հետո։
  2. .then-ում վերադարձրու խոստումը։ Առանց return-ի շղթան կոտրվում է։
  3. Միշտ գրիր .catch։ Առանց դրա սխալը լուռ կկորչի։

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել՝ ինչու setTimeout(fn, 0)-ը վերջում է աշխատում
  • Գիտե՞մ խոստումի երեք վիճակը
  • Հասկանո՞ւմ եմ, ինչո՞ւ է callback դժոխքը վատ
  • Կարո՞ղ եմ գրել ֆունկցիա, որ վերադարձնում է Promise
  • Գիտե՞մ տարբերությունը all-ի և allSettled-ի միջև
Հաջորդ դասին

fetch և async/await։ Վերջապես կխոսենք իսկական սերվերի հետ և կստանանք իրական տվյալներ։ Իսկ async/await-ը թույլ կտա խոստումները գրել այնպես, ասես սովորական սինխրոն կոդ լինեն։
