01Գաղտնաբառեր

Հիմնական կանոն

Գաղտնաբառը երբեք չի պահվում։ Պահվում է դրա հեշը։

Հեշավորումը միակողմանի է․ գաղտնաբառից հեշ ստանալը հեշտ է, հակառակը՝ գործնականում անհնար։

Դա նշանակում է, որ բազայի արտահոսքի դեպքում հարձակվողը գաղտնաբառեր չի ստանում։ Իսկ արտահոսքերը լինում են՝ SQL injection, աշխատակցի սխալ, չպաշտպանված պահուստավորում։

ՄոտեցումԳնահատական
Բաց տեքստ❌ Աններելի
md5(), sha1()❌ Կոտրվում է վայրկյաններում
sha256 + աղ⚠️ Ավելի լավ, բայց չափազանց արագ
password_hash()✅ Ճիշտ ⭐
Ինչո՞ւ «չափազանց արագ»-ը վատ է

Սովորական հեշ-ֆունկցիաները նախագծված են արագության համար։ Ժամանակակից վիդեոքարտը հաշվում է միլիարդավոր sha256 վայրկյանում։

Դա նշանակում է, որ հարձակվողը կարող է փորձել բոլոր տարածված գաղտնաբառերը րոպեների ընթացքում։

password_hash()-ը օգտագործում է bcrypt՝ դիտավորյալ դանդաղ ալգորիթմ։ Մեկ ստուգումը ~50-100 մվ է։ Օգտագործողի համար անտեսանելի, հարձակվողի համար՝ միլիոնավոր անգամ ավելի թանկ։

օգտագործում
// Գրանցում
$hash = password_hash($password, PASSWORD_DEFAULT);

// $2y$12$LQv3c1yqBw...  — 60 նիշ
// Աղն արդեն ներսում է, առանձին պահել պետք չէ ⭐

// Ստուգում
if (password_verify($password, $hash)) {
    // ճիշտ է
}

// Նույն գաղտնաբառը → ամեն անգամ ՏԱՐԲԵՐ հեշ
// (պատահական աղի պատճառով)
// Ուստի հեշերը երբեք չեն համեմատվում === -ով ⚠️

Ինչո՞ւ VARCHAR(190) և VARCHAR(255)

190 — utf8mb4-ում ամեն նիշը մինչև 4 բայտ է, իսկ ինդեքսի սահմանը հին MySQL-ում 767 բայտ էր։ 190 × 4 = 760։ Անվտանգ ընտրություն UNIQUE դաշտի համար։

255 — bcrypt-ը 60 նիշ է, բայց PASSWORD_DEFAULT-ը ապագայում կարող է փոխվել ավելի երկար ալգորիթմի։ Դաշտը պիտի պատրաստ լինի։

02Auth ծառայությունը

src/Services/Auth.php
<?php
declare(strict_types=1);

namespace App\Services;

use App\Models\User;
use App\Repositories\UserRepository;

class Auth
{
    private ?User $cached = null;

    public function __construct(
        private readonly UserRepository $users,
    ) {}

    public function attempt(string $email, string $password): bool
    {
        $user = $this->users->findByEmail($email);

        // ⭐ Ստուգում ենք ՄԻՇՏ, նույնիսկ երբ օգտագործողը չկա
        $hash = $user?->passwordHash
            ?? '$2y$12$' . str_repeat('x', 53);

        if (!password_verify($password, $hash) || $user === null) {
            return false;
        }

        // Հեշը հնացե՞լ է — վերահեշավորում ենք
        if (password_needs_rehash($hash, PASSWORD_DEFAULT)) {
            $this->users->updatePassword(
                $user->id,
                password_hash($password, PASSWORD_DEFAULT)
            );
        }

        // ⭐ Նոր ID սեսիայի համար — session fixation-ի դեմ
        session_regenerate_id(true);

        $_SESSION['user_id']    = $user->id;
        $_SESSION['login_time'] = time();

        return true;
    }

    public function logout(): void
    {
        $_SESSION = [];

        if (ini_get('session.use_cookies')) {
            $p = session_get_cookie_params();
            setcookie(session_name(), '', time() - 42000,
                $p['path'], $p['domain'], $p['secure'], $p['httponly']);
        }

        session_destroy();
    }

    public function user(): ?User
    {
        if ($this->cached !== null) {
            return $this->cached;
        }

        $id = $_SESSION['user_id'] ?? null;

        return $this->cached = $id
            ? $this->users->find((int) $id)
            : null;
    }

    public function check(): bool
    {
        return $this->user() !== null;
    }

    public function isAdmin(): bool
    {
        return $this->user()?->role === 'admin';
    }
}
Խորացված · ինչու ստուգում ենք հեշը, նույնիսկ երբ օգտագործողը չկա

Եթե անմիջապես վերադարձնեինք false, պատասխանը կգար ~1 մվ-ում։ Իսկ գոյություն ունեցող հասցեի դեպքում՝ ~80 մվ (bcrypt)։

Հարձակվողը չափում է ժամանակը և պարզում է, որ էլ. հասցեն գրանցված է։ Դա կոչվում է timing attack, և այն թույլ է տալիս հավաքել գործող հասցեների ցուցակ։

Կեղծ հեշի ստուգումը հավասարեցնում է ժամանակը։ Նույն պատճառով սխալի հաղորդագրությունը պիտի լինի ընդհանուր՝ «Սխալ էլ. հասցե կամ գաղտնաբառ», ոչ «Այդպիսի օգտագործող չկա»։

session_regenerate_id(true) — session fixation

Հարձակումը՝ հարձակվողը քեզ ուղարկում է հղում ?PHPSESSID=abc123։ Դու մուտք ես գործում, և այդ ID-ն դառնում է քո նույնականացված սեսիան։ Հարձակվողն արդեն գիտի ID-ն։

Դաս 7-ի use_strict_mode-ը մասամբ պաշտպանում է։ Բայց հուսալի լուծումը՝ մուտքի պահին տալ նոր ID։

true արգումենտը ջնջում է հին սեսիայի ֆայլը։ Առանց դրա այն մնում է սերվերում։

Խորացված · password_needs_rehash

PHP-ի հետագա տարբերակներում PASSWORD_DEFAULT-ը կարող է փոխվել ավելի ուժեղ ալգորիթմի։

Այս երեք տողը ապահովում է, որ հին օգտագործողների գաղտնաբառերը աստիճանաբար թարմացվեն՝ ամեն մուտքի ժամանակ, առանց նրանց մասնակցության։

Առանց դրա տասը տարի հետո բազայում կլինեն 2015-ի ալգորիթմով հեշեր։

03Ստուգումը՝ մեկ տեղում

Խնդիրը, որ լուծում ենք

Ադմինի ուղիները չորսն են։ Ստուգումը կարելի էր գրել ամեն կոնտրոլերի մեթոդի սկզբում։

Բայց վաղը ավելացնում ես հինգերորդը և մոռանում ես։ Դա նկատելի չէ․ էջն աշխատում է։ Այն պարզապես բաց է բոլորի համար։

Դաս 15-ում ամեն հարցում սկսեցինք անցկացնել մեկ ֆայլով՝ public/index.php։ Հիմա դա օգտագործում ենք․ ստուգումը դնում ենք Router-ից առաջ, և այն աշխատում է ամեն /admin-ով սկսվող հասցեի համար, նաև այն ուղիների, որոնք դեռ չեն գրվել։

Հարցում │ ▼ public/index.php │ ├─ POST է՞ → _token-ը ճի՞շտ է ոչ → 419 │ ├─ /admin… է՞ → մուտք գործա՞ծ է ոչ → /login │ ▼ Router → Կոնտրոլեր → Response
public/index.php — երկու ստուգում Router-ից առաջ
$request = Request::fromGlobals();

try {

Սա middleware-ի գաղափարն է

Կոդ, որը կատարվում է ամեն հարցումից առաջ և կարող է կանգնեցնել այն, կոչվում է middleware։ Մենք գրեցինք այն ամենապարզ ձևով՝ երկու if front controller-ում։

Կանոնը, որ այստեղից պետք է հիշել․ պաշտպանությունը կապվում է հասցեի հետ, ոչ թե կոնտրոլերի ներսի կոդի։ Այդպես այն մոռանալ հնարավոր չէ։

Laravel-ում նույնը գրվում է Route::middleware('auth')->group(...), և ամեն ստուգում առանձին դաս է (Դաս 21)։ Գաղափարը նույնն է, պարզապես ավելի կազմակերպված։

04CSRF

Հարձակում, որի մասին շատերը չգիտեն

Cross-Site Request Forgery։ Այն չի գողանում տվյալները — այն ստիպում է քեզ կատարել գործողություն առանց քո իմացության։

Սցենար 1. Դու մուտք ես գործում technomarket.test/admin Բրաուզերում պահվում է PHPSESSID cookie-ն 2. Չես դուրս գալիս — պարզապես բացում ես նոր ներդիր 3. Բացում ես հարձակվողի էջը (գովազդ, ֆորում, նամակ) 4. Այդ էջում կա թաքնված ֆորմա՝ <form action="https://technomarket.test/admin/delete" method="post"> <input type="hidden" name="id" value="5"> </form> <script>document.forms[0].submit()</script> 5. Բրաուզերը ուղարկում է հարցումը ԵՎ ԿՑՈՒՄ Է ՔՈ COOKIE-Ն ⚠️ 6. Սերվերը տեսնում է վավեր սեսիա → ջնջում է ապրանքը Դու ոչինչ չես նկատել։
Ինչո՞ւ սեսիան չի փրկում

Սեսիան պատասխանում է հարցին «ո՞վ է սա»։ CSRF-ը այդ հարցի պատասխանը չի կեղծում — հարցումն իսկապես քեզնից է։

Բացակայում է պատասխանը մյուս հարցին՝ «իսկապե՞ս դու ես ուզել այս գործողությունը»։

Հենց դա է ապացուցում CSRF token-ը։

Ինչպես է աշխատում token-ը Սերվերը գեներացնում է պատահական տող և պահում այն ՍԵՍԻԱՅՈՒՄ Ամեն ֆորմա կրում է նույն տողը hidden դաշտում POST-ի ժամանակ սերվերը համեմատում է դրանք Հարձակվողի էջը ՉԻ ԿԱՐՈՂ իմանալ token-ը՝ · այն սեսիայում է, ոչ cookie-ում · JS-ը ուրիշ դոմենից չի կարդում քո էջը (CORS) → հարցումը մերժվում է
src/helpers.php — երեք ֆունկցիա
function csrf_token(): string
{
    return $_SESSION['_csrf_token'] ??= bin2hex(random_bytes(32));
}

function csrf_field(): string
{
    return '<input type="hidden" name="_token" value="' . e(csrf_token()) . '">';
}

function csrf_verify(?string $token): bool
{
    $stored = $_SESSION['_csrf_token'] ?? '';

    if ($stored === '' || $token === null || $token === '') {
        return false;
    }

    return hash_equals($stored, $token);   // ⭐ ոչ ===
}
hash_equals, ոչ ===

===-ը կանգ է առնում առաջին տարբերվող նիշի վրա։ Դա նշանակում է, որ համեմատման տևողությունը կախված է նրանից, քանի նիշ է համընկել։

Հարձակվողը կարող է նիշ առ նիշ վերականգնել token-ը՝ չափելով պատասխանի ժամանակը։

hash_equals-ը միշտ համեմատում է ամբողջ երկարությունը։ Դա նույն timing attack-ի դեմ պաշտպանությունն է, ինչ բաժին 02-ի խորացված նշումում։

Հիշո՞ւմ ես Դաս 2-ը՝ «PHP-ում ==-ը անվտանգության խոցելիություն է»։ Ահա դեպք, որտեղ նույնիսկ ===-ը բավարար չէ։

Ինչո՞ւ միայն POST

Բաժին 03-ի ստուգումը աշխատում է միայն POST-ի համար։ GET-ը ըստ սահմանման չպիտի փոխի վիճակը (Դաս 6)։ Եթե GET-ը ինչ-որ բան ջնջում է, խնդիրը CSRF-ը չէ, այլ ճարտարապետությունը։

Հենց դրա համար Դաս 11-ում ջնջումը դարձրինք POST։ Այն ժամանակ պատճառը բոտերն էին — հիմա ավելանում է երկրորդը՝ CSRF-ից պաշտպանվելու հնարավորությունը։

պարզ պաշտպանություն brute force-ից
class LoginThrottle
{
    private const MAX = 5;
    private const LOCK_SECONDS = 900;   // 15 րոպե

    public function tooManyAttempts(): bool
    {
        $data = $_SESSION['login_attempts'] ?? ['count' => 0, 'until' => 0];

        return $data['count'] >= self::MAX && $data['until'] > time();
    }

    public function hit(): void
    {
        $data = $_SESSION['login_attempts'] ?? ['count' => 0, 'until' => 0];
        $data['count']++;
        $data['until'] = time() + self::LOCK_SECONDS;

        $_SESSION['login_attempts'] = $data;
    }

    public function clear(): void
    {
        unset($_SESSION['login_attempts']);
    }
}

06AuthController

src/Controllers/AuthController.php
class AuthController
{
    public function __construct(
        private readonly Auth $auth,
        private readonly LoginThrottle $throttle,
    ) {}

    public function form(Request $request): Response
    {
        if ($this->auth->check()) {
            return Response::redirect($this->auth->isStaff() ? '/admin' : '/');
        }

        return Response::html(
            page('pages/login', ['error' => flash('auth_error')], 'Մուտք')
        );
    }

    public function login(Request $request): Response
    {
        if ($this->throttle->tooManyAttempts()) {
            flash('auth_error', 'Չափազանց շատ փորձեր։ Փորձեք 15 րոպեից։');
            return Response::redirect('/login');
        }

        $email    = $request->input('email');
        $password = $request->input('password');

        if (!$this->auth->attempt($email, $password)) {
            $this->throttle->hit();

            // ⭐ Ընդհանուր հաղորդագրություն
            flash('auth_error', 'Սխալ էլ. հասցե կամ գաղտնաբառ');

            return Response::redirect('/login');
        }

        $this->throttle->clear();

        // Աշխատակիցը՝ ադմին վահանակ, հաճախորդը՝ գլխավոր էջ (բաժին 07)
        $to = $_SESSION['intended'] ?? ($this->auth->isStaff() ? '/admin' : '/');
        unset($_SESSION['intended']);

        return Response::redirect($to);
    }

    public function logout(Request $request): Response
    {
        $this->auth->logout();

        return Response::redirect('/');
    }
}

public function create(string $email, string $passwordHash, string $name): int
{
    $stmt = $this->pdo->prepare(
        'INSERT INTO users (email, password_hash, name)
         VALUES (:email, :hash, :name)'
    );

    $stmt->execute([
        'email' => $email,
        'hash'  => $passwordHash,
        'name'  => $name,
    ]);

    return (int) $this->pdo->lastInsertId();
}

src/Controllers/RegisterController.php
class RegisterController
{
    public function __construct(
        private readonly UserRepository $users,
        private readonly Auth $auth,
    ) {}

    public function form(Request $request): Response
    {
        if ($this->auth->check()) {
            return Response::redirect('/');
        }

        return Response::html(page('pages/register', ['errors' => [], 'old' => []], 'Գրանցում'));
    }

    public function store(Request $request): Response
    {
        $data = [
            'name'  => $request->input('name'),
            'email' => mb_strtolower($request->input('email')),
        ];
        $password = $request->input('password');

        // 1. Վալիդացիա
        $errors = $this->validate($data, $password, $request->input('password_confirmation'));

        // 2. Հասցեն արդեն կա՞
        if ($errors === [] && $this->users->findByEmail($data['email']) !== null) {
            $errors['email'] = 'Այս էլ. հասցեով արդեն կա հաշիվ';
        }

        // 3–4. Պահպանում
        if ($errors === []) {
            try {
                $id = $this->users->create(
                    $data['email'],
                    password_hash($password, PASSWORD_DEFAULT),
                    $data['name'],
                );
            } catch (\PDOException $e) {
                if ($e->getCode() !== '23000') {   // 23000 = UNIQUE-ի խախտում
                    throw $e;
                }
                $errors['email'] = 'Այս էլ. հասցեով արդեն կա հաշիվ';
            }
        }

        if ($errors !== []) {
            // ⭐ Գաղտնաբառը երբեք չի վերադառնում ֆորմա
            return Response::html(page('pages/register', ['errors' => $errors, 'old' => $data], 'Գրանցում'));
        }

        // 5. Մուտք
        $this->auth->login($this->users->find($id));
        flash('success', 'Բարի գալուստ, ' . $data['name']);

        return Response::redirect('/');
    }

    private function validate(array $data, string $password, string $confirm): array
    {
        $errors = [];

        $len = mb_strlen($data['name']);
        if ($len < 2 || $len > 100) {
            $errors['name'] = 'Անունը՝ 2-ից 100 նիշ';
        }

        if (!filter_var($data['email'], FILTER_VALIDATE_EMAIL) || mb_strlen($data['email']) > 190) {
            $errors['email'] = 'Ստուգիր էլ. հասցեն';
        }

        if (mb_strlen($password) < 8) {
            $errors['password'] = 'Գաղտնաբառը՝ առնվազն 8 նիշ';
        } elseif ($password !== $confirm) {
            $errors['password_confirmation'] = 'Գաղտնաբառերը չեն համընկնում';
        }

        return $errors;
    }
}

findByEmail()-ը տալիս է հասկանալի հաղորդագրություն, բայց չի երաշխավորում ոչինչ։ Եթե երկու հարցում գան նույն միլիվայրկյանին, երկուսն էլ կտեսնեն, որ հասցեն ազատ է։ Իսկական երաշխիքը բազայի UNIQUE-ն է․ երկրորդ INSERT-ը կմերժվի 23000 կոդով, և մենք այն վերածում ենք նույն հաղորդագրության։ Կանոն՝ եզակիությունը ապահովում է բազան, ոչ PHP-ն։

09Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Մուտք, դերեր, CSRF», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր դասարանականի 6 քայլը և յոթ փորձարկումը։
  2. Ուղարկիր սքրինշոթներ CSRF հարձակման՝ պաշտպանությամբ և առանց դրա։
  3. Ցուցադրիր, որ նույն գաղտնաբառը տալիս է տարբեր հեշեր։ Բացատրիր, ինչու ===-ը այստեղ չի աշխատի։
  4. Չափիր password_hash()-ի ժամանակը microtime()-ով։ Համեմատիր md5()-ի հետ։
  5. Խորացված․ ցուցադրիր timing attack-ի հնարավորությունը՝ չափիր մուտքի ժամանակը գոյություն ունեցող և չգոյություն ունեցող հասցեով, կեղծ հեշի ստուգումով և առանց դրա։
  6. Ցուցադրիր session_regenerate_id()-ի աշխատանքը՝ համեմատիր PHPSESSID-ն մուտքից առաջ և հետո։
  7. Ավելացրու երրորդ ստուգում․ /admin/delete-ը թույլատրված է միայն role = 'admin'-ին ($auth->isAdmin()), մյուսներին՝ 403։
  8. Ավելացրու ադմին էջ /manage/orders հասցեով (առանց /admin-ի)։ Պաշտպանվա՞ծ է։ Ինչ կանոն պետք է պահել, որ սա չպատահի։
  9. Ավելացրու «Բարև, [անուն]» և ելքի կոճակ header-ում։
  10. DevTools-ով ավելացրու գրանցման ֆորմային <input name="role" value="admin"> և ուղարկիր։ HeidiSQL-ում ստուգիր նոր օգտագործողի դերը։ Բացատրիր, ինչու այն չփոխվեց։
  11. Ավելացրու գրանցմանը փորձերի սահմանափակում LoginThrottle-ի օրինակով՝ առավելագույնը 3 նոր հաշիվ մեկ սեսիայից ժամում։ Ինչու՞ է սա ուսումնական, ոչ իրական պաշտպանություն (բաժին 05)։
  12. Խնդրիր AI-ին վերանայել Auth-ը։ Ի՞նչ գտավ։ Համաձա՞յն ես։
