03Request և Response

Երկու փոքր դաս։ Առաջինը հավաքում է այն, ինչ եկել է բրաուզերից, երկրորդը՝ այն, ինչ պետք է հետ ուղարկել։

src/Http/Request.php
<?php
declare(strict_types=1);

namespace App\Http;

class Request
{
    public function __construct(
        public readonly string $method,
        public readonly string $path,
        private readonly array $query,
        private readonly array $body,
    ) {}

    public static function fromGlobals(): self
    {
        $path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';

        return new self(
            $_SERVER['REQUEST_METHOD'] ?? 'GET',
            '/' . trim($path, '/'),
            $_GET,
            $_POST,
        );
    }

    public function query(string $key, string $default = ''): string
    {
        return trim((string) ($this->query[$key] ?? $default));
    }

    public function input(string $key, string $default = ''): string
    {
        return trim((string) ($this->body[$key] ?? $default));
    }

    public function int(string $key, int $default = 0): int
    {
        $value = filter_var($this->body[$key] ?? $this->query[$key] ?? null, FILTER_VALIDATE_INT);

        return $value === false ? $default : $value;
    }

    /** Ցուցակից մեկը — Դաս 6-ի սպիտակ ցուցակը ⭐ */
    public function oneOf(string $key, array $allowed, string $default): string
    {
        $value = $this->query($key);

        return in_array($value, $allowed, true) ? $value : $default;
    }
}

oneOf()-ը Դաս 6-ի սպիտակ ցուցակն է՝ դարձած մեթոդ։ Տեսակավորման պարամետրը այլ ձևով կարդալու տարբերակ պարզապես չկա, ուստի այն մոռանալ հնարավոր չէ։

src/Http/Response.php
<?php
declare(strict_types=1);

namespace App\Http;

class Response
{
    public function __construct(
        private readonly string $body = '',
        private readonly int $status = 200,
        private readonly array $headers = [],
    ) {}

    public static function html(string $body, int $status = 200): self
    {
        return new self($body, $status, ['Content-Type' => 'text/html; charset=utf-8']);
    }

    public static function redirect(string $to): self
    {
        return new self('', 303, ['Location' => $to]);    // PRG — Դաս 6
    }

    public function send(): void
    {
        http_response_code($this->status);

        foreach ($this->headers as $name => $value) {
            header("{$name}: {$value}");
        }

        echo $this->body;
    }
}

04Router

src/Http/Router.php
<?php
declare(strict_types=1);

namespace App\Http;

use App\Exceptions\NotFoundException;

class Router
{
    private array $routes = [];     // ['GET' => ['/product/{slug}' => callable]]

    public function get(string $path, callable $handler): void
    {
        $this->routes['GET'][$path] = $handler;
    }

    public function post(string $path, callable $handler): void
    {
        $this->routes['POST'][$path] = $handler;
    }

    public function dispatch(Request $request): Response
    {
        foreach ($this->routes[$request->method] ?? [] as $route => $handler) {

            // /product/{slug} → #^/product/([^/]+)$#
            $pattern = '#^' . preg_replace('#\{[a-z_]+\}#', '([^/]+)', $route) . '$#u';

            if (preg_match($pattern, $request->path, $matches)) {
                array_shift($matches);                         // առաջինը՝ ամբողջ հասցեն
                $params = array_map('urldecode', $matches);    // հայերեն slug-երի համար

                return $handler($request, ...$params);
            }
        }

        throw new NotFoundException();
    }
}
dispatch()-ը քայլ առ քայլ Հարցում․ GET /product/monitor-24 1. Վերցնում է GET-ի ուղիները 2. '/' → #^/$# → չի համընկնում 3. '/product/{slug}' → #^/product/([^/]+)$# → համընկնում է ✅ $matches = ['monitor-24'] 4. $handler($request, 'monitor-24') ↓ ProductController::show($request, 'monitor-24')

Առանց 404-ի /product/գոյություն-չունի-ն կվերադարձներ դատարկ էջ 200 կոդով։ Google-ը այն կինդեքսավորեր որպես իրական էջ (դա կոչվում է soft 404), իսկ սխալները չէին երևա ոչ մի հաշվետվությունում։

05Կոնտրոլերներ

MVC՝ ի՞նչ է որտեղ CONTROLLER · կարդում է Request-ը · կանչում է Repository-ն · ընտրում է շաբլոնը ⚠️ ՉԻ պարունակում SQL, ՉԻ պարունակում HTML │ ┌───┴────────────┐ ▼ ▼ MODEL VIEW Product, views/*.php Repository, e(), ցիկլեր SQL ⚠️ ՉԻ դիմում բազային
src/helpers.php — Դաս 5-ի render()-ի կողքին
/** Դաս 5-ի render()-ը՝ երկու ուղղմամբ, որ խոստացել էինք */
function render(string $view, array $data = []): string
{
    extract($data, EXTR_SKIP);          // ⭐ 1. չի վերագրում $view-ը

    ob_start();

    try {
        require __DIR__ . "/../views/{$view}.php";
    } catch (\Throwable $e) {
        ob_end_clean();                   // ⭐ 2. չթողնել կիսատ HTML
        throw $e;
    }

    return ob_get_clean();
}

/** Էջ՝ layout-ի մեջ դրված — Դաս 5-ի երկու render()-ը մեկ կանչով */
function page(string $view, array $data = [], string $title = 'ՏեխնոՄարկետ'): string
{
    return render('layout', [
        'title'   => $title,
        'content' => render($view, $data),
    ]);
}

/** Հասցե՝ query պարամետրերով, ապահով */
function url(string $path, array $query = []): string
{
    $url = '/' . ltrim($path, '/');

    return $query === [] ? $url : $url . '?' . http_build_query($query);
}

/** Ընթացիկ էջ՞ — aria-current-ի համար */
function isCurrent(string $path): bool
{
    $current = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';

    return '/' . trim($current, '/') === '/' . trim($path, '/');
}
Դաս 5-ի երկու թույլ տեղը՝ ուղղված

EXTR_SKIP. եթե $data-ում լինի 'view' բանալի, առանց այս դրոշի այն կվերագրեր ֆունկցիայի $view-ը, և կբեռնվեր այլ ֆայլ։

ob_end_clean() catch-ում. եթե շաբլոնում սխալ լինի, բուֆերում արդեն կա կիսատ HTML։ Առանց մաքրման՝ բաժին 06-ի 500 էջը կհայտնվի կիսատ <div>-ների մեջ։ Հիմա, երբ սխալները մշակվում են մեկ տեղում, այս ուղղումը դառնում է անհրաժեշտ։

src/Controllers/ProductController.php
<?php
declare(strict_types=1);

namespace App\Controllers;

use App\Exceptions\NotFoundException;
use App\Http\Request;
use App\Http\Response;
use App\Repositories\CategoryRepository;
use App\Repositories\ProductRepository;

class ProductController
{
    private const SORTS    = ['price_asc', 'price_desc', 'name', 'newest'];
    private const PER_PAGE = 12;

    public function __construct(
        private readonly ProductRepository $products,
        private readonly CategoryRepository $categories,
    ) {}

    public function index(Request $request): Response
    {
        $sort = $request->oneOf('sort', self::SORTS, 'price_asc');
        $page = max(1, $request->int('page', 1));

        return Response::html(page('pages/home', [
            'products'   => $this->products->all(null, $sort, self::PER_PAGE, ($page - 1) * self::PER_PAGE),
            'total'      => $this->products->count(),
            'categories' => $this->categories->all(),
            'sort'       => $sort,
            'page'       => $page,
            'perPage'    => self::PER_PAGE,
        ], 'ՏեխնոՄարկետ — տեխնիկա Երևանում'));
    }

    public function show(Request $request, string $slug): Response
    {
        $product = $this->products->findBySlug($slug);

        if ($product === null) {
            throw new NotFoundException('Ապրանքը չի գտնվել');   // ⭐
        }

        return Response::html(page('pages/product', ['product' => $product], $product->name));
    }
}
Զրո SQL, զրո HTML

Կոնտրոլերը կարդում է պարամետրերը, կանչում Repository-ն, ընտրում շաբլոնը։ Վերջ։ Կանոն Դաս 14-ից․ կախվածությունները երևում են կոնստրուկտորում։ Եթե դրանք դառնան յոթ, կոնտրոլերը շատ բան է անում։ Եվ մեկ այլ կանոն․ եթե կոնտրոլերը 40 տողից երկար է, այնտեղ ընկել է մի բան, որի տեղը այնտեղ չէ։

Իսկ $request-ը փոխանցվում է մեթոդին, ոչ կոնստրուկտորին, որովհետև այն նոր է ամեն հարցման հետ։

06Ամեն ինչ միասին

bootstrap.php — ավելացնում ենք կոնտրոլերները
// ... Դաս 14-ի bootstrap.php-ն՝ autoload, .env, session, $pdo, Repository-ներ ...

use App\Controllers\ProductController;
use App\Controllers\CartController;
use App\Controllers\ContactController;
use App\Controllers\AdminController;

$productController = new ProductController($products, $categories);
$cartController    = new CartController($products, $cart);
$contactController = new ContactController();
$adminController   = new AdminController($products, $categories);
routes.php — կայքի քարտեզը
<?php
declare(strict_types=1);

// Հանրային
$router->get('/',                [$productController, 'index']);
$router->get('/product/{slug}',  [$productController, 'show']);

$router->get('/contact',         [$contactController, 'form']);
$router->post('/contact',        [$contactController, 'submit']);

// Զամբյուղ
$router->get('/cart',            [$cartController, 'index']);
$router->post('/cart/add',       [$cartController, 'add']);
$router->post('/cart/remove',    [$cartController, 'remove']);

// Ադմին ⚠️ դեռ առանց պաշտպանության — Դաս 16
$router->get('/admin',           [$adminController, 'index']);
$router->get('/admin/edit/{id}', [$adminController, 'form']);
$router->post('/admin/edit/{id}',[$adminController, 'save']);
routes.php-ը փաստաթուղթ է

Կարդալով այն՝ 30 վայրկյանում հասկանում ես, ինչ էջեր ունի կայքը։ Առաջ դրա համար պետք էր բացել ամեն ֆայլ։ Եվ մեկ հայացքով երևում է, որ /admin-ը պաշտպանված չէ։

[$productController, 'show']-ը PHP-ում «callable» է՝ «կանչիր այս օբյեկտի այս մեթոդը»։ Laravel-ում գրառումը համարյա նույնն է՝ [ProductController::class, 'show']։ Տարբերությունը կտեսնես Դաս 19-ում։

public/index.php — ամբողջ ծրագիրը մեկ ֆայլում
<?php
declare(strict_types=1);

use App\Exceptions\HttpException;
use App\Http\Request;
use App\Http\Response;
use App\Http\Router;

require_once __DIR__ . '/../bootstrap.php';

$router = new Router();
require __DIR__ . '/../routes.php';

try {
    $response = $router->dispatch(Request::fromGlobals());

} catch (HttpException $e) {
    // Սպասված սխալ՝ 404 (և Դաս 16-ում՝ 403)
    $response = Response::html(
        page('pages/error', ['code' => $e->status, 'message' => $e->getMessage()], (string) $e->status),
        $e->status
    );

} catch (\Throwable $e) {
    // Անսպասելի սխալ
    error_log((string) $e);                          // ⭐ մանրամասները՝ լոգում
    $debug = ($_ENV['APP_DEBUG'] ?? 'false') === 'true';

    $response = Response::html(
        page('pages/error', ['code' => 500, 'message' => $debug ? $e->getMessage() : 'Տեխնիկական խնդիր'], '500'),
        500
    );
}

$response->send();
Հարցման ուղին Բրաուզեր GET /product/monitor-24 ▼ Apache + .htaccess ֆայլ չկա → public/index.php ▼ bootstrap.php .env, session, PDO, Repository-ներ, կոնտրոլերներ ▼ Router::dispatch() համեմատում է routes.php-ի հետ ▼ ProductController::show() Repository → Product → page() ▼ Response → send()
APP_DEBUG-ը՝ մեկ տեղում

Զարգացման ժամանակ սխալի հաղորդագրությունը էկրանին խնայում է ժամանակ։ Production-ում այն կարող է պարունակել SQL հարցում և ֆայլերի ուղիներ։

Դա Դաս 1-ի display_errors-ի և Դաս 10-ի PDOException-ի նույն կանոնն է։ Այս անգամ այն ամրագրված է մեկ տեղում ամբողջ ծրագրի համար, և ոչ մի կոնտրոլեր չի կարող մոռանալ այն։

Խորացված · ինչ է անում Laravel-ը այստեղ

Մենք ստեղծում ենք բոլոր չորս կոնտրոլերը ամեն հարցման ժամանակ, թեև օգտագործվում է մեկը։ Մեր չափի նախագծում դա խնդիր չէ։

Laravel-ը կոնտրոլերը ստեղծում է միայն այն ժամանակ, երբ ուղին համընկնում է, և ինքն է կարդում կոնստրուկտորի տիպերը՝ որոշելու, թե ինչ փոխանցել։ Դա կոչվում է autowiring։ Մեր bootstrap.php-ի ձեռքով գրված new ProductController($products, $categories)-ը հենց այն է, ինչ Laravel-ը անում է ավտոմատ (Դաս 19)։

07Դասարանական

  1. Ստեղծիր public/, տեղափոխիր css/, js/, images/, փոխիր Laragon-ի DocumentRoot-ը և ավելացրու .htaccess
  2. Գրիր Request, Response, Router և երկու բացառությունը
  3. Ավելացրու page(), url(), isCurrent() helper-ները և views/pages/error.php
  4. Վերաշարադրիր ProductController-ը դասով, ապա նույն ձևով մնացած երեքը
  5. Գրիր routes.php և public/index.php, ջնջիր հին էջերի ֆայլերը
Հինգ պարտադիր փորձարկում

1. /product/noutbuq-pro-14 → ապրանքի էջը
2. /product/գոյություն-չունի → 404 էջ, ոչ դատարկ էջ
3. /css/main.css → բացվում է ուղիղ
4. Դիտմամբ գցիր throw new \Exception('test') կոնտրոլերում → 500 էջ։ Ապա APP_DEBUG=false → հաղորդագրությունը թաքնվե՞ց
5. /.env → 404 կամ 403, ոչ ֆայլի պարունակություն ⭐

Հինգերորդը ամենակարևորն է․ այն ստուգում է, որ public/-ը ճիշտ է կարգավորված։ Դաս 22-ում նույն ստուգումը կանենք իրական սերվերում։

08Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Սեփական MVC», Անիի գրքերի ակումբում՝ քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի կետերը, որ նշված չեն որպես «մինի-նախագծում», կատարիր ՏեխնոՄարկետում։

  1. Ավարտիր դասարանականը և հինգ փորձարկումը։ Ուղարկիր /.env-ի պատասխանի սքրինշոթը։
  2. Ավելացրու /category/{slug} ուղին և ProductController::category() մեթոդը։
  3. Ավելացրու /search ուղին ?q= պարամետրով։
  4. Ավելացրու aria-current="page" մենյուում isCurrent()-ով (HTML Դաս 18)։
  5. Փորձարկիր հայերեն slug՝ /product/նոութբուք։ Աշխատո՞ւմ է urldecode-ը։
  6. Հաշվիր՝ քանի տող է ProductController-ը։ Կա՞ դրանում SQL կամ HTML։
  7. Խորացված․ ավելացրու 405-ի ստուգումը Router-ում։
  8. Խնդրիր AI-ին վերանայել Router-ը անվտանգության տեսանկյունից։ Ի՞նչ գտավ։ Համաձա՞յն ես։

09Հիշելու թերթիկ

Ֆայլ / դասՊատասխանատվություն
public/index.phpՄիակ մուտքը, սխալների մշակում
.htaccessԱմեն ինչ → index.php, բացի իրական ֆայլերից
routes.phpՀասցե → կոնտրոլեր
RouterԳտնել ուղին, կանչել, կամ 404
RequestՄուտքային տվյալներ
ResponseԵլքային տվյալներ
ControllerԿապում է Repository-ն և շաբլոնը
NotFoundException404՝ առանց exit-ի

Կոնտրոլերի կանոնները

✅ Կարդալ Request-ից ❌ SQL ✅ Կանչել Repository ❌ HTML ✅ Վերադարձնել Response ❌ echo, exit, header() ✅ Գցել բացառություն

Ստուգաթերթ

  • Վեբ-արմատը public/-ն է
  • /.env-ը հասանելի չէ բրաուզերից
  • Չգտնված ուղին տալիս է 404, ոչ 200
  • Ուղիների ցուցակը մեկ ֆայլում է
  • Կոնտրոլերները դասեր են, կախվածությունները՝ կոնստրուկտորում
  • Բոլոր սխալները մշակվում են public/index.php-ում
  • Հին էջերի ֆայլերը ջնջված են

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել, ինչու մեկ մուտք
  • Հասկանո՞ւմ եմ, ինչու public/-ը անվտանգության հարց է
  • Կարո՞ղ եմ նկարագրել հարցման ուղին բրաուզերից մինչև send()
  • Գիտե՞մ, ինչու 404-ը բացառություն է, ոչ echo
  • Կարո՞ղ եմ ասել, ինչ չպետք է լինի կոնտրոլերում
Հաջորդ դասին

Նույնականացում և CSRF՝ անվտանգության չորրորդ դասը։ Մի քանի դաս ասում էինք «ադմինը պաշտպանված չէ»։ Հիմա ունենք այն, ինչ պետք է դա շտկելու համար․ մեկ մուտք, ուղիների ցուցակ, կենտրոնացված սխալներ։ Կսովորենք password_hash, մուտք և ելք, CSRF token և փորձերի սահմանափակում։
