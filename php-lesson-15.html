02Request

Ինչո՞ւ չօգտագործել $_GET-ը ուղիղ

1. Ամեն տեղ պիտի գրես ?? '' և trim()
2. Ամեն արժեք տող է — (int) ամեն տեղ
3. Չես կարող փորձարկել․ $_GET-ը գլոբալ է
4. Կոնտրոլերի ստորագրությունը չի ասում, ինչից է կախված

src/Http/Request.php
<?php
declare(strict_types=1);

namespace App\Http;

class Request
{
    public function __construct(
        public readonly string $method,
        public readonly string $path,
        private readonly array $query,
        private readonly array $body,
        private readonly array $files = [],
    ) {}

    public static function fromGlobals(): self
    {
        $uri  = $_SERVER['REQUEST_URI'] ?? '/';
        $path = parse_url($uri, PHP_URL_PATH) ?: '/';

        return new self(
            method: $_SERVER['REQUEST_METHOD'] ?? 'GET',
            path:   '/' . trim($path, '/'),
            query:  $_GET,
            body:   $_POST,
            files:  $_FILES,
        );
    }

    // ---------- Ընթերցում ----------

    public function query(string $key, string $default = ''): string
    {
        return trim((string) ($this->query[$key] ?? $default));
    }

    public function input(string $key, string $default = ''): string
    {
        return trim((string) ($this->body[$key] ?? $default));
    }

    public function int(string $key, int $default = 0): int
    {
        $value = $this->body[$key] ?? $this->query[$key] ?? null;
        $int   = filter_var($value, FILTER_VALIDATE_INT);

        return $int === false ? $default : $int;
    }

    /** Ցուցակից մեկը — Դաս 6-ի օրինաչափությունը ⭐ */
    public function oneOf(string $key, array $allowed, string $default): string
    {
        $value = (string) ($this->query[$key] ?? $this->body[$key] ?? '');

        return in_array($value, $allowed, true) ? $value : $default;
    }

    public function has(string $key): bool
    {
        return isset($this->body[$key]) || isset($this->query[$key]);
    }

    public function all(): array
    {
        return $this->body;
    }

    public function file(string $key): ?array
    {
        return $this->files[$key] ?? null;
    }

    public function isPost(): bool
    {
        return $this->method === 'POST';
    }
}
Ի՞նչ ձեռք բերեցինք

$request->int('page', 1) — մեկ կանչ՝ լռելյայն արժեքով, փոխարկումով և վավերացումով միասին։

Առաջ դա էր՝ max(1, (int) ($_GET['page'] ?? 1))։

oneOf()-ը Դաս 6-ի սպիտակ ցուցակի օրինաչափությունն է՝ դարձած մեթոդ։ Հիմա անհնար է մոռանալ այն․ պարզապես չկա այլ ձև տեսակավորման պարամետր կարդալու։

03Response

src/Http/Response.php
<?php
declare(strict_types=1);

namespace App\Http;

class Response
{
    public function __construct(
        private readonly string $body = '',
        private readonly int $status = 200,
        private readonly array $headers = [],
    ) {}

    public static function html(string $body, int $status = 200): self
    {
        return new self($body, $status, [
            'Content-Type' => 'text/html; charset=utf-8',
        ]);
    }

    public static function json(array $data, int $status = 200): self
    {
        return new self(
            json_encode($data, JSON_UNESCAPED_UNICODE | JSON_THROW_ON_ERROR),
            $status,
            ['Content-Type' => 'application/json; charset=utf-8']
        );
    }

    public static function redirect(string $to, int $status = 303): self
    {
        return new self('', $status, ['Location' => $to]);
    }

    public function send(): void
    {
        http_response_code($this->status);

        foreach ($this->headers as $name => $value) {
            header("{$name}: {$value}");
        }

        echo $this->body;
    }
}
Ինչո՞ւ օբյեկտ, ոչ echo

1. Կոնտրոլերը ոչինչ չի տպում — այն վերադարձնում է պատասխան։ Դա նշանակում է, որ այն կարելի է փորձարկել․ կանչում ես մեթոդը և ստուգում վերադարձված օբյեկտը։

2. Վերնագրերը ուղարկվում են մեկ տեղում — վերջ Դաս 6-ի «headers already sent» խնդրին։

3. Կոնտրոլերը չգիտի ձևաչափը — Response::json()-ը կանչելով նույն կոնտրոլերը կարող է սպասարկել և՛ էջ, և՛ API (Դաս 17)։

JSON_THROW_ON_ERROR-ը կարևոր է․ առանց դրա կոտրված UTF-8-ի դեպքում json_encode-ը լուռ վերադարձնում է false, և հաճախորդը ստանում է դատարկ պատասխան։

04View

src/View.php
<?php
declare(strict_types=1);

namespace App;

use RuntimeException;

class View
{
    public function __construct(
        private readonly string $basePath,
        private readonly string $layout = 'layout',
    ) {}

    /** Միայն մասնակի շաբլոն, առանց layout-ի */
    public function partial(string $name, array $data = []): string
    {
        $file = $this->basePath . '/' . $name . '.php';

        if (!is_file($file)) {
            throw new RuntimeException("Շաբլոնը չի գտնվել՝ {$name}");
        }

        extract($data, EXTR_SKIP);      // ⭐ չի վերագրում $this-ը

        ob_start();

        try {
            require $file;
        } catch (\Throwable $e) {
            ob_end_clean();          // ⭐ չթողնել կիսատ HTML
            throw $e;
        }

        return ob_get_clean();
    }

    /** Էջ՝ layout-ի մեջ դրված */
    public function page(string $name, array $data = [], string $title = ''): string
    {
        return $this->partial($this->layout, [
            'title'   => $title,
            'content' => $this->partial($name, $data),
        ]);
    }

    /** Ադմինի layout */
    public function withLayout(string $layout): self
    {
        return new self($this->basePath, $layout);
    }
}
Երկու կարևոր դետալ

EXTR_SKIP — եթե $data-ում լինի 'this' կամ 'file' բանալի, առանց այս դրոշի այն կվերագրեր տեղական փոփոխականները և կկոտրեր մեթոդը։ Դա հազվադեպ է, բայց ախտորոշելը՝ մղձավանջ։

ob_end_clean() catch-ում — եթե շաբլոնում բացառություն է լինում, բուֆերում արդեն կա կիսատ HTML։ Առանց մաքրման՝ սխալի էջը կհայտնվի կիսատ <div>-ների մեջ, և դասավորությունը կփլվի։

Երկուսն էլ այն դեպքերն են, որ ձեռնարկներում չկան, բայց իրական նախագծերում պատահում են։

withLayout() — անփոփոխելի օբյեկտ

Այն չի փոխում ընթացիկ View-ն, այլ վերադարձնում է նոր օրինակ։

$view->withLayout('layout-admin')->page('admin/list', $data)

Այդպես նույն View-ն կարող է օգտագործվել և՛ հանրային էջերում, և՛ ադմինում՝ առանց վիճակի փոփոխության։ Laravel-ում շատ դասեր հենց այս ձևով են աշխատում։

05Կոնտրոլեր-դասեր

src/Controllers/ProductController.php
<?php
declare(strict_types=1);

namespace App\Controllers;

use App\Http\Request;
use App\Http\Response;
use App\Repositories\ProductRepository;
use App\Repositories\CategoryRepository;
use App\View;
use App\Exceptions\NotFoundException;

class ProductController
{
    private const SORTS = ['price_asc', 'price_desc', 'name', 'newest'];
    private const PER_PAGE = 12;

    public function __construct(
        private readonly ProductRepository $products,
        private readonly CategoryRepository $categories,
        private readonly View $view,
    ) {}

    public function index(Request $request): Response
    {
        $sort = $request->oneOf('sort', self::SORTS, 'price_asc');
        $page = max(1, $request->int('page', 1));

        $products = $this->products->findAll(
            null,
            $sort,
            self::PER_PAGE,
            ($page - 1) * self::PER_PAGE
        );

        return Response::html(
            $this->view->page('pages/home', [
                'products'   => $products,
                'categories' => $this->categories->findAll(),
                'total'      => $this->products->count(),
                'sort'       => $sort,
                'page'       => $page,
                'perPage'    => self::PER_PAGE,
            ], 'ՏեխնոՄարկետ — տեխնիկա Երևանում')
        );
    }

    public function show(Request $request, string $slug): Response
    {
        $product = $this->products->findBySlug($slug);

        if ($product === null) {
            throw new NotFoundException('Ապրանքը չի գտնվել');   // ⭐
        }

        return Response::html(
            $this->view->page(
                'pages/product',
                ['product' => $product],
                $product->name . ' — ՏեխնոՄարկետ'
            )
        );
    }
}
Ուշադրություն՝ 30 տող, զրո SQL, զրո HTML

Կոնտրոլերը կարդում է պարամետրերը, կանչում Repository-ն, ընտրում շաբլոնը։ Վերջ։

Կախվածությունները երևում են կոնստրուկտորում՝ երեք հատ։ Եթե դրանք դառնան յոթ, դա ազդանշան է, որ կոնտրոլերը շատ բան է անում։

Իսկ $request-ը փոխանցվում է մեթոդին, ոչ կոնստրուկտորին — որովհետև այն փոխվում է ամեն կանչի հետ, իսկ Repository-ն՝ ոչ։

Բացառություն, ոչ abort()

Դաս 14-ում abort(404)-ը ուղիղ տպում էր էջը և կանչում exit։ Դա աշխատում է, բայց երկու խնդիր ունի․

· Կոնտրոլերը դառնում է չփորձարկելի — exit-ը սպանում է թեստը
· Կոնտրոլերը որոշում է, ինչպես ցուցադրել սխալը — դա իր գործը չէ

throw new NotFoundException()-ը պարզապես ասում է․ «այս բանը չկա»։ Իսկ ինչպես դա ցույց տալ — HTML էջ, թե JSON — որոշում է front controller-ը։

Դաս 17-ում այս տարբերությունը կրիտիկական կդառնա․ API-ն 404-ը պիտի վերադարձնի JSON-ով, ոչ HTML-ով։

Router::dispatch — վերադարձնում է Response
public function dispatch(Request $request): Response
{
    $method = $request->method;
    $path   = $request->path;

    foreach ($this->routes[$method] ?? [] as $route => $action) {

        $params = $this->matchRoute($route, $path);

        if ($params === null) {
            continue;
        }

        [$class, $action] = $action;

        $controller = $this->container->make($class);   // ⭐

        return $controller->$action($request, ...$params);
    }

    if ($this->existsWithOtherMethod($method, $path)) {
        throw new MethodNotAllowedException();
    }

    throw new NotFoundException();
}

            $this->view(),
        ),

        default => throw new \RuntimeException("Անհայտ դաս՝ {$class}"),
    };
}
Ազնիվ նշում

Այս match-ը ձեռքով է։ Ամեն նոր կոնտրոլերի հետ պիտի ավելացնես ճյուղ։

Laravel-ը դա անում է ավտոմատ՝ reflection-ով կարդում է կոնստրուկտորի տիպերը և ինքն է հավաքում կախվածությունները։ Դա կոչվում է autowiring։

Մենք դա չենք գրի — չափազանց բարդ է այս փուլում։ Բայց հիմա հասկանում ես, ինչ է անում Laravel-ը այնտեղ, և Դաս 18-ում այն կախարդանք չի թվա։

07Front controller

public/index.php — ամբողջ ծրագիրը 40 տողում
<?php
declare(strict_types=1);

use App\Exceptions\HttpException;
use App\Http\Request;
use App\Http\Response;
use App\Http\Router;

require_once __DIR__ . '/../bootstrap.php';

$request = Request::fromGlobals();
$router  = new Router($app);

require __DIR__ . '/../routes.php';

try {
    $response = $router->dispatch($request);

} catch (HttpException $e) {

    // Սպասված սխալ՝ 404, 405, 403
    $response = Response::html(
        $app->view()->page('pages/error', [
            'code'    => $e->status,
            'message' => $e->getMessage(),
        ], (string) $e->status),
        $e->status
    );

} catch (\Throwable $e) {

    // Անսպասելի սխալ
    error_log((string) $e);       // ⭐ մանրամասները՝ լոգում

    $debug = ($_ENV['APP_DEBUG'] ?? 'false') === 'true';

    $response = Response::html(
        $app->view()->page('pages/error', [
            'code'    => 500,
            'message' => $debug ? $e->getMessage() : 'Տեխնիկական խնդիր',
        ], '500'),
        500
    );
}

$response->send();
Ահա ամբողջ ծրագրի կմախքը

Հարցում → Router → Կոնտրոլեր → Պատասխան → ուղարկում։

Եվ երկու catch՝ սպասված սխալների և անսպասելիների համար։

Ամենակարևորը՝ ամեն սխալ մշակվում է մեկ տեղում։ Ցանկացած կոնտրոլերում գցված բացառությունը հայտնվում է այստեղ։ Չկա էջ, որտեղ սխալը կվերածվի սպիտակ էկրանի կամ կցուցադրի stack trace օգտագործողին։

APP_DEBUG-ը կրիտիկական է

Զարգացման ժամանակ $e->getMessage()-ը էկրանին խնայում է ժամանակ։

Production-ում այն կարող է պարունակել SQL հարցում, ֆայլերի ուղիներ, բազայի կառուցվածք։

Դա Դաս 1-ի display_errors-ի և Դաս 9-ի PDOException-ի նույն կանոնն է՝ երրորդ անգամ։ Այս անգամ այն ամրագրված է մեկ տեղում ամբողջ ծրագրի համար։

Նույնականացում և CSRF — անվտանգության չորրորդ և վերջին դասը։

Հինգ դաս ասում էինք «ադմինը պաշտպանված չէ»։ Այժմ ֆրեյմվորքն ունի այն ամենը, ինչ պետք է դա շտկելու համար․ մեկ մուտք, ուղիների ցուցակ, կենտրոնացված բացառություններ։

Կսովորենք password_hash, մուտք և ելք, middleware-ի գաղափարը և CSRF token-ը։
