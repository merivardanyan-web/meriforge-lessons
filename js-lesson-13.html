const form    = document.querySelector(".form");
const success = document.querySelector(".form__success");

const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

// Ամեն դաշտի համար՝ ֆունկցիա, որ վերադարձնում է սխալի տեքստ կամ ""
const rules = {
  name(value) {
    if (value.trim() === "")      return "Անունը պարտադիր է";
    if (value.trim().length < 2) return "Առնվազն 2 նիշ";
    return "";
  },

  email(value) {
    if (value.trim() === "")            return "Էլ. հասցեն պարտադիր է";
    if (!emailPattern.test(value.trim())) return "Սխալ ձևաչափ։ Օրինակ՝ ani@mail.com";
    return "";
  },

  age(value) {
    if (value.trim() === "") return "Տարիքը պարտադիր է";
    const n = Number(value);
    if (Number.isNaN(n))     return "Մուտքագրեք թիվ";
    if (n < 14 || n > 100)  return "Տարիքը պիտի լինի 14-ից 100";
    return "";
  },

  password(value) {
    if (value.length < 8)                  return "Առնվազն 8 նիշ";
    if (value === value.toLowerCase())     return "Ավելացրեք մեծատառ";
    if (!/\d/.test(value))                 return "Ավելացրեք թիվ";
    return "";
  }
};
main.js — ստուգում և ուղարկում
function showError(input, message) {
  const field = input.closest(".field");
  field.classList.add("has-error");
  field.querySelector(".field__error").textContent = message;
}

function clearError(input) {
  const field = input.closest(".field");
  field.classList.remove("has-error");
  field.querySelector(".field__error").textContent = "";
}

// Ստուգում է մեկ դաշտ, վերադարձնում true/false
function validateField(input) {
  const rule = rules[input.name];
  if (!rule) return true;

  const error = rule(input.value);

  if (error) {
    showError(input, error);
    return false;
  }

  clearError(input);
  return true;
}

// ★ Իրական ժամանակի ստուգում — դելեգացիա (Դաս 12)
form.addEventListener("blur", (e) => {
  if (e.target.name in rules) validateField(e.target);
}, true);   // blur-ը չի փուչիկանում → պետք է capture փուլը

// Սխալը վերանում է անմիջապես, երբ ուղղում են
form.addEventListener("input", (e) => {
  const field = e.target.closest(".field");
  if (field?.classList.contains("has-error")) {
    validateField(e.target);
  }
});

// Ուղարկում
form.addEventListener("submit", (e) => {
  e.preventDefault();
  success.textContent = "";

  // Ստուգում ենք ԲՈԼՈՐ դաշտերը (ոչ some — որ բոլոր սխալները երևան)
  const inputs = [...form.querySelectorAll("input[name]")];
  const results = inputs.map(validateField);
  let isValid = results.every(r => r);      // Դաս 6

  // Checkbox-ը առանձին
  const terms = form.terms;
  if (!terms.checked) {
    showError(terms, "Անհրաժեշտ է համաձայնությունը");
    isValid = false;
  } else {
    clearError(terms);
  }

  if (!isValid) {
    form.querySelector(".has-error input")?.focus();
    return;
  }

  // Ամեն ինչ կարգին է
  const data = Object.fromEntries(new FormData(form));
  console.log("Ուղարկվող տվյալները՝", data);

  success.textContent = `Շնորհակալություն, ${data.name}։ Գրանցումը հաջողվեց։`;
  form.reset();
});

Երկու նուրբ մանրուք

1. blur-ը և focus-ը չեն փուչիկանում։ Դելեգացիայի համար պիտի օգտագործես երրորդ արգումենտը՝ true (Դաս 12՝ որսման փուլ)։
2. Օգտագործում ենք map + every, ոչ թե some-ով վաղաժամ ելք։ Այդպես բոլոր սխալ դաշտերը միանգամից նշվում են, ոչ միայն առաջինը։

UX կանոն

Սխալը ցույց տուր դաշտից դուրս գալիս (blur), ոչ թե առաջին իսկ նիշից։ Հակառակ դեպքում մարդը դեռ «a» է գրել, արդեն կարմիր է ամեն ինչ։ Բայց երբ սխալն արդեն ցուցադրված է — ուղղիր անմիջապես (input)։

09Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Նոր գրքի ֆորմա», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

  1. Ֆորմա մեկ դաշտով և կոճակով։ Սեղմելիս տպիր դաշտի արժեքը։ Համոզվիր, որ էջը չի վերաբեռնվում։
  2. Դաշտ և պարբերություն։ input-ով ցույց տուր մուտքագրվածը իրական ժամանակում և նիշերի քանակը։
  3. Checkbox՝ «Ցույց տալ գաղտնաբառը»։ Նշելիս type-ը դառնա text, հանելիս՝ նորից password։
  4. Radio կոճակներ՝ վճարման եղանակի ընտրություն։ Ընտրվածը ցույց տուր էկրանին։
  5. Select երկրների ցուցակով։ Ընտրվածի փոփոխման ժամանակ ցույց տուր համապատասխան հեռախոսի կոդը։
  6. Գրիր validators.js ֆայլ՝ isEmpty, minLength, isEmail, isPhone, isNumber, inRange ֆունկցիաներով։ Ամեն մեկը ստուգիր 3 արժեքով։
  7. Կապի ֆորմա՝ անուն, էլ. հասցե, հաղորդագրություն (textarea, առնվազն 10 նիշ)։ Սխալները ցույց տուր դաշտի տակ։
  8. Ամբողջական գրանցման ֆորմա դասարանական օրինակի հիման վրա։ Ավելացրու «Կրկնել գաղտնաբառը» դաշտ և ստուգիր համընկնումը։
  9. Ավելացրու գաղտնաբառի ուժի ցուցիչ՝ թույլ / միջին / ուժեղ, գունավոր գծով (classList-ով)։
  10. Ֆորմա, որը հաջող ուղարկումից հետո ավելացնում է նոր տող աղյուսակում կամ քարտ ցուցակում (միացրու Դաս 10-ի render-ը և Դաս 12-ի դելեգացիան՝ ջնջելու համար)։
