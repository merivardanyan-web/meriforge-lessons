01Թվաբանական օպերատորներ

Չորսը դու արդեն գիտես դպրոցից։ Երկուսը՝ նոր են։

console.log(a / b);   // բաժանում
console.log(a % b);   // մնացորդ  ← նոր
console.log(a ** 2);  // աստիճան  ← նոր

Ի՞նչ է %-ը (մնացորդ)

Պատկերացրու՝ ունես 17 կոնֆետ և ուզում ես բաժանել 5 երեխայի հավասար։ Ամեն երեխա կստանա 3 կոնֆետ (17 / 5 = 3, ամբողջը), իսկ ձեռքիդ կմնա 2 կոնֆետ։ Ահա այդ 2-ը հենց 17 % 5-ն է։

Ամենակարևոր կիրառությունը

Զույգ թիվը միշտ 2-ի բաժանվում է առանց մնացորդի։ Ուրեմն՝ եթե x % 2 === 0, թիվը զույգ է։ Եթե x % 2 === 1 — կենտ։

03Համեմատման օպերատորներ

Համեմատությունը միշտ վերադարձնում է միայն երկու հնարավոր արժեքից մեկը՝ true կամ false։ Ուրիշ բան լինել չի կարող։

ՕպերատորՆշանակությունՕրինակԱրդյունք
>մեծ է10 > 5true
<փոքր է10 < 5false
>=մեծ է կամ հավասար5 >= 5true
<=փոքր է կամ հավասար4 <= 3false
===խիստ հավասար է5 === "5"false
!==խիստ հավասար չէ5 !== "5"true
արդյունքը միշտ boolean է
const age = 16;

console.log(age >= 18);
console.log(age < 18);
console.log(typeof (age >= 18));

// արդյունքը կարելի է պահել փոփոխականում
const isAdult = age >= 18;
console.log(isAdult);

04if / else — ծրագիրը որոշում է կայացնում

Ամեն օր դու ինքդ էլ այսպես ես մտածում․ «ԵԹԵ անձրև է, հովանոց վերցնեմ, ԱՅԼԱՊԵՍ արևային ակնոց»։ JS-ում նույն բանն է։

կառուցվածքը
if (պայման) {
  // այս կոդն աշխատում է, եթե պայմանը true է
} else {
  // այս կոդն աշխատում է, եթե պայմանը false է
}
իրական օրինակ
const age = 16;

if (age >= 18) {
  console.log("Կարող ես մեքենա վարել։");
} else {
  console.log("Դեռ վաղ է։ Սպասիր մի քիչ։");
}
Console
Դեռ վաղ է։ Սպասիր մի քիչ։

else if — երբ տարբերակները մեկից շատ են

JS-ը ստուգում է պայմանները վերևից ներքև և կանգ է առնում առաջին ճշմարիտի վրա։ Մնացածը նույնիսկ չի նայում։

գնահատականի ստուգում
const mark = 17;

if (mark >= 18) {
  console.log("Գերազանց");
} else if (mark >= 14) {
  console.log("Լավ");
} else if (mark >= 10) {
  console.log("Բավարար");
} else {
  console.log("Անբավարար");
}
Console
Լավ
Հերթականությունը կարևոր է

Եթե վերևում գրես mark >= 10, ապա 17-ը կընկնի այնտեղ և կտպի «Բավարար»։ Միշտ սկսիր ամենախիստ պայմանից և գնա դեպի ամենաթույլը։

Ոճ

Ձևավոր փակագծերը { } գրիր միշտ, նույնիսկ եթե ներսում մեկ տող է։ Այսօր մեկ տող է, վաղը կավելացնես երկրորդը և կսկսվեն դժվար գտնվող սխալները։

05Truthy և falsy

if-ի փակագծերում պարտադիր չէ համեմատություն գրել։ Կարելի է դնել ցանկացած արժեք, և JS-ը ինքը կորոշի՝ դա «այո» է, թե «ոչ»։

Falsy են ընդամենը 6 արժեք։ Հիշիր դրանք, և կիմանաս ամեն ինչ․

6 falsy արժեքները
false
0
""          // դատարկ տեքստ
null
undefined
NaN         // Not a Number — «թիվ չէ»

Մնացած ամեն ինչ truthy է։ Այո՛, նույնիսկ "0" տեքստը և "false" տեքստը՝ որովհետև դրանք դատարկ չեն։

ստուգենք
const userName = "";

if (userName) {
  console.log(`Բարև, ${userName}։`);
} else {
  console.log("Անունը լրացված չէ։");
}

const basket = 0;
if (basket) {
  console.log("Զամբյուղում ապրանք կա։");
} else {
  console.log("Զամբյուղը դատարկ է։");
}
Console
Անունը լրացված չէ։
Զամբյուղը դատարկ է։
Ինչի՞ համար է սա

if (userName)-ը շատ ավելի կարճ է, քան if (userName !== "" && userName !== undefined && userName !== null)։ Նույն բանն է անում։

06Տրամաբանական օպերատորներ

Երբ մեկ պայմանը քիչ է։ Երեքն են՝ && (ԵՎ), || (ԿԱՄ), ! (ՈՉ)։

&& — ԵՎ · բոլորը պիտի ճիշտ լինեն

«Կինո կգնամ, ԵԹԵ փող ունեմ ԵՎ ազատ եմ»։ Եթե թեկուզ մեկը չկա — կինո չկա։

&&
const hasMoney = true;
const isFree   = false;

console.log(hasMoney && isFree);

if (hasMoney && isFree) {
  console.log("Գնում ենք կինո։");
} else {
  console.log("Այսօր չի ստացվի։");
}
Console
false
Այսօր չի ստացվի։

|| — ԿԱՄ · գոնե մեկը բավական է

«Տանը կմնամ, ԵԹԵ անձրև է ԿԱՄ հիվանդ եմ»։ Մեկը բավական է։

||
const isRaining = true;
const isSick    = false;

if (isRaining || isSick) {
  console.log("Մնում եմ տանը։");
} else {
  console.log("Դուրս եմ գնում։");
}
Console
Մնում եմ տանը։

! — ՈՉ · շրջում է

!
const isLoggedIn = false;

console.log(isLoggedIn);
console.log(!isLoggedIn);

if (!isLoggedIn) {
  console.log("Խնդրում ենք մուտք գործել։");
}
Console
false
true
Խնդրում ենք մուտք գործել։

Ճշմարտության աղյուսակ

ABA && BA || B
truetruetruetrue
truefalsefalsetrue
falsetruefalsetrue
falsefalsefalsefalse

Մի քանի պայման միասին

ֆորմայի ստուգում
const age = 25;
const hasTicket = true;
const isBanned  = false;

if (age >= 18 && hasTicket && !isBanned) {
  console.log("Բարի գալուստ համերգ։");
} else {
  console.log("Մուտքն արգելված է։");
}
Console
Բարի գալուստ համերգ։
Կարճ միացում (short-circuit)

&&-ը ծույլ է․ եթե առաջին պայմանը false է, նա մնացածը նույնիսկ չի կարդում — արդյունքն արդեն պարզ է։ Նույնը ||-ի դեպքում՝ եթե առաջինը true է, մնացածին չի նայում։

07?? — լռելյայն արժեք

Շատ հաճախ մեզ պետք է․ «վերցրու այս արժեքը, իսկ եթե չկա — վերցրու պահեստայինը»։

?? nullish coalescing
let nickname = null;

const displayName = nickname ?? "Հյուր";
console.log(displayName);

nickname = "Ani_2010";
console.log(nickname ?? "Հյուր");
Console
Հյուր
Ani_2010
?? ընդդեմ ||

||-ը փոխարինում է բոլոր falsy արժեքները (նաև 0 և "")։ ??-ը փոխարինում է միայն null-ը և undefined-ը։

տարբերությունը
const messages = 0;   // իսկապես 0 նամակ կա

console.log(messages || "տվյալ չկա");
console.log(messages ?? "տվյալ չկա");
Console
տվյալ չկա   ← սխալ, 0-ը իրական արժեք էր
0            ← ճիշտ

08Ternary — պայմանը մեկ տողով

Երբ if / else-ը շատ պարզ է, կարելի է գրել մեկ տողով։ Կառուցվածքը՝

ձևը
պայման ? եթե_true : եթե_false
համեմատենք
const age = 20;

// երկար ձևը
let status;
if (age >= 18) {
  status = "չափահաս";
} else {
  status = "անչափահաս";
}
console.log(status);

// նույնը ternary-ով
const status2 = age >= 18 ? "չափահաս" : "անչափահաս";
console.log(status2);

// շատ հարմար է template literal-ի ներսում
const count = 1;
console.log(`Զամբյուղում ${count} ապրանք ${count === 1 ? "է" : "են"}`);
Console
չափահաս
չափահաս
Զամբյուղում 1 ապրանք է
Չափի զգացում

Ternary-ն օգտագործիր միայն պարզ դեպքերում։ Ternary-ի մեջ ternary գրելը կոդը դարձնում է անընթեռնելի։ Երբ դժվարանում ես կարդալ — վերադարձիր if / else-ին։

09switch

Երբ մեկ փոփոխականը համեմատում ենք շատ կոնկրետ արժեքների հետ, switch-ը ավելի մաքուր է, քան երկար else if շղթան։

շաբաթվա օրը
const day = 3;

switch (day) {
  case 1:
    console.log("Երկուշաբթի");
    break;
  case 2:
    console.log("Երեքշաբթի");
    break;
  case 3:
    console.log("Չորեքշաբթի");
    break;
  default:
    console.log("Անհայտ օր");
}
Console
Չորեքշաբթի
Մի՛ մոռացիր break-ը

Առանց break-ի JS-ը շարունակում է կատարել հաջորդ բոլոր case-երը՝ առանց պայմանը ստուգելու։ Սա ամենահաճախ հանդիպող սխալն է switch-ում։

ինչ է լինում առանց break-ի
const day = 1;

12Տնային աշխատանք

  1. prompt-ով վերցրու թիվ և տպիր՝ դրական է, բացասական, թե զրո։
  2. Վերցրու թիվ և տպիր՝ զույգ է թե կենտ։ Օգտագործիր %։
  3. Երեք թիվ՝ a, b, c։ Գտիր և տպիր ամենամեծը (միայն if-երով, առանց Math.max)։
  4. Ջրի ջերմաստիճանը՝ փոփոխական։ 0-ից ցածր՝ «սառույց», 0–99՝ «հեղուկ», 100 և ավելի՝ «գոլորշի»։
  5. Խանութում գնումը 20 000 դրամից բարձր է — 10% զեղչ։ 50 000-ից բարձր — 20%։ Հաշվիր և տպիր վերջնական գումարը։
  6. Օգտագործողից վերցրու օրվա համարը (1–7) և switch-ով տպիր օրվա անունը։ Սխալ թվի դեպքում՝ «Սխալ մուտք»։
  7. Մուտքի ստուգում․ login === "admin" ԵՎ password === "1234" → «Բարի գալուստ», այլապես՝ «Սխալ տվյալներ»։
  8. Գրիր ternary-ով․ եթե տարիքը 18-ից մեծ է — «մուտքն ազատ է», այլապես — «մուտքն արգելված է»։
  9. Բացատրիր մեկնաբանության մեջ՝ ինչո՞ւ է 0 || "պահեստային"-ը տալիս «պահեստային», իսկ 0 ?? "պահեստային"-ը՝ 0։
Հանձնում
