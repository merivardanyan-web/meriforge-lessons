toggle — ամենաօգտագործվողը

06Render զանգվածից

Ահա այս դասի գագաթնակետը։ Դաս 6-ի map-ը, Դաս 1-ի template literal-ը և DOM-ը միանում են մեկ օրինաչափության մեջ։

map + join + innerHTML
const products = [
  { id: 1, name: "Նոութբուք", price: 450000 },
  { id: 2, name: "Մկնիկ",     price: 8000   },
  { id: 3, name: "Մոնիտոր",   price: 120000 }
];

const row = document.querySelector(".row");

row.innerHTML = products.map(p => `
  <div class="col">
    <div class="wrapper">
      <h3 class="title">${p.name}</h3>
      <span class="price">${p.price.toLocaleString("ru-RU")} ֏</span>
      <button class="btn" data-id="${p.id}">Գնել</button>
    </div>
  </div>
`).join("");

Երեք տող՝ ցանկացած քանակի քարտի համար։ 3 ապրանք լինի, թե 300 — կոդը նույնն է։

07Դասարանական աշխատանք

Ամբողջական render ֆունկցիա՝ ֆիլտրով և վիճակագրությամբ։ Այն ամենը, ինչ սովորել ենք 10 դասում, մեկ ֆայլում։

main.js
const products = [
  { id: 1, name: "Նոութբուք", price: 450000, inStock: true  },
  { id: 2, name: "Մկնիկ",     price: 8000,   inStock: true  },
  { id: 3, name: "Ստեղնաշար", price: 25000,  inStock: false },
  { id: 4, name: "Մոնիտոր",   price: 120000, inStock: true  }
];

const container = document.querySelector("#products");
const stats     = document.querySelector(".stats");

// Մեկ քարտի HTML-ը
function createCard(product) {
  const { id, name, price, inStock } = product;   // Դաս 7

  return `
    <div class="col">
      <div class="wrapper ${inStock ? "" : "is-disabled"}">
        <h3 class="title">${name}</h3>
        <span class="price">${price.toLocaleString("ru-RU")} ֏</span>
        <span class="badge">${inStock ? "Առկա է" : "Սպառված"}</span>
        <button class="btn" data-id="${id}" ${inStock ? "" : "disabled"}>
          Գնել
        </button>
      </div>
    </div>
  `;
}

// Ամբողջ ցուցակի ցուցադրում
function render(list) {
  if (list.length === 0) {
    container.innerHTML = `<p class="empty">Ապրանք չգտնվեց։</p>`;
    return;
  }

  container.innerHTML = list.map(createCard).join("");
  updateStats(list);
}

// Վիճակագրություն (Դաս 6)
function updateStats(list) {
  const available = list.filter(p => p.inStock);
  const total = available.reduce((sum, p) => sum + p.price, 0);

  stats.textContent =
    `${list.length} ապրանք · ${available.length} առկա · ${total.toLocaleString("ru-RU")} ֏`;
}

// Գործարկում
render(products);

// Ֆիլտրված տարբերակներ — փորձիր Console-ում
// render(products.filter(p => p.inStock));
// render(products.filter(p => p.price < 100000));
// render([...products].sort((a, b) => a.price - b.price));
Ուշադրություն կառուցվածքին

createCard-ը միայն HTML է պատրաստում։ render-ը միայն էջ է դնում։ updateStats-ը միայն հաշվում է։ Մեկ ֆունկցիա = մեկ գործ (Դաս 4)։ Դաս 16-ում այս ֆայլը կբաժանենք մոդուլների։

09Տնային աշխատանք

Մինի-նախագիծ

Այս տնայինի գլխավոր մասը՝ «Դարակը JS-ով», քայլ առ քայլ, starter ֆայլերով և պատրաստի օրինակով։ Ներքևի ցուցակի մնացած կետերը առանձին վարժություններ են։

  1. Ստեղծիր էջ մեկ h1-ով և JS-ով փոխիր դրա տեքստը։ Ապա փորձիր նույնը innerHTML-ով՝ <em> թեգով ներսում։
  2. Ստեղծիր .box դիվ։ JS-ով ավելացրու .active դասը, հետո հանիր, հետո toggle արա երկու անգամ։ Ամեն քայլից հետո տպիր className-ը։
  3. CSS-ում գրիր .hidden { display: none; }։ JS-ով թաքցրու և ցույց տուր որևէ բլոկ՝ միայն classList-ով։
  4. Ստեղծիր առաջընթացի գիծ (progress bar)։ JS-ով դինամիկ սահմանիր դրա լայնությունը՝ style.width-ով ըստ տոկոսի փոփոխականի։
  5. Ստեղծիր <img> և JS-ով սահմանիր src և alt։ Ապա setAttribute-ով ավելացրու title։
  6. Կոճակին ավելացրու data-id և data-name։ JS-ով կարդա դրանք dataset-ով և տպիր։
  7. Ունես անունների զանգված։ map + join + innerHTML-ով գեներացրու <ul> ցուցակ։
  8. Նույն զանգվածով՝ արա նույնը, բայց createElement և append-ով։ Համեմատիր՝ որ ձևն է քեզ ավելի հարմար և գրիր մեկնաբանության մեջ։
  9. Վերցրու դասարանական products զանգվածը։ Render արա միայն 100 000-ից էժան ապրանքները, դասավորած գնով։
  10. Ավելացրու renderEmpty() դեպքի մշակում․ եթե ֆիլտրից հետո ապրանք չմնա, ցույց տուր «Ապրանք չգտնվեց» հաղորդագրություն։ Ստուգիր՝ ֆիլտրելով 1 000 000-ից թանկերը։

10Հիշելու թերթիկ

ԳրառումԻնչ է անում
el.textContent = "x"Փոխում է տեքստը (ապահով) ✅
el.innerHTML = "<b>x</b>"Դնում է HTML (զգույշ՝ XSS)
el.src / el.hrefՈւղիղ ատրիբուտներ
el.setAttribute("a", "b")Ցանկացած ատրիբուտ
el.dataset.idԿարդում է data-id-ն (միշտ տող)
el.classList.add("x")Ավելացնում է դաս
el.classList.remove("x")Հանում է դաս
el.classList.toggle("x")Կա → հանում, չկա → ավելացնում
el.classList.contains("x")true / false
el.style.backgroundColorՈւղիղ ոճ (camelCase, միավորով)
document.createElement("div")Ստեղծում է նոր էլեմենտ
parent.append(el)Դնում է ներսում՝ վերջում
parent.prepend(el)Դնում է ներսում՝ սկզբում
el.remove()Ջնջում է
el.insertAdjacentHTML("beforeend", html)Ավելացնում է HTML՝ առանց վերակառուցելու
arr.map(fn).join("")Զանգված → HTML տող

Ինքնաստուգում

  • Կարո՞ղ եմ բացատրել՝ ինչու օգտագործողի տվյալները չեն դրվում innerHTML-ի մեջ
  • Հասկանո՞ւմ եմ, ինչու է classList-ը գերադասելի style-ից
  • Գիտե՞մ, ինչու է innerHTML +=-ը վատ
  • Կարո՞ղ եմ առանց նայելու գրել map(...).join("") render
  • Կարո՞ղ եմ կառուցել .row > .col > .wrapper կառուցվածք JS-ով
Հաջորդ դասին

Իրադարձություններ։ Այսօր քարտերը ցուցադրվում են, բայց կոճակները դեռ մեռած են։ Հաջորդ դասին սովորում ենք addEventListener-ը, և էջը վերջապես կսկսի արձագանքել՝ մենյու, մոդալ, ակորդեոն, թաբեր։
