js/ui/header.js
function initBurger() {
    const toggle = document.querySelector(".nav-toggle");
    const nav = document.querySelector(".header nav");
    if (!toggle || !nav) return;

    document.addEventListener("keydown", (e) => {
        if (e.key === "Escape" && toggle.checked) {
            toggle.checked = false;
            toggle.focus();                      // ֆոկուսը վերադարձնում ենք մենյուի կոճակին
        }
    });

    nav.addEventListener("click", (e) => {
        if (e.target.closest("a")) toggle.checked = false;
    });
}

  • product.html?id=1…12-ը ցույց է տալիս ճիշտ ապրանքը, ?id=999՝ հաղորդագրություն
  • Պատկերասրահի փոքր նկարները փոխում են մեծը
  • Պատվերի ֆորման ցույց է տալիս բոլոր սխալները միանգամից, «Վերցնել խանութից»-ի դեպքում հասցեն պարտադիր չէ
  • Հաջող պատվերից հետո՝ շնորհակալության էջ պատվերի համարով, զամբյուղը դատարկ է
  • Գրանցումը ստուգում է գաղտնաբառերի համընկնումը և ցույց է տալիս ուժը
  • Ստուգման տրամաբանությունը գրված է մեկ անգամ՝ validation.js-ում
