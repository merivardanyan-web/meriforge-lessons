js/api.js
const cache = new Map();

async function fetchJson(url) {
    const response = await fetch(url);
    if (!response.ok) {                              // 404-ը catch չի ընկնում ինքնաբերաբար (JS Դաս 20)
        throw new ApiError(response.status, `Հարցումը ձախողվեց՝ ${url}`);
    }
    return response.json();
}

function request(url) {
    if (!cache.has(url)) {
        const promise = fetchJson(url).catch(error => {
            cache.delete(url);                       // ձախողվածը չենք պահում՝ «Կրկին փորձել»-ը նորից կհարցնի
            throw error;
        });
        cache.set(url, promise);
    }
    return cache.get(url);
}

export const api = {
    getProducts() {
        return request(PRODUCTS_URL);
    },

    async getProduct(id) {
        const products = await request(PRODUCTS_URL);
        const product = products.find(p => p.id === id);
        if (!product) throw new NotFoundError("Այդպիսի ապրանք չկա։");
        return product;
    }
};

export class ApiError extends Error {
    constructor(status, message) {
        super(message);
        this.name = "ApiError";
        this.status = status;
    }

    get isServerError() {
        return this.status >= 500;
    }
}

export class NotFoundError extends Error {
    constructor(message) {
        super(message);
        this.name = "NotFoundError";
    }
}

export class Cart {
    #items;                      // փակ — դրսից ոչ ոք չի փչացնի (JS Դաս 22)
    #storageKey;
    #listeners = [];

    constructor(storageKey) {
        this.#storageKey = storageKey;
        this.#items = load(storageKey, []);
    }

    // ---------- Կարդալու համար ----------
    get count() {
        return this.#items.reduce((sum, item) => sum + item.qty, 0);
    }

    // ...

    add(id, qty = 1) {
        const existing = this.#items.find(item => item.id === id);
        // Անփոփոխ թարմացում՝ նոր զանգված, ոչ push (JS Դաս 23)
        this.#items = existing
            ? this.#items.map(item => item.id === id ? { ...item, qty: Math.min(item.qty + qty, MAX_QTY) } : item)
            : [...this.#items, { id, qty: Math.min(qty, MAX_QTY) }];
        this.#save();
        return this;
    }

    // ...

    onChange(listener) {
        this.#listeners.push(listener);
    }

    // ---------- Ներքին ----------
    #save() {
        save(this.#storageKey, this.#items);
        this.#listeners.forEach(listener => listener(this));
    }
}

js/models/Wishlist.js
export class Wishlist {
    #ids;
    #storageKey;

    constructor(storageKey) {
        this.#storageKey = storageKey;
        this.#ids = new Set(load(storageKey, []));   // JSON-ը Set չգիտի՝ պահում ենք զանգված
    }

    get size() {
        return this.#ids.size;
    }

    has(id) {
        return this.#ids.has(id);
    }

    // Վերադարձնում է նոր վիճակը՝ true, եթե հիմա հավանած է
    toggle(id) {
        if (this.#ids.has(id)) {
            this.#ids.delete(id);
        } else {
            this.#ids.add(id);
        }
        save(this.#storageKey, [...this.#ids]);
        return this.#ids.has(id);
    }
}

export function countBy(products, key) {
    const counts = new Map();
    products.forEach(p => counts.set(p[key], (counts.get(p[key]) ?? 0) + 1));
    return counts;
}

export function getVisibleProducts(products, state, wishlist) {
    const query = state.search.toLowerCase();

    const result = products
        .filter(p => state.categories.length === 0 || state.categories.includes(p.category))
        .filter(p => state.brands.length === 0 || state.brands.includes(p.brand))
        .filter(p => state.min === null || p.price >= state.min)
        .filter(p => state.max === null || p.price <= state.max)
        .filter(p => state.stock !== "in" || p.inStock)
        .filter(p => state.stock !== "sale" || p.oldPrice !== null)
        .filter(p => state.stock !== "fav" || wishlist.has(p.id))
        .filter(p => query === "" ||
            p.name.toLowerCase().includes(query) ||
            CATEGORIES[p.category].toLowerCase().includes(query));

    return [...result].sort(SORTERS[state.sort] ?? SORTERS.popular);
}

404-ի դեպքում էջում հայտնվում է undefinedresponse.ok-ը չի ստուգվում, fetch-ը 404-ը սխալ չի համարում (Դաս 20)
«Կրկին փորձել»-ը ոչինչ չի անումՔեշում մնացել է ձախողված Promise-ը։ catch-ում ջնջիր այն cache.delete(url)
cart.#items-ը Console-ում սխալ է տալիսԱյդպես էլ պետք է․ փակ դաշտ է (Դաս 22)։ Օգտագործիր cart.count
Հավանածները կորչում են թարմացնելիսJSON.stringify(new Set([1]))-ը "{}" է։ Պահիր [...set]
this-ը undefined է forEach-ի ներսումՍովորական function-ի փոխարեն՝ arrow (Դաս 22)
catch-ը թաքցնում է ծրագրի իսկական սխալներըՍտուգիր instanceof ValidationError, մնացածը throw error (Դաս 23)

04Համեմատիր օրինակի հետ

Live Server-ով

Բացիր example-2-final-ը Live Server-ով (file://-ով մոդուլները և fetch-ը չեն աշխատի)։ Համեմատիր․

1. Ֆայլերի ծառը. Քանի՞ մոդուլ ունես, և ո՞րն է ամենամեծը։ Եթե մի ֆայլ 250 տողից երկար է, գուցե այն երկու գործ է անում։
2. Կախվածություններ. Նկարիր քո import-ների գրաֆը (Դաս 18)։ Օղակ կա՞։
3. Սխալներ. Network → Offline։ Ամեն էջ ի՞նչ է ցույց տալիս։
4. API-ի փոխարինում. Եթե վաղը ապրանքները գան /api/products-ից, քանի՞ ֆայլ պետք է փոխես։ Օրինակում՝ մեկ (config.js-ի PRODUCTS_URL-ը)։

05Ընդունման ստուգաթերթ

Ֆունկցիոնալություն

  • Console-ը մաքուր է բոլոր 9 էջում
  • Կատալոգ, ապրանք, զամբյուղ, պատվեր՝ տվյալները products.json-ից
  • Ամեն էջ ունի բեռնման և սխալի վիճակ, «Կրկին փորձել»-ը աշխատում է
  • Հավանածները պահպանվում են, «Միայն հավանածները» ֆիլտրը աշխատում է
  • Ֆիլտրերի կատեգորիաները և ապրանքանիշերը կառուցվում են տվյալներից՝ քանակներով
  • Ֆորմաները մերժում են սխալ տվյալները, սխալները՝ դաշտերի տակ

Կոդ (JS Դաս 24, բաժին 05)

  • Ոչ մի var, console.log, debugger
  • Ամեն fetch ստուգում է response.ok, ամեն catch ինչ-որ բան անում է
  • Ֆունկցիաների անունները սկսվում են բայով, կախարդական թվեր չկան (config.js)
  • Զամբյուղը և հավանածները՝ դասեր #private դաշտերով
  • Օգտագործողի տեքստը (որոնում, անուն) էջում է հայտնվում textContent-ով

Եզրային դեպքեր

  • Դատարկ զամբյուղ, դատարկ ֆիլտր, ?id=999
  • Network → Offline և սխալ ֆայլի անուն
  • Շատ երկար տեքստ որոնման դաշտում և անուն դաշտում
  • «Գնել»-ի կրկնակի արագ սեղմում․ քանակը 2 է, ոչ սխալ
  • Հեռախոս 320px, մուգ թեմա, միայն ստեղնաշար

Git

  • Առնվազն 10 իմաստալից commit, կա .gitignore
  • README՝ հասցեով, հնարավորություններով և Բլոկ 1-ի սահմանափակումների թարմացված ցուցակով
  • Հրապարակված տարբերակն աշխատում է

Հնարավոր հարցեր

  • Ինչու՞ ես զամբյուղում պահում միայն id և qty (Դաս 16)
  • Ինչու՞ է «Գնել»-ի listener-ը document-ի վրա, ոչ ամեն կոճակի
  • Ինչ կլինի, եթե fetch-ը վերադարձնի 404, և որտեղ է դա մշակվում
  • Ինչու՞ է քեշում պահվում Promise-ը, ոչ պատասխանը
  • Ինչով է Cart դասը ավելի լավ, քան Առաջադրանք 1-ի ֆունկցիաները, և ինչով՝ ոչ
  • Ինչու՞ Set հավանածների, բայց զանգված զամբյուղի համար
  • Որտեղ է այս կոդը փակույթ, և ինչ է այն «հիշում»
  • Ինչ կփոխվի PHP բլոկում, երբ ապրանքները գան իսկական API-ից
