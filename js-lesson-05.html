01Ինչո՞ւ են պետք զանգվածները

Պիտի պահես դասարանի 5 աշակերտի անունները։ Առանց զանգվածի՝

const student2 = "Դավիթ";
const student3 = "Մարիամ";

02Ինդեքսներ · հաշվարկը սկսվում է 0-ից

Սա ամենակարևոր բանն է, որ պիտի հիշես այսօր։ Առաջին էլեմենտի համարը 0 է, ոչ թե 1։

03Ավելացնել և հեռացնել

Չորս հիմնական մեթոդ։ Երկուսը աշխատում են վերջից, երկուսը՝ սկզբից։

ՄեթոդԻնչ է անումՈրտեղՎերադարձնում է
push(x)ավելացնում էվերջումնոր երկարությունը
pop()հեռացնում էվերջիցհեռացվածը
unshift(x)ավելացնում էսկզբումնոր երկարությունը
shift()հեռացնում էսկզբիցհեռացվածը
push և pop — վերջում
const cart = ["հաց", "կաթ"];

cart.push("պանիր");
console.log(cart);

cart.push("ձու", "կարագ");   // մի քանիսը միանգամից
console.log(cart);

const removed = cart.pop();
console.log(removed);
console.log(cart);
Console
["հաց", "կաթ", "պանիր"]
["հաց", "կաթ", "պանիր", "ձու", "կարագ"]
կարագ
["հաց", "կաթ", "պանիր", "ձու"]
unshift և shift — սկզբում
const queue = ["Դավիթ", "Մարիամ"];

queue.unshift("Անի");      // Անին անցավ առաջ
console.log(queue);

const first = queue.shift();  // առաջինը սպասարկվեց
console.log(first);
console.log(queue);
Console
["Անի", "Դավիթ", "Մարիամ"]
Անի
["Դավիթ", "Մարիամ"]
Հիշելու հնարք

push / pop — վերջում, ինչպես ափսեների կույտ։
shift / unshift — սկզբում, ինչպես հերթ խանութում։ shift-ը «տեղաշարժում» է բոլորին առաջ։

04slice — կտրել կրկնօրինակ

slice()-ը վերցնում է զանգվածի մի կտոր և վերադարձնում նոր զանգված։ Բնօրինակը չի փոխվում։

slice(սկիզբ, մինչև)
const nums = [10, 20, 30, 40, 50];

console.log(nums.slice(1, 3));   // ինդեքս 1-ից մինչև 3 (3-ը ՉԻ մտնում)
console.log(nums.slice(2));      // 2-ից մինչև վերջ
console.log(nums.slice(-2));     // վերջին երկուսը
console.log(nums);                // բնօրինակը անփոփոխ է
Console
[20, 30]
[30, 40, 50]
[40, 50]
[10, 20, 30, 40, 50]
Հիշիր

Երկրորդ թիվը չի մտնում արդյունքի մեջ։ slice(1, 3) վերցնում է ինդեքս 1 և 2։ Սա շփոթեցնող է, բայց նույն կանոնը գործում է JS-ի շատ տեղերում։

splice — վիրահատում է բնօրինակը

splice()-ը փոխում է ինքը զանգվածը՝ կարող է հեռացնել, ավելացնել կամ փոխարինել ցանկացած տեղում։

splice(որտեղից, քանիսը, ինչ_ավելացնել)
const fruits = ["խնձոր", "բանան", "նարինջ", "տանձ"];

// հեռացնել 1 հատ ինդեքս 1-ից
fruits.splice(1, 1);
console.log(fruits);

// ավելացնել՝ առանց հեռացնելու (քանակը = 0)
fruits.splice(1, 0, "ելակ", "կիվի");
console.log(fruits);

// փոխարինել՝ հեռացնել 2, դնել 1
fruits.splice(0, 2, "սալոր");
console.log(fruits);
Console
["խնձոր", "նարինջ", "տանձ"]
["խնձոր", "ելակ", "կիվի", "նարինջ", "տանձ"]
["սալոր", "կիվի", "նարինջ", "տանձ"]
slice ընդդեմ splice

slice = «պատճենել» → նոր զանգված, բնօրինակը անվնաս։
splice = «վիրահատել» → փոխում է բնօրինակը։
Անունները նման են, գործողությունը՝ բոլորովին տարբեր։ Ուշադիր եղիր։

05Փնտրել զանգվածում

indexOf և includes
const colors = ["կարմիր", "կանաչ", "կապույտ", "կանաչ"];

console.log(colors.indexOf("կապույտ"));
console.log(colors.indexOf("կանաչ"));       // ԱՌԱՋԻՆ գտնվածը
console.log(colors.lastIndexOf("կանաչ"));   // ՎԵՐՋԻՆ գտնվածը
console.log(colors.indexOf("սև"));          // չկա → -1

console.log(colors.includes("կարմիր"));
console.log(colors.includes("սև"));
Console
2
1
3
-1
true
false
Ինչո՞ւ -1

Եթե indexOf-ը վերադարձներ 0, մենք չէինք հասկանա՝ գտավ առաջին էլեմենտում, թե ընդհանրապես չգտավ։ Դրա համար «չգտա»-ի նշանը -1-ն է։

գործնական օգտագործում
const cart = ["հաց", "կաթ", "պանիր"];

function addToCart(item) {
  if (cart.includes(item)) {
    return `${item}-ը արդեն զամբյուղում է։`;
  }
  cart.push(item);
  return `${item}-ը ավելացվեց։`;
}

console.log(addToCart("ձու"));
console.log(addToCart("կաթ"));
console.log(cart);
Console
ձու-ը ավելացվեց։
կաթ-ը արդեն զամբյուղում է։
["հաց", "կաթ", "պանիր", "ձու"]

06join և split

join-ը զանգվածը դարձնում է տեքստ։ split-ը՝ հակառակը (այն տողի մեթոդ է, ոչ զանգվածի)։

join
const students = ["Անի", "Դավիթ", "Մարիամ"];

console.log(students.join(", "));
console.log(students.join(" — "));
console.log(students.join(""));
console.log(students.join());        // լռելյայն՝ ստորակետ
Console
Անի, Դավիթ, Մարիամ
Անի — Դավիթ — Մարիամ
ԱնիԴավիթՄարիամ
Անի,Դավիթ,Մարիամ
split
const text = "խնձոր,բանան,նարինջ";
const fruits = text.split(",");
console.log(fruits);
console.log(fruits.length);

const sentence = "Ես սովորում եմ JavaScript";
console.log(sentence.split(" "));

const word = "կատու";
console.log(word.split(""));       // տառ առ տառ
Console
["խնձոր", "բանան", "նարինջ"]
3
["Ես", "սովորում", "եմ", "JavaScript"]
["կ", "ա", "տ", "ու"]
Գործնական հնարք

Բառը շրջելու համար՝ word.split("").reverse().join("")։ Դաս 4-ում դա ցիկլով էինք անում — հիմա մեկ տողով է։

07Անցնել զանգվածով

Ահա այն պահը, երբ Դաս 3-ի ցիկլերը և զանգվածները միանում են։

for ցիկլով
const students = ["Անի", "Դավիթ", "Մարիամ"];

for (let i = 0; i < students.length; i++) {
  console.log(`${i + 1}. ${students[i]}`);
}
Console
1. Անի
2. Դավիթ
3. Մարիամ
Երկու կարևոր մանրուք

let i = 0 — սկսում ենք 0-ից, ոչ 1-ից։
i < length — խիստ փոքր, ոչ թե <=։ Հակառակ դեպքում վերջում կստանաս undefined։

for..of — ավելի պարզ ձև

Երբ ինդեքսը պետք չէ, այլ միայն արժեքները։

for..of
const prices = [1200, 450, 3000, 800];
let total = 0;

for (const price of prices) {
  total += price;
}

console.log(`Ընդամենը՝ ${total} դրամ`);
console.log(`Միջինը՝ ${total / prices.length} դրամ`);
Console
Ընդամենը՝ 5450 դրամ
Միջինը՝ 1362.5 դրամ
ամենամեծը գտնելը
const marks = [14, 19, 8, 17, 20, 11];
let max = marks[0];   // ենթադրում ենք՝ առաջինն է ամենամեծը

for (const mark of marks) {
  if (mark > max) {
    max = mark;
  }
}

console.log(`Ամենաբարձր գնահատականը՝ ${max}`);
Console
Ամենաբարձր գնահատականը՝ 20

08Պատճենել զանգված · spread

Ահա մի բան, որի վրա սայթաքում են գրեթե բոլոր սկսնակները։

անսպասելի վարքագիծ
const original = ["ա", "բ", "գ"];
const copy = original;      // «պատճենեցինք»

copy.push("դ");

console.log(copy);
console.log(original);        // ← բնօրինակն էլ փոխվեց
Console
["ա", "բ", "գ", "դ"]
["ա", "բ", "գ", "դ"]   ← ?!

Ինչո՞ւ։ Որովհետև զանգվածը փոփոխականում չի պահվում ամբողջությամբ։ Փոփոխականում պահվում է միայն հասցեն՝ ուր է այն ընկած հիշողության մեջ։ copy = original գրելով դու պատճենեցիր ոչ թե տունը, այլ միայն հասցեն թղթի վրա։ Երկու թուղթը ցույց են տալիս նույն տունը։

todo.js
const todos = [];

function addTodo(task) {
  if (todos.includes(task)) {
    return "Այս գործն արդեն ցուցակում է։";
  }
  todos.push(task);
  return `Ավելացվեց՝ ${task}`;
}

function removeTodo(task) {
  const index = todos.indexOf(task);
  if (index === -1) {
    return "Այդպիսի գործ չկա։";
  }
  todos.splice(index, 1);
  return `Ջնջվեց՝ ${task}`;
}

function showTodos() {
  if (todos.length === 0) {
    console.log("Ցուցակը դատարկ է։");
    return;
  }
  console.log(`--- ${todos.length} գործ ---`);
  for (let i = 0; i < todos.length; i++) {
    console.log(`${i + 1}. ${todos[i]}`);
  }
}

console.log(addTodo("Տնային գրել"));
console.log(addTodo("Հաց գնել"));
console.log(addTodo("Հաց գնել"));
showTodos();
console.log(removeTodo("Հաց գնել"));
showTodos();
